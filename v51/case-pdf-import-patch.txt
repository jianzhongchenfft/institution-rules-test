  function importSupervisors(){ return (state.caseStaff||[]).filter(s=>s.role==='supervisor'); }
  function importMaskId(v=''){ const s=String(v||'').trim(); return s.length>=6?s.slice(0,5)+'*****':(s||'—'); }
  function importFmtDate(v){ return v?rocDate(v+'T00:00:00'):'—'; }
  function importRocToIso(v){
    if(!v)return '';
    const m=String(v).trim().match(/(\d{1,4})\s*[\/.-]\s*(\d{1,2})\s*[\/.-]\s*(\d{1,2})/); if(!m)return '';
    let y=Number(m[1]); if(y<1911)y+=1911;
    return `${String(y).padStart(4,'0')}-${String(Number(m[2])).padStart(2,'0')}-${String(Number(m[3])).padStart(2,'0')}`;
  }
  function importCleanText(s=''){ return String(s).replace(/[｜|]/g,' ').replace(/[\t\u00a0]+/g,' ').replace(/ +/g,' ').trim(); }
  function importLines(text=''){ return String(text).split(/\r?\n/).map(importCleanText).filter(Boolean); }
  function importFindLineValue(text,labels){
    const lines=importLines(text);
    for(const label of labels){
      for(let i=0;i<lines.length;i++){
        const p=lines[i].indexOf(label); if(p<0)continue;
        let rest=lines[i].slice(p+label.length).replace(/^[\s:：*＊]+/,'').trim();
        if(rest)return rest;
        if(lines[i+1])return lines[i+1];
      }
    }
    return '';
  }
  function importFirstMatch(text,patterns,group=1){ for(const re of patterns){const m=String(text).match(re);if(m&&m[group])return importCleanText(m[group]);} return ''; }
  function importTrimAtLabels(v,labels){ let out=String(v||''); labels.forEach(x=>{const p=out.indexOf(x);if(p>=0)out=out.slice(0,p);}); return out.replace(/^[\s:：*＊]+|[\s*＊]+$/g,'').trim(); }

  async function importLoadLib(src,test){
    if(test())return;
    await new Promise((resolve,reject)=>{const s=document.createElement('script');s.src=src;s.async=true;s.onload=resolve;s.onerror=()=>reject(new Error('外部解析元件載入失敗'));document.head.appendChild(s);});
  }

  async function importPrepareOcr(){
    await importLoadLib('https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js',()=>!!window.pdfjsLib);
    window.pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
    await importLoadLib('https://cdn.jsdelivr.net/npm/tesseract.js@5.1.1/dist/tesseract.min.js',()=>!!window.Tesseract);
  }

  async function importOcrPdf(file,onProgress){
    await importPrepareOcr();
    const pdf=await window.pdfjsLib.getDocument({data:new Uint8Array(await file.arrayBuffer())}).promise;
    const wanted=[1,2,14,19,20].filter(n=>n<=pdf.numPages);
    const result={};
    const worker=await window.Tesseract.createWorker('chi_tra+eng',1,{logger:m=>{if(m.status==='recognizing text'&&onProgress)onProgress(`文字辨識中 ${Math.round((m.progress||0)*100)}%`);}});
    try{
      for(let i=0;i<wanted.length;i++){
        const n=wanted[i]; if(onProgress)onProgress(`正在解析第 ${n} 頁（${i+1}/${wanted.length}）…`);
        const page=await pdf.getPage(n); const viewport=page.getViewport({scale:1.65});
        const canvas=document.createElement('canvas'); canvas.width=Math.ceil(viewport.width); canvas.height=Math.ceil(viewport.height);
        const ctx=canvas.getContext('2d',{willReadFrequently:true}); await page.render({canvasContext:ctx,viewport}).promise;
        const r=await worker.recognize(canvas); result[n]=r?.data?.text||'';
      }
    }finally{ await worker.terminate(); }
    return result;
  }

  function importParseServices(text){
    const map=new Map();
    for(let raw of importLines(text)){
      raw=raw.replace(/\b(BA|GA)\s+(\d)/gi,'$1$2');
      const cm=raw.match(/\b(BA\d{2}(?:-\d+)?|GA\d{2})\b/i); if(!cm)continue;
      const code=cm[1].toUpperCase(); const group=code.startsWith('GA')?'GA':'BA';
      const after=raw.slice(cm.index+cm[0].length);
      const price=after.match(/\d+(?:\.\d+)?\s*元/);
      let name=price?after.slice(0,price.index):after;
      name=name.replace(/[\[\]【】]/g,'').replace(/[（(]\s*\d{3,}\s*[）)]/g,'').replace(/^[\s:：*-]+/,'').trim();
      const nums=after.match(/\d+(?:\.\d+)?/g)||[];
      let qty=''; if(nums.length>=2)qty=nums[nums.length-2];
      if(!map.has(code))map.set(code,{service_group:group,service_code:code,service_name:name,approved_quantity:qty});
    }
    return [...map.values()].sort((a,b)=>a.service_code.localeCompare(b.service_code));
  }

  function importParsePages(pages){
    const p1=pages[1]||'',p2=pages[2]||'',p14=pages[14]||'',p19=pages[19]||'',p20=pages[20]||'';
    const all=[p1,p2,p14,p19,p20].join('\n');
    let nationalId=importFirstMatch(p1,[/([A-Z][12]\d{8})/i]);
    let caseName=importFirstMatch(p1,[/個案姓名(?:\s*[\/／]\s*身分證字號)?\s*[：:]?\s*([\u4e00-\u9fff]{2,5})/]);
    if(!caseName&&nationalId){const idx=p1.indexOf(nationalId);const before=p1.slice(Math.max(0,idx-40),idx);caseName=importFirstMatch(before,[/([\u4e00-\u9fff]{2,5})\s*[\/／]?\s*$/]);}
    let birth=importFirstMatch(p1,[/個案生日\s*[：:]?\s*(\d{1,3}\s*[\/.-]\s*\d{1,2}\s*[\/.-]\s*\d{1,2})/]);
    let gender=importFirstMatch(p1,[/性別\s*[：:]?\s*([男女])/]);
    let phone=importFirstMatch(p1,[/個案電話\s*[：:]?\s*(0\d[\d\-\s]{7,14})/]);
    let address=importFindLineValue(p1,['居住(通訊)地址','居住（通訊）地址','居住地址','通訊地址']);
    address=importTrimAtLabels(address,['戶籍地址','常用語言','個案電話','性別']);
    let living=importFindLineValue(p1,['目前居住狀況']);
    let livesAlone=null; if(/獨居/.test(living))livesAlone=true; else if(/同住|與他人/.test(living))livesAlone=false;

    let contactName=importFindLineValue(p2,['聯絡人姓名']); contactName=importTrimAtLabels(contactName,['聯絡人電話','聯絡人手機','與個案']);
    let contactRel=importFindLineValue(p2,['與個案服務者關係或身分','與個案關係','服務者關係或身分']); contactRel=importTrimAtLabels(contactRel,['其他關係說明','聯絡人Email','聯絡人地址']);
    let contactPhone=importFirstMatch(p2,[/聯絡人手機\s*[：:]?\s*(0\d[\d\-\s]{7,14})/,/聯絡人電話\s*[：:]?\s*(0\d[\d\-\s]{7,14})/]);

    let assessment=importFirstMatch(all,[/(?:評估完成日期|評估完成日|評估日期)\s*[：:]?\s*(\d{1,3}[\/.-]\d{1,2}[\/.-]\d{1,2})/]);
    let planApproval=importFirstMatch(p14+'\n'+all,[/(?:計畫核定日期|計畫核定日|核定日期)\s*[：:]?\s*(\d{1,3}[\/.-]\d{1,2}[\/.-]\d{1,2})/]);
    let cms=importFirstMatch(p14+'\n'+p19+'\n'+all,[/CMS\s*(?:第)?\s*([1-8])\s*級/i]);
    let aUnit=importFindLineValue(p19,['A單位名稱']); aUnit=importTrimAtLabels(aUnit,['A個管姓名','聯絡電話','聯絡Email']);
    let manager=importFindLineValue(p19,['A個管姓名']); manager=importTrimAtLabels(manager,['聯絡電話','聯絡Email']);
    let managerPhone=importFirstMatch(p19,[/聯絡電話\s*[：:]?\s*(0\d[\d\-\s]{7,14})/]);
    let assessor=importFindLineValue(p1,['承辦人','評估人員','照顧專員']); assessor=importTrimAtLabels(assessor,['原照管者','原照管員','是否為A輔干','備註']);
    const services=importParseServices(p19+'\n'+p20);
    return {
      case_name:caseName,national_id:nationalId.toUpperCase(),birth_date:importRocToIso(birth),gender,
      address,phone:phone.replace(/\s/g,''),lives_alone:livesAlone,
      contact:{contact_name:contactName,relationship:contactRel,phone:contactPhone.replace(/\s/g,'')},
      assessment_date:importRocToIso(assessment),cms_level:cms,plan_approval_date:importRocToIso(planApproval),
      a_unit_name:aUnit,case_manager_name:manager,case_manager_phone:managerPhone.replace(/\s/g,''),assessor_name:assessor,
      services
    };
  }

  async function importSave(data){
    if(!data.case_name){toast('個案姓名不可空白','warn');return false;}
    showLoading(true);
    try{
      const {data:r,error}=await db.rpc('import_case_pdf_data',{payload:data}); if(error)throw error;
      closeModal(); toast(r?.action==='updated'?'已更新既有個案資料':'個案已建立','success'); navigate('cases'); return true;
    }catch(err){console.error(err);toast('個案匯入失敗：'+(err?.message||err),'error');return false;} finally{showLoading(false);}
  }
