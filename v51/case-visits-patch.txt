  function canUseCases(){
    return !!state.profile && ['supervisor','business_manager','organization_manager','admin'].includes(state.profile.role);
  }

  function caseStaffName(id){
    return state.caseStaff.find(s=>s.id===id)?.display_name || '尚未指派';
  }

  function caseTypeLabel(type){
    return ({home:'家訪',phone:'電訪',new_case:'新案家訪',other:'其他'})[type] || '其他';
  }

  function resultLabel(v){
    return ({improved:'已改善',ongoing:'持續追蹤',worse:'情況惡化',referred:'已轉介／處理'})[v] || '—';
  }

  function dateOnly(value){
    if(!value) return null;
    const d = new Date(value + (String(value).length===10?'T00:00:00':''));
    return Number.isNaN(d.getTime()) ? null : d;
  }

  function visitListFor(caseId){
    return state.caseVisits.filter(v=>v.case_id===caseId).sort((a,b)=>{
      const da = new Date((a.visit_date||'1970-01-01')+'T00:00:00');
      const dbb = new Date((b.visit_date||'1970-01-01')+'T00:00:00');
      return dbb-da || new Date(b.created_at)-new Date(a.created_at);
    });
  }

  function latestVisitFor(caseId){
    return visitListFor(caseId)[0] || null;
  }

  function nextVisitFor(caseId){
    const v = visitListFor(caseId).find(x=>x.next_visit_date);
    return v?.next_visit_date || null;
  }

  function isFollowUpResolved(visitId){
    return state.caseVisits.some(v=>v.previous_follow_up_visit_id===visitId && !!v.previous_follow_up_result);
  }

  function pendingFollowUpsFor(caseId){
    return visitListFor(caseId)
      .filter(v=>v.follow_up_needed && !isFollowUpResolved(v.id))
      .sort((a,b)=>String(a.follow_up_date||'9999-12-31').localeCompare(String(b.follow_up_date||'9999-12-31')));
  }

  function caseStatusInfo(c){
    const today = new Date(); today.setHours(0,0,0,0);
    const pending = pendingFollowUpsFor(c.id);
    const overdueFollow = pending.find(v=>v.follow_up_date && dateOnly(v.follow_up_date)<today);
    if(overdueFollow) return {key:'overdue',text:'追蹤逾期'};
    if(pending.length) return {key:'followup',text:'待追蹤'};
    const next = dateOnly(nextVisitFor(c.id));
    if(!next) return {key:'unplanned',text:'未排定'};
    if(next<today) return {key:'overdue',text:'訪視逾期'};
    const soon = new Date(today); soon.setDate(soon.getDate()+30);
    if(next<=soon) return {key:'due',text:'即將到期'};
    return {key:'normal',text:'正常'};
  }

  function currentCaseScope(){
    let list = state.cases.filter(c=>c.service_status!=='closed');
    const filter = state.caseSupervisorFilter || (state.profile?.role==='supervisor'?'mine':'all');
    if(filter==='mine') list = list.filter(c=>c.supervisor_id===state.profile?.id);
    else if(filter!=='all') list = list.filter(c=>c.supervisor_id===filter);
    return list;
  }

  async function loadCaseVisitData(silent=false){
    if(!canUseCases()) return;
    if(!silent) showLoading(true);
    try{
      const [staffRes,casesRes,visitsRes] = await Promise.all([
        db.rpc('get_case_staff_directory'),
        db.from('care_cases').select('id,case_name,supervisor_id,care_worker_name,open_date,cms_level,service_status,notes,created_at').order('case_name'),
        db.from('case_visits').select('*').order('visit_date',{ascending:false}).order('created_at',{ascending:false})
      ]);
      if(staffRes.error) throw staffRes.error;
      if(casesRes.error) throw casesRes.error;
      if(visitsRes.error) throw visitsRes.error;
      state.caseStaff = staffRes.data || [];
      state.cases = casesRes.data || [];
      state.caseVisits = visitsRes.data || [];
      if(!state.caseSupervisorFilter) state.caseSupervisorFilter = state.profile?.role==='supervisor' ? 'mine' : 'all';
      state.caseDataLoaded = true;
    }catch(err){
      console.error(err);
      toast('讀取個案訪視資料失敗：'+(err?.message||err),'error');
    }finally{
      if(!silent) showLoading(false);
    }
  }

  async function saveCase(){
    const caseName=$('caseName').value.trim();
    if(!caseName){toast('請輸入個案姓名','warn');return false;}
    showLoading(true);
    try{
      const {error}=await db.from('care_cases').insert({
        case_name:caseName,
        supervisor_id:$('caseSupervisor').value||null,
        care_worker_name:$('caseWorker').value.trim()||null,
        open_date:$('caseOpenDate').value||null,
        cms_level:$('caseCms').value.trim()||null,
        notes:$('caseNotes').value.trim()||null,
        created_by:state.user.id
      });
      if(error) throw error;
      closeModal();
      await loadCaseVisitData(true);
      renderCaseVisitsPage();
      toast('個案已建立','success');
      return true;
    }catch(err){toast('新增個案失敗：'+(err?.message||err),'error');return false;}
    finally{showLoading(false);}
  }

  function buildVisitSummary(c,prior){
    const type=$('visitType').value;
    const date=$('visitDate').value;
    const contact=$('contactPerson').value.trim();
    const intro = type==='phone' ? `${date?rocDate(date+'T00:00:00'):''}電聯${contact||'個案／家屬'}了解近期生活及居家服務情形。` : `${date?rocDate(date+'T00:00:00'):''}進行${caseTypeLabel(type)}，了解個案近期狀況及居家服務情形。`;
    const parts=[intro];
    if($('physicalStatus').value==='stable') parts.push('個案近期身體狀況穩定。'); else parts.push(`身體狀況有變化${$('physicalNote').value.trim()?`，${$('physicalNote').value.trim()}`:''}。`);
    if($('mentalStatus').value==='changed') parts.push(`精神或認知狀況有變化${$('mentalNote').value.trim()?`，${$('mentalNote').value.trim()}`:''}。`);
    if($('mobilityStatus').value==='changed') parts.push(`行動能力有變化${$('mobilityNote').value.trim()?`，${$('mobilityNote').value.trim()}`:''}。`);
    if($('dailyStatus').value==='abnormal') parts.push(`飲食、排泄或睡眠有異常${$('dailyNote').value.trim()?`，${$('dailyNote').value.trim()}`:''}。`);
    const fit=$('serviceFit').value;
    parts.push(fit==='fit'?'目前服務內容符合需求。':fit==='adjust'?'目前服務內容部分需調整，後續將再確認需求。':'目前狀況建議重新評估服務需求。');
    parts.push($('workerStatus').value==='stable'?'居服員服務情形穩定。':'居服員服務有需改善事項，後續將持續追蹤。');
    if($('familyFeedback').value==='none') parts.push('個案／家屬目前無其他意見。'); else parts.push(`個案／家屬有提出意見${$('familyFeedbackNote').value.trim()?`，${$('familyFeedbackNote').value.trim()}`:''}。`);
    if(prior && $('resolvePrior')?.checked) parts.push(`另追蹤前次事項，本次結果為「${resultLabel($('priorResult').value)}」。`);
    if($('followNeeded').checked){const item=$('followItem').value.trim();parts.push(`本次仍需後續追蹤${item?`「${item}」`:''}。`);} else parts.push('目前無新增追蹤事項。');
    return parts.join('');
  }

  async function saveVisit(c,prior){
    const visitDate=$('visitDate').value;
    if(!visitDate){toast('請填寫訪視日期','warn');return false;}
    const followNeeded=$('followNeeded').checked;
    if(followNeeded && !$('followItem').value.trim()){toast('已勾選需要追蹤，請填寫追蹤事項','warn');return false;}
    let summary=$('visitSummary').value.trim();
    if(!summary) summary=buildVisitSummary(c,prior);
    showLoading(true);
    try{
      const {error}=await db.from('case_visits').insert({
        case_id:c.id,
        visit_type:$('visitType').value,
        visit_date:visitDate,
        visitor_staff_id:state.profile.id,
        contact_person:$('contactPerson').value.trim()||null,
        physical_status:$('physicalStatus').value,
        physical_note:$('physicalNote').value.trim()||null,
        mental_status:$('mentalStatus').value,
        mental_note:$('mentalNote').value.trim()||null,
        mobility_status:$('mobilityStatus').value,
        mobility_note:$('mobilityNote').value.trim()||null,
        daily_status:$('dailyStatus').value,
        daily_note:$('dailyNote').value.trim()||null,
        service_fit:$('serviceFit').value,
        worker_service_status:$('workerStatus').value,
        family_feedback:$('familyFeedback').value,
        family_feedback_note:$('familyFeedbackNote').value.trim()||null,
        summary,
        previous_follow_up_visit_id:prior && $('resolvePrior')?.checked ? prior.id : null,
        previous_follow_up_result:prior && $('resolvePrior')?.checked ? $('priorResult').value : null,
        follow_up_needed:followNeeded,
        follow_up_item:followNeeded ? ($('followItem').value.trim()||null) : null,
        follow_up_date:followNeeded ? ($('followDate').value||null) : null,
        next_visit_date:$('nextVisitDate').value||null,
        created_by:state.user.id
      });
      if(error) throw error;
      closeModal();
      await loadCaseVisitData(true);
      renderCaseVisitsPage();
      toast('訪視紀錄已儲存','success');
      return true;
    }catch(err){toast('儲存訪視紀錄失敗：'+(err?.message||err),'error');return false;}
    finally{showLoading(false);}
  }
