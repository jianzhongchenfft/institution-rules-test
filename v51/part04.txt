  const el = (tag, attrs={}, children=[]) => {
    const node = document.createElement(tag);
    Object.entries(attrs).forEach(([k,v]) => {
      if(k === 'class') node.className = v;
      else if(k === 'text') node.textContent = v;
      else if(k.startsWith('on') && typeof v === 'function') node.addEventListener(k.slice(2).toLowerCase(), v);
      else if(v !== null && v !== undefined) node.setAttribute(k,v);
    });
    (Array.isArray(children)?children:[children]).forEach(c => node.append(c?.nodeType ? c : document.createTextNode(String(c ?? ''))));
    return node;
  };

  async function init(){
    wireStaticEvents();
    const { data, error } = await db.auth.getSession();
    if(error) console.error(error);
    await applySession(data?.session || null);
    db.auth.onAuthStateChange((event, session) => {
      if(event === 'SIGNED_IN' || event === 'SIGNED_OUT' || event === 'TOKEN_REFRESHED'){
        setTimeout(()=>applySession(session),0);
      }
    });
  }

  function wireStaticEvents(){
    $('googleLoginBtn').addEventListener('click', loginGoogle);
    $('mobileMenu').addEventListener('click', () => $('sidebar').classList.toggle('open'));
    $('userChip').addEventListener('click', async () => { if(confirm('要登出內部管理系統嗎？')) await db.auth.signOut(); });
    $('notifyBtn').addEventListener('click', () => toast('目前沒有新通知。','success'));
    document.addEventListener('click', (e) => {
      if(window.innerWidth <= 820 && $('sidebar').classList.contains('open') && !e.target.closest('#sidebar') && !e.target.closest('#mobileMenu')) $('sidebar').classList.remove('open');
    });
  }

  async function loginGoogle(){
    $('loginError').classList.add('hidden');
    try{
      const redirectTo = window.location.origin + window.location.pathname;
      const { error } = await db.auth.signInWithOAuth({ provider:'google', options:{ redirectTo } });
      if(error) throw error;
    }catch(err){
      $('loginError').textContent = '登入失敗：' + (err?.message || err);
      $('loginError').classList.remove('hidden');
    }
  }

  async function applySession(session){
    state.session = session;
    state.user = session?.user || null;
    if(!session){
      state.profile = null;
      $('appView').classList.add('hidden');
      $('loginView').classList.remove('hidden');
      return;
    }
    showLoading(true);
    try{
      const email = session.user.email || '';
      const { data:profile, error } = await db.from('staff_users').select('id,email,display_name,role,is_active').ilike('email', email).maybeSingle();
      if(error) throw error;
      if(!profile || !profile.is_active){
        await db.auth.signOut();
        $('loginError').textContent = '此 Google 帳號尚未被加入機構授權名單。';
        $('loginError').classList.remove('hidden');
        return;
      }
      state.profile = profile;
      $('loginView').classList.add('hidden');
      $('appView').classList.remove('hidden');
      $('userName').textContent = profile.display_name || email;
      $('userRole').textContent = roleLabels[profile.role] || profile.role;
      $('avatar').textContent = (profile.display_name || email || '使').slice(0,1);
      renderNav();
      await loadRegulations();
      navigate(state.page || 'dashboard');
    }catch(err){
      console.error(err);
      toast('讀取使用者資料失敗：' + (err?.message || err),'error');
    }finally{ showLoading(false); }
  }

  function renderNav(){
    const nav = $('nav'); nav.innerHTML='';
    let section='';
    modules.forEach(m => {
      if(m.section !== section){ section=m.section; nav.append(el('div',{class:'nav-section',text:section})); }
      const btn = el('button',{class:`nav-btn ${state.page===m.id?'active':''}`,onClick:()=>navigate(m.id)},[
        el('span',{class:'nav-icon',text:m.icon}), el('span',{text:m.label})
      ]);
      if(m.badge) btn.append(el('span',{class:'nav-badge',text:m.badge}));
      nav.append(btn);
    });
  }
