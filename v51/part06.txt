nt?.status||'draft';

      <div class="reg-actions"></div>`;
    const actions=card.querySelector('.reg-actions');
    if(current?.file_path) actions.append(el('button',{class:'btn sm primary',text:'下載最新版',onClick:()=>downloadVersion(current)}));
    actions.append(el('button',{class:'btn sm',text:`版本紀錄 (${versions.length})`,onClick:()=>openVersionsModal(reg)}));
    if(canManage()) actions.append(el('button',{class:'btn sm',text:'＋ 上傳新版本',onClick:()=>openRegulationModal(reg)}));
    if(canManage() && drafts[0]) actions.append(el('button',{class:'btn sm',text:'發布草稿',onClick:()=>publishVersion(reg,drafts[0])}));
    return card;
  }

  async function downloadVersion(v){
    if(!v.file_path){ toast('此版本沒有附件檔案','warn'); return; }
    showLoading(true);
    try{
      const { data, error }=await db.storage.from(STORAGE_BUCKET).download(v.file_path);
      if(error) throw error;
      const url=URL.createObjectURL(data); const a=document.createElement('a'); a.href=url; a.download=v.file_name||'內規文件'; document.body.append(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),1500);
      toast('檔案下載已開始','success');
    }catch(err){ toast('下載失敗：'+(err?.message||err),'error'); }
    finally{showLoading(false);}
  }

  async function saveRegulation(existing){
    const title=existing?.title || $('fTitle')?.value.trim();
    const category=existing?.category || $('fCategory')?.value.trim();
    const version=$('fVersion').value.trim();
    const effectiveDate=$('fDate').value || null;
    const notes=$('fNotes').value.trim();
    const file=$('fFile').files[0];
    const publishNow=$('fPublish').checked;
    if(!title || !category || !version || !file){ toast('請填妥名稱、分類、版本並選擇附件檔案','warn'); return false; }
    const allowed=['application/pdf','application/msword','application/vnd.openxmlformats-officedocument.wordprocessingml.document'];
    if(!allowed.includes(file.type) && !/\.(pdf|doc|docx)$/i.test(file.name)){ toast('附件只允許 PDF、DOC 或 DOCX','warn'); return false; }
    if(file.size>25*1024*1024){ toast('檔案不可超過 25 MB','warn'); return false; }
    showLoading(true);
    let uploadedPath=null;
    try{
      const regId=existing?.id || crypto.randomUUID();
      const ext=(file.name.split('.').pop()||'bin').toLowerCase().replace(/[^a-z0-9]/g,'');
      const path=`${regId}/${crypto.randomUUID()}.${ext}`;
      const { error:upErr }=await db.storage.from(STORAGE_BUCKET).upload(path,file,{contentType:file.type||undefined,upsert:false});
      if(upErr) throw upErr; uploadedPath=path;
      if(!existing){
        const slug='rule-'+Date.now()+'-'+crypto.randomUUID().slice(0,8);
        const { error:rErr }=await db.from('regulations').insert({id:regId,slug,title,category,is_active:true});
        if(rErr) throw rErr;
      }
      const versionId=crypto.randomUUID();
      const { error:vErr }=await db.from('regulation_versions').insert({
        id:versionId,regulation_id:regId,version_label:version,effective_date:effectiveDate,summary:'',revision_notes:notes,
        status:'draft',created_by:state.user.id,file_path:path,file_name:file.name,file_mime_type:file.type||null,file_size:file.size
      });
      if(vErr) throw vErr;
      if(publishNow){ await publishVersionByIds(regId,versionId,false); }
      closeModal(); await loadRegulations(); renderRegulations(); toast(publishNow?'內規已上傳並發布':'內規草稿已上傳','success'); return true;
    }catch(err){
      console.error(err);
      if(uploadedPath){ try{ await db.storage.from(STORAGE_BUCKET).remove([uploadedPath]); }catch(_){} }
      toast('上傳失敗：'+(err?.message||err),'error'); return false;
    }finally{showLoading(false);}
  }

  async function publishVersion(reg,version){
    if(!confirm(`確定要發布「${reg.title}」V${version.version_label} 嗎？\n原已發布版本會改為歷史版本。`)) return;
    showLoading(true);
    try{ await publishVersionByIds(reg.id,version.id,true); await loadRegulations(); renderRegulations(); toast('版本已正式發布','success'); }
    catch(err){toast('發布失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  async function publishVersionByIds(regId,versionId,throwErrors=true){
    const { error:e1 }=await db.from('regulation_versions').update({status:'superseded'}).eq('regulation_id',regId).eq('status','published').neq('id',versionId);
    if(e1){ if(throwErrors) throw e1; else throw e1; }
    const { error:e2 }=await db.from('regulation_versions').update({status:'published',published_at:new Date().toISOString()}).eq('id',versionId);
    if(e2) throw e2;
  }
