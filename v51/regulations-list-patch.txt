  function renderRegulations(){
    const cats=[...new Set(state.regulations.map(r=>r.category).filter(Boolean))].sort();
    const q=state.regulationQuery.trim().toLowerCase();
    const filtered=state.regulations.filter(r=>{
      const matchQ=!q || r.title.toLowerCase().includes(q) || (r.category||'').toLowerCase().includes(q);
      const matchCat=state.regulationCategory==='all' || r.category===state.regulationCategory;
      return matchQ&&matchCat;
    });
    $('content').innerHTML=`
      <div class="toolbar"><div class="searchbox"><input id="regSearch" placeholder="搜尋內規名稱或分類" value="${escapeHTML(state.regulationQuery)}"></div><select id="regCat"><option value="all">全部分類</option>${cats.map(c=>`<option ${state.regulationCategory===c?'selected':''} value="${escapeHTML(c)}">${escapeHTML(c)}</option>`).join('')}</select><button id="reloadRegs" class="btn">↻ 重新整理</button>${canManage()?'<button id="addRegBtn" class="btn primary">＋ 上傳新內規</button>':''}</div>
      <div id="regListShell" class="reg-list-shell"><div class="reg-list-head"><span>文件名稱</span><span>版本</span><span>生效日期</span><span>狀態</span></div><div id="regList" class="reg-grid reg-list"></div></div>`;
    $('regSearch').addEventListener('input',e=>{ const pos=e.target.selectionStart; state.regulationQuery=e.target.value; renderRegulations(); const input=$('regSearch'); if(input){ input.focus(); try{input.setSelectionRange(pos,pos);}catch(_){}} });
    $('regCat').addEventListener('change',e=>{state.regulationCategory=e.target.value; renderRegulations();});
    $('reloadRegs').addEventListener('click',async()=>{showLoading(true);await loadRegulations();showLoading(false);renderRegulations();toast('已重新整理內規資料','success');});
    $('addRegBtn')?.addEventListener('click',()=>openRegulationModal());
    const list=$('regList');
    if(!filtered.length){ const shell=$('regListShell'); shell.className=''; shell.innerHTML='<div class="empty"><div class="icon">📂</div><h3>沒有符合條件的內規</h3><p>請調整搜尋條件，或由有管理權限的人員上傳新的內規文件。</p></div>'; return; }
    filtered.forEach(reg=>list.append(renderRegCard(reg)));
  }

  function renderRegCard(reg){
    const {current}=getVersionArrays(reg);
    const status=current?.status||'draft';
    const statusText=status==='published'?'已發布':status==='draft'?'草稿':'歷史版本';
    const card=el('article',{class:'reg-card reg-row'});
    card.innerHTML=`
      <div class="reg-cell title-cell"><button type="button" class="reg-title-btn">${escapeHTML(reg.title)}</button></div>
      <div class="reg-cell version-cell">${current?'V'+escapeHTML(current.version_label):'—'}</div>
      <div class="reg-cell date-cell">${current?.effective_date?rocDate(current.effective_date+'T00:00:00'):'—'}</div>
      <div class="reg-cell status-cell"><span class="status ${status}">${statusText}</span></div>`;
    card.querySelector('.reg-title-btn').addEventListener('click',()=>openRegulationDetailModal(reg));
    return card;
  }

  function openRegulationDetailModal(reg){
    const {versions,current,drafts}=getVersionArrays(reg);
    const status=current?.status||'draft';
    const statusText=status==='published'?'已發布':status==='draft'?'草稿':'歷史版本';
    const html=`
      <div class="reg-detail-summary">
        <div class="reg-detail-item"><div class="reg-detail-label">分類</div><div class="reg-detail-value">${escapeHTML(reg.category||'未分類')}</div></div>
        <div class="reg-detail-item"><div class="reg-detail-label">目前版本</div><div class="reg-detail-value">${current?'V'+escapeHTML(current.version_label):'—'}</div></div>
        <div class="reg-detail-item"><div class="reg-detail-label">生效日期</div><div class="reg-detail-value">${current?.effective_date?rocDate(current.effective_date+'T00:00:00'):'—'}</div></div>
        <div class="reg-detail-item"><div class="reg-detail-label">狀態</div><div class="reg-detail-value"><span class="status ${status}">${statusText}</span></div></div>
        <div class="reg-detail-item full"><div class="reg-detail-label">最新檔案</div><div class="reg-detail-value">${current?.file_name?escapeHTML(current.file_name):'尚未上傳檔案'}</div></div>
        <div class="reg-detail-item"><div class="reg-detail-label">版本數</div><div class="reg-detail-value">${versions.length} 個版本${drafts.length?`・${drafts.length} 草稿`:''}</div></div>
      </div>
      <div id="regDetailActions" class="reg-detail-actions"></div>`;
    openModal({title:reg.title,html,showConfirm:false});
    const actions=$('regDetailActions');
    if(current?.file_path) actions.append(el('button',{class:'btn primary',text:'下載最新版',onClick:()=>downloadVersion(current)}));
    actions.append(el('button',{class:'btn',text:`版本紀錄 (${versions.length})`,onClick:()=>openVersionsModal(reg)}));
    if(canManage()) actions.append(el('button',{class:'btn',text:'＋ 上傳新版本',onClick:()=>openRegulationModal(reg)}));
    if(canManage()&&drafts[0]) actions.append(el('button',{class:'btn',text:'發布草稿',onClick:()=>{closeModal();publishVersion(reg,drafts[0]);}}));
  }
