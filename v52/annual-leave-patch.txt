  function alCanSubmit(){ return !!state.profile; }
  function alCanManage(){ return !!state.profile && ['business_manager','organization_manager','admin'].includes(state.profile.role); }
  function alCanReview(){ return !!state.profile && ['organization_manager','admin'].includes(state.profile.role); }
  function alRoleLabel(role){ return roleLabels[role] || role || '—'; }
  function alStatusLabel(status){ return ({draft:'草稿',pending:'待審核',approved:'已核准',rejected:'已退回',withdrawn:'已撤回',voided:'已作廢'})[status] || status || '—'; }
  function alDuration(minutes){
    const n=Math.max(0,Number(minutes)||0), h=Math.floor(n/60), m=n%60;
    if(h&&m) return `${h}小時${m}分`;
    if(h) return `${h}小時`;
    return `${m}分鐘`;
  }
  function alDate(value){
    if(!value) return '—';
    const d=new Date(`${String(value).slice(0,10)}T00:00:00`);
    return Number.isNaN(d.getTime()) ? String(value) : `${d.getFullYear()-1911}/${String(d.getMonth()+1).padStart(2,'0')}/${String(d.getDate()).padStart(2,'0')}`;
  }
  function alTime(value){ return value ? String(value).slice(0,5) : '—'; }
  function alMinutesBetween(start,end){
    if(!/^([01]\d|2[0-3]):[0-5]\d$/.test(start||'') || !/^([01]\d|2[0-3]):[0-5]\d$/.test(end||'')) return 0;
    const [sh,sm]=start.split(':').map(Number), [eh,em]=end.split(':').map(Number);
    return Math.max(0,(eh*60+em)-(sh*60+sm));
  }
  function alTimeOptions(selected){
    let html='<option value="">時</option>';
    for(let i=0;i<24;i++){
      const v=String(i).padStart(2,'0');
      html+=`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`;
    }
    return html;
  }
  function alTimeParts(value){
    const v=String(value||'').slice(0,5);
    if(!/^([01]\d|2[0-3]):[0-5]\d$/.test(v)) return {hour:'',minute:'00'};
    const [hour,minute]=v.split(':');
    return {hour,minute};
  }
  function alSelectedTime(prefix){
    const h=$(prefix+'Hour')?.value||'', m=$(prefix+'Minute')?.value||'';
    return h && m ? `${h}:${m}` : '';
  }
  function alMonthRange(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    const [y,m]=safe.split('-').map(Number);
    const start=`${y}-${String(m).padStart(2,'0')}-01`;
    const end=isoDate(new Date(y,m,0));
    return {start,end};
  }
  function alOwnRows(){ return (state.annualLeaveRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile?.id)); }
  function alOwnCredits(){ return (state.annualLeaveCredits||[]).filter(c=>String(c.staff_id)===String(state.profile?.id)); }
  function alBalance(staffId,date=isoDate(new Date())){
    const credits=(state.annualLeaveCredits||[]).filter(c=>String(c.staff_id)===String(staffId) && String(c.period_start)<=date && String(c.period_end)>=date);
    const granted=credits.reduce((s,c)=>s+(Number(c.granted_minutes)||0),0);
    const remaining=credits.reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const pending=(state.annualLeaveRows||[]).filter(r=>String(r.applicant_staff_id)===String(staffId) && r.status==='pending' && credits.some(c=>String(r.leave_date)>=String(c.period_start)&&String(r.leave_date)<=String(c.period_end))).reduce((s,r)=>s+(Number(r.minutes)||0),0);
    return {granted,used:Math.max(0,granted-remaining),remaining,pending,available:Math.max(0,remaining-pending)};
  }

  async function loadAnnualLeaveData(){
    if(!state.profile) return;
    showLoading(true);
    try{
      const tasks=[
        db.from('annual_leave_requests').select('id,applicant_staff_id,applicant_name,applicant_role,leave_date,start_time,end_time,minutes,reason,notes,status,review_note,submitted_at,reviewed_at,created_at,updated_at').order('leave_date',{ascending:false}).order('created_at',{ascending:false}),
        db.from('annual_leave_credits').select('id,staff_id,period_start,period_end,granted_minutes,remaining_minutes,label,note,created_by_name,created_at,updated_at').order('period_end',{ascending:true}).order('created_at',{ascending:true}),
        db.from('annual_leave_allocations').select('id,credit_id,request_id,allocation_date,minutes,is_active,reversed_at,created_at').order('created_at',{ascending:false})
      ];
      if(alCanManage()) tasks.push(db.from('staff_users').select('id,display_name,role,hire_date,is_active').eq('is_active',true).order('display_name',{ascending:true}));
      const results=await Promise.all(tasks);
      for(const r of results) if(r.error) throw r.error;
      state.annualLeaveRows=results[0].data||[];
      state.annualLeaveCredits=results[1].data||[];
      state.annualLeaveAllocations=results[2].data||[];
      state.annualLeaveStaff=alCanManage()?(results[3]?.data||[]):[];
      state.annualLeaveLoaded=true;
    }catch(err){
      console.error('讀取特休資料失敗',err);
      state.annualLeaveLoaded=false;
      toast('讀取特休資料失敗：'+(err?.message||err),'error');
    }finally{ showLoading(false); }
  }

  function openAnnualLeaveModal(existing=null){
    if(!alCanSubmit()){toast('目前帳號沒有特休申請權限','warn');return;}
    openModal({
      title:existing?'編輯特休申請':'申請特休',
      html:annualLeaveFormHTML(existing),
      confirmText:'送出審核',
      onConfirm:()=>saveAnnualLeave(existing,'pending'),
      large:true
    });
    if(!existing){
      const foot=$('modalRoot').querySelector('.modal-foot');
      if(foot){
        const draft=el('button',{class:'btn',text:'儲存草稿',onClick:()=>saveAnnualLeave(null,'draft')});
        foot.insertBefore(draft,$('modalConfirm'));
      }
    }
    const update=()=>{
      const start=alSelectedTime('alStart'), end=alSelectedTime('alEnd'), mins=alMinutesBetween(start,end);
      $('alDurationHelp').textContent=start&&end&&mins>0?alDuration(mins):'請選擇完整且正確的起訖時間';
    };
    const updateBalance=()=>{
      const b=alBalance(state.profile?.id,$('alDate').value||isoDate(new Date()));
      $('alBalanceValue').value=alDuration(b.available);
    };
    ['alStartHour','alStartMinute','alEndHour','alEndMinute'].forEach(id=>$(id)?.addEventListener('change',update));
    $('alDate')?.addEventListener('change',updateBalance);
    update();
  }

  async function saveAnnualLeave(existing,status){
    const leaveDate=$('alDate').value;
    const start=alSelectedTime('alStart'), end=alSelectedTime('alEnd');
    const minutes=alMinutesBetween(start,end);
    const reason=$('alReason').value.trim(), notes=$('alNotes').value.trim();
    if(!leaveDate){toast('請選擇特休日期','warn');return false;}
    if(!start||!end){toast('請選擇完整的開始與結束時間','warn');return false;}
    if(!['00','30'].includes(start.slice(3,5))||!['00','30'].includes(end.slice(3,5))){toast('特休時間請以30分鐘為間隔','warn');return false;}
    if(minutes<=0){toast('結束時間須晚於開始時間','warn');return false;}
    const balance=alBalance(state.profile.id,leaveDate);
    if(status==='pending' && minutes>balance.available){toast(`所選日期目前可用特休僅 ${alDuration(balance.available)}`,'warn');return false;}
    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      leave_date:leaveDate,
      start_time:start,
      end_time:end,
      minutes,
      reason,
      notes:notes||null,
      status,
      submitted_at:status==='pending'?new Date().toISOString():null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:new Date().toISOString()
    };
    showLoading(true);
    try{
      let error;
      if(existing){ ({error}=await db.from('annual_leave_requests').update(payload).eq('id',existing.id)); }
      else { payload.created_by=state.user.id; ({error}=await db.from('annual_leave_requests').insert(payload)); }
      if(error) throw error;
      closeModal();
      await loadAnnualLeaveData();
      state.annualLeaveView='mine';
      renderAnnualLeavePage();
      toast(status==='pending'?'特休申請已送出審核':'特休草稿已儲存','success');
      return true;
    }catch(err){
      console.error('儲存特休失敗',err);
      toast('儲存特休申請失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  function openAnnualCreditModal(){
    if(!alCanManage()){toast('目前帳號沒有特休額度管理權限','warn');return;}
    if(!(state.annualLeaveStaff||[]).length){toast('目前沒有可設定額度的員工資料','warn');return;}
    const options=state.annualLeaveStaff.map(s=>`<option value="${s.id}">${escapeHTML(s.display_name)}｜${escapeHTML(alRoleLabel(s.role))}</option>`).join('');
    openModal({
      title:'設定特休額度',
      html:`<div class="form-grid">
        <div class="field full"><label>員工 *</label><select id="alCreditStaff"><option value="" disabled selected>請選擇</option>${options}</select></div>
        <div class="field"><label>額度起日 *</label><input id="alCreditStart" type="date"></div>
        <div class="field"><label>額度迄日 *</label><input id="alCreditEnd" type="date"></div>
        <div class="field"><label>特休總時數 *</label><input id="alCreditHours" type="number" min="0.5" max="400" step="0.5" placeholder="例如 56"></div>
        <div class="field"><label>額度名稱</label><input id="alCreditLabel" type="text" maxlength="80" placeholder="例如：115年度特休"></div>
        <div class="field full"><label>備註</label><textarea id="alCreditNote" placeholder="選填"></textarea></div>
      </div>`,
      confirmText:'建立額度',
      onConfirm:saveAnnualCredit,
      large:true
    });
  }

  async function saveAnnualCredit(){
    const staffId=$('alCreditStaff').value, start=$('alCreditStart').value, end=$('alCreditEnd').value;
    const hours=Number($('alCreditHours').value), label=$('alCreditLabel').value.trim(), note=$('alCreditNote').value.trim();
    if(!staffId){toast('請選擇員工','warn');return false;}
    if(!start||!end){toast('請填寫額度起訖日期','warn');return false;}
    if(end<start){toast('額度迄日不可早於起日','warn');return false;}
    if(!Number.isFinite(hours)||hours<=0||Math.round(hours*2)!==hours*2){toast('特休總時數請以0.5小時為單位','warn');return false;}
    const minutes=Math.round(hours*60);
    const staff=state.annualLeaveStaff.find(s=>String(s.id)===String(staffId));
    const duplicate=(state.annualLeaveCredits||[]).some(c=>String(c.staff_id)===String(staffId)&&c.period_start===start&&c.period_end===end);
    if(duplicate && !confirm('此員工已有相同期間的特休額度。若這是補登或追加額度可繼續；否則請先取消並確認資料。\n\n仍要建立嗎？')) return false;
    showLoading(true);
    try{
      const {error}=await db.from('annual_leave_credits').insert({
        staff_id:staffId,
        period_start:start,
        period_end:end,
        granted_minutes:minutes,
        remaining_minutes:minutes,
        label:label||null,
        note:note||null,
        created_by:state.user.id,
        created_by_name:state.profile.display_name
      });
      if(error) throw error;
      closeModal();
      await loadAnnualLeaveData();
      state.annualLeaveView='credits';
      renderAnnualLeavePage();
      toast(`已建立 ${staff?.display_name||'員工'} 的特休額度`,'success');
      return true;
    }catch(err){
      console.error('建立特休額度失敗',err);
      toast('建立特休額度失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  async function reviewAnnualLeave(id,status,note=null){
    if(!alCanReview()){toast('目前帳號沒有特休審核權限','warn');return;}
    const row=(state.annualLeaveRows||[]).find(r=>String(r.id)===String(id));
    if(!row||row.status!=='pending'){toast('此特休申請目前不是待審核狀態','warn');return;}
    if(status==='approved'&&!confirm(`確定核准「${row.applicant_name}」${alDate(row.leave_date)} ${alDuration(row.minutes)} 特休嗎？`)) return;
    showLoading(true);
    try{
      const {error}=await db.from('annual_leave_requests').update({
        status,
        review_note:note||null,
        reviewed_by:state.user.id,
        reviewed_at:new Date().toISOString(),
        updated_at:new Date().toISOString()
      }).eq('id',id).eq('status','pending');
      if(error) throw error;
      closeModal();
      await loadAnnualLeaveData();
      renderAnnualLeavePage();
      toast(status==='approved'?'特休申請已核准':'特休申請已退回','success');
    }catch(err){
      console.error('審核特休失敗',err);
      toast('審核特休失敗：'+(err?.message||err),'error');
    }finally{showLoading(false);}
  }

  function openAnnualRejectModal(id){
    openModal({
      title:'退回特休申請',
      html:'<div class="field"><label>退回原因 *</label><textarea id="alRejectReason" placeholder="請說明需要補充或修正的內容"></textarea></div>',
      confirmText:'確認退回',
      onConfirm:async()=>{
        const note=$('alRejectReason').value.trim();
        if(!note){toast('請填寫退回原因','warn');return false;}
        await reviewAnnualLeave(id,'rejected',note);
        return true;
      }
    });
  }

  async function withdrawAnnualLeave(id){
    const row=alOwnRows().find(r=>String(r.id)===String(id));
    if(!row||row.status!=='pending'){toast('只有待審核中的特休申請可以撤回','warn');return;}
    if(!confirm(`確定撤回 ${alDate(row.leave_date)} 的特休申請嗎？`)) return;
    showLoading(true);
    try{
      const {error}=await db.from('annual_leave_requests').update({
        status:'withdrawn',
        withdraw_reason:'申請人撤回',
        withdrawn_by:state.user.id,
        withdrawn_at:new Date().toISOString(),
        updated_at:new Date().toISOString()
      }).eq('id',id).eq('status','pending');
      if(error) throw error;
      await loadAnnualLeaveData();
      renderAnnualLeavePage();
      toast('特休申請已撤回','success');
    }catch(err){
      console.error('撤回特休失敗',err);
      toast('撤回特休失敗：'+(err?.message||err),'error');
    }finally{showLoading(false);}
  }

  function wireAnnualLeaveActions(){
    document.querySelectorAll('[data-al-edit]').forEach(btn=>btn.addEventListener('click',()=>{
      const row=alOwnRows().find(r=>String(r.id)===String(btn.dataset.alEdit));
      if(row) openAnnualLeaveModal(row);
    }));
    document.querySelectorAll('[data-al-withdraw]').forEach(btn=>btn.addEventListener('click',()=>withdrawAnnualLeave(btn.dataset.alWithdraw)));
    document.querySelectorAll('[data-al-approve]').forEach(btn=>btn.addEventListener('click',()=>reviewAnnualLeave(btn.dataset.alApprove,'approved')));
    document.querySelectorAll('[data-al-reject]').forEach(btn=>btn.addEventListener('click',()=>openAnnualRejectModal(btn.dataset.alReject)));
  }

  const overtimeNavigateForAnnual = navigate;
  navigate = function(id){
    if(id!=='overtime') return overtimeNavigateForAnnual(id);
    if(state.attendanceSection==='annual'){
      state.page='overtime';
      renderNav();
      $('sidebar').classList.remove('open');
      window.scrollTo({top:0,behavior:'auto'});
      $('content').innerHTML='<div class="empty compact"><div class="icon">🌿</div><h3>正在載入特休資料</h3><p>請稍候。</p></div>';
      return loadAnnualLeaveData().then(renderAnnualLeavePage);
    }
    return overtimeNavigateForAnnual(id);
  };

  const observer=new MutationObserver(()=>{
    if(state.page==='overtime'&&state.attendanceSection==='overtime') decorateOvertimePage();
  });
  observer.observe($('content'),{childList:true,subtree:true});

  if(state.profile){
    renderNav();
    if(state.page==='overtime'&&state.attendanceSection==='overtime') setTimeout(decorateOvertimePage,0);
  }
})();
