  const levelFromScore=score=>{
    if(score===null||score===undefined)return null;
    if(score<=20)return 'complete_dependence';
    if(score<=60)return 'severe_dependence';
    if(score<=90)return 'moderate_dependence';
    if(score<=99)return 'mild_dependence';
    return 'independent';
  };
  const diffText=(now,prev)=>{
    if(now===null||prev===null)return '—';
    const d=now-prev;
    if(d===0)return '無變化';
    return (d>0?'+':'')+d+' 分';
  };
  const isChanged=key=>{
    const now=selectedScore(key);
    const prev=previousScore(key);
    return now!==null&&prev!==null&&now!==prev;
  };

  async function loadAdl(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');
    const [currentRes,historyRes]=await Promise.all([
      db.from('assessment_adl_records')
        .select('*')
        .eq('assessment_event_id',eventId)
        .maybeSingle(),
      db.from('assessment_adl_records')
        .select('*')
        .eq('case_id',row.case_id)
        .not('confirmed_at','is',null)
        .neq('assessment_event_id',eventId)
        .order('confirmed_at',{ascending:false})
    ]);
    if(currentRes.error)throw currentRes.error;
    if(historyRes.error)throw historyRes.error;

    local.eventId=eventId;
    local.record=currentRes.data||null;
    local.answers={...safeObj(local.record?.answers)};
    local.notes={...safeObj(local.record?.notes)};
    local.expandedNotes=new Set();

    const allCompleted=historyRes.data||[];
    const dated=allCompleted
      .map(record=>({record,event:eventById(record.assessment_event_id)}))
      .filter(x=>x.event?.planned_date);

    local.hasLaterCompleted=dated.some(x=>x.event.planned_date>row.planned_date);

    const candidates=dated
      .filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>{
        const byDate=String(b.event.planned_date).localeCompare(String(a.event.planned_date));
        if(byDate!==0)return byDate;
        return String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||''));
      });

    local.previous=candidates[0]?.record||null;
    local.previousEvent=candidates[0]?.event||null;

    const copiedSourceIsValid=!!local.record?.copied_from_id
      && local.record.copied_from_id===local.previous?.id;
    local.pendingCopyFromId=copiedSourceIsValid?local.record.copied_from_id:null;
  }

  function renderItem(item){
    const current=selectedScore(item.key);
    const prev=previousScore(item.key);
    const changed=isChanged(item.key);
    const noteValue=String(local.notes?.[item.key]||'');
    const showNote=changed||!!noteValue||local.expandedNotes.has(item.key);
    return `
      <section class="adl-card ${changed?'changed':''}" data-adl-card="${item.key}">
        <div class="adl-card-head">
          <div class="adl-card-title">${escapeHTML(item.label)}</div>
          <div class="adl-card-meta">
            <span class="adl-max">最高 ${item.max} 分</span>
            ${prev!==null?'<span class="adl-prev">前次 '+prev+' 分</span>':''}
            ${changed?'<span class="adl-change">較前次 '+escapeHTML(diffText(current,prev))+'</span>':''}
          </div>
        </div>
        <div class="adl-options">
          ${item.options.map(([score,text])=>`
            <label class="adl-option ${current===score?'selected':''}">
              <input type="radio" name="adl_${item.key}" value="${score}" ${current===score?'checked':''}>
              <span class="adl-score">${score}</span>
              <span class="adl-option-text">${escapeHTML(text)}</span>
            </label>
          `).join('')}
        </div>
        ${window.v52AssessmentToolSummary?.previousNote(local.previous?.notes?.[item.key])||''}
        ${!showNote?`
          <div class="adl-note"><button type="button" class="btn sm" data-adl-add-note="${item.key}">＋ 補充備註</button></div>
        `:''}
        ${showNote?`
          <div class="adl-note">
            <label>${changed?'本項較前次有變化，可補充原因或觀察：':'備註（選填）：'}</label>
            <textarea data-adl-note="${item.key}" placeholder="可記錄輔具使用、協助程度或本次觀察">${escapeHTML(noteValue)}</textarea>
          </div>
        `:''}
      </section>
    `;
  }

  function refreshAdlDerivedUI(){

  }

  function currentToolSummarySource(){return ({answers:local.answers,notes:local.notes});}

  async function saveAdl(finalize){
    const row=eventById(local.eventId);
    if(!row||!canEditEvent(row))return toast('您沒有此個案的評估編輯權限','error');

    document.querySelectorAll('[data-adl-note]').forEach(textarea=>{
      local.notes[textarea.dataset.adlNote]=textarea.value;
    });

    const count=answeredCount();
    if(finalize&&count!==ADL_ITEMS.length){
      const missing=ADL_ITEMS.filter(x=>selectedScore(x.key)===null).map(x=>x.label).join('、');
      return toast('尚有未完成項目：'+missing,'warn');
    }
    if(!finalize&&count===0){
      return toast('至少填寫一項後再儲存草稿','warn');
    }

    showLoading(true);
    try{
      const {data,error}=await db.rpc('save_assessment_adl',{
        p_event_id:row.id,
        p_answers:local.answers,
        p_notes:local.notes,
        p_finalize:!!finalize,
        p_copied_from_id:local.pendingCopyFromId||local.record?.copied_from_id||null
      });
      if(error)throw error;
      await window.v52AssessmentToolSummary.saveAfterTool(row.id,'adl',currentToolSummarySource(),row.case_id);

      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){
        window.v52AssessmentApi?.renderAssessmentDetail?.(row.id);
        toast('ADL本次確認已完成','success');
      }else{
        await loadAdl(row.id);
        renderAdlPage();
        toast('ADL草稿已儲存','success');
      }
    }catch(err){
      console.error(err);
      const msg=String(err?.message||err);
      if(msg.includes('ADL_INCOMPLETE'))toast('尚有ADL項目未完成','warn');
      else toast('儲存ADL失敗：'+msg,'error');
    }finally{
      showLoading(false);
    }
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(!adlForm(eventId))return toast('本次評估未選用ADL','warn');
    showLoading(true);
    try{
      await loadAdl(eventId);
      await window.v52AssessmentToolSummary.load(eventId,'adl',currentToolSummarySource());
      renderAdlPage();
    }catch(err){
      console.error(err);
      toast('讀取ADL失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  window.v52AssessmentAdl={open};
})();
