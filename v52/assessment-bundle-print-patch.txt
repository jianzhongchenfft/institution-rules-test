  async function maybeRecord(table,eventId){
    const r=await db.from(table).select('*').eq('assessment_event_id',eventId).maybeSingle();
    if(r.error)throw r.error;
    return r.data||null;
  }

  async function previousRecord(table,row,predicate=()=>true){
    const r=await db.from(table).select('*').eq('case_id',row.case_id).not('confirmed_at','is',null).neq('assessment_event_id',row.id).order('confirmed_at',{ascending:false});
    if(r.error)throw r.error;
    const candidates=(r.data||[])
      .filter(predicate)
      .map(record=>({record,event:(state.assessmentEvents||[]).find(e=>e.id===record.assessment_event_id)}))
      .filter(x=>x.event?.planned_date&&x.event.planned_date<row.planned_date)
      .sort((a,b)=>String(b.event.planned_date).localeCompare(String(a.event.planned_date))||String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||'')));
    return candidates[0]||null;
  }

    if(!c)throw new Error('找不到個案資料');
    const forms=(state.assessmentEventForms||[]).filter(x=>x.assessment_event_id===eventId).sort((a,b)=>(a.sort_order||0)-(b.sort_order||0));

    const [
      adl,iadl,health,homeSafety,support,spmsq,gds15,caregiver,summary,

      maybeRecord('assessment_adl_records',eventId),

      previousRecord('assessment_health_records',row),
      previousRecord('assessment_home_safety_records',row),

      previousRecord('assessment_gds15_records',row,x=>x.completion_mode==='completed'),

    local.records={adl,iadl,health_medication:health,home_safety:homeSafety,support,spmsq,gds15,caregiver_burden:caregiver};

      adl:prevAdl?.record||null,iadl:prevIadl?.record||null,health_medication:prevHealth?.record||null,

    };
  }

  function comparison(code,current,previous){
    if(!previous)return '首次評估／無前次可比較資料';
    if(code==='adl'){
      const d=Number(current?.total_score||0)-Number(previous.total_score||0);
      return `前次 ${previous.total_score??'—'} 分 → 本次 ${current?.total_score??'—'} 分${d===0?'（無變化）':`（${d>0?'+':''}${d} 分）`}`;
    }
    if(code==='iadl'){
      const d=Number(current?.total_score||0)-Number(previous.total_score||0);
      return `前次 ${previous.total_score??'—'} 分 → 本次 ${current?.total_score??'—'} 分${d===0?'（無變化）':`（${d>0?'+':''}${d} 分）`}`;
    }
    if(code==='spmsq'){
      if(current?.is_unable)return '本次無法評估';
      const a=Number(previous.error_count||0),b=Number(current?.error_count||0);
      return `錯誤題數 ${a} → ${b} 題`;
    }
    if(code==='gds15'){
      if(current?.completion_mode!=='completed')return '本次無可比較分數';
      return `前次 ${previous.total_score??'—'} 分 → 本次 ${current?.total_score??'—'} 分`;
    }
    if(code==='health_medication'){
      const count=r=>Object.values(safe(r?.health_items)).filter(v=>v==='observe'||v==='refer').length;
      return `需觀察／轉介項目 ${count(previous)} → ${count(current)} 項`;
    }
    if(code==='home_safety'){
      const count=r=>Object.values(safe(r?.answers)).filter(v=>v==='needs_improvement').length;
      return `需改善項目 ${count(previous)} → ${count(current)} 項`;
    }
    if(code==='support'){
      return `整體支持：${OVERALL_SUPPORT[safe(previous.summary).overall_support]||'—'} → ${OVERALL_SUPPORT[safe(current?.summary).overall_support]||'—'}`;
    }
    if(code==='caregiver_burden'){
      const count=r=>Number(safe(r?.summary).positive_count??Object.values(safe(r?.answers)).filter(v=>v==='yes').length);
      return `符合指標 ${count(previous)} → ${count(current)} 項`;
    }
    return '—';
  }

        <table class="meta"><tbody>

          <tr><th>負責督導</th><td>${html(staffName(row.responsible_staff_id||c.supervisor_id))}</td><th>表單狀態</th><td>${form?html(formStatusLabel(form.status)):'已完成'}</td></tr>
          ${form?`<tr><th>前次比較</th><td colspan="3">${html(compare)}${prevEvent?.planned_date?`｜前次日期：${rocDate(prevEvent.planned_date+'T00:00:00')}`:''}</td></tr>`:''}
        </tbody></table>
      </header>`;

  function renderTool(form){
    if(form.form_code==='adl')return renderAdl(form);
    if(form.form_code==='iadl')return renderIadl(form);
    if(form.form_code==='health_medication')return renderHealth(form);
    if(form.form_code==='home_safety')return renderHome(form);
    if(form.form_code==='support')return renderSupport(form);
    if(form.form_code==='spmsq')return renderSpmsq(form);
    if(form.form_code==='gds15')return renderGds(form);
    if(form.form_code==='caregiver_burden')return renderCaregiver(form);
    return sectionMissing(form.form_name);
  }

  function openPreview(selected){
    const w=window.open('','_blank');
    if(!w){
      toast('瀏覽器阻擋了列印預覽視窗，請允許此網站開啟彈出式視窗。','warn');
      return;
    }
    const doc=buildHtml(selected);
    w.document.open();w.document.write(doc);w.document.close();
  }

  function showSelection(){
    openModal({
      title:'列印完整評估套表',
      html:selectionHtml(),
      large:true,
      confirmText:'產生列印預覽',
      onConfirm:async()=>{
        const selected=[...document.querySelectorAll('[data-bundle-section]:checked')].map(x=>x.dataset.bundleSection);
        if(!selected.length)return toast('請至少勾選一項列印內容','warn');
        closeModal();
        openPreview(selected);
      }
    });
    $('bundleSelectAll')?.addEventListener('click',()=>document.querySelectorAll('[data-bundle-section]').forEach(x=>x.checked=true));
    $('bundleClearAll')?.addEventListener('click',()=>document.querySelectorAll('[data-bundle-section]').forEach(x=>x.checked=false));
  }

  async function open(eventId){
    const row=(state.assessmentEvents||[]).find(x=>x.id===eventId);
    if(!row)return toast('找不到評估事件','error');
    const forms=(state.assessmentEventForms||[]).filter(x=>x.assessment_event_id===eventId);
    if(!forms.length)return toast('本次評估尚未建立評估工具','warn');
    if(forms.some(x=>!['completed','unable','not_applicable'].includes(x.status))){
      return toast('尚有評估工具未完成，完成後才能列印完整評估套表','warn');
    }
    showLoading(true);
    try{
      await load(eventId);
      showSelection();
    }catch(err){
      console.error(err);
      toast('讀取完整評估套表失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  window.v52AssessmentBundlePrint={open};
