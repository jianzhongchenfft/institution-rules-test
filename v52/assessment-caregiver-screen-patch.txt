  const local={

  const safe=v=>(v&&typeof v==='object'&&!Array.isArray(v))?v:{};
  const eventById=id=>state.assessmentEvents?.find(x=>x.id===id)||null;
  const caseById=id=>state.assessmentCases?.find(x=>x.id===id)||null;
  const eventForms=id=>(state.assessmentEventForms||[]).filter(x=>x.assessment_event_id===id);

  const answer=key=>local.answers?.[key]||null;
  const previousAnswer=key=>local.previous?.answers?.[key]||null;
  const positiveCount=()=>INDICATORS.filter(([key])=>answer(key)==='yes').length;
  const hasHighRisk=()=>[...HIGH_RISK_KEYS].some(key=>answer(key)==='yes');
  const professionalReferral=()=>local.summary.professional_referral===true;
  const referralRequired=()=>positiveCount()>=2||hasHighRisk()||professionalReferral();

  function indicatorChange(key){
    const prev=previousAnswer(key),cur=answer(key);
    if(!prev||!cur)return null;
    if(prev==='no'&&cur==='yes')return {kind:'new-risk',label:'新增風險'};
    if(prev==='yes'&&cur==='yes')return {kind:'persist',label:'持續風險'};
    if(prev==='yes'&&cur==='no')return {kind:'better',label:'已改善'};
    return null;
  }

  function indicatorChangeCounts(){
    const counts={newRisk:0,persist:0,better:0};
    INDICATORS.forEach(([key])=>{
      const change=indicatorChange(key);
      if(change?.kind==='new-risk')counts.newRisk++;
      if(change?.kind==='persist')counts.persist++;
      if(change?.kind==='better')counts.better++;
    });
    return counts;
  }

  async function load(eventId){
    const row=eventById(eventId);if(!row)throw new Error('找不到評估事件');
    const [cur,hist,support]=await Promise.all([
      db.from('assessment_caregiver_screen_records').select('*').eq('assessment_event_id',eventId).maybeSingle(),
      db.from('assessment_caregiver_screen_records').select('*').eq('case_id',row.case_id).not('confirmed_at','is',null).neq('assessment_event_id',eventId).order('confirmed_at',{ascending:false}),
      db.from('assessment_support_records').select('*').eq('assessment_event_id',eventId).maybeSingle()
    ]);
    if(cur.error)throw cur.error;if(hist.error)throw hist.error;
    if(support.error&&support.error.code!=='PGRST116')throw support.error;

    local.eventId=eventId;local.record=cur.data||null;local.supportRecord=support.data||null;
    local.applicability=local.record?.applicability||'applicable';
    local.caregiverInfo={...safe(local.record?.caregiver_info)};
    local.answers={...safe(local.record?.answers)};
    local.notes={...safe(local.record?.notes)};
    local.summary={...safe(local.record?.summary)};
    local.expandedNotes=new Set();

    const dated=(hist.data||[]).map(record=>({record,event:eventById(record.assessment_event_id)})).filter(x=>x.event?.planned_date);
    const candidates=dated.filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>String(b.event.planned_date).localeCompare(String(a.event.planned_date))||String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||'')));
    local.previous=candidates[0]?.record||null;
    local.previousEvent=candidates[0]?.event||null;
  }

  function renderSummary(){
    const count=local.applicability==='applicable'?positiveCount():0;
    const prevCount=local.previous?.applicability==='applicable'?Number(local.previous?.summary?.positive_count||0):null;
    const changes=indicatorChangeCounts();
    return `<section class="cg-summary">
      <div class="cg-summary-card"><label>本次符合指標</label><strong>${local.applicability==='applicable'?count+'／10':'—'}</strong><small>此工具不計總分，以符合項目判定</small></div>
      <div class="cg-summary-card"><label>高風險指標</label><strong>${local.applicability==='applicable'?(hasHighRisk()?'有':'無'):'—'}</strong><small>暴力／疏忽或自殺相關風險</small></div>
      <div class="cg-summary-card"><label>轉介判定</label><strong>${local.applicability==='applicable'?(referralRequired()?'達轉介標準':'目前未達'):'—'}</strong><small>符合任兩項，或高風險指標任一項</small></div>
      <div class="cg-summary-card"><label>前次／變化</label><strong>${prevCount===null?'—':prevCount+' 項 → '+count+' 項'}</strong><small>${prevCount===null?'無前次完成紀錄':'新增 '+changes.newRisk+'｜持續 '+changes.persist+'｜改善 '+changes.better}</small></div>
    </section>`;
  }

    const h=safe(local.supportRecord?.household);

    return {name:h.primary_caregiver_name,relation:h.primary_caregiver_relation||''};
  }

  function renderIndicator([key,title,help],index){
    const cur=answer(key),prev=previousAnswer(key),note=String(local.notes[key]||'');
    const positive=cur==='yes',showNote=positive||!!note||local.expandedNotes.has(key);
    const high=HIGH_RISK_KEYS.has(key);
    const change=indicatorChange(key);
    return `<div class="cg-card ${positive?'positive':''} ${positive&&high?'high-risk':''}">
      <div class="cg-card-head">
        <div class="cg-card-title">${index+1}. ${escapeHTML(title)}<small>${escapeHTML(help)}</small></div>
        <div>${prev?`<span class="cg-tag prev">前次：${prev==='yes'?'符合':'未符合'}</span>`:''}${change?`<span class="cg-tag ${change.kind}" style="margin-left:5px">${change.label}</span>`:''}${high?`<span class="cg-tag risk" style="margin-left:5px">高風險指標</span>`:''}</div>
      </div>
      <div class="cg-options">
        <label class="cg-option no ${cur==='no'?'selected':''}"><input type="radio" name="cg_${key}" value="no" ${cur==='no'?'checked':''}>未符合</label>
        <label class="cg-option yes ${cur==='yes'?'selected':''}"><input type="radio" name="cg_${key}" value="yes" ${cur==='yes'?'checked':''}>符合</label>
      </div>

    $('content').innerHTML=`<div class="cg-page">

  function rerenderPreservingScroll(){const y=window.scrollY;render();requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));}

  function bind(row){
    $('cgBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));
    $('cgApplicability')?.addEventListener('change',e=>{
      local.applicability=e.target.value;
      local.expandedNotes=new Set();
      if(local.applicability!=='applicable'){
        local.answers={};
        local.notes={};
        local.caregiverInfo={};
        local.summary={};
      }else{
        delete local.summary.unable_reason;
      }
      rerenderPreservingScroll();
    });
    $('cgUseSupportCaregiver')?.addEventListener('click',()=>{
      const cg=supportCaregiver();if(!cg)return;
      local.caregiverInfo={...local.caregiverInfo,name:cg.name,relation:cg.relation};
      rerenderPreservingScroll();toast('已帶入本次支持系統的主要照顧者','success');
    });
    document.querySelectorAll('[data-cg-info]').forEach(el=>el.addEventListener('input',()=>{local.caregiverInfo[el.dataset.cgInfo]=el.value;}));
    document.querySelectorAll('input[name^="cg_"]').forEach(input=>input.addEventListener('change',()=>{
      const key=input.name.replace(/^cg_/,'');local.answers[key]=input.value;rerenderPreservingScroll();
    }));
    document.querySelectorAll('[data-cg-add-note]').forEach(btn=>btn.addEventListener('click',()=>{local.expandedNotes.add(btn.dataset.cgAddNote);rerenderPreservingScroll();}));
    document.querySelectorAll('[data-cg-note]').forEach(el=>el.addEventListener('input',()=>{local.notes[el.dataset.cgNote]=el.value;}));
    $('cgProfessionalReferral')?.addEventListener('change',e=>{local.summary.professional_referral=e.target.checked;rerenderPreservingScroll();});
    document.querySelectorAll('[data-cg-summary]').forEach(el=>el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{local.summary[el.dataset.cgSummary]=el.value;}));
    $('cgSaveDraft')?.addEventListener('click',()=>save(false));
    $('cgConfirm')?.addEventListener('click',()=>save(true));
  }

  function validate(finalize){
    if(!finalize)return null;
    if(local.applicability==='unable'&&!String(local.summary.unable_reason||'').trim())return '請填寫本次無法評估原因';
    if(local.applicability!=='applicable')return null;
    if(!String(local.caregiverInfo.name||'').trim())return '請填寫主要照顧者姓名／稱謂';
    if(!String(local.caregiverInfo.relation||'').trim())return '請填寫主要照顧者與個案關係';
    const missing=INDICATORS.filter(([key])=>!['yes','no'].includes(answer(key)));
    if(missing.length)return '尚有未完成初篩項目：'+missing.map(([,label])=>label).join('、');
    const noteMissing=INDICATORS.filter(([key])=>answer(key)==='yes'&&!String(local.notes[key]||'').trim());
    if(noteMissing.length)return '以下項目已勾選「符合」，請填寫說明：'+noteMissing.map(([,label])=>label).join('、');
    if(professionalReferral()&&!String(local.summary.professional_reason||'').trim())return '請填寫專業判斷轉介理由';
    if(referralRequired()){
      if(!local.summary.referral_action)return '本次達轉介標準，請選擇轉介／處理狀態';
      if(!String(local.summary.action_note||'').trim())return '本次達轉介標準，請填寫處理紀錄';
    }
    return null;
  }

  async function save(finalize){
    const row=eventById(local.eventId);if(!row||!canEdit(row))return toast('您沒有此個案的評估編輯權限','error');
    const problem=validate(finalize);if(problem)return toast(problem,'warn');
    const y=window.scrollY;showLoading(true);
    try{
      const {error}=await db.rpc('save_assessment_caregiver_screen',{
        p_event_id:row.id,p_applicability:local.applicability,p_caregiver_info:local.caregiverInfo,
        p_answers:local.answers,p_notes:local.notes,p_summary:local.summary,p_finalize:!!finalize
      });
      if(error)throw error;
      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){window.v52AssessmentApi?.renderAssessmentDetail?.(row.id);toast('高負荷家庭照顧者初篩已完成','success');}
      else{await load(row.id);render();requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));toast('初篩草稿已儲存','success');}
    }catch(err){console.error(err);toast('儲存高負荷家庭照顧者初篩失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  async function open(eventId){
    const row=eventById(eventId);if(!row)return toast('找不到評估事件','error');
    if(row.status==='pending')return toast('請先開始本次評估','warn');
    if(!formById(eventId))return toast('本次評估未選用高負荷家庭照顧者初篩','warn');
    showLoading(true);
    try{await load(eventId);render();window.scrollTo({top:0,behavior:'auto'});}
    catch(err){console.error(err);toast('讀取高負荷家庭照顧者初篩失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  window.v52AssessmentCaregiverScreen={open};
})();
