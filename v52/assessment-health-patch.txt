  async function load(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');

    const [cur,hist]=await Promise.all([
      db.from('assessment_health_records').select('*').eq('assessment_event_id',eventId).maybeSingle(),
      db.from('assessment_health_records').select('*').eq('case_id',row.case_id).not('confirmed_at','is',null).neq('assessment_event_id',eventId).order('confirmed_at',{ascending:false})
    ]);
    if(cur.error)throw cur.error;
    if(hist.error)throw hist.error;

    local.eventId=eventId;
    local.record=cur.data||null;
    local.medicalInfo={...safe(local.record?.medical_info)};
    const currentCase=caseById(row.case_id);
    if(!Object.prototype.hasOwnProperty.call(local.medicalInfo,'baseline_important_conditions')){
      local.medicalInfo.baseline_important_conditions=currentCase?.important_conditions||'';
    }
    if(!Object.prototype.hasOwnProperty.call(local.medicalInfo,'baseline_ongoing_treatments')){
      local.medicalInfo.baseline_ongoing_treatments=currentCase?.ongoing_treatments||'';
    }
    local.medicalInfo.healthcare_events=safeArray(local.medicalInfo.healthcare_events).map(x=>({...safe(x)}));
    local.medicalInfo.medication_changes=safeArray(local.medicalInfo.medication_changes).map((x,i)=>({
      id:x?.id||('legacy-med-'+i),
      action:x?.action||'',
      duration_type:x?.duration_type||'',
      medication_before_entry_id:x?.medication_before_entry_id||'',
      medication_before:x?.medication_before||'',
      medication_after:x?.medication_after||'',
      note:x?.note||''
    }));
    local.medicalInfo.medication_corrections=safeArray(local.medicalInfo.medication_corrections).map((x,i)=>({
      id:x?.id||('legacy-fix-'+i),
      action:x?.action||'',
      reason:x?.reason||'',
      target_entry_id:x?.target_entry_id||'',
      medication_before:x?.medication_before||'',
      medication_after:x?.medication_after||'',
      note:x?.note||''
    }));
    local.medicationChecks={...safe(local.record?.medication_checks)};
    local.healthItems={...safe(local.record?.health_items)};
    local.healthNotes={...safe(local.record?.health_notes)};

    const dated=(hist.data||[]).map(record=>({record,event:eventById(record.assessment_event_id)})).filter(x=>x.event?.planned_date);
    local.hasLaterCompleted=dated.some(x=>x.event.planned_date>row.planned_date);
    const candidates=dated
      .filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>String(b.event.planned_date).localeCompare(String(a.event.planned_date))||String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||'')));

    local.previous=candidates[0]?.record||null;
    local.previousEvent=candidates[0]?.event||null;
    local.pendingCopyFromId=(local.record?.copied_from_id&&local.record.copied_from_id===local.previous?.id)?local.record.copied_from_id:null;

    if(!local.record&&local.previous){
      const prevMed=safe(local.previous.medical_info);
      local.medicalInfo.medication_method=prevMed.medication_method||'';
      local.medicalInfo.medication_assisted_by=prevMed.medication_assisted_by||'';
      local.medicalInfo.medication_regular=prevMed.medication_regular||'';
      local.medicationChecks=JSON.parse(JSON.stringify(safe(local.previous.medication_checks)));
      local.medicalInfo.medication_changes=[];
      local.medicalInfo.medication_corrections=[];
      local.medicalInfo.no_medication_changes=false;
    }

    const prevMed=safe(local.previous?.medical_info);
    const savedBase=safeArray(local.medicalInfo.fixed_medication_base_entries);
    const savedEffective=safeArray(local.medicalInfo.fixed_medication_entries);
    const previousEffective=safeArray(prevMed.fixed_medication_entries);

    if(savedBase.length){
      local.fixedMedicationBaseEntries=savedBase.map(x=>normalizeMedicationEntry(x,row));
    }else if(!local.record&&previousEffective.length){
      local.fixedMedicationBaseEntries=previousEffective.map(x=>normalizeMedicationEntry(x,local.previousEvent));
    }else if(!local.record&&local.previous){
      local.fixedMedicationBaseEntries=legacyEntriesFromText(prevMed.fixed_medications||'',local.previousEvent,'legacy');
    }else if(savedEffective.length&&row.assessment_type!=='opening'){
      local.fixedMedicationBaseEntries=savedEffective.map(x=>normalizeMedicationEntry(x,row));
    }else{
      local.fixedMedicationBaseEntries=legacyEntriesFromText(
        local.medicalInfo.fixed_medications||'',
        row,
        row.assessment_type==='opening'?'opening_baseline':'legacy'
      );
    }

    rebuildFixedMedicationEntries();
  }

  function syncInputs(){
    document.querySelectorAll('[data-health-medical]').forEach(el=>{
      local.medicalInfo[el.dataset.healthMedical]=el.type==='checkbox'?el.checked:el.value;
    });

    const events=[];
    document.querySelectorAll('[data-health-event]').forEach(row=>{
      const item={};
      row.querySelectorAll('[data-health-event-field]').forEach(el=>item[el.dataset.healthEventField]=el.value);
      if(Object.values(item).some(v=>String(v||'').trim()!==''))events.push(item);
    });
    local.medicalInfo.healthcare_events=events;

    const medicationChanges=[];
    document.querySelectorAll('[data-med-change]').forEach(rowEl=>{
      const item={id:rowEl.dataset.medChange||newMedicationChange().id};
      rowEl.querySelectorAll('[data-med-change-field]').forEach(el=>item[el.dataset.medChangeField]=el.value);
      if(item.medication_before_entry_id){
        const target=entryById(rebuildFixedMedicationEntries(),item.medication_before_entry_id);
        if(target)item.medication_before=target.medication;
      }
      if(Object.entries(item).some(([k,v])=>k!=='id'&&String(v||'').trim()!==''))medicationChanges.push(item);
    });
    local.medicalInfo.medication_changes=medicationChanges;

    const medicationCorrections=[];
    document.querySelectorAll('[data-med-correction]').forEach(rowEl=>{
      const item={id:rowEl.dataset.medCorrection||newMedicationCorrection().id};
      rowEl.querySelectorAll('[data-med-correction-field]').forEach(el=>item[el.dataset.medCorrectionField]=el.value);
      if(item.target_entry_id){
        const target=entryById(local.fixedMedicationBaseEntries,item.target_entry_id)
          ||entryById(rebuildFixedMedicationEntries(),item.target_entry_id);
        if(target)item.medication_before=target.medication;
      }
      if(Object.entries(item).some(([k,v])=>k!=='id'&&String(v||'').trim()!==''))medicationCorrections.push(item);
    });
    local.medicalInfo.medication_corrections=medicationCorrections;

    document.querySelectorAll('[data-health-med-note]').forEach(el=>{
      const key=el.dataset.healthMedNote;
      local.medicationChecks[key]={...(local.medicationChecks[key]||{}),note:el.value};
    });

    document.querySelectorAll('[data-health-note]').forEach(el=>{
      local.healthNotes[el.dataset.healthNote]=el.value;
    });
  }

  function renderMedicationChange(item,index){
    const entries=rebuildFixedMedicationEntries();
    const action=item.action||'',duration=item.duration_type||'';
    const selectedId=item.medication_before_entry_id||'';
    const beforeOptions=[...entries];
    if(selectedId&&!beforeOptions.some(x=>x.id===selectedId)&&item.medication_before){
      beforeOptions.push({
        id:selectedId,
        medication:item.medication_before,
        origin_type:'legacy',
        origin_date:null,
        origin_event_id:null,
        origin_note:''
      });
    }
    const beforeField=action==='stop'||action==='adjust'
      ?(duration==='long_term'
        ?`<select data-med-change-field="medication_before_entry_id">
            <option value="">請選擇目前固定用藥</option>
            ${beforeOptions.map(entry=>`<option value="${escapeHTML(entry.id)}" ${selectedId===entry.id?'selected':''}>${escapeHTML(entry.medication)}｜${escapeHTML(medicationEntryTag(entry))}</option>`).join('')}
          </select>`
        :`<input data-med-change-field="medication_before" value="${escapeHTML(item.medication_before||'')}" placeholder="例如：感冒藥A">`)
      :'';
    const afterField=action==='add'||action==='adjust'
      ?`<input data-med-change-field="medication_after" value="${escapeHTML(item.medication_after||'')}" placeholder="${action==='add'?'例如：高血壓藥A，每日1次':'請填調整後藥物與用法'}">`
      :'';

    return `
      <div class="health-event" data-med-change="${escapeHTML(item.id||'')}">
        <div class="health-event-head">
          <strong>用藥異動 ${index+1}</strong>
          <button type="button" class="btn sm" data-remove-med-change="${index}">移除</button>
        </div>
        <div class="health-event-grid">
          <div>
            <label>異動類型</label>
            <select data-med-change-field="action">
              <option value="">請選擇</option>

          </div>

                <button id="healthCopyPrevious" class="btn sm">帶入前次</button>
              </div>
            `:''}
            <div class="health-items">${HEALTH_ITEMS.map(renderHealthItem).join('')}</div>
          </div>
        </section>

        ${canEditRow?`
          <div class="health-actions">
            ${form.status!=='completed'?'<button id="healthDraft" class="btn">儲存草稿</button>':''}
            <button id="healthConfirm" class="btn primary">${form.status==='completed'?'儲存修正並確認':'完成本次確認'}</button>
          </div>
        `:''}
      </div>
    `;

    $('healthBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));

    document.querySelectorAll('[data-health-medical]').forEach(el=>{
      const evt=el.tagName==='SELECT'||el.type==='checkbox'?'change':'input';
      el.addEventListener(evt,()=>{
        syncInputs();
        const key=el.dataset.healthMedical;
        if(key==='fixed_medications'){
          setFixedMedicationBaseFromCurrent();
        }
        if(key==='no_medication_changes'){
          rebuildFixedMedicationsFromChanges();
        }
        if(['no_healthcare_events','no_fixed_medications','unvisited_health_change','no_medication_changes','side_effect'].includes(key)){
          const y=window.scrollY;
          render();
          window.scrollTo({top:y,behavior:'auto'});
        }
      });
    });

    document.querySelectorAll('input[name^="health_med_"]').forEach(input=>input.addEventListener('change',()=>{
      syncInputs();
      const key=input.name.replace('health_med_','');
      local.medicationChecks[key]={...(local.medicationChecks[key]||{}),status:input.value};
      const y=window.scrollY;render();window.scrollTo({top:y,behavior:'auto'});
    }));

    document.querySelectorAll('input[name^="health_item_"]').forEach(input=>input.addEventListener('change',()=>{
      syncInputs();
      local.healthItems[input.name.replace('health_item_','')]=input.value;
      const y=window.scrollY;render();window.scrollTo({top:y,behavior:'auto'});
    }));

    $('healthAddEvent')?.addEventListener('click',()=>{
      syncInputs();
      local.medicalInfo.no_healthcare_events=false;
      local.medicalInfo.healthcare_events=[...safeArray(local.medicalInfo.healthcare_events),emptyEvent()];
      render();
    });

    document.querySelectorAll('[data-remove-health-event]').forEach(btn=>btn.addEventListener('click',()=>{
      syncInputs();
      const index=Number(btn.dataset.removeHealthEvent);
      const events=safeArray(local.medicalInfo.healthcare_events);
      events.splice(index,1);
      local.medicalInfo.healthcare_events=events;
      render();
    }));

    $('healthAddMedicationChange')?.addEventListener('click',()=>{
      syncInputs();
      local.medicalInfo.no_medication_changes=false;
      local.medicalInfo.medication_changes=[...safeArray(local.medicalInfo.medication_changes),newMedicationChange()];
      render();
    });

    document.querySelectorAll('[data-remove-med-change]').forEach(btn=>btn.addEventListener('click',()=>{
      syncInputs();
      const index=Number(btn.dataset.removeMedChange);
      const items=safeArray(local.medicalInfo.medication_changes);
      items.splice(index,1);
      local.medicalInfo.medication_changes=items;
      rebuildFixedMedicationsFromChanges();
      render();
    }));

    document.querySelectorAll('[data-med-change-field]').forEach(el=>{
      const evt=el.tagName==='SELECT'?'change':'input';
      el.addEventListener(evt,()=>{
        syncInputs();
        rebuildFixedMedicationsFromChanges();
        const fixedEl=document.querySelector('[data-health-medical="fixed_medications"]');
        if(fixedEl)fixedEl.value=local.medicalInfo.fixed_medications||'';
        const noneEl=document.querySelector('[data-health-medical="no_fixed_medications"]');
        if(noneEl)noneEl.checked=local.medicalInfo.no_fixed_medications===true;
        if(el.tagName==='SELECT'){
          const y=window.scrollY;render();window.scrollTo({top:y,behavior:'auto'});
        }
      });
    });

    $('healthCopyPrevious')?.addEventListener('click',()=>{
      if(!local.previous)return;
      local.healthItems={...safe(local.previous.health_items)};
      local.healthNotes={...safe(local.previous.health_notes)};
      local.pendingCopyFromId=local.previous.id;
      render();
      toast('已帶入前次身體與健康狀況','success');
    });

    $('healthDraft')?.addEventListener('click',()=>save(false));
    $('healthConfirm')?.addEventListener('click',()=>save(true));
    window.scrollTo({top:0,behavior:'auto'});
  }

  async function save(finalize){
    const row=eventById(local.eventId);
    if(!row||!canEdit(row))return toast('您沒有此個案的評估編輯權限','error');

    syncInputs();

    if(finalize&&answeredCount()!==10){
      const missing=HEALTH_ITEMS
        .filter(([k])=>!HEALTH_OPTIONS.some(x=>x[0]===local.healthItems[k]))
        .map(([,l])=>l)
        .join('、');
      return toast('尚有未完成項目：'+missing,'warn');
    }

    if(!finalize&&answeredCount()===0&&Object.keys(local.medicalInfo).length===0){
      return toast('至少填寫一項後再儲存草稿','warn');
    }

    const medicalInfoForSave={...local.medicalInfo};
    delete medicalInfoForSave.diagnosis_change;
    delete medicalInfoForSave.diagnosis_change_note;

    const c=caseById(row.case_id);
    const savedMedical=safe(local.record?.medical_info);
    const baselineReference=local.record?{
      important_conditions:savedMedical.baseline_important_conditions||'',
      ongoing_treatments:savedMedical.baseline_ongoing_treatments||''
    }:{
      important_conditions:c?.important_conditions||'',
      ongoing_treatments:c?.ongoing_treatments||''
    };
    const healthMasterFieldsChanged=
      String(medicalInfoForSave.baseline_important_conditions||'').trim()!==String(baselineReference.important_conditions||'').trim()
      ||String(medicalInfoForSave.baseline_ongoing_treatments||'').trim()!==String(baselineReference.ongoing_treatments||'').trim();

    if(local.hasLaterCompleted&&healthMasterFieldsChanged){
      const ok=window.confirm('你正在修改較早日期的健康評估，此項修改也會同步更新目前個案主檔，確定儲存嗎？');
      if(!ok)return;
    }

    const masterWillChange=
      healthMasterFieldsChanged
      &&(
        String(medicalInfoForSave.baseline_important_conditions||'').trim()!==String(c?.important_conditions||'').trim()
        ||String(medicalInfoForSave.baseline_ongoing_treatments||'').trim()!==String(c?.ongoing_treatments||'').trim()
      );

    if(medicalInfoForSave.no_healthcare_events===true){
      medicalInfoForSave.healthcare_events=[];
    }else{
      medicalInfoForSave.healthcare_events=safeArray(medicalInfoForSave.healthcare_events)
        .filter(item=>Object.values(safe(item)).some(v=>String(v||'').trim()!==''));
    }

    if(medicalInfoForSave.unvisited_health_change!=='yes'){
      medicalInfoForSave.unvisited_health_change_note='';
    }

    if(medicalInfoForSave.no_fixed_medications===true){
      medicalInfoForSave.fixed_medications='';
    }

    let medicationChanges=medicalInfoForSave.no_medication_changes===true
      ?[]
      :safeArray(medicalInfoForSave.medication_changes).map(x=>({...x}));

    if(finalize){
      const incomplete=medicationChanges.find(x=>!isCompleteMedicationChange(x));
      if(incomplete){
        return toast('請完成本期用藥異動的「異動類型、用藥性質及藥物內容」','warn');
      }
    }

    local.medicalInfo.medication_changes=medicationChanges;
    local.medicalInfo.no_medication_changes=medicationChanges.length===0;
    const fixedLines=rebuildFixedMedicationsFromChanges();

    medicalInfoForSave.fixed_medications=fixedLines.join('\n');
    medicalInfoForSave.no_fixed_medications=fixedLines.length===0;
    medicalInfoForSave.medication_changes=medicationChanges;
    medicalInfoForSave.no_medication_changes=medicationChanges.length===0;
    medicalInfoForSave.medication_change=medicationChanges.length?'changed':'none';
    medicalInfoForSave.medication_change_note=medicationChanges.filter(isCompleteMedicationChange).map(item=>{
      const action=item.action==='add'?'新增':item.action==='stop'?'停用':'調整';
      const duration=item.duration_type==='long_term'?'長期／固定':'短期／臨時';
      const detail=item.action==='add'
        ?item.medication_after
        :item.action==='stop'
          ?item.medication_before
          :item.medication_before+' → '+item.medication_after;
      return action+'｜'+duration+'｜'+detail+(item.note?'｜'+item.note:'');
    }).join('\n');

    if(medicalInfoForSave.side_effect!=='present'){
      medicalInfoForSave.side_effect_note='';
    }

    showLoading(true);
    try{
      const {error}=await db.rpc('save_assessment_health',{
        p_event_id:row.id,
        p_medical_info:medicalInfoForSave,
        p_medication_checks:local.medicationChecks,
        p_health_items:local.healthItems,
        p_health_notes:local.healthNotes,
        p_finalize:!!finalize,
        p_copied_from_id:local.pendingCopyFromId||null
      });
      if(error)throw error;

      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      await load(row.id);
      render();
      toast(
        masterWillChange
          ?(finalize?'身體與健康狀況評估已完成，個案健康資料已更新':'健康評估草稿已儲存，個案健康資料已更新')
          :(finalize?'身體與健康狀況評估已完成':'健康評估草稿已儲存'),
        'success'
      );
    }catch(err){
      console.error(err);
      toast('儲存健康評估失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(row.status==='pending')return toast('請先開始本次評估','warn');
    if(!formById(eventId))return toast('本次評估未選用身體與健康狀況評估','warn');

    showLoading(true);
    try{await load(eventId);render();}
    catch(err){console.error(err);toast('讀取健康評估失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  window.v52AssessmentHealth={open};
})();
