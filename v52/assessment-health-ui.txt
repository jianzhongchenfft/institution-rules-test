  function renderMedCheck([key,label]){
    const cur=local.medicationChecks[key]||{};
    const prev=local.previous?.medication_checks?.[key]||{};
    const changed=cur.status&&prev.status&&cur.status!==prev.status;
    const noteValue=String(cur.note||'');
    const noteRequired=cur.status==='improve'||cur.status==='not_applicable';
    const showNote=noteRequired||!!noteValue||local.expandedMedicationNotes.has(key);
    const notePlaceholder=cur.status==='improve'
      ?'請說明需改善的狀況'
      :cur.status==='not_applicable'
        ?'請說明不適用原因'
        :'補充備註（選填）';

    return `
      <div class="health-card ${changed?'changed':''}">
        <div class="health-card-head">
          <div class="health-card-title">${escapeHTML(label)}</div>
          <div class="health-tags">
            ${prev.status?`<span class="health-tag prev">前次：${escapeHTML(medStatusLabel(prev.status))}</span>`:''}
            ${changed?'<span class="health-tag change">本次有變化</span>':''}
          </div>
        </div>
        <div class="health-options">
          ${[['good','狀況良好'],['improve','需改善'],['not_applicable','不適用']].map(([v,l])=>`
            <label class="health-option ${cur.status===v?'selected':''}">
              <input type="radio" name="health_med_${key}" value="${v}" ${cur.status===v?'checked':''}>${l}
            </label>
          `).join('')}
        </div>
        ${!showNote&&!['improve','not_applicable'].includes(cur.status)
          ?`<div class="health-note"><button type="button" class="btn sm" data-health-med-add-note="${key}">＋ 補充備註</button></div>`
          :''}
        ${showNote?`<div class="health-note">
          <label style="display:block;font-size:10px;font-weight:850;color:#667085;margin-bottom:4px">備註${noteRequired?' *':''}</label>
          <textarea data-health-med-note="${key}" placeholder="${escapeHTML(notePlaceholder)}">${escapeHTML(noteValue)}</textarea>
        </div>`:''}
      </div>
    `;
  }

  function renderFixedMedicationList(entries){
    if(!entries.length){
      return '<div class="health-notice" style="margin:0;background:#f9fafb;border-color:#eaecf0;color:#667085">目前無固定用藥</div>';
    }
    return `
      <div class="health-med-checks">

  function renderMedicationChange(item,index){
    const entries=medicationEntriesBeforeChange(index);
    const action=item.action||'',duration=item.duration_type||'';
    const selectedId=item.medication_before_entry_id||'';
    const beforeOptions=[...entries];
    if(selectedId&&!beforeOptions.some(x=>x.id===selectedId)&&item.medication_before){
      beforeOptions.push({
        id:selectedId,
        medication:item.medication_before,
        origin_type:'legacy',
        origin_date:null,
        origin_event_id:null,
        origin_note:''
      });
    }
    const beforeField=action==='stop'||action==='adjust'
      ?(duration==='long_term'
        ?`<select data-med-change-field="medication_before_entry_id">
            <option value="">請選擇目前固定用藥</option>
            ${beforeOptions.map(entry=>`<option value="${escapeHTML(entry.id)}" ${selectedId===entry.id?'selected':''}>${escapeHTML(entry.medication)}｜${escapeHTML(medicationEntryTag(entry))}</option>`).join('')}
          </select>`
        :`<input data-med-change-field="medication_before" value="${escapeHTML(item.medication_before||'')}" placeholder="例如：感冒藥A">`)
      :'';
    const afterField=action==='add'||action==='adjust'
      ?`<input data-med-change-field="medication_after" value="${escapeHTML(item.medication_after||'')}" placeholder="${action==='add'?'例如：高血壓藥A，每日1次':'請填調整後藥物與用法'}">`
      :'';

    return `
      <div class="health-event" data-med-change="${escapeHTML(item.id||'')}">
        <div class="health-event-head">
          <strong>用藥異動 ${index+1}</strong>
          <button type="button" class="btn sm" data-remove-med-change="${index}">移除</button>
        </div>
        <div class="health-event-grid">
          <div>
            <label>異動類型</label>
            <select data-med-change-field="action">
              <option value="">請選擇</option>
              <option value="add" ${action==='add'?'selected':''}>新增</option>
              <option value="stop" ${action==='stop'?'selected':''}>停用</option>
              <option value="adjust" ${action==='adjust'?'selected':''}>調整</option>
            </select>
          </div>
          <div>
            <label>用藥性質</label>
            <select data-med-change-field="duration_type">
              <option value="">請選擇</option>
              <option value="long_term" ${duration==='long_term'?'selected':''}>長期／固定</option>
              <option value="short_term" ${duration==='short_term'?'selected':''}>短期／臨時</option>
            </select>
          </div>
          ${(action==='stop'||action==='adjust')?`
            <div class="wide">
              <label>${action==='stop'?'停用藥物':'調整前藥物'}</label>
              ${beforeField}
            </div>`:''}
          ${(action==='add'||action==='adjust')?`
            <div class="wide">
              <label>${action==='add'?'新增藥物／用法':'調整後藥物／用法'}</label>
              ${afterField}
            </div>`:''}
          <div class="wide">
            <label>備註（選填）</label>
            <textarea data-med-change-field="note">${escapeHTML(item.note||'')}</textarea>
          </div>
        </div>
      </div>
    `;
  }
