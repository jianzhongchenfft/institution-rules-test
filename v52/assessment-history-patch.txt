  function dateTime(value){
    if(!value)return '—';
    const d=new Date(value);
    if(Number.isNaN(d.getTime()))return String(value);
    return d.toLocaleString('zh-TW',{
      timeZone:'Asia/Taipei',
      year:'numeric',month:'2-digit',day:'2-digit',
      hour:'2-digit',minute:'2-digit',hour12:false
    });
  }

  function staffLabel(id){
    if(!id)return '—';
    return local.staffMap.get(id)
      ||state.assessmentStaff?.find(x=>x.id===id)?.display_name
      ||(state.profile?.id===id?state.profile.display_name:null)
      ||('使用者 '+String(id).slice(0,8));
  }

  function valueText(value){
    if(value===null||value===undefined)return '空白';
    if(typeof value==='boolean')return value?'是':'否';
    if(typeof value==='number')return String(value);
    if(typeof value==='string'){
      const trimmed=value.trim();
      if(!trimmed)return '空白';
      return VALUE_LABELS[trimmed]||trimmed;
    }
    if(Array.isArray(value)){
      if(!value.length)return '無';
      return `共 ${value.length} 筆資料`;
    }
    if(typeof value==='object'){
      if(Object.keys(value).length===1&&Object.prototype.hasOwnProperty.call(value,'status')){
        return valueText(value.status);
      }
      const raw=JSON.stringify(value);
      return raw.length>140?raw.slice(0,137)+'…':raw;
    }
    return String(value);
  }

  function sameValue(a,b){
    return JSON.stringify(a)===JSON.stringify(b);
  }

  function fieldLabel(tool,path){

      const m=path.match(/^(answers|notes)\.(.+)$/);
      if(m)return (ADL_LABELS[m[2]]||m[2])+(m[1]==='notes'?'－備註':'－分數');
      if(path==='total_score')return 'ADL 總分';
      if(path==='dependency_level')return '依賴程度';
    }
    if(tool==='iadl'){
      const m=path.match(/^(answers|notes)\.(.+)$/);
      if(m)return (IADL_LABELS[m[2]]||m[2])+(m[1]==='notes'?'－備註':'－分數');
      if(path==='total_score')return 'IADL 總分';
    }
    if(tool==='spmsq'){
      let m=path.match(/^judgments\.q(\d+)$/);
      if(m)return `第 ${m[1]} 題－判定`;
      m=path.match(/^notes\.q(\d+)$/);
      if(m)return `第 ${m[1]} 題－備註`;
      if(path==='error_count')return '錯誤題數';
      if(path==='is_unable')return '是否無法評估';
      if(path==='unable_reason')return '無法評估原因';
    }
    if(tool==='gds15'){
      const m=path.match(/^answers\.q(\d+)$/);
      if(m)return `第 ${m[1]} 題`;
      if(path==='total_score')return 'GDS-15 總分';
      if(path==='completion_mode')return '完成方式';
      if(path==='exception_reason')return '例外原因';
    }
    if(tool==='health'){
      let m=path.match(/^health_items\.(.+)$/);
      if(m)return HEALTH_LABELS[m[1]]||m[1];
      m=path.match(/^health_notes\.(.+)$/);
      if(m)return (HEALTH_LABELS[m[1]]||m[1])+'－備註';
      m=path.match(/^medication_checks\.([^.]+)(?:\.status)?$/);
      if(m)return '用藥安全－'+(MED_LABELS[m[1]]||m[1]);
      if(path==='total_score')return '健康狀態總分';
      if(path.startsWith('medical_info.'))return '健康／用藥資訊－'+path.replace('medical_info.','');
    }
    if(tool==='home_safety'){
      let m=path.match(/^answers\.(.+)$/);
      if(m)return HOME_LABELS[m[1]]||m[1];
      m=path.match(/^notes\.(.+)$/);
      if(m)return (HOME_LABELS[m[1]]||m[1])+'－備註';
      if(path.startsWith('home_profile.'))return '住宅概況－'+path.replace('home_profile.','');
      if(path.startsWith('summary.'))return '環境安全總結－'+path.replace('summary.','');
    }

      let m=path.match(/^support_domains\.([^.]+)\.(status|note)$/);
      if(m)return (SUPPORT_DOMAIN_LABELS[m[1]]||m[1])+(m[2]==='note'?'－說明':'');

      if(path.startsWith('summary.'))return '支持系統總結－'+path.replace('summary.','');

  function historyValueText(tool,path,value){
    if(tool==='summary'&&path==='previous_followup_status'){
      return {done:'已完成',partial:'部分完成',undone:'未完成',closed:'不再需要追蹤'}[value]||valueText(value);
    }
    if(tool==='health'&&/^medication_checks\.[^.]+\.status$/.test(path)){
      return {good:'狀況良好',improve:'需改善',not_applicable:'不適用'}[value]||valueText(value);
    }
    if(tool==='support'&&path==='resources.economic_status'){
      return {stable:'經濟狀況無虞',watch:'尚可但需留意',difficulty:'有明顯經濟困難'}[value]||valueText(value);
    }
    return valueText(value);
  }

  function flattenChanges(tool,path,oldValue,newValue,out){
    if(sameValue(oldValue,newValue))return;
    if(
      oldValue&&newValue
      &&typeof oldValue==='object'&&!Array.isArray(oldValue)
      &&typeof newValue==='object'&&!Array.isArray(newValue)
    ){
      const keys=new Set([...Object.keys(oldValue),...Object.keys(newValue)]);
      for(const key of keys){
        flattenChanges(tool,path?path+'.'+key:key,oldValue[key],newValue[key],out);
      }
      return;
    }
    out.push({
      label:fieldLabel(tool,path),
      oldValue:historyValueText(tool,path,oldValue),
      newValue:historyValueText(tool,path,newValue)
    });
  }

  function normalizeFieldRow(tool,row){
    const out=[];
    flattenChanges(tool,row.field_name||'',row.old_value,row.new_value,out);
    return out;
  }

  function normalizeSummaryRow(row){
    const oldSnap=row.old_snapshot&&typeof row.old_snapshot==='object'?row.old_snapshot:{};
    const newSnap=row.new_snapshot&&typeof row.new_snapshot==='object'?row.new_snapshot:{};
    const keys=new Set([...Object.keys(oldSnap),...Object.keys(newSnap)]);
    const out=[];
    const hidden=new Set([
      'previous_summary_id','previous_key_findings_snapshot','previous_care_recommendations_snapshot',
      'completed_at'
    ]);
    for(const key of keys){
      if(hidden.has(key))continue;
      flattenChanges('summary',key,oldSnap[key],newSnap[key],out);
    }
    return out;
  }

  function groupRows(tool,label,rows){
    if(tool==='summary'){
      return rows.map(row=>({
        tool,label,changedAt:row.changed_at,changedBy:row.changed_by,
        changes:normalizeSummaryRow(row)
      })).filter(x=>x.changes.length);
    }

    const map=new Map();
    for(const row of rows){
      const key=`${row.changed_at}|${row.changed_by||''}`;
      if(!map.has(key)){
        map.set(key,{tool,label,changedAt:row.changed_at,changedBy:row.changed_by,changes:[]});
      }
      map.get(key).changes.push(...normalizeFieldRow(tool,row));
    }
    return [...map.values()].filter(x=>x.changes.length);
  }

  async function load(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');

    const queries=HISTORY_SOURCES.map(async([tool,label,table])=>{
      const res=await db.from(table)
        .select('*')
        .eq('assessment_event_id',eventId)
        .order('changed_at',{ascending:false});
      if(res.error)throw res.error;
      return groupRows(tool,label,res.data||[]);
    });

    const [groupsBySource,staffRes]=await Promise.all([
      Promise.all(queries),
      db.from('staff_users').select('id,display_name')
    ]);
    if(staffRes.error)throw staffRes.error;

    local.eventId=eventId;
    local.groups=groupsBySource.flat().sort((a,b)=>String(b.changedAt||'').localeCompare(String(a.changedAt||'')));
    local.staffMap=new Map((staffRes.data||[]).map(x=>[x.id,x.display_name]));
  }

  function lifecycleCard(label,value){
    return `<div class="ah-time-card"><label>${escapeHTML(label)}</label><strong>${escapeHTML(dateTime(value))}</strong></div>`;
  }

    window.v52AssessmentApi?.setAssessmentFormHeader?.(row,c,'修改歷程');
    $('content').innerHTML=`<div class="ah-page">
      <div class="page-head">

          <h1>${escapeHTML(c.case_name)}｜評估修改歷程</h1>

        </div>
        <div><button id="ahBack" type="button" class="btn">← 返回評估事件</button></div>
      </div>

      ${window.v52AssessmentApi?.renderAssessmentContextBar?.(row,c)||''}

      <div class="ah-timeline">
        ${lifecycleCard('建立事件',row.created_at)}
        ${lifecycleCard('開始評估',row.started_at)}
        ${lifecycleCard('表單完成',row.forms_completed_at)}
        ${lifecycleCard('整體完成',row.completed_at)}
        ${lifecycleCard('最後更新',row.updated_at)}
      </div>

      <div class="ah-note">
        這裡不是每一次按「儲存」都新增一份完整副本，而是顯示正式紀錄在建立後曾被修改的內容。第一次建立／完成本身不算「修正」；因此沒有修改紀錄是正常情況。
      </div>

      ${local.groups.length?`
        <div class="ah-list">
          ${local.groups.map(group=>`
            <section class="ah-entry">
              <div class="ah-entry-head">
                <div>
                  <strong>${escapeHTML(group.label)}</strong>
                  <small>${escapeHTML(dateTime(group.changedAt))}｜修改人：${escapeHTML(staffLabel(group.changedBy))}</small>
                </div>
                <span class="ah-entry-count">${group.changes.length} 項變更</span>
              </div>
              <div class="ah-entry-body">
                ${group.changes.map(change=>`
                  <div class="ah-change">
                    <div class="ah-change-label">${escapeHTML(change.label)}</div>
                    <div class="ah-value">${escapeHTML(change.oldValue)}</div>
                    <div class="ah-arrow">→</div>
                    <div class="ah-value">${escapeHTML(change.newValue)}</div>
                  </div>
                `).join('')}
              </div>
            </section>
          `).join('')}
        </div>
      `:'<div class="ah-empty">目前沒有修改紀錄。這表示本次評估的正式紀錄建立後尚未再次修正。</div>'}
    </div>`;

    $('ahBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));
    window.scrollTo({top:0,behavior:'auto'});
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(!state.profile||!['business_manager','organization_manager','admin'].includes(state.profile.role)){
      return toast('修改歷程僅供管理者查閱','warn');
    }
    showLoading(true);
    try{
      await load(eventId);
      render();
    }catch(err){
      console.error(err);
      toast('讀取評估修改歷程失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  window.v52AssessmentHistory={open};
})();
