  function changeKind(key){
    const now=answerStatus(key);
    const prev=previousStatus(key);
    if(!now||!prev)return null;
    if(now==='needs_improvement'&&prev==='needs_improvement')return {key:'persistent',label:'持續風險'};
    if(now==='needs_improvement'&&prev!=='needs_improvement')return {key:'new-risk',label:'新增風險'};
    if(prev==='needs_improvement'&&now==='safe')return {key:'improved',label:'已改善'};
    if(prev==='needs_improvement'&&now==='not_applicable')return {key:'no-longer',label:'不再適用'};
    return null;
  }
  function countChange(kind){return HOME_SAFETY_ITEMS.filter(item=>changeKind(item.key)?.key===kind).length;}

  }

  async function loadHomeSafety(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');

    const [currentRes,historyRes]=await Promise.all([
      db.from('assessment_home_safety_records')
        .select('*')
        .eq('assessment_event_id',eventId)
        .maybeSingle(),
      db.from('assessment_home_safety_records')
        .select('*')
        .eq('case_id',row.case_id)
        .not('confirmed_at','is',null)
        .neq('assessment_event_id',eventId)
        .order('confirmed_at',{ascending:false})
    ]);
    if(currentRes.error)throw currentRes.error;
    if(historyRes.error)throw historyRes.error;

    local.eventId=eventId;
    local.record=currentRes.data||null;
    local.homeProfile={...safeObj(local.record?.home_profile)};
    local.answers={...safeObj(local.record?.answers)};
    local.notes={...safeObj(local.record?.notes)};
    local.summary={...safeObj(local.record?.summary)};
    local.expandedNotes=new Set();

    const dated=(historyRes.data||[])
      .map(record=>({record,event:eventById(record.assessment_event_id)}))
      .filter(x=>x.event?.planned_date);

    local.hasLaterCompleted=dated.some(x=>x.event.planned_date>row.planned_date);
    const candidates=dated
      .filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>{
        const byDate=String(b.event.planned_date).localeCompare(String(a.event.planned_date));
        if(byDate!==0)return byDate;
        return String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||''));
      });

    local.previous=candidates[0]?.record||null;
    local.previousEvent=candidates[0]?.event||null;
    const copiedSourceIsValid=!!local.record?.copied_from_id&&local.record.copied_from_id===local.previous?.id;
    local.pendingCopyFromId=copiedSourceIsValid?local.record.copied_from_id:null;
  }

  function renderItem(key,label,disabled){
    const current=answerStatus(key);
    const prev=previousStatus(key);
    const change=changeKind(key);
    const noteValue=String(local.notes?.[key]||'');
    const manualNotApplicable=current==='not_applicable'&&!isAutoNotApplicable(key);
    const noteRequired=current==='needs_improvement'||manualNotApplicable;
    const showNote=noteRequired||!!noteValue||local.expandedNotes.has(key);
    const notePlaceholder=current==='needs_improvement'
      ?'請說明需改善的狀況'
      :manualNotApplicable
        ?'請說明不適用原因'
        :'補充備註（選填）';
    return `
      <div class="hs-item ${current==='needs_improvement'?'risk':''}" data-hs-item="${key}">
        <div class="hs-item-top">
          <div class="hs-item-title">${escapeHTML(label)}</div>
          <div>
            ${prev?`<span style="font-size:9.5px;color:#667085;margin-right:6px">前次：${escapeHTML(STATUS_LABELS[prev]||prev)}</span>`:''}
            ${change?`<span class="hs-change ${change.key}">${change.label}</span>`:''}
          </div>
        </div>
        <div class="hs-options">
          ${['safe','needs_improvement','not_applicable'].map(status=>`
            <label class="hs-option ${status} ${current===status?'selected':''}">
              <input type="radio" name="hs_${key}" value="${status}" ${current===status?'checked':''} ${disabled?'disabled':''}>
              ${STATUS_LABELS[status]}
            </label>
          `).join('')}
        </div>
        ${!showNote&&!disabled&&!['needs_improvement','not_applicable'].includes(current)
          ?`<button type="button" class="btn sm" data-hs-add-note="${key}">＋ 補充備註</button>`

            <input class="hs-item-note" data-hs-note="${key}" value="${escapeHTML(noteValue)}" placeholder="${escapeHTML(notePlaceholder)}">
          </div>
        `:''}
      </div>
    `;
  }

        ?'已選擇不自行進入廚房或備餐，本區自動列為不適用'
        :'';
    return `

        <div class="hs-section-head">
          <strong>${escapeHTML(section.title)}</strong>
          ${note?`<small>${escapeHTML(note)}</small>`:''}
        </div>
        ${section.items.map(([key,label])=>renderItem(key,label,disabled)).join('')}
      </section>
    `;

      return;

    $('content').innerHTML=`
      <div class="hs-page">

  function rerenderPreservingScroll(){
    const y=window.scrollY;
    renderHomeSafetyPage();
    requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));
  }

  function setConditionalGroup(profileKey,value){
    const keys=profileKey==='uses_stairs'?STAIR_KEYS:profileKey==='uses_kitchen'?KITCHEN_KEYS:[];
    if(value==='no'){
      keys.forEach(key=>{local.answers[key]='not_applicable';});
    }else if(value==='yes'){
      keys.forEach(key=>{
        if(local.answers[key]==='not_applicable')delete local.answers[key];
      });
    }
  }

  function bindHomeSafetyEvents(row){
    $('homeSafetyBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));

    document.querySelectorAll('[data-hs-profile]').forEach(el=>{
      const key=el.dataset.hsProfile;
      const rerenderKeys=['housing_type','has_level_difference','uses_stairs','uses_kitchen'];
      el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{
        local.homeProfile[key]=el.value;
        if(key==='uses_stairs'||key==='uses_kitchen')setConditionalGroup(key,el.value);
        if(rerenderKeys.includes(key))rerenderPreservingScroll();
      });
    });

    document.querySelectorAll('[data-hs-mobility]').forEach(input=>{
      input.addEventListener('change',()=>{
        const selected=[...document.querySelectorAll('[data-hs-mobility]:checked')].map(x=>x.dataset.hsMobility);
        local.homeProfile.mobility_modes=selected;
        rerenderPreservingScroll();
      });
    });

    document.querySelectorAll('input[name^="hs_"]').forEach(input=>{
      input.addEventListener('change',()=>{
        const key=input.name.replace(/^hs_/,'');
        local.answers[key]=input.value;
        rerenderPreservingScroll();
      });
    });

    document.querySelectorAll('[data-hs-add-note]').forEach(btn=>{
      btn.addEventListener('click',()=>{
        local.expandedNotes.add(btn.dataset.hsAddNote);
        rerenderPreservingScroll();
      });
    });

    document.querySelectorAll('[data-hs-note]').forEach(input=>{
      input.addEventListener('input',()=>{
        local.notes[input.dataset.hsNote]=input.value;
      });
    });

    document.querySelectorAll('[data-hs-summary]').forEach(el=>{
      const key=el.dataset.hsSummary;
      const rerenderKeys=['explained_to','family_response','followup_required'];
      el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{
        local.summary[key]=el.value;
        if(rerenderKeys.includes(key))rerenderPreservingScroll();
      });
    });

    $('homeSafetyCopyPrevious')?.addEventListener('click',()=>{
      if(!local.previous)return;
      const hasCurrent=Object.keys(local.answers||{}).length||Object.keys(local.homeProfile||{}).length;
      if(hasCurrent&&!confirm('帶入前次評估會覆蓋目前尚未儲存的住宅概況與21項結果，是否繼續？'))return;
      local.homeProfile={...safeObj(local.previous.home_profile)};
      local.answers={...safeObj(local.previous.answers)};
      local.notes={...safeObj(local.previous.notes)};
      local.summary={};
      local.pendingCopyFromId=local.previous.id;
      rerenderPreservingScroll();
      toast('已帶入前次居家環境安全評估，請逐項確認本次狀況','success');
    });

    $('homeSafetySaveDraft')?.addEventListener('click',()=>saveHomeSafety(false));
    $('homeSafetyConfirm')?.addEventListener('click',()=>saveHomeSafety(true));
  }

  function validateFinalize(){
    if(!local.homeProfile.housing_type)return '請選擇住宅類型';
    if(!String(local.homeProfile.primary_floor||'').trim())return '請填寫個案主要活動樓層';
    if(!local.homeProfile.uses_stairs)return '請選擇是否需經常上下樓梯';
    if(!local.homeProfile.has_level_difference)return '請選擇是否有明顯門檻或高低差';
    if(!Array.isArray(local.homeProfile.mobility_modes)||!local.homeProfile.mobility_modes.length)return '請至少選擇一項主要移動方式';
    if(!local.homeProfile.uses_kitchen)return '請選擇個案是否會自行進入廚房或備餐';

    const missing=HOME_SAFETY_ITEMS.filter(item=>!answerStatus(item.key));
    if(missing.length)return '尚有未完成項目：'+missing.map(x=>x.label).join('、');

    const missingRiskNotes=HOME_SAFETY_ITEMS.filter(item=>
      answerStatus(item.key)==='needs_improvement'
      && !String(local.notes?.[item.key]||'').trim()
    );
    if(missingRiskNotes.length)return '以下項目選擇「需改善」，請填寫備註：'+missingRiskNotes.map(x=>x.label).join('、');

    const missingNaNotes=HOME_SAFETY_ITEMS.filter(item=>
      answerStatus(item.key)==='not_applicable'
      && !isAutoNotApplicable(item.key)
      && !String(local.notes?.[item.key]||'').trim()
    );
    if(missingNaNotes.length)return '以下項目選擇「不適用」，請填寫原因：'+missingNaNotes.map(x=>x.label).join('、');

    if(riskKeys().length){
      if(!String(local.summary.recommendations||'').trim())return '本次有需改善項目，請填寫主要環境安全問題／改善建議';
      if(!local.summary.explained_to)return '本次有需改善項目，請選擇說明對象';
      if(local.summary.explained_to==='not_explained'&&!String(local.summary.not_explained_reason||'').trim())return '請填寫未說明原因';
      if(!local.summary.family_response)return '本次有需改善項目，請選擇個案／家屬回應';
      if(!local.summary.followup_required)return '本次有需改善項目，請選擇是否需後續追蹤';
    }
    return null;
  }

  async function saveHomeSafety(finalize){
    const row=eventById(local.eventId);
    if(!row||!canEditEvent(row))return toast('您沒有此個案的評估編輯權限','error');

    if(finalize){
      const problem=validateFinalize();
      if(problem)return toast(problem,'warn');
    }else{
      const hasData=Object.keys(local.homeProfile||{}).length||Object.keys(local.answers||{}).length||Object.keys(local.notes||{}).length||Object.keys(local.summary||{}).length;
      if(!hasData)return toast('至少填寫一項後再儲存草稿','warn');
    }

    const scrollY=window.scrollY;
    showLoading(true);
    try{
      const {error}=await db.rpc('save_assessment_home_safety',{
        p_event_id:row.id,
        p_home_profile:local.homeProfile,
        p_answers:local.answers,
        p_notes:local.notes,
        p_summary:local.summary,
        p_finalize:!!finalize,
        p_copied_from_id:local.pendingCopyFromId||local.record?.copied_from_id||null
      });
      if(error)throw error;

      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){
        window.v52AssessmentApi?.renderAssessmentDetail?.(row.id);
        toast('居家環境安全評估已完成','success');
      }else{
        await loadHomeSafety(row.id);
        renderHomeSafetyPage();
        requestAnimationFrame(()=>window.scrollTo({top:scrollY,behavior:'auto'}));
        toast('居家環境安全評估草稿已儲存','success');
      }
    }catch(err){
      console.error(err);
      const msg=String(err?.message||err);
      if(msg.includes('HOME_SAFETY_INCOMPLETE'))toast('尚有居家環境安全評估項目未完成','warn');
      else toast('儲存居家環境安全評估失敗：'+msg,'error');
    }finally{
      showLoading(false);
    }
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(row.status==='pending')return toast('請先開始本次評估','warn');
    if(!homeSafetyForm(eventId))return toast('本次評估未選用居家環境安全評估','warn');

    showLoading(true);
    try{
      await loadHomeSafety(eventId);
      renderHomeSafetyPage();
      window.scrollTo({top:0,behavior:'auto'});
    }catch(err){
      console.error(err);
      toast('讀取居家環境安全評估失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  window.v52AssessmentHomeSafety={open};
})();
