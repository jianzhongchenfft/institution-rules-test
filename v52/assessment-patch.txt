  async function loadAssessmentData(showBusy=true){
    if(!canUseAssessments())return;
    if(showBusy)showLoading(true);

          .order('created_at',{ascending:false}),
        db.from('assessment_event_forms')
          .select('*')
          .order('sort_order',{ascending:true})

      state.assessmentCases=casesRes.data||[];
      state.assessmentStaff=staffRes.data||[];
      state.assessmentEvents=eventRes.data||[];

    }finally{
      if(showBusy)showLoading(false);
    }
  }

  const ASSESSMENT_STATUS_SORT_ORDER={
    pending:0,
    in_progress:1,
    forms_completed:2,
    completed:3,
    voided:4
  };

  function assessmentSortIndicator(key){
    if(state.assessmentSortKey!==key)return '<span class="assessment-sort-indicator"></span>';
    return '<span class="assessment-sort-indicator">'+(state.assessmentSortDir==='asc'?'↑':'↓')+'</span>';
  }

  function assessmentSortValue(row,key){
    const c=assessmentCase(row.case_id);
    if(key==='planned_date')return row.planned_date||'9999-12-31';
    if(key==='case_name')return c?.case_name||'';
    if(key==='assessment_type')return assessmentTypeLabel(row.assessment_type);
    if(key==='reason')return assessmentReasonText(row);
    if(key==='supervisor')return assessmentStaffName(row.responsible_staff_id||c?.supervisor_id);
    if(key==='progress'){
      const p=assessmentProgress(row.id);
      return {ratio:p.total?p.done/p.total:0,done:p.done,total:p.total};
    }
    if(key==='status')return ASSESSMENT_STATUS_SORT_ORDER[row.status]??99;
    return '';
  }

  function assessmentCompareRows(a,b,key){
    if(key==='progress'){
      const av=assessmentSortValue(a,key),bv=assessmentSortValue(b,key);
      if(av.ratio!==bv.ratio)return av.ratio-bv.ratio;
      if(av.done!==bv.done)return av.done-bv.done;
      return av.total-bv.total;
    }
    const av=assessmentSortValue(a,key),bv=assessmentSortValue(b,key);
    if(typeof av==='number'&&typeof bv==='number')return av-bv;
    return String(av).localeCompare(String(bv),'zh-Hant',{numeric:true,sensitivity:'base'});
  }

  function assessmentScopedRows(){
    let rows=[...state.assessmentEvents];
    if(state.assessmentOwner==='mine'){
      rows=rows.filter(row=>assessmentCase(row.case_id)?.supervisor_id===state.profile.id);
    }
    if(state.assessmentListScope==='open'){
      rows=rows.filter(row=>!['completed','voided'].includes(row.status));
    }
    const bounds=assessmentMonthBounds();
    if(state.assessmentFilter==='this_month'){
      rows=rows.filter(row=>!['completed','voided'].includes(row.status)&&row.planned_date>=bounds.first&&row.planned_date<bounds.next);
    }else if(state.assessmentFilter==='overdue'){
      rows=rows.filter(assessmentIsOverdue);
    }else if(state.assessmentFilter==='in_progress'){
      rows=rows.filter(row=>row.status==='in_progress');
    }else if(state.assessmentFilter==='forms_completed'){
      rows=rows.filter(row=>row.status==='forms_completed');
    }
    if(state.assessmentSortKey){
      const dir=state.assessmentSortDir==='desc'?-1:1;
      rows.sort((a,b)=>{
        const primary=assessmentCompareRows(a,b,state.assessmentSortKey);
        if(primary!==0)return primary*dir;
        return String(a.planned_date||'').localeCompare(String(b.planned_date||''))*dir;
      });
    }else{
      rows.sort((a,b)=>{
        const aOpen=!['completed','voided'].includes(a.status);
        const bOpen=!['completed','voided'].includes(b.status);
        if(aOpen!==bOpen)return aOpen?-1:1;
        if(aOpen)return String(a.planned_date||'9999-12-31').localeCompare(String(b.planned_date||'9999-12-31'));
        return String(b.planned_date||'').localeCompare(String(a.planned_date||''));
      });
    }
    return rows;
  }

  function assessmentEditableCases(existing){

  function assessmentReasonOptions(selected){
    return '<option value="">請選擇</option>'+ASSESSMENT_REASON_OPTIONS
      .map(([value,label])=>'<option value="'+value+'" '+(value===selected?'selected':'')+'>'+escapeHTML(label)+'</option>')
      .join('');
  }

  function assessmentCaseOptions(cases,selected){
    return '<option value="">請選擇個案</option>'+cases.map(c=>
      '<option value="'+c.id+'" '+(c.id===selected?'selected':'')+'>'+escapeHTML(c.case_name)+(c.case_no?'｜'+escapeHTML(c.case_no):'')+'</option>'
    ).join('');
  }

  function assessmentTypeOptions(selected){
    return Object.entries(ASSESSMENT_TYPES).map(([value,label])=>
      '<option value="'+value+'" '+(value===selected?'selected':'')+'>'+escapeHTML(label)+'</option>'
    ).join('');
  }

  function assessmentFormAvailable(def,type){
    return (def.availableFor||ALL_ASSESSMENT_TYPES).includes(type);
  }

  function assessmentFormRequired(def,type){
    return (def.requiredFor||[]).includes(type);
  }

  function assessmentFormKindLabel(def,type){
    if(assessmentFormRequired(def,type)){
      if(def.kind==='health_medication')return type==='opening'?'開案初始版':'後續追蹤版';
      if(def.kind==='baseline_carry')return type==='opening'?'開案建立基準':'沿用現況並確認';
      return '完整評估核心工具';
    }
    if(def.kind==='health_medication')return '依本次問題選用';
    return '依個案狀況選用';
  }

  function assessmentSelectedCodesForNew(type){
    return ASSESSMENT_FORMS
      .filter(def=>assessmentFormAvailable(def,type)&&assessmentFormRequired(def,type))
      .map(def=>def.code);
  }

  function assessmentFormStatusLabel(v){
    return ({
      pending:'待進行',
      in_progress:'填寫中',
      completed:'已完成',
      unable:'無法完成',
      not_applicable:'不適用'
    })[v]||v||'—';
  }

  function renderAssessmentFormOptions(existing,type){
    const currentForms=existing?assessmentFormsFor(existing.id):[];
    const currentMap=new Map(currentForms.map(x=>[x.form_code,x]));
    const eventLocked=!!existing&&['completed','voided'].includes(existing.status);
    const selected=new Set(
      existing
        ?currentForms
          .filter(row=>{
            const def=ASSESSMENT_FORMS.find(x=>x.code===row.form_code);
            return def&&assessmentFormAvailable(def,type);
          })
          .map(x=>x.form_code)
        :assessmentSelectedCodesForNew(type)
    );

    return ASSESSMENT_FORMS
      .filter(def=>assessmentFormAvailable(def,type))
      .map(def=>{
        const row=currentMap.get(def.code);
        const locked=!!row&&row.status!=='pending';
        const required=assessmentFormRequired(def,type);
        const checked=selected.has(def.code)||required;
        return `
          <label class="assessment-form-option">
            <input type="checkbox"
              data-assessment-form="${def.code}"
              data-status-locked="${locked?'1':'0'}"
              data-required="${required?'1':'0'}"
              ${checked?'checked':''}
              ${(eventLocked||locked||required)?'disabled':''}>
            <span>
              <strong>${escapeHTML(def.name)}</strong>
              <small>${escapeHTML(assessmentFormKindLabel(def,type))}${eventLocked?'｜評估事件已完成，工具組合不可變更':locked?'｜已開始填寫，無法由此移除':''}</small>
            </span>
          </label>
        `;
      }).join('');
  }

  function applyAssessmentFormRules(){
    document.querySelectorAll('[data-assessment-form]').forEach(input=>{
      const locked=input.dataset.statusLocked==='1';
      const required=input.dataset.required==='1';
      if(locked||required){
        input.checked=true;
        input.disabled=true;
      }else{
        input.disabled=false;
      }
    });
  }

  }

      condition_change:'post_hospital_change',

  }

            <div id="assessmentSupervisorPreview" class="assessment-supervisor-preview"></div>

            <div class="assessment-form-grid" id="assessmentFormGrid">
              ${renderAssessmentFormOptions(existing,initialType)}
            </div>
          </div>
        </div>
      `,
      onConfirm:async()=>{
        const caseId=existing?.case_id||$('assessmentCase')?.value;
        const type=$('assessmentType')?.value;
        const plannedDate=$('assessmentPlannedDate')?.value;
        const reasonCode=$('assessmentReason')?.value;
        if(!caseId)return toast('請選擇個案','warn');
        if(!type)return toast('請選擇評估類型','warn');
        if(!plannedDate)return toast('請填寫預定評估日期','warn');
        if(!reasonCode)return toast('請選擇評估原因','warn');

        const newerCompletedAssessment=[...state.assessmentEvents]
          .filter(row=>row.case_id===caseId
            && row.id!==existing?.id
            && ['forms_completed','completed'].includes(row.status)
            && row.planned_date
            && row.planned_date>plannedDate)
          .sort((a,b)=>String(b.planned_date).localeCompare(String(a.planned_date)))[0];

        if(newerCompletedAssessment){
          const ok=window.confirm(
            '本次評估日期早於個案較新的已完成評估紀錄，系統將視為補登歷史評估，前次資料會依評估日期重新判定。\n\n確定仍要繼續儲存嗎？'
          );
          if(!ok)return false;
        }

        if(!$('assessmentWithVisit')?.checked){
          const ok=window.confirm('本次評估未勾選「與家訪同日進行」，確定仍要繼續儲存嗎？');
          if(!ok)return false;
        }

        const selectedCodes=[...document.querySelectorAll('[data-assessment-form]')]
          .filter(x=>x.checked)
          .map(x=>x.dataset.assessmentForm);
        if(!selectedCodes.length)return toast('請至少選擇一項評估工具','warn');

        const c=assessmentCase(caseId);
        if(!c||!assessmentCanEditCase(caseId))return toast('您沒有此個案的評估編輯權限','error');

        showLoading(true);
        let createdEvent=null;
        try{
          const payload={
            planned_date:plannedDate,
            reason_code:reasonCode,
            reason_note:$('assessmentReasonNote')?.value.trim()||null,
            combined_with_home_visit:!!$('assessmentWithVisit')?.checked,
            problem_followup_date:['condition_change','specific_problem'].includes(type)
              ?($('assessmentProblemFollowupDate')?.value||null)
              :null,
            updated_by:state.user.id,
            updated_at:new Date().toISOString()
          };

          if(existing){
            if(existing.status==='pending')payload.assessment_type=type;
            const {error}=await db.from('assessment_events').update(payload).eq('id',existing.id);
            if(error)throw error;

            if(!['completed','voided'].includes(existing.status)){
              const current=assessmentFormsFor(existing.id);
              const currentMap=new Map(current.map(x=>[x.form_code,x]));
              const now=new Date().toISOString();
              const existingRows=[];
              const newRows=[];

              selectedCodes.forEach(code=>{
                const def=ASSESSMENT_FORMS.find(x=>x.code===code);
                const old=currentMap.get(code);
                const row={
                  assessment_event_id:existing.id,
                  form_code:def.code,
                  form_name:def.name,
                  sort_order:def.sort,
                  is_required:assessmentFormRequired(def,type),
                  status:old?.status||'pending',
                  updated_at:now
                };
                if(old?.id){
                  existingRows.push({
                    ...row,
                    id:old.id,
                    created_at:old.created_at||now
                  });
                }else{
                  newRows.push(row);
                }
              });

              if(existingRows.length){
                const {error:upsertError}=await db.from('assessment_event_forms')
                  .upsert(existingRows,{onConflict:'assessment_event_id,form_code'});
                if(upsertError)throw upsertError;
              }
              if(newRows.length){
                const {error:insertError}=await db.from('assessment_event_forms').insert(newRows);
                if(insertError)throw insertError;
              }

              const removable=current
                .filter(x=>x.status==='pending'&&!selectedCodes.includes(x.form_code))
                .map(x=>x.id);
              if(removable.length){
                const {error:deleteError}=await db.from('assessment_event_forms').delete().in('id',removable);
                if(deleteError)throw deleteError;
              }

              if(existing.status!=='pending'){
                const doneStatuses=new Set(['completed','unable','not_applicable']);
                const allFormsDone=selectedCodes.every(code=>{
                  const old=currentMap.get(code);
                  return !!old&&doneStatuses.has(old.status);
                });
                const nextStatus=allFormsDone?'forms_completed':'in_progress';
                const statusPayload={
                  status:nextStatus,
                  forms_completed_at:allFormsDone?(existing.forms_completed_at||now):null,
                  updated_by:state.user.id,
                  updated_at:now
                };
                const {error:statusError}=await db.from('assessment_events')
                  .update(statusPayload)
                  .eq('id',existing.id);
                if(statusError)throw statusError;
              }
            }

            closeModal();
            await loadAssessmentData(false);
            renderAssessmentDetail(existing.id);
            toast('評估事件已更新','success');
            return true;
          }

          const insertPayload={
            case_id:caseId,
            assessment_type:type,
            planned_date:plannedDate,
            reason_code:reasonCode,
            reason_note:$('assessmentReasonNote')?.value.trim()||null,
            combined_with_home_visit:!!$('assessmentWithVisit')?.checked,
            problem_followup_date:['condition_change','specific_problem'].includes(type)
              ?($('assessmentProblemFollowupDate')?.value||null)
              :null,
            responsible_staff_id:c.supervisor_id||null,
            status:'pending',
            created_by:state.user.id,
            updated_by:state.user.id
          };
          const {data:newEvent,error:eventError}=await db.from('assessment_events')
            .insert(insertPayload)
            .select()
            .single();
          if(eventError)throw eventError;
          createdEvent=newEvent;

          const formRows=selectedCodes.map(code=>{
            const def=ASSESSMENT_FORMS.find(x=>x.code===code);
            return {
              assessment_event_id:newEvent.id,
              form_code:def.code,
              form_name:def.name,
              sort_order:def.sort,
              is_required:assessmentFormRequired(def,type),
              status:'pending'
            };
          });
          const {error:formsError}=await db.from('assessment_event_forms').insert(formRows);
          if(formsError)throw formsError;

          if(creationHook){
            try{await creationHook(newEvent);}
            catch(hookErr){
              console.error('assessment create hook failed',hookErr);
              toast('評估已建立，但來源紀錄連結失敗，請稍後從家、電訪紀錄重新連結。','warn');
            }
          }

          closeModal();
          await loadAssessmentData(false);
          renderAssessmentDetail(newEvent.id);
          toast('評估事件已建立','success');
          return true;
        }catch(err){
          if(createdEvent){
            try{
              await db.from('assessment_events').update({
                status:'voided',
                voided_at:new Date().toISOString(),
                void_reason:'建立評估工具時發生錯誤，系統自動作廢',
                updated_by:state.user.id,
                updated_at:new Date().toISOString()
              }).eq('id',createdEvent.id);
            }catch(_){}
          }
          console.error(err);
          toast('儲存評估事件失敗：'+(err?.message||err),'error');
          return false;
        }finally{
          showLoading(false);
        }
      }
    });

    const toggleProblemFollowupField=()=>{
      const type=$('assessmentType')?.value;
      const field=$('assessmentProblemFollowupField');
      if(field)field.style.display=['condition_change','specific_problem'].includes(type)?'block':'none';
    };

    let previousType=initialType;
    $('assessmentType')?.addEventListener('change',()=>{
      const next=$('assessmentType').value;
      const grid=$('assessmentFormGrid');
      if(grid)grid.innerHTML=renderAssessmentFormOptions(existing,next);
      applyAssessmentFormRules();

      if(reason&&(!reason.value||reason.value===defaultAssessmentReasonForType(previousType))){
        reason.value=defaultAssessmentReasonForType(next);
      }
      previousType=next;
      toggleProblemFollowupField();
    });
    $('assessmentCase')?.addEventListener('change',updateAssessmentSupervisorPreview);
    updateAssessmentSupervisorPreview();
    applyAssessmentFormRules();
    toggleProblemFollowupField();
  }

  async function startAssessment(row){
    if(!assessmentCanEditCase(row.case_id))return toast('您沒有此個案的評估編輯權限','error');
    showLoading(true);
    try{
      const {error}=await db.from('assessment_events').update({
        status:'in_progress',
        started_at:row.started_at||new Date().toISOString(),
        updated_by:state.user.id,
        updated_at:new Date().toISOString()
      }).eq('id',row.id).eq('status','pending');
      if(error)throw error;
      await loadAssessmentData(false);
      renderAssessmentDetail(row.id);
      toast('評估事件已開始','success');
    }catch(err){
      toast('開始評估失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  function voidAssessment(row){
    if(!assessmentCanEditCase(row.case_id))return toast('您沒有此個案的評估編輯權限','error');
    openModal({
      title:'作廢評估事件',
      confirmText:'確認作廢',
      html:`
        ${row.status==='completed'?'<div class="assessment-detail-note" style="margin-bottom:10px">此評估已完成。作廢後仍保留原始紀錄與歷程，但不再列入最新有效評估及三個月評估週期。</div>':''}
        <div class="field">
          <label>作廢原因 *</label>
          <textarea id="assessmentVoidReason" rows="4" placeholder="請說明為何作廢此評估事件"></textarea>
        </div>
      `,
      onConfirm:async()=>{
        const reason=$('assessmentVoidReason')?.value.trim();
        if(!reason)return toast('請填寫作廢原因','warn');
        showLoading(true);
        try{
          const {error}=await db.from('assessment_events').update({
            status:'voided',
            voided_at:new Date().toISOString(),
            void_reason:reason,
            updated_by:state.user.id,
            updated_at:new Date().toISOString()
          }).eq('id',row.id);
          if(error)throw error;
          closeModal();
          await loadAssessmentData(false);
          if(window.caseContactApi&&typeof window.caseContactApi.loadData==='function'){
            await window.caseContactApi.loadData(true);
          }
          const returnCaseId=window.__assessmentReturnCaseId;
          if(returnCaseId&&typeof window.openCaseDetailPage==='function'){
            window.__assessmentReturnCaseId=null;
            await window.openCaseDetailPage(returnCaseId,'assessment',true);
          }else{
            renderAssessmentPage();
          }
          toast('評估事件已作廢','success');
          return true;
        }catch(err){
          toast('作廢失敗：'+(err?.message||err),'error');
          return false;
        }finally{
          showLoading(false);
        }
      }
    });
  }

  function assessmentFormActionButton(row,form){
    const config=ASSESSMENT_FORM_HANDLERS[form.form_code];
    if(row.status==='pending')return '<button class="btn sm" disabled>待開始</button>';
    if(row.status==='voided')return '<button class="btn sm" disabled>下一階段開放</button>';
    if(!config||!window[config.api])return '<button class="btn sm" disabled>下一階段開放</button>';

    const done=(config.done||['completed']).includes(form.status);
    return '<button class="btn sm '+(done?'assessment-view-btn':'primary')+'" data-assessment-open="'+escapeHTML(form.form_code)+'">'
      +(done?'查看／修正':'開始／繼續')
      +'</button>';
  }

  function bindAssessmentFormActions(row){
    document.querySelectorAll('[data-assessment-open]').forEach(btn=>{
      const code=btn.dataset.assessmentOpen;
      const config=ASSESSMENT_FORM_HANDLERS[code];
      const api=config?window[config.api]:null;
      if(!api?.open)return;
      btn.addEventListener('click',()=>{
        if(config.section)api.open(row.id,config.section);
        else api.open(row.id);
      });
    });
  }

  function renderAssessmentDetail(id){
    const row=state.assessmentEvents.find(x=>x.id===id);
    if(!row){
      renderAssessmentPage();
      return;
    }
    const c=assessmentCase(row.case_id);
    const forms=assessmentFormsFor(row.id);
    const p=assessmentProgress(row.id);
    const canEdit=assessmentCanEditCase(row.case_id)&&row.status!=='voided';
    $('topTitle').textContent='評估管理';
    $('crumb').textContent='評估事件';
    $('content').innerHTML=`
      <div class="page-head">
        <div>
          <h1>${escapeHTML(c?.case_name||'未知個案')}｜${escapeHTML(assessmentTypeLabel(row.assessment_type))}</h1>
          <p>預定評估日：${row.planned_date?rocDate(row.planned_date+'T00:00:00'):'—'}　${assessmentStatusBadge(row)}</p>
        </div>
        <div class="assessment-head-actions">
          <button id="assessmentBack" class="btn">← 返回評估列表</button>
          ${p.total&&p.done===p.total&&row.status!=='voided'
            ?'<button id="assessmentBundlePrintBtn" class="btn">列印完整評估套表</button>'
            :''}
          ${p.total&&p.done===p.total&&row.status!=='voided'
            ?'<button id="assessmentSummaryBtn" class="btn '+(row.status==='forms_completed'?'primary':'')+'">'+(row.status==='completed'?'查看總結／列印':'填寫評估總結')+'</button>'
            :''}
          ${canEdit?'<button id="assessmentEdit" class="btn">編輯事件</button>':''}
          ${canEdit&&row.status==='pending'?'<button id="assessmentStart" class="btn primary">開始評估</button>':''}
          ${canEdit&&row.status!=='voided'?'<button id="assessmentVoid" class="btn">作廢</button>':''}
        </div>
      </div>

      ${renderAssessmentContextBar(row,c)}

      <div class="assessment-detail-grid">
        <div class="assessment-detail-cell"><label>個案</label><div>${escapeHTML(c?.case_name||'—')}</div></div>
        <div class="assessment-detail-cell"><label>評估類型</label><div>${escapeHTML(assessmentTypeLabel(row.assessment_type))}</div></div>
        <div class="assessment-detail-cell"><label>負責督導</label><div>${escapeHTML(assessmentStaffName(row.responsible_staff_id||c?.supervisor_id))}</div></div>
        <div class="assessment-detail-cell"><label>評估原因</label><div>${escapeHTML(assessmentReasonText(row))}</div></div>
        <div class="assessment-detail-cell"><label>與家訪同日</label><div>${row.combined_with_home_visit?'是':'否'}</div></div>
        <div class="assessment-detail-cell"><label>表單進度</label><div>${p.done}／${p.total}</div></div>
        ${['condition_change','specific_problem'].includes(row.assessment_type)?'<div class="assessment-detail-cell"><label>問題追蹤日期</label><div>'+(row.problem_followup_date?rocDate(row.problem_followup_date+'T00:00:00'):'未設定')+'</div></div>':''}
        <div class="assessment-detail-cell"><label>完整評估週期</label><div>${
          ['opening','periodic'].includes(row.assessment_type)
            ?'本次完成後作為完整評估基準'
            :(row.counts_as_full_reassessment?'已確認視同完整再評估':'目前不影響原定完整評估週期')
        }</div></div>
        ${row.status==='voided'?'<div class="assessment-detail-cell" style="grid-column:1/-1"><label>作廢原因</label><div>'+escapeHTML(row.void_reason||'—')+'</div></div>':''}
      </div>

      <div class="assessment-detail-note">
        評估工具會依評估類型顯示：開案先建立基準資料，後續評估追蹤期間變化；ADL、IADL、認知、情緒與身體健康等則記錄本次當下狀態。
      </div>
      ${canManageAllAssessments()?`
        <div style="text-align:right;margin:-4px 2px 10px;font-size:10px;color:#98a2b3">
          管理查核｜<button id="assessmentHistoryBtn" type="button" style="border:0;background:none;padding:0;color:#667085;font:inherit;text-decoration:underline;cursor:pointer">查看修改歷程</button>
        </div>
      `:''}

      <section class="panel">
        <div class="panel-head">
          <div><h3>本次評估工具</h3><small>${p.total?p.done+'/'+p.total+' 已完成':'尚未選擇工具'}</small></div>
        </div>
        <div class="panel-body">
          ${forms.length?`
            <div class="assessment-form-cards">
              ${forms.map(form=>`
                <div class="assessment-form-card ${escapeHTML(form.status||'pending')}">
                  <div>
                    <strong>${escapeHTML(form.form_name)}</strong>
                    <small>${form.is_required?'核心工具｜':''}${form.status==='completed'?'✓ 已完成':form.status==='in_progress'?'填寫中':escapeHTML(assessmentFormStatusLabel(form.status))}</small>
                  </div>
                  ${assessmentFormActionButton(row,form)}
                </div>
              `).join('')}
            </div>
          `:'<div class="assessment-empty">此評估事件目前沒有評估工具。</div>'}
        </div>
      </section>
    `;

    $('assessmentBack')?.addEventListener('click',()=>{
      const returnCaseId=window.__assessmentReturnCaseId;
      if(returnCaseId&&typeof window.openCaseDetailPage==='function'){
        window.__assessmentReturnCaseId=null;
        window.openCaseDetailPage(returnCaseId,'assessment',true);
        return;
      }
      renderAssessmentPage();
    });
    $('assessmentHistoryBtn')?.addEventListener('click',()=>{
      const api=window.v52AssessmentHistory;
      if(!api?.open)return toast('評估修改歷程模組尚未載入','error');
      api.open(row.id);
    });
    $('assessmentBundlePrintBtn')?.addEventListener('click',()=>{
      const api=window.v52AssessmentBundlePrint;
      if(!api?.open)return toast('完整評估套表列印模組尚未載入','error');
      api.open(row.id);
    });
    $('assessmentSummaryBtn')?.addEventListener('click',()=>{
      const api=window.v52AssessmentSummary;
      if(!api?.open)return toast('評估總結模組尚未載入','error');
      api.open(row.id);
    });
    $('assessmentEdit')?.addEventListener('click',()=>openAssessmentEventModal(row));
    $('assessmentStart')?.addEventListener('click',()=>startAssessment(row));
    $('assessmentVoid')?.addEventListener('click',()=>voidAssessment(row));
    bindAssessmentFormActions(row);
    window.scrollTo({top:0,behavior:'auto'});
  }

  function renderCaseAssessmentTab(caseId){
    const c=assessmentCase(caseId);
    const rows=state.assessmentEvents
      .filter(row=>row.case_id===caseId)
      .sort((a,b)=>String(b.planned_date||'').localeCompare(String(a.planned_date||''))||new Date(b.created_at)-new Date(a.created_at));
    const completed=rows.filter(row=>row.status==='completed');
    const openRows=rows.filter(row=>!['completed','voided'].includes(row.status))
      .sort((a,b)=>String(a.planned_date||'9999-12-31').localeCompare(String(b.planned_date||'9999-12-31')));
    const latest=completed[0]||null;
    const next=openRows[0]||null;
    const canEdit=assessmentCanEditCase(caseId);
    const history=rows.length?rows.map(row=>{
      const p=assessmentProgress(row.id);
      return '<tr>'+
        '<td>'+(row.planned_date?rocDate(row.planned_date+'T00:00:00'):'—')+'</td>'+
        '<td>'+escapeHTML(assessmentTypeLabel(row.assessment_type))+'</td>'+
        '<td>'+escapeHTML(assessmentReasonText(row))+'</td>'+
        '<td>'+escapeHTML(assessmentStaffName(row.responsible_staff_id||c?.supervisor_id))+'</td>'+
        '<td><div class="assessment-progress"><div class="assessment-progress-bar"><span style="width:'+p.pct+'%"></span></div><small>'+p.done+'/'+p.total+'</small></div></td>'+
        '<td>'+assessmentStatusBadge(row)+'</td>'+
        '<td><button class="btn sm" data-case-assessment-id="'+row.id+'">查看評估</button></td>'+
      '</tr>';
    }).join(''):'';
    return [
      '<div class="case-overview-grid">',
        '<div class="case-overview-card"><label>最近完成評估</label><strong>'+(latest&&latest.planned_date?rocDate(latest.planned_date+'T00:00:00'):'尚無')+'</strong><small>'+(latest?escapeHTML(assessmentTypeLabel(latest.assessment_type)):'尚無已完成評估')+'</small></div>',
        '<div class="case-overview-card"><label>進行中／待評估</label><strong>'+openRows.length+' 筆</strong><small>不含已完成及已作廢</small></div>',
        '<div class="case-overview-card"><label>下一筆預定</label><strong>'+(next&&next.planned_date?rocDate(next.planned_date+'T00:00:00'):'尚未安排')+'</strong><small>'+(next?escapeHTML(assessmentTypeLabel(next.assessment_type)):'目前無待辦評估')+'</small></div>',
      '</div>',
      '<div class="case-panel-card">',
        '<div class="case-panel-head"><div><strong>歷次評估紀錄</strong><div class="case-panel-note">此處與「評估管理」共用同一份資料，不另外建立第二套評估紀錄。</div></div>',
          (canEdit?'<button class="btn sm primary" data-case-assessment-new>＋ 新增評估</button>':''),
        '</div>',
        rows.length
          ?'<div style="overflow-x:auto"><table class="case-table-simple"><thead><tr><th>評估日期</th><th>類型</th><th>原因</th><th>負責人</th><th>進度</th><th>狀態</th><th>操作</th></tr></thead><tbody>'+history+'</tbody></table></div>'
          :'<div class="case-empty-inline">此個案目前尚無評估紀錄。</div>',
      '</div>'
    ].join('');
  }

  function bindCaseAssessmentTab(caseId){
    document.querySelectorAll('[data-case-assessment-id]').forEach(btn=>{
      btn.addEventListener('click',()=>{
        window.__assessmentReturnCaseId=caseId;
        renderAssessmentDetail(btn.dataset.caseAssessmentId);
      });
    });
    document.querySelector('[data-case-assessment-new]')?.addEventListener('click',()=>{
      if(!assessmentCanEditCase(caseId))return;
      window.__assessmentReturnCaseId=caseId;
      openAssessmentEventModal();
      setTimeout(()=>{
        const caseSelect=$('assessmentCase');
        if(caseSelect){
          caseSelect.value=caseId;
          caseSelect.dispatchEvent(new Event('change',{bubbles:true}));
        }
      },0);
    });
  }

  function clearAssessmentReturnContext(){
    window.__assessmentReturnCaseId=null;
  }

  window.v52AssessmentApi={
    loadAssessmentData,
    renderAssessmentPage,
    renderAssessmentDetail,
    openAssessmentEventModal,
    renderAssessmentContextBar,
    setAssessmentFormHeader,
    canEditCase:assessmentCanEditCase,
    renderCaseAssessmentTab,
    bindCaseAssessmentTab,
    clearReturnContext:clearAssessmentReturnContext
  };

  const assessmentBaseNavigate=navigate;
  navigate=function(id){
    if(id!=='assessments')return assessmentBaseNavigate(id);
    window.__assessmentReturnCaseId=null;
    const m=modules.find(x=>x.id==='assessments');
    state.page='assessments';
    renderNav();
    $('topTitle').textContent=m?.label||'評估管理';
    $('crumb').textContent=m?.crumb||'評估管理';
    $('sidebar').classList.remove('open');
    $('content').innerHTML='<div class="empty compact"><div class="icon">🩺</div><h3>正在載入評估資料</h3><p>請稍候。</p></div>';
    if(!canUseAssessments()){
      renderAssessmentPage();
      return;
    }
    loadAssessmentData(true).then(renderAssessmentPage);
    window.scrollTo({top:0,behavior:'auto'});
  };
