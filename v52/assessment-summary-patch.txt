  async function maybeRecord(table,eventId){
    const res=await db.from(table).select('*').eq('assessment_event_id',eventId).maybeSingle();
    if(res.error)throw res.error;
    return res.data||null;
  }

  async function load(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');

    const [summary,adl,iadl,spmsq,gds15,health,homeSafety,support,caregiver]=await Promise.all([
      db.from('assessment_event_summaries').select('*').eq('assessment_event_id',eventId).maybeSingle(),
      maybeRecord('assessment_adl_records',eventId),
      maybeRecord('assessment_iadl_records',eventId),
      maybeRecord('assessment_spmsq_records',eventId),
      maybeRecord('assessment_gds15_records',eventId),
      maybeRecord('assessment_health_records',eventId),
      maybeRecord('assessment_home_safety_records',eventId),
      maybeRecord('assessment_support_records',eventId),
      maybeRecord('assessment_caregiver_screen_records',eventId)
    ]);
    if(summary.error)throw summary.error;

    local.eventId=eventId;
    local.summary=summary.data||{
      overall_status:null,
      key_findings:'',
      care_recommendations:'',
      followup_required:null,
      followup_plan:'',
      service_adjustment_required:null,
      service_adjustment_note:'',
      external_coordination_required:null,
      external_coordination_note:''
    };
    local.records={adl,iadl,spmsq,gds15,health,home_safety:homeSafety,support,caregiver};
  }

  function gdsResult(score){
    if(score===null||score===undefined)return '—';
    if(Number(score)<=6)return '適應狀況良好';
    if(Number(score)<=10)return '中度情緒困擾';
    return '高度情緒困擾';
  }

  function toolSummary(code,form){
    const r=local.records;
    if(code==='adl'){
      const x=r.adl;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      return {title:form.form_name,main:`${x.total_score??'—'} 分｜${ADL_LEVELS[x.dependency_level]||'—'}`,detail:'滿分 100 分；分數越高代表日常生活功能越獨立。'};
    }
    if(code==='iadl'){
      const x=r.iadl;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      return {title:form.form_name,main:`${x.total_score??'—'}／8 分`,detail:'工具性日常生活活動；分數越高代表獨立性越好。'};
    }
    if(code==='spmsq'){
      const x=r.spmsq;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      if(x.is_unable)return {title:form.form_name,main:'本次無法評估',detail:text(x.unable_reason)};
      const err=Number(x.error_count||0);
      return {title:form.form_name,main:`錯誤 ${err} 題`,detail:err<=2?'多屬正常範圍':'3 題以上；建議持續觀察並視需要進一步評估。'};
    }
    if(code==='gds15'){
      const x=r.gds15;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      if(x.completion_mode!=='completed'){
        const mode=x.completion_mode==='unable'?'無法評估':'不適用';
        return {title:form.form_name,main:mode,detail:text(x.exception_reason)};
      }
      return {title:form.form_name,main:`${x.total_score??'—'}／15 分｜${gdsResult(x.total_score)}`,detail:''};
    }
    if(code==='health_medication'){
      const x=r.health;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      const items=safeObj(x.health_items);
      const watch=Object.entries(items).filter(([,v])=>v==='observe').map(([k])=>HEALTH_ITEMS[k]||k);
      const refer=Object.entries(items).filter(([,v])=>v==='refer').map(([k])=>HEALTH_ITEMS[k]||k);
      const notes=safeObj(x.health_notes);
      const details=[];
      if(watch.length)details.push('需觀察：'+joinList(watch));
      if(refer.length)details.push('異常轉介：'+joinList(refer));
      const noteList=Object.entries(notes).filter(([,v])=>String(v||'').trim()).map(([k,v])=>`${HEALTH_ITEMS[k]||k}：${v}`);
      if(noteList.length)details.push(noteList.join('；'));
      if(!details.length)details.push('10 項健康狀況未見需特別列示之異常。');
      return {title:form.form_name,main:`健康狀態 ${x.total_score??'—'}／20 分`,detail:details.join('\n')};
    }
    if(code==='home_safety'){
      const x=r.home_safety;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      const answers=safeObj(x.answers);
      const risks=Object.entries(answers).filter(([,v])=>v==='needs_improvement').map(([k])=>HOME_ITEMS[k]||k);
      const summary=safeObj(x.summary);
      const details=[];
      if(risks.length)details.push('需改善：'+joinList(risks));
      if(String(summary.recommendations||'').trim())details.push('改善建議：'+summary.recommendations);
      return {title:form.form_name,main:risks.length?`需改善 ${risks.length} 項`:'未發現需改善項目',detail:details.join('\n')};
    }
    if(code==='support'){
      const x=r.support;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      const domains=safeObj(x.support_domains);
      const concerns=Object.entries(domains)
        .filter(([,v])=>['partial','insufficient'].includes(v?.status))
        .map(([k])=>SUPPORT_DOMAINS[k]||k);
      const resources=safeObj(x.resources);
      const unmet=safeArr(resources.unmet_needs).filter(v=>!['completed','declined'].includes(v?.status));
      const sum=safeObj(x.summary);
      const details=[];
      if(concerns.length)details.push('支持需留意：'+joinList(concerns));
      if(unmet.length)details.push('待處理資源需求：'+unmet.length+' 項');
      if(String(sum.support_plan||'').trim())details.push('支持計畫：'+sum.support_plan);
      return {title:form.form_name,main:SUPPORT_OVERALL[sum.overall_support]||'尚未評估',detail:details.join('\n')};
    }
    if(code==='caregiver_burden'){
      const x=r.caregiver;
      if(!x)return {title:form.form_name,main:'無已儲存紀錄',detail:''};
      if(x.applicability!=='applicable'){
        const sum=safeObj(x.summary);
        return {title:form.form_name,main:CAREGIVER_APPLICABILITY[x.applicability]||'不適用',detail:x.applicability==='unable'?text(sum.unable_reason):''};
      }
      const answers=safeObj(x.answers);
      const positive=Object.entries(answers).filter(([,v])=>v==='yes').map(([k])=>CG_INDICATORS[k]||k);
      const sum=safeObj(x.summary);
      const details=[];
      if(positive.length)details.push('符合：'+joinList(positive));
      if(sum.referral_required===true)details.push('已達高負荷家庭照顧者轉介標準。');
      return {title:form.form_name,main:`符合 ${Number(sum.positive_count||positive.length)}／10 項｜${sum.referral_required===true?'達轉介標準':'目前未達轉介標準'}`,detail:details.join('\n')};
    }
    return {title:form.form_name,main:'已完成',detail:''};
  }

  function toolSummaries(){
    return formsByEvent(local.eventId).map(form=>({
      code:form.form_code,
      form,
      ...toolSummary(form.form_code,form)
    }));
  }

  function renderToolCards(){
    return `<div class="asum-grid">${toolSummaries().map(item=>`
      <div class="asum-card">
        <div class="asum-card-head"><strong>${escapeHTML(item.title)}</strong><span>${escapeHTML(item.form.status==='completed'?'已完成':item.form.status==='unable'?'無法完成':'不適用')}</span></div>
        <div class="asum-card-main">${escapeHTML(item.main)}</div>
        <div class="asum-card-detail">${escapeHTML(item.detail||'')}</div>
      </div>
    `).join('')}</div>`;
  }

  function boolSelect(value){
    const val=value===true?'yes':value===false?'no':'';
    return `<option value="">請選擇</option>
      <option value="no" ${val==='no'?'selected':''}>否</option>
      <option value="yes" ${val==='yes'?'selected':''}>是</option>`;
  }

  function captureText(){
    local.summary.overall_status=$('asumOverall')?.value||null;
    local.summary.key_findings=$('asumFindings')?.value||'';
    local.summary.care_recommendations=$('asumRecommendations')?.value||'';
    if($('asumFollowupPlan'))local.summary.followup_plan=$('asumFollowupPlan').value;
    if($('asumAdjustNote'))local.summary.service_adjustment_note=$('asumAdjustNote').value;
    if($('asumExternalNote'))local.summary.external_coordination_note=$('asumExternalNote').value;
  }

  function rerenderWithScroll(){
    const y=window.scrollY;
    render();
    requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));
  }

    $('asumBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));
    $('asumPrint')?.addEventListener('click',()=>openPrintPreview());

    $('asumFindings')?.addEventListener('input',e=>local.summary.key_findings=e.target.value);
    $('asumRecommendations')?.addEventListener('input',e=>local.summary.care_recommendations=e.target.value);
    $('asumFollowup')?.addEventListener('change',e=>{
      captureText();
      local.summary.followup_required=e.target.value==='yes'?true:e.target.value==='no'?false:null;
      if(local.summary.followup_required!==true)local.summary.followup_plan='';
      rerenderWithScroll();
    });

      captureText();
      local.summary.service_adjustment_required=e.target.value==='yes'?true:e.target.value==='no'?false:null;
      if(local.summary.service_adjustment_required!==true)local.summary.service_adjustment_note='';
      rerenderWithScroll();
    });
    $('asumExternal')?.addEventListener('change',e=>{
      captureText();
      local.summary.external_coordination_required=e.target.value==='yes'?true:e.target.value==='no'?false:null;
      if(local.summary.external_coordination_required!==true)local.summary.external_coordination_note='';
      rerenderWithScroll();
    });
    $('asumFollowupPlan')?.addEventListener('input',e=>local.summary.followup_plan=e.target.value);
    $('asumAdjustNote')?.addEventListener('input',e=>local.summary.service_adjustment_note=e.target.value);
    $('asumExternalNote')?.addEventListener('input',e=>local.summary.external_coordination_note=e.target.value);
    $('asumDraft')?.addEventListener('click',()=>save(false));
    $('asumConfirm')?.addEventListener('click',()=>save(true));
  }

  function validate(finalize){
    if(!finalize)return null;
    const s=local.summary;
    if(!s.overall_status)return '請選擇整體狀況';
    if(!String(s.key_findings||'').trim())return '請填寫主要評估發現／問題與需求';
    if(!String(s.care_recommendations||'').trim())return '請填寫照顧服務重點／建議';
    if(s.followup_required===null||s.followup_required===undefined)return '請選擇是否需後續追蹤';
    if(s.service_adjustment_required===null||s.service_adjustment_required===undefined)return '請選擇是否需調整服務／照顧安排';
    if(s.external_coordination_required===null||s.external_coordination_required===undefined)return '請選擇是否需對外協調／轉介';
    if(s.followup_required&&!String(s.followup_plan||'').trim())return '請填寫後續追蹤計畫';
    if(s.service_adjustment_required&&!String(s.service_adjustment_note||'').trim())return '請填寫服務／照顧調整內容';
    if(s.external_coordination_required&&!String(s.external_coordination_note||'').trim())return '請填寫對外協調／轉介內容';
    return null;
  }

  async function save(finalize){
    const row=eventById(local.eventId);
    if(!row||!canEdit(row))return toast('您沒有此個案的評估編輯權限','error');
    captureText();
    if(!allFormsDone(row.id))return toast('尚有評估工具未完成，無法進行整體總結','warn');
    const problem=validate(finalize);
    if(problem)return toast(problem,'warn');

    showLoading(true);
    try{
      const s=local.summary;
      const {error}=await db.rpc('save_assessment_event_summary',{
        p_event_id:row.id,
        p_overall_status:s.overall_status||null,
        p_key_findings:String(s.key_findings||'').trim()||null,
        p_care_recommendations:String(s.care_recommendations||'').trim()||null,
        p_followup_required:s.followup_required,
        p_followup_plan:String(s.followup_plan||'').trim()||null,
        p_service_adjustment_required:s.service_adjustment_required,
        p_service_adjustment_note:String(s.service_adjustment_note||'').trim()||null,
        p_external_coordination_required:s.external_coordination_required,
        p_external_coordination_note:String(s.external_coordination_note||'').trim()||null,
        p_finalize:!!finalize
      });
      if(error)throw error;
      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){
        await load(row.id);
        render();
        toast('本次整體評估已完成','success');
      }else{
        await load(row.id);
        render();
        toast('評估總結草稿已儲存','success');
      }
    }catch(err){
      console.error(err);
      toast('儲存評估總結失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  function printText(value){
    return escapeHTML(value===null||value===undefined||value===''?'—':String(value)).replace(/\n/g,'<br>');
  }

  function printBool(value){
    return value===true?'是':value===false?'否':'—';
  }

  function openPrintPreview(){
    try{
      const html=buildPrintHtml();
      const w=window.open('','_blank');
      if(!w){
        toast('瀏覽器阻擋了列印預覽視窗，請允許此網站開啟彈出式視窗。','warn');
        return;
      }
      w.document.open();
      w.document.write(html);
      w.document.close();
    }catch(err){
      console.error(err);
      toast('建立評估列印預覽失敗：'+(err?.message||err),'error');
    }
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(!allFormsDone(eventId))return toast('尚有評估工具未完成，完成後才能進行整體總結','warn');
    showLoading(true);
    try{
      await load(eventId);
      render();
      window.scrollTo({top:0,behavior:'auto'});
    }catch(err){
      console.error(err);
      toast('讀取評估總結失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }
