  function domainChange(key){
    const now=domainStatus(key),prev=previousDomainStatus(key);
    if(!now||!prev)return null;
    if(now===prev)return ['partial','insufficient'].includes(now)?{kind:'persist',label:'持續需留意'}:null;
    const rank={insufficient:1,partial:2,adequate:3};
    if(!(now in rank)||!(prev in rank))return null;
    return rank[now]>rank[prev]?{kind:'better',label:'支持改善'}:{kind:'worse',label:'支持減弱'};
  }

  async function loadSupport(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');
    const [cur,hist]=await Promise.all([
      db.from('assessment_support_records').select('*').eq('assessment_event_id',eventId).maybeSingle(),
      db.from('assessment_support_records').select('*').eq('case_id',row.case_id).not('confirmed_at','is',null).neq('assessment_event_id',eventId).order('confirmed_at',{ascending:false})
    ]);
    if(cur.error)throw cur.error;if(hist.error)throw hist.error;

    local.eventId=eventId;local.record=cur.data||null;
    local.household={...safe(local.record?.household)};
    local.familyMembers=safeArray(local.record?.family_members).map((x,i)=>({
      id:x?.id||('family-'+i),name:x?.name||'',relation:x?.relation||'',co_resident:x?.co_resident||'',
      contact_frequency:x?.contact_frequency||'',support_role:x?.support_role||'',note:x?.note||''
    }));
    local.supportDomains={...safe(local.record?.support_domains)};
    local.resources={...safe(local.record?.resources)};
    local.resources.social_resources=safeArray(local.resources.social_resources).map((x,i)=>{
      let type=x?.type||'';
      if(type==='other')type=x?.nature==='informal'?'other_informal':'other_formal';
      return {
        id:x?.id||('resource-'+i),nature:resourceNatureForType(type),type,name:x?.name||'',
        assistance:x?.assistance||'',usage_status:x?.usage_status||'',sufficiency:x?.sufficiency||'',insufficiency_note:x?.insufficiency_note||''
      };
    });
    local.resources.unmet_needs=safeArray(local.resources.unmet_needs).map((x,i)=>({
      id:x?.id||('need-'+i),type:x?.type||'',need:x?.need||'',action:x?.action||'',status:x?.status||''
    }));
    local.summary={...safe(local.record?.summary)};
    local.expandedDomainNotes=new Set();

    const dated=(hist.data||[]).map(record=>({record,event:eventById(record.assessment_event_id)})).filter(x=>x.event?.planned_date);
    local.hasLaterCompleted=dated.some(x=>x.event.planned_date>row.planned_date);
    const candidates=dated.filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>String(b.event.planned_date).localeCompare(String(a.event.planned_date))||String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||'')));
    local.previous=candidates[0]?.record||null;local.previousEvent=candidates[0]?.event||null;
    local.pendingCopyFromId=(local.record?.copied_from_id&&local.record.copied_from_id===local.previous?.id)?local.record.copied_from_id:null;
  }

  function validateFinalize(){
    const h=local.household,r=local.resources,s=local.summary;
    if(!h.living_arrangement)return '請選擇居住安排';
    if(!h.family_change_status)return '請選擇近期是否有重大家庭變化';
    if(h.family_change_status==='yes'&&!String(h.family_change_note||'').trim())return '請填寫重大家庭變化說明';
    if(!h.primary_caregiver_status)return '請選擇是否有主要照顧者';
    if(h.primary_caregiver_status==='present'){
      if(!String(h.primary_caregiver_name||'').trim())return '請填寫主要照顧者姓名／稱謂';
      if(!String(h.primary_caregiver_relation||'').trim())return '請填寫主要照顧者與個案關係';
    }
    if(!h.backup_available)return '請選擇是否有替代照顧者／其他重要支持者';
    if(h.backup_available==='yes'&&local.familyMembers.length===0)return '已選擇有替代照顧者／其他重要支持者，請至少新增一位支持者';
    if(h.backup_available==='no'&&local.familyMembers.length>0)return '目前已新增重要支持者，請將「是否有替代照顧者／其他重要支持者」改為「有」';

    const incompleteSupporter=local.familyMembers.find(x=>
      !String(x.name||'').trim()
      ||!String(x.relation||'').trim()
      ||!String(x.support_role||'').trim()
    );
    if(incompleteSupporter)return '重要支持者請至少填寫「姓名／稱謂、與個案關係、可提供哪些協助」';

    const missingDomains=SUPPORT_DOMAINS.filter(([key])=>!domainStatus(key));
    if(missingDomains.length)return '尚有未完成支持面向：'+missingDomains.map(([,label])=>label).join('、');
    const missingNotes=SUPPORT_DOMAINS.filter(([key])=>['partial','insufficient','not_applicable'].includes(domainStatus(key))&&!String(local.supportDomains?.[key]?.note||'').trim());
    if(missingNotes.length)return '以下支持面向請填寫說明：'+missingNotes.map(([,label])=>label).join('、');

    if(!r.economic_status)return '請選擇目前經濟狀況';
    if(['watch','difficulty'].includes(r.economic_status)&&!String(r.economic_note||'').trim())return '請填寫經濟狀況說明';
    if(!r.economic_care_impact)return '請選擇經濟因素是否影響照顧或醫療';
    if(r.economic_care_impact==='yes'&&!String(r.economic_care_impact_note||'').trim())return '請填寫經濟因素影響情形';

    const incompleteResource=safeArray(r.social_resources).find(x=>!x.type||!resourceNatureForType(x.type)||!String(x.name||'').trim()||!String(x.assistance||'').trim()||!x.usage_status||!x.sufficiency);
    if(incompleteResource)return '請完整填寫目前社會資源的「類型、名稱、提供協助、使用狀況、是否足夠」';
    const resourceNeedNote=safeArray(r.social_resources).find(x=>['partial','insufficient'].includes(x.sufficiency)&&!String(x.insufficiency_note||'').trim());
    if(resourceNeedNote)return '社會資源選擇「部分不足／不足」時，請填寫不足原因';

    if(!r.social_interaction_status)return '請選擇目前社會互動情形';
    if(['limited','isolated','unable'].includes(r.social_interaction_status)&&!String(r.social_interaction_note||'').trim())return '請填寫社會互動情形說明';
    if(!r.community_participation_status)return '請選擇是否參與社區、宗教、據點或其他活動';

    if(!r.unmet_needs_status)return '請選擇目前是否有尚未滿足的資源需求';
    if(r.unmet_needs_status==='yes'&&safeArray(r.unmet_needs).length===0)return '已選擇有尚未滿足的資源需求，請至少新增一筆需求';
    const incompleteNeed=safeArray(r.unmet_needs).find(x=>!x.type||!String(x.need||'').trim()||!String(x.action||'').trim()||!x.status);
    if(incompleteNeed)return '請完整填寫尚缺資源的「需求類型、需求內容、處理／轉介方式、處理狀態」';

    if(!s.overall_support)return '請選擇整體支持狀況';
    if(['needs_attention','weak'].includes(s.overall_support)&&!String(s.key_issues||'').trim())return '請填寫主要支持問題';
    if(!s.followup_required)return '請選擇是否需後續追蹤';
    if(s.followup_required==='yes'&&!String(s.followup_note||'').trim())return '請填寫後續追蹤重點';
    return null;
  }

  async function saveSupport(finalize){
    const row=eventById(local.eventId);if(!row||!canEditEvent(row))return toast('您沒有此個案的評估編輯權限','error');
    syncDynamicRows();
    if(finalize){const problem=validateFinalize();if(problem)return toast(problem,'warn');}
    else{
      const hasData=Object.keys(local.household||{}).length||local.familyMembers.length||Object.keys(local.supportDomains||{}).length||Object.keys(local.resources||{}).length||Object.keys(local.summary||{}).length;
      if(!hasData)return toast('至少填寫一項後再儲存草稿','warn');
    }
    const y=window.scrollY;showLoading(true);
    try{
      const {error}=await db.rpc('save_assessment_support',{
        p_event_id:row.id,p_household:local.household,p_family_members:local.familyMembers,p_support_domains:local.supportDomains,
        p_resources:local.resources,p_summary:local.summary,p_finalize:!!finalize,p_copied_from_id:local.pendingCopyFromId||local.record?.copied_from_id||null
      });
      if(error)throw error;
      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){window.v52AssessmentApi?.renderAssessmentDetail?.(row.id);toast('支持系統評估已完成','success');}
      else{await loadSupport(row.id);renderSupportPage();requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));toast('支持系統草稿已儲存','success');}
    }catch(err){console.error(err);toast('儲存支持系統失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  async function open(eventId){
    const row=eventById(eventId);if(!row)return toast('找不到評估事件','error');
    if(row.status==='pending')return toast('請先開始本次評估','warn');
    if(!supportForm(eventId))return toast('本次評估未選用支持系統','warn');
    showLoading(true);
    try{await loadSupport(eventId);renderSupportPage();window.scrollTo({top:0,behavior:'auto'});}
    catch(err){console.error(err);toast('讀取支持系統失敗：'+(err?.message||err),'error');}
    finally{showLoading(false);}
  }

  window.v52AssessmentSupport={open};
})();
