  function domainChange(key){
    const now=domainStatus(key),prev=previousDomainStatus(key);
    if(!now||!prev||now===prev)return now&&prev&&['partial','insufficient'].includes(now)?{kind:'persist',label:'持續需留意'}:null;
    const rank={insufficient:1,partial:2,adequate:3};
    if(!(now in rank)||!(prev in rank))return null;
    return rank[now]>rank[prev]
      ?{kind:'better',label:'支持改善'}
      :{kind:'worse',label:'支持減弱'};
  }

  async function loadSupport(eventId){
    const row=eventById(eventId);
    if(!row)throw new Error('找不到評估事件');

    const [cur,hist]=await Promise.all([
      db.from('assessment_support_records').select('*').eq('assessment_event_id',eventId).maybeSingle(),
      db.from('assessment_support_records').select('*').eq('case_id',row.case_id).not('confirmed_at','is',null).neq('assessment_event_id',eventId).order('confirmed_at',{ascending:false})
    ]);
    if(cur.error)throw cur.error;
    if(hist.error)throw hist.error;

    local.eventId=eventId;
    local.record=cur.data||null;
    local.household={...safe(local.record?.household)};
    local.familyMembers=safeArray(local.record?.family_members).map((x,i)=>({
      id:x?.id||('family-'+i),
      name:x?.name||'',
      relation:x?.relation||'',
      co_resident:x?.co_resident||'',
      contact_frequency:x?.contact_frequency||'',
      support_role:x?.support_role||'',
      note:x?.note||''
    }));
    local.supportDomains={...safe(local.record?.support_domains)};
    local.resources={...safe(local.record?.resources)};
    local.summary={...safe(local.record?.summary)};
    local.expandedDomainNotes=new Set();

    const dated=(hist.data||[])
      .map(record=>({record,event:eventById(record.assessment_event_id)}))
      .filter(x=>x.event?.planned_date);

    local.hasLaterCompleted=dated.some(x=>x.event.planned_date>row.planned_date);
    const candidates=dated
      .filter(x=>x.event.planned_date<row.planned_date)
      .sort((a,b)=>String(b.event.planned_date).localeCompare(String(a.event.planned_date))||String(b.record.confirmed_at||'').localeCompare(String(a.record.confirmed_at||'')));

    local.previous=candidates[0]?.record||null;
    local.previousEvent=candidates[0]?.event||null;
    local.pendingCopyFromId=(local.record?.copied_from_id&&local.record.copied_from_id===local.previous?.id)?local.record.copied_from_id:null;
  }

  function renderDomain([key,label]){
    const cur=local.supportDomains[key]||{};
    const prev=local.previous?.support_domains?.[key]||{};
    const change=domainChange(key);
    const noteValue=String(cur.note||'');
    const noteRequired=['partial','insufficient','not_applicable'].includes(cur.status);
    const showNote=noteRequired||!!noteValue||local.expandedDomainNotes.has(key);
    const placeholder=cur.status==='partial'
      ?'請說明目前不足之處'
      :cur.status==='insufficient'
        ?'請說明明顯不足或需介入的情形'
        :cur.status==='not_applicable'
          ?'請說明不適用原因'
          :'補充備註（選填）';
    return `
      <div class="support-domain ${['partial','insufficient'].includes(cur.status)?'concern':''}">
        <div class="support-domain-head">
          <div class="support-domain-title">${escapeHTML(label)}</div>
          <div class="support-tags">
            ${prev.status?`<span class="support-tag prev">前次：${escapeHTML(statusLabel(prev.status))}</span>`:''}
            ${change?`<span class="support-tag ${change.kind}">${escapeHTML(change.label)}</span>`:''}
          </div>
        </div>
        <div class="support-options">
          ${DOMAIN_STATUS.map(([v,l])=>`
            <label class="support-option ${cur.status===v?'selected':''}">
              <input type="radio" name="support_domain_${key}" value="${v}" ${cur.status===v?'checked':''}>
              ${l}
            </label>
          `).join('')}
        </div>
        ${!showNote&&!['partial','insufficient','not_applicable'].includes(cur.status)?`<div class="support-note"><button type="button" class="btn sm" data-support-domain-add-note="${key}">＋ 補充備註</button></div>`:''}
        ${showNote?`
          <div class="support-note">
            <label>備註${noteRequired?' *':''}</label>
            <textarea data-support-domain-note="${key}" placeholder="${escapeHTML(placeholder)}">${escapeHTML(noteValue)}</textarea>
          </div>
        `:''}
      </div>
    `;
  }

  function renderOverall(){
    const s=local.summary;
    return `
      <section class="support-section">
        <div class="support-section-head"><strong>五、整體支持評估與後續</strong></div>
        <div class="support-grid">
          <div class="support-field">
            <label>整體支持狀況 *</label>
            <select data-support-summary="overall_support">
              <option value="">請選擇</option>
              <option value="adequate" ${s.overall_support==='adequate'?'selected':''}>支持充足</option>
              <option value="needs_attention" ${s.overall_support==='needs_attention'?'selected':''}>部分不足／需追蹤</option>
              <option value="weak" ${s.overall_support==='weak'?'selected':''}>支持薄弱／需介入</option>
            </select>
          </div>
          <div class="support-field">
            <label>是否需後續追蹤 *</label>
            <select data-support-summary="followup_required">
              <option value="">請選擇</option>
              <option value="yes" ${s.followup_required==='yes'?'selected':''}>是</option>
              <option value="no" ${s.followup_required==='no'?'selected':''}>否</option>
            </select>
          </div>
          ${['needs_attention','weak'].includes(s.overall_support)?`
            <div class="support-field full">
              <label>主要支持問題 *</label>
              <textarea data-support-summary="key_issues" placeholder="請說明目前最需要注意的支持缺口。">${escapeHTML(s.key_issues||'')}</textarea>
            </div>
          `:''}
          <div class="support-field full">
            <label>目前支持安排／建議</label>
            <textarea data-support-summary="support_plan" placeholder="例如：持續由案次子負責就醫接送，假日由案長子替代；必要時協助連結社福資源。">${escapeHTML(s.support_plan||'')}</textarea>
          </div>
          ${s.followup_required==='yes'?`
            <div class="support-field full">
              <label>後續追蹤重點 *</label>
              <textarea data-support-summary="followup_note" placeholder="例如：下次家訪確認主要照顧者工作時間變動後的替代照顧安排。">${escapeHTML(s.followup_note||'')}</textarea>
            </div>
          `:''}
        </div>
      </section>
    `;
  }

  function renderSupportPage(){
    const row=eventById(local.eventId);
    const c=row?caseById(row.case_id):null;
    const form=row?supportForm(row.id):null;
    if(!row||!c||!form){
      window.v52AssessmentApi?.renderAssessmentPage?.();
      return;
    }

    const canEdit=canEditEvent(row)&&row.status!=='voided';
    window.v52AssessmentApi?.setAssessmentFormHeader?.(row,c,'支持系統');

    $('content').innerHTML=`
      <div class="support-page">
        <div class="page-head">
          <div>
            <h1>${escapeHTML(c.case_name)}｜支持系統</h1>
            <p>評估日期：${row.planned_date?rocDate(row.planned_date+'T00:00:00'):'—'}　狀態：${escapeHTML(({pending:'待進行',in_progress:'評估中',forms_completed:'表單完成／待總結',completed:'已完成',voided:'作廢'})[row.status]||row.status)}</p>
          </div>
          <div class="assessment-head-actions"><button id="supportBack" type="button" class="btn">← 返回評估事件</button></div>
        </div>

        ${window.v52AssessmentApi?.renderAssessmentContextBar?.(row,c)||''}
        ${renderSummary()}
        ${renderCopyBox()}
        ${local.hasLaterCompleted?'<div class="support-notice">此筆評估日期早於個案較新的已完成支持系統紀錄；前次比較仍以本次日期以前最近一筆已完成紀錄為準。</div>':''}
        ${local.pendingCopyFromId&&!local.record?.confirmed_at?'<div class="support-notice">已帶入前次資料，請確認目前家庭、支持面向與資源是否仍相同。</div>':''}

        ${renderHousehold()}
        ${renderFamilyMembers()}
        ${renderDomains()}
        ${renderResources()}
        ${renderOverall()}

        ${canEdit?`
          <div class="support-actions">
            ${form.status!=='completed'?'<button id="supportSaveDraft" type="button" class="btn">儲存草稿</button>':''}
            <button id="supportConfirm" type="button" class="btn primary">${form.status==='completed'?'儲存修正並確認':'完成本次確認'}</button>
          </div>
        `:''}
      </div>
    `;

    bindEvents(row);
  }

  function rerenderPreservingScroll(){
    const y=window.scrollY;
    renderSupportPage();
    requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));
  }

  function syncFamilyMembers(){
    const items=[];
    document.querySelectorAll('[data-support-family]').forEach(rowEl=>{
      const item={id:rowEl.dataset.supportFamily||newId('family')};
      rowEl.querySelectorAll('[data-support-family-field]').forEach(el=>item[el.dataset.supportFamilyField]=el.value);
      if(Object.entries(item).some(([k,v])=>k!=='id'&&String(v||'').trim()!==''))items.push(item);
    });
    local.familyMembers=items;
  }

  function bindEvents(row){
    $('supportBack')?.addEventListener('click',()=>window.v52AssessmentApi?.renderAssessmentDetail?.(row.id));

    document.querySelectorAll('[data-support-household]').forEach(el=>{
      const key=el.dataset.supportHousehold;
      const rerenderKeys=['living_arrangement','primary_caregiver_status','backup_available'];
      el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{
        local.household[key]=el.value;
        if(rerenderKeys.includes(key))rerenderPreservingScroll();
      });
    });

    $('supportAddFamilyMember')?.addEventListener('click',()=>{
      syncFamilyMembers();
      local.familyMembers=[...local.familyMembers,{id:newId('family'),name:'',relation:'',co_resident:'',contact_frequency:'',support_role:'',note:''}];
      rerenderPreservingScroll();
    });

    document.querySelectorAll('[data-remove-support-family]').forEach(btn=>btn.addEventListener('click',()=>{
      syncFamilyMembers();
      local.familyMembers=local.familyMembers.filter(x=>x.id!==btn.dataset.removeSupportFamily);
      rerenderPreservingScroll();
    }));

    document.querySelectorAll('[data-support-family-field]').forEach(el=>{
      el.addEventListener(el.tagName==='SELECT'?'change':'input',syncFamilyMembers);
    });

    document.querySelectorAll('input[name^="support_domain_"]').forEach(input=>input.addEventListener('change',()=>{
      const key=input.name.replace('support_domain_','');
      local.supportDomains[key]={...(local.supportDomains[key]||{}),status:input.value};
      rerenderPreservingScroll();
    }));

    document.querySelectorAll('[data-support-domain-add-note]').forEach(btn=>btn.addEventListener('click',()=>{
      local.expandedDomainNotes.add(btn.dataset.supportDomainAddNote);
      rerenderPreservingScroll();
    }));

    document.querySelectorAll('[data-support-domain-note]').forEach(el=>el.addEventListener('input',()=>{
      const key=el.dataset.supportDomainNote;
      local.supportDomains[key]={...(local.supportDomains[key]||{}),note:el.value};
    }));

    document.querySelectorAll('[data-support-resource]').forEach(el=>{
      const key=el.dataset.supportResource;
      el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{
        local.resources[key]=el.value;
        if(key==='economic_status')rerenderPreservingScroll();
      });
    });

    document.querySelectorAll('[data-support-resource-type]').forEach(input=>input.addEventListener('change',()=>{
      local.resources.resource_types=[...document.querySelectorAll('[data-support-resource-type]:checked')].map(x=>x.dataset.supportResourceType);
    }));

    document.querySelectorAll('[data-support-summary]').forEach(el=>{
      const key=el.dataset.supportSummary;
      el.addEventListener(el.tagName==='SELECT'?'change':'input',()=>{
        local.summary[key]=el.value;
        if(['overall_support','followup_required'].includes(key))rerenderPreservingScroll();
      });
    });

    $('supportCopyPrevious')?.addEventListener('click',()=>{
      if(!local.previous)return;
      const hasCurrent=Object.keys(local.household||{}).length||local.familyMembers.length||Object.keys(local.supportDomains||{}).length||Object.keys(local.resources||{}).length||Object.keys(local.summary||{}).length;
      if(hasCurrent&&!confirm('帶入前次資料會覆蓋目前尚未儲存的支持系統內容，是否繼續？'))return;
      local.household={...safe(local.previous.household)};
      local.familyMembers=safeArray(local.previous.family_members).map(x=>({...safe(x),id:x?.id||newId('family')}));
      local.supportDomains=JSON.parse(JSON.stringify(safe(local.previous.support_domains)));
      local.resources={...safe(local.previous.resources),resource_types:[...safeArray(local.previous?.resources?.resource_types)]};
      local.summary={...safe(local.previous.summary)};
      local.pendingCopyFromId=local.previous.id;
      rerenderPreservingScroll();
      toast('已帶入前次支持系統資料，請確認本次狀況','success');
    });

    $('supportSaveDraft')?.addEventListener('click',()=>saveSupport(false));
    $('supportConfirm')?.addEventListener('click',()=>saveSupport(true));
  }

  function validateFinalize(){
    if(!local.household.living_arrangement)return '請選擇居住安排';
    if(!local.household.primary_caregiver_status)return '請選擇是否有主要照顧者';
    if(local.household.primary_caregiver_status==='present'){
      if(!String(local.household.primary_caregiver_name||'').trim())return '請填寫主要照顧者姓名／稱謂';
      if(!String(local.household.primary_caregiver_relation||'').trim())return '請填寫主要照顧者與個案關係';
    }

    const missingDomains=SUPPORT_DOMAINS.filter(([key])=>!domainStatus(key));
    if(missingDomains.length)return '尚有未完成支持面向：'+missingDomains.map(([,label])=>label).join('、');

    const missingNotes=SUPPORT_DOMAINS.filter(([key])=>
      ['partial','insufficient','not_applicable'].includes(domainStatus(key))
      && !String(local.supportDomains?.[key]?.note||'').trim()
    );
    if(missingNotes.length)return '以下支持面向請填寫說明：'+missingNotes.map(([,label])=>label).join('、');

    if(!local.resources.economic_status)return '請選擇經濟狀況';
    if(['watch','difficulty'].includes(local.resources.economic_status)&&!String(local.resources.economic_note||'').trim())return '請填寫經濟狀況說明';

    if(!local.summary.overall_support)return '請選擇整體支持狀況';
    if(['needs_attention','weak'].includes(local.summary.overall_support)&&!String(local.summary.key_issues||'').trim())return '請填寫主要支持問題';
    if(!local.summary.followup_required)return '請選擇是否需後續追蹤';
    if(local.summary.followup_required==='yes'&&!String(local.summary.followup_note||'').trim())return '請填寫後續追蹤重點';
    return null;
  }

  async function saveSupport(finalize){
    const row=eventById(local.eventId);
    if(!row||!canEditEvent(row))return toast('您沒有此個案的評估編輯權限','error');
    syncFamilyMembers();

    if(finalize){
      const problem=validateFinalize();
      if(problem)return toast(problem,'warn');
    }else{
      const hasData=Object.keys(local.household||{}).length||local.familyMembers.length||Object.keys(local.supportDomains||{}).length||Object.keys(local.resources||{}).length||Object.keys(local.summary||{}).length;
      if(!hasData)return toast('至少填寫一項後再儲存草稿','warn');
    }

    const y=window.scrollY;
    showLoading(true);
    try{
      const {error}=await db.rpc('save_assessment_support',{
        p_event_id:row.id,
        p_household:local.household,
        p_family_members:local.familyMembers,
        p_support_domains:local.supportDomains,
        p_resources:local.resources,
        p_summary:local.summary,
        p_finalize:!!finalize,
        p_copied_from_id:local.pendingCopyFromId||local.record?.copied_from_id||null
      });
      if(error)throw error;

      await window.v52AssessmentApi?.loadAssessmentData?.(false);
      if(finalize){
        window.v52AssessmentApi?.renderAssessmentDetail?.(row.id);
        toast('支持系統評估已完成','success');
      }else{
        await loadSupport(row.id);
        renderSupportPage();
        requestAnimationFrame(()=>window.scrollTo({top:y,behavior:'auto'}));
        toast('支持系統草稿已儲存','success');
      }
    }catch(err){
      console.error(err);
      toast('儲存支持系統失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  async function open(eventId){
    const row=eventById(eventId);
    if(!row)return toast('找不到評估事件','error');
    if(row.status==='pending')return toast('請先開始本次評估','warn');
    if(!supportForm(eventId))return toast('本次評估未選用支持系統','warn');

    showLoading(true);
    try{
      await loadSupport(eventId);
      renderSupportPage();
      window.scrollTo({top:0,behavior:'auto'});
    }catch(err){
      console.error(err);
      toast('讀取支持系統失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  window.v52AssessmentSupport={open};
})();
