  async function load(eventId,code,record){

      db.from(TOOL_TABLE[code]).select('*').eq('assessment_event_id',eventId).maybeSingle()

    if(summaryRes.error)throw summaryRes.error;
    if(toolRes.error)throw toolRes.error;

    const hash=sourceFingerprint(code,src);
    const prior=local.get(key(eventId,code));
    const keepDraft=!!prior?.edited;
    const state={
      eventId,code,savedSource:saved,source:src,pendingFields:keepDraft?new Set(prior.pendingFields):new Set(),
      findings:keepDraft?prior.findings:(data?.findings??''),
      recommendations:keepDraft?prior.recommendations:(data?.recommendations??''),
      sig:keepDraft?{...prior.sig}:(data?hashes(data.source_signature):{findings:'',recommendations:''}),
      generated:keepDraft?prior.generated:!!data?.has_generated,
      exists:!!data,edited:keepDraft,preview:false
    };
    local.set(key(eventId,code),state);return state;
  }

  function mount(eventId,code,recordOrProvider,rootSelector,actionSelector){
    const root=document.querySelector(rootSelector);
    if(!root)return;
    const s=local.get(key(eventId,code));
    if(!s)return;
    const provider=typeof recordOrProvider==='function'?recordOrProvider:()=>recordOrProvider;
    s.readSource=provider;
    const view=render(eventId,code,provider());
    if(!view)return;
    const action=root.querySelector(actionSelector);
    if(action)action.insertAdjacentHTML('beforebegin',view);
    else root.insertAdjacentHTML('beforeend',view);
    bind(eventId,code,provider());
  }
  function bind(eventId,code,record){
    const root=document.querySelector('[data-ats-root="'+code+'"]');if(!root)return;
    const s=get(eventId,code,record);if(!s)return;
    const refresh=()=>{
      root.outerHTML=render(eventId,code,s.source);
      bind(eventId,code,s.source);
    };
    root.querySelectorAll('[data-ats-input]').forEach(el=>el.addEventListener('input',()=>{
      s[el.dataset.atsInput]=el.value;s.edited=true;s.generated=true;
    }));
    root.querySelector('[data-ats-preview]')?.addEventListener('click',()=>{
      s.source=source(code,typeof s.readSource==='function'?s.readSource():s.source);
      if(!s.generated&&!s.edited){
        const suggested=generate(code,s.source);
        s.findings=suggested.findings;s.recommendations=suggested.recommendations;
        const h=sourceFingerprint(code,s.source);
        s.sig={findings:h,recommendations:h};
        s.pendingFields.add('findings');s.pendingFields.add('recommendations');
        s.generated=true;s.edited=true;s.preview=false;
        refresh();toast('已生成本項總結，可自由修改，請記得儲存','success');
      }else{
        s.preview=true;refresh();
        document.querySelector('[data-ats-root="'+code+'"] .ats-preview-modal')?.focus();
      }
    });
    root.querySelector('[data-ats-cancel]')?.addEventListener('click',()=>{s.preview=false;refresh();});
    root.querySelector('[data-ats-confirm]')?.addEventListener('click',()=>{
      s.source=source(code,typeof s.readSource==='function'?s.readSource():s.source);
      const suggested=generate(code,s.source),h=sourceFingerprint(code,s.source);

      const selected=[...root.querySelectorAll('[data-ats-replace]:checked')];
      if(!selected.length)return toast('請先勾選要取代的欄位','warn');
      selected.forEach(el=>{
        const f=el.dataset.atsReplace;s[f]=suggested[f];s.sig[f]=h;s.pendingFields.add(f);
      });
      s.generated=true;s.edited=true;s.preview=false;refresh();
      toast('已更新選取的總結欄位，請儲存本項評估','success');
    });
    root.querySelector('[data-ats-ack]')?.addEventListener('click',async()=>{
      // 只確認已成功儲存的來源，避免未儲存的修改被誤認為已確認。
      const h=sourceFingerprint(code,s.savedSource);
      s.sig={findings:h,recommendations:h};s.preview=false;s.generated=true;
      try{await persist(s,(state.assessmentEvents||[]).find(x=>x.id===eventId)?.case_id);refresh();
        toast('已確認保留目前總結','success');
      }catch(e){toast('確認失敗：'+(e?.message||e),'error');}
    });
  }
  async function persist(s,caseId){
    if(!s||!caseId)throw new Error('缺少評估事件或個案資料');
    const payload={assessment_event_id:s.eventId,tool_code:s.code,case_id:caseId,
      findings:s.findings,recommendations:s.recommendations,source_signature:JSON.stringify(s.sig),has_generated:!!s.generated,
      updated_at:new Date().toISOString()};
    const {error}=await db.from(TABLE).upsert(payload,{onConflict:'assessment_event_id,tool_code'});
    if(error)throw error;s.exists=true;s.edited=false;
  }
  async function saveAfterTool(eventId,code,record,caseId){
    const s=get(eventId,code,record);
    if(!s)throw new Error('本項評估總結尚未載入，請重新開啟表單');
    const {data,error}=await db.from(TOOL_TABLE[code]).select('*').eq('assessment_event_id',eventId).maybeSingle();
    if(error)throw error;
    if(!data)throw new Error('原始評估紀錄尚未成功儲存');
    const saved=source(code,data),h=sourceFingerprint(code,saved);
    // 未按生成的表單僅儲存空白單項總結，不主動產生文字。
    if(!s.exists&&!s.generated)s.sig={findings:'',recommendations:''};
    for(const field of s.pendingFields){s.sig[field]=h;}
    await persist(s,caseId);
    s.pendingFields.clear();
    s.savedSource=saved;
  }
  async function trackOverall(eventId,caseId,findings,recommendations,signature){
    const payload={assessment_event_id:eventId,case_id:caseId,tool_code:'overall',
      findings:findings||'',recommendations:recommendations||'',
      source_signature:JSON.stringify(signature),has_generated:true};
    const {error}=await db.from(TABLE).upsert(payload,{onConflict:'assessment_event_id,tool_code'});
    if(error)throw error;
  }
  function merged(rows,forms){
    const byCode=new Map(arr(rows).map(x=>[x.tool_code,x]));
    const parts={findings:[],recommendations:[]};
    const missing=[];
    for(const form of arr(forms)){
      if(!['completed','unable','not_applicable'].includes(form.status))continue;
      const code=form.form_code,row=byCode.get(code);
      if(!row||!row.has_generated){missing.push(TITLES[code]||form.form_name);continue;}
      for(const f of ['findings','recommendations']){
        if(value(row[f]))parts[f].push('【'+(TITLES[code]||form.form_name)+'】\n'+value(row[f]));
      }
    }
    return {findings:parts.findings.join('\n\n'),recommendations:parts.recommendations.join('\n\n'),missing};
  }
