    const n=Math.max(0,Number(minutes)||0), h=Math.floor(n/60), m=n%60;

  function alWorkHoursOn(staffId,date){
    const rows=(state.annualLeaveWorkHours||[])
      .filter(x=>String(x.staff_id)===String(staffId)&&String(x.effective_from)<=String(date))
      .sort((a,b)=>String(b.effective_from).localeCompare(String(a.effective_from))||String(b.created_at||'').localeCompare(String(a.created_at||'')));
    const fromTimeline=Number(rows[0]?.agreed_daily_hours);
    if(Number.isFinite(fromTimeline)&&fromTimeline>0)return fromTimeline;
    const staff=(state.annualLeaveStaff||[]).find(x=>String(x.id)===String(staffId));
    const current=Number(staff?.agreed_daily_hours);
    if(Number.isFinite(current)&&current>0)return current;
    const credit=(state.annualLeaveCredits||[]).find(x=>String(x.staff_id)===String(staffId)&&String(x.period_start)<=String(date)&&String(x.period_end)>=String(date));
    const snapshot=Number(credit?.daily_hours_snapshot);
    return Number.isFinite(snapshot)&&snapshot>0?snapshot:null;
  }

  function alTimeToMinutes(value){
    if(!/^([01]\d|2[0-3]):[0-5]\d$/.test(String(value||'')))return null;
    const [h,m]=String(value).split(':').map(Number);
    return h*60+m;
  }
  function alChargeMinutesFor(start,end,date,staffId=state.profile?.id){
    const row=alWorkScheduleOn(staffId,date);
    const s=alTimeToMinutes(start),e=alTimeToMinutes(end);
    if(!row||s===null||e===null||e<=s)return null;
    const ws=alTimeToMinutes(String(row.work_start_time).slice(0,5));
    const we=alTimeToMinutes(String(row.work_end_time).slice(0,5));
    if(ws===null||we===null)return null;
    const overlap=(a,b,c,d)=>Math.max(0,Math.min(b,d)-Math.max(a,c));
    if(row.break_start_time&&row.break_end_time){
      const bs=alTimeToMinutes(String(row.break_start_time).slice(0,5));
      const be=alTimeToMinutes(String(row.break_end_time).slice(0,5));
      if(bs===null||be===null)return null;
      return overlap(s,e,ws,bs)+overlap(s,e,be,we);
    }
    return overlap(s,e,ws,we);
  }
  function alLeaveDaysFor(minutes,date,staffId=state.profile?.id,start=null,end=null){
    const hours=alWorkHoursOn(staffId,date);
    if(!hours)return 0;
    const charge=start&&end?alChargeMinutesFor(start,end,date,staffId):Math.max(0,Number(minutes)||0);
    if(charge===null)return 0;
    return Math.round(Math.min(1,charge/(hours*60))*1000000)/1000000;
  }

  function alMinutesBetween(start,end){
    if(!/^([01]\d|2[0-3]):[0-5]\d$/.test(start||'') || !/^([01]\d|2[0-3]):[0-5]\d$/.test(end||'')) return 0;
    const [sh,sm]=start.split(':').map(Number), [eh,em]=end.split(':').map(Number);
    return Math.max(0,(eh*60+em)-(sh*60+sm));
  }

  function alMonthRange(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    const [y,m]=safe.split('-').map(Number);

  function alBalance(staffId,date=isoDate(new Date())){
    const credits=(state.annualLeaveCredits||[]).filter(c=>String(c.staff_id)===String(staffId) && String(c.period_start)<=date && String(c.period_end)>=date && c.settlement_status!=='settled' && c.settlement_status!=='carried_over');
    const granted=credits.reduce((s,c)=>s+(Number(c.granted_days)||Number(c.entitlement_days)||0),0);
    const remaining=credits.reduce((s,c)=>s+(Number(c.remaining_days)||0),0);
    const pending=(state.annualLeaveRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId) && r.status==='pending' && credits.some(c=>String(r.leave_date)>=String(c.period_start)&&String(r.leave_date)<=String(c.period_end)))
      .reduce((s,r)=>s+(Number(r.leave_days)||alLeaveDaysFor(r.minutes,r.leave_date,staffId,r.start_time,r.end_time)||0),0);
    return {granted,used:Math.max(0,granted-remaining),remaining,pending,available:Math.max(0,remaining-pending)};
  }

  function alUsageRows(creditId=null){
    const today=isoDate(new Date());
    let credits=[];
    if(creditId){
      const credit=(state.annualLeaveCredits||[]).find(c=>String(c.id)===String(creditId));
      if(credit)credits=[credit];
    }else{
      credits=alCurrentCredits(state.profile?.id,today);
    }
    const creditMap=new Map(credits.map(c=>[String(c.id),c]));
    const requestMap=new Map((state.annualLeaveRows||[]).map(r=>[String(r.id),r]));
    return (state.annualLeaveAllocations||[])
      .filter(a=>a.is_active===true && creditMap.has(String(a.credit_id)))
      .map(a=>({
        allocation:a,
        credit:creditMap.get(String(a.credit_id)),
        request:requestMap.get(String(a.request_id))||null
      }))
      .sort((a,b)=>{
        const ad=String(a.request?.leave_date||a.allocation?.allocation_date||'');
        const bd=String(b.request?.leave_date||b.allocation?.allocation_date||'');
        return bd.localeCompare(ad)||new Date(b.allocation?.created_at||0)-new Date(a.allocation?.created_at||0);
      });
  }

  async function loadAnnualLeaveData(){
    if(!state.profile) return;
    showLoading(true);
    try{
      const staffQuery=alCanManage()
        ? db.from('staff_users').select('id,display_name,role,hire_date,is_active,agreed_daily_hours').eq('is_active',true).order('display_name',{ascending:true})
        : db.from('staff_users').select('id,display_name,role,hire_date,is_active,agreed_daily_hours').eq('id',state.profile.id).limit(1);
      const tasks=[
        db.from('annual_leave_requests').select('id,applicant_staff_id,applicant_name,applicant_role,leave_date,start_time,end_time,minutes,charge_minutes,work_hours_snapshot,leave_days,reason,notes,status,review_note,submitted_at,reviewed_at,created_at,updated_at').order('leave_date',{ascending:false}).order('created_at',{ascending:false}),
        db.from('annual_leave_credits').select('id,staff_id,period_start,period_end,granted_days,remaining_days,granted_minutes,remaining_minutes,label,note,created_by_name,entitlement_days,daily_hours_snapshot,generation_source,settlement_status,created_at,updated_at').order('period_end',{ascending:true}).order('created_at',{ascending:true}),
        db.from('annual_leave_allocations').select('id,credit_id,request_id,allocation_date,minutes,days,is_active,reversed_at,created_at').order('created_at',{ascending:false}),
        db.from('leave_cancellation_requests').select('id,source_type,source_id,applicant_staff_id,applicant_name,leave_date,minutes,reason,status,requested_at,reviewed_at,review_note,created_at').eq('source_type','annual_leave').order('requested_at',{ascending:false}),
        db.from('staff_work_hours_history').select('id,staff_id,effective_from,agreed_daily_hours,work_start_time,work_end_time,break_start_time,break_end_time,note,created_at').order('effective_from',{ascending:false}).order('created_at',{ascending:false}),
        staffQuery
      ];
      const results=await Promise.all(tasks);
      for(const r of results) if(r.error) throw r.error;
      state.annualLeaveRows=results[0].data||[];
      state.annualLeaveCredits=results[1].data||[];
      state.annualLeaveAllocations=results[2].data||[];
      state.annualLeaveCancellationRows=results[3].data||[];
      state.annualLeaveWorkHours=results[4].data||[];
      state.annualLeaveStaff=results[5].data||[];
      state.annualLeaveLoaded=true;
    }catch(err){
      console.error('讀取特休資料失敗',err);
      state.annualLeaveLoaded=false;
      toast('讀取特休資料失敗：'+(err?.message||err),'error');
    }finally{ showLoading(false); }
  }

  function alCancellationNote(r){

    if(!cancel)return '';
    if(cancel.status==='pending')return `<div class="al-detail">銷假原因：${escapeHTML(cancel.reason||'—')}</div>`;
    if(cancel.status==='rejected')return `<div class="al-detail" style="color:#b42318">銷假退回：${escapeHTML(cancel.review_note||'未填寫原因')}</div>`;
    if(cancel.status==='approved')return `<div class="al-detail">銷假已核准：${escapeHTML(cancel.reason||'—')}</div>`;
    return '';
  }

  function alCanRequestCancellation(r){
    if(!alCanSubmit()||String(r.applicant_staff_id)!==String(state.profile?.id)||r.status!=='approved')return false;
    const cancel=alCancellationFor(r.id);
    if(cancel&&['pending','approved'].includes(cancel.status))return false;
    return String(r.leave_date||'')>isoDate(new Date());
  }
