// Attendance V2 lifecycle core
(() => {
  if(window.AttendanceV2Lifecycle) return;

  const hooks=new Map();

  function register(event,key,fn){
    if(!event||!key||typeof fn!=='function') return;
    if(!hooks.has(event)) hooks.set(event,new Map());
    hooks.get(event).set(key,fn);
  }

  function unregister(event,key){
    hooks.get(event)?.delete(key);
  }

  function run(event,context={}){
    const list=[...(hooks.get(event)?.entries()||[])];
    for(const [key,fn] of list){
      try{
        const out=fn(context);
        if(out&&typeof out.then==='function'){
          out.catch(err=>console.error(`AttendanceV2 hook failed: ${event}/${key}`,err));
        }
      }catch(err){
        console.error(`AttendanceV2 hook failed: ${event}/${key}`,err);
      }
    }
  }

  function runSoon(event,context={}){
    queueMicrotask(()=>run(event,context));
  }

  window.AttendanceV2Lifecycle={register,unregister,run,runSoon};

  if(!document.getElementById('attendance-v2-shared-style')){
    const style=document.createElement('style');
    style.id='attendance-v2-shared-style';
    style.textContent='.attendance-expiry-reminder{margin:-2px 0 12px;border:1px solid #eaded9;background:#fffaf8;color:#667085;border-radius:10px;padding:9px 12px;font-size:11.5px;line-height:1.55}.attendance-expiry-reminder strong{color:#344054}.attendance-expiry-reminder .muted{color:#98a2b3}';
    document.head.appendChild(style);
  }
})();
