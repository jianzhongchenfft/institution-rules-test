  function overtimeMonthRange(month){
    const safe = /^\d{4}-\d{2}$/.test(month || '') ? month : isoDate(new Date()).slice(0,7);
    const [year,mon] = safe.split('-').map(Number);
    const start = `${year}-${String(mon).padStart(2,'0')}-01`;
    const next = new Date(year, mon, 1);
    const end = isoDate(new Date(next.getFullYear(), next.getMonth(), 0));
    return {start,end};
  }

  function overtimeNormalizeTime(value){
    const raw=String(value||'').trim().replace(/：/g,':');
    const match=raw.match(/^(\d{1,2}):(\d{1,2})$/);
    if(!match) return raw;
    const h=Number(match[1]),m=Number(match[2]);
    if(h>23 || m>59) return raw;
    return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`;
  }

  function overtimeValid24h(value){
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(String(value||''));
  }

  function overtimeMinutesBetween(start,end){
    const s=overtimeNormalizeTime(start),e=overtimeNormalizeTime(end);
    if(!overtimeValid24h(s) || !overtimeValid24h(e)) return 0;
    const toMin = value => {
      const [h,m] = String(value).split(':').map(Number);
      return h*60 + m;
    };
    const startMin=toMin(s),endMin=toMin(e);
    if(endMin===0 && startMin>0) return 1440-startMin;
    return endMin>startMin ? endMin-startMin : 0;
  }

  function overtimeCrossesMidnight(start,end){
    const s=overtimeNormalizeTime(start),e=overtimeNormalizeTime(end);
    if(!overtimeValid24h(s) || !overtimeValid24h(e)) return false;
    const [sh,sm]=s.split(':').map(Number),[eh,em]=e.split(':').map(Number);
    const startMin=sh*60+sm,endMin=eh*60+em;
    return endMin<startMin && endMin!==0;
  }

  function overtimeTimeParts(value){
    const normalized=overtimeNormalizeTime(value);
    if(!overtimeValid24h(normalized)) return {hour:'',minute:''};
    const [hour,minute]=normalized.split(':');
    return {hour,minute};
  }

  function compLeaveBalance(staffId,asOfDate=null){
    const fallback=isoDate(new Date());
    const domDate=document.getElementById('clDate')?.value||'';
    const rawDate=String(asOfDate||domDate||fallback);
    const asOf=/^\d{4}-\d{2}-\d{2}$/.test(rawDate)?rawDate:fallback;
    const credits=(state.compLeaveCredits||[]).filter(c=>String(c.staff_id)===String(staffId));
    const eligible=credits.filter(c=>String(c.earned_date||'')<=asOf&&(!c.expires_on||String(c.expires_on)>=asOf));
    const earned=eligible.reduce((sum,c)=>sum+(Number(c.earned_minutes)||0),0);
    const ledgerRemaining=eligible.reduce((sum,c)=>sum+(Number(c.remaining_minutes)||0),0);
    const used=compLeaveApprovedUsedMinutes(staffId);
    const pending=compLeavePendingMinutes(staffId);
    const available=Math.max(0,ledgerRemaining-pending);
    return {
      earned,
      used,
      pending,
      ledgerRemaining,
      remaining:available,
      requestable:available
    };
  }

  function compLeaveIsHolidayVisitCredit(c){
    return String(c?.source_label||'').includes('假日家訪');
  }

  function compLeaveSourceBreakdown(staffId,asOfDate=null){
    const fallback=isoDate(new Date());
    const rawDate=String(asOfDate||fallback);
    const asOf=/^\d{4}-\d{2}-\d{2}$/.test(rawDate)?rawDate:fallback;
    const credits=(state.compLeaveCredits||[])
      .filter(x=>String(x.staff_id)===String(staffId))
      .filter(x=>String(x.earned_date||'')<=asOf&&(!x.expires_on||String(x.expires_on)>=asOf));
    const ordinary=credits.filter(x=>!compLeaveIsHolidayVisitCredit(x)).reduce((s,x)=>s+(Number(x.remaining_minutes)||0),0);
    const holidayVisit=credits.filter(compLeaveIsHolidayVisitCredit).reduce((s,x)=>s+(Number(x.remaining_minutes)||0),0);
    return {ordinary,holidayVisit,total:ordinary+holidayVisit};
  }

  function compLeaveSourceAvailability(staffId,asOfDate=null,excludeUsageId=null){
    const base=compLeaveSourceBreakdown(staffId,asOfDate);
    const pending=(state.compLeaveRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='pending')
      .filter(r=>!excludeUsageId||String(r.id)!==String(excludeUsageId))
      .reduce((sum,r)=>sum+(Number(r.minutes)||0),0);
    let ordinary=base.ordinary;
    let holidayVisit=base.holidayVisit;
    let reserve=pending;
    const fromOrdinary=Math.min(ordinary,reserve);
    ordinary-=fromOrdinary;
    reserve-=fromOrdinary;
    const fromHoliday=Math.min(holidayVisit,reserve);
    holidayVisit-=fromHoliday;
    return {ordinary:Math.max(0,ordinary),holidayVisit:Math.max(0,holidayVisit),total:Math.max(0,ordinary+holidayVisit),pending};
  }

  function compLeavePlannedAllocation(staffId,asOfDate,minutes,excludeUsageId=null){
    const available=compLeaveSourceAvailability(staffId,asOfDate,excludeUsageId);
    let need=Math.max(0,Number(minutes)||0);
    const ordinary=Math.min(need,available.ordinary);
    need-=ordinary;
    const holidayVisit=Math.min(need,available.holidayVisit);
    need-=holidayVisit;
    return {ordinary,holidayVisit,shortage:Math.max(0,need),total:Math.max(0,Number(minutes)||0),available};
  }

  async function loadOvertimeRows(){
    if(!state.profile) return;
    showLoading(true);
    try{
      const [overtimeResult,leaveResult,settlementResult,creditResult,cancellationResult] = await Promise.all([
        db.from('overtime_requests')
          .select('id,applicant_staff_id,applicant_name,applicant_role,work_date,request_type,start_time,end_time,overtime_minutes,compensation_method,reason,holiday_case_count,comp_minutes,notes,status,review_note,submitted_at,reviewed_at,regular_day_weekday_snapshot,regular_day_warning,created_at,updated_at')
          .order('work_date',{ascending:false})
          .order('created_at',{ascending:false}),
        db.from('comp_leave_usages')
          .select('id,applicant_staff_id,applicant_name,applicant_role,leave_date,start_time,end_time,minutes,reason,notes,status,review_note,submitted_at,reviewed_at,created_at,updated_at')
          .order('leave_date',{ascending:false})
          .order('created_at',{ascending:false}),
        db.from('payroll_month_settlements')
          .select('id,settlement_month,summary,total_overtime_pay_minutes,total_comp_leave_earned_minutes,total_comp_leave_used_minutes,settled_by,settled_by_name,settled_at,created_at')
          .order('settlement_month',{ascending:false}),
        db.from('comp_leave_credits')
          .select('id,staff_id,source_type,source_id,earned_date,earned_minutes,remaining_minutes,expires_on,source_label,created_at,updated_at')
          .order('earned_date',{ascending:true})
          .order('created_at',{ascending:true}),
        db.from('leave_cancellation_requests')
          .select('id,source_type,source_id,applicant_staff_id,applicant_name,leave_date,minutes,reason,status,requested_at,reviewed_at,review_note,created_at')
          .eq('source_type','comp_leave')
          .order('requested_at',{ascending:false})
      ]);
      if(overtimeResult.error) throw overtimeResult.error;
      if(leaveResult.error) throw leaveResult.error;
      if(settlementResult.error) throw settlementResult.error;
      if(creditResult.error) throw creditResult.error;
      if(cancellationResult.error) throw cancellationResult.error;
      state.overtimeRows = overtimeResult.data || [];
      state.compLeaveRows = leaveResult.data || [];
      state.payrollSettlements = settlementResult.data || [];
      state.compLeaveCredits = creditResult.data || [];
      state.leaveCancellationRows = cancellationResult.data || [];

  }

  function renderCompLeaveOverview(){
    const staffId=state.profile?.id;
    const today=isoDate(new Date());
    const allCredits=(state.compLeaveCredits||[])
      .filter(c=>String(c.staff_id)===String(staffId))
      .filter(c=>(Number(c.remaining_minutes)||0)>0);

    const usable=allCredits
      .filter(c=>String(c.earned_date||'')<=today)
      .filter(c=>!c.expires_on||String(c.expires_on)>=today)
      .sort((a,b)=>{
        const aExpiry=compLeaveIsHolidayVisitCredit(a)?'9999-12-31':String(a.expires_on||'9999-12-31');
        const bExpiry=compLeaveIsHolidayVisitCredit(b)?'9999-12-31':String(b.expires_on||'9999-12-31');
        return aExpiry.localeCompare(bExpiry)
          ||String(a.earned_date||'').localeCompare(String(b.earned_date||''))
          ||String(a.created_at||'').localeCompare(String(b.created_at||''));
      });

    const expiredMinutes=allCredits
      .filter(c=>!compLeaveIsHolidayVisitCredit(c)&&c.expires_on&&String(c.expires_on)<today)
      .reduce((sum,c)=>sum+(Number(c.remaining_minutes)||0),0);

    const soonMinutes=usable
      .filter(c=>{
        if(compLeaveIsHolidayVisitCredit(c)||!c.expires_on)return false;
        const d=Math.round((new Date(String(c.expires_on)+'T00:00:00')-new Date(today+'T00:00:00'))/86400000);
        return d>=0&&d<=30;
      })
      .reduce((sum,c)=>sum+(Number(c.remaining_minutes)||0),0);

    const missingExpiry=allCredits.some(c=>
      !compLeaveIsHolidayVisitCredit(c)
      && String(c.earned_date||'')<=today
      && !c.expires_on
    );

    return `<section class="ot-rule-card">
      <h3>換休期限與剩餘明細</h3>
      <p>僅列出目前仍可使用的換休額度；一般加班換休依到期日優先使用，假日家訪換休無到期限制。</p>
      ${missingExpiry?'<div class="ot-rule-highlight">部分一般加班換休尚未產生到期日，請先確認人員管理中的到職日資料。</div>':''}
      ${soonMinutes>0?`<div class="ot-rule-highlight">未來30天內有 <strong>${escapeHTML(overtimeDurationLabel(soonMinutes))}</strong> 一般加班換休即將到期，建議優先安排。</div>`:''}
      ${expiredMinutes>0?`<div class="ot-rule-highlight">已有 <strong>${escapeHTML(overtimeDurationLabel(expiredMinutes))}</strong> 一般加班換休到期未休，不再列入可用額度，請於薪資作業中辦理結清。</div>`:''}
      <div style="margin-top:12px">
        ${usable.length
          ? `<div class="ot-table-shell"><table class="ot-table"><thead><tr><th>取得日期</th><th>來源</th><th>原始額度</th><th>剩餘</th><th>到期日</th><th>狀態</th></tr></thead><tbody>${usable.map(c=>{
              const holiday=compLeaveIsHolidayVisitCredit(c);
              const expiryText=holiday?'無到期限制':(c.expires_on?rocDate(String(c.expires_on)+'T00:00:00'):'待設定');
              let status='有效';
              if(holiday)status='無到期限制';
              else if(c.expires_on){
                const d=Math.round((new Date(String(c.expires_on)+'T00:00:00')-new Date(today+'T00:00:00'))/86400000);
                if(d===0)status='今天到期';
                else if(d<=30)status=`${d}天內到期`;
              }
              return `<tr>
                <td>${rocDate(String(c.earned_date)+'T00:00:00')}</td>
                <td>${escapeHTML(c.source_label||'換休')}</td>
                <td>${escapeHTML(overtimeDurationLabel(c.earned_minutes))}</td>
                <td><strong>${escapeHTML(overtimeDurationLabel(c.remaining_minutes))}</strong></td>
                <td>${escapeHTML(expiryText)}</td>
                <td>${escapeHTML(status)}</td>
              </tr>`;
            }).join('')}</tbody></table></div>`
          : '<div class="ot-empty"><div class="icon">🕒</div><h3>目前沒有可用換休額度</h3></div>'}
      </div>
    </section>`;
  }

  function payrollMonthLabel(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    const [year,mon]=safe.split('-').map(Number);
    return `民國 ${year-1911} 年 ${mon} 月`;
  }

  function payrollBuildRows(){
    const {start,end}=overtimeMonthRange(state.overtimeMonth);
    const approvedOvertime=state.overtimeRows.filter(r=>r.status==='approved' && r.work_date<=end);
    const approvedLeave=state.compLeaveRows.filter(r=>r.status==='approved' && r.leave_date<=end);
    const staffIds=new Set([
      ...approvedOvertime.map(r=>r.applicant_staff_id),
      ...approvedLeave.map(r=>r.applicant_staff_id)
    ]);
    const rows=[];

    staffIds.forEach(staffId=>{
      const monthOvertime=approvedOvertime.filter(r=>r.applicant_staff_id===staffId && r.work_date>=start && r.work_date<=end);
      const monthLeave=approvedLeave.filter(r=>r.applicant_staff_id===staffId && r.leave_date>=start && r.leave_date<=end);
      const overtimePayItems=monthOvertime
        .filter(r=>r.compensation_method==='overtime_pay')
        .map(r=>({
          id:r.id,
          date:r.work_date,
          type:r.request_type,
          startTime:r.start_time?String(r.start_time).slice(0,5):null,
          endTime:r.end_time?String(r.end_time).slice(0,5):null,
          minutes:Number(r.overtime_minutes)||0,
          reason:r.reason||''
        }));
      const compLeaveEarnedItems=monthOvertime
        .filter(r=>r.compensation_method==='comp_leave')
        .map(r=>({
          id:r.id,
          date:r.work_date,
          type:r.request_type,
          startTime:r.start_time?String(r.start_time).slice(0,5):null,
          endTime:r.end_time?String(r.end_time).slice(0,5):null,
          minutes:r.request_type==='holiday_visit_comp'?(Number(r.comp_minutes)||0):(Number(r.overtime_minutes)||0),
          reason:r.reason||''
        }));
      const compLeaveUsedItems=monthLeave.map(r=>({
        id:r.id,
        date:r.leave_date,
        startTime:String(r.start_time||'').slice(0,5),
        endTime:String(r.end_time||'').slice(0,5),
        minutes:Number(r.minutes)||0,
        reason:r.reason||''
      }));

      const overtimePayMinutes=overtimePayItems.reduce((sum,r)=>sum+r.minutes,0);
      const compLeaveEarnedMinutes=compLeaveEarnedItems.reduce((sum,r)=>sum+r.minutes,0);
      const compLeaveUsedMinutes=compLeaveUsedItems.reduce((sum,r)=>sum+r.minutes,0);

      if(overtimePayMinutes===0 && compLeaveEarnedMinutes===0 && compLeaveUsedMinutes===0) return;

      const meta=state.overtimeRows.find(r=>r.applicant_staff_id===staffId)
        || state.compLeaveRows.find(r=>r.applicant_staff_id===staffId)
        || {};
      rows.push({
        staffId,
        name:meta.applicant_name||'未命名',
        role:meta.applicant_role||'',
        overtimePayMinutes,
        compLeaveEarnedMinutes,
        compLeaveUsedMinutes,
        overtimePayItems,
        compLeaveEarnedItems,
        compLeaveUsedItems
      });
    });

    return rows.sort((a,b)=>String(a.name).localeCompare(String(b.name),'zh-Hant'));
  }

  function payrollReportRows(){
    const settlement=payrollSettlementForMonth();
    const snapshotRows=settlement?.summary?.staff;
    return settlement && Array.isArray(snapshotRows) ? snapshotRows : payrollBuildRows();
  }

  function payrollTotals(rows,settlement=null){
    if(settlement){
      return {
        overtimePayMinutes:Number(settlement.total_overtime_pay_minutes)||0,
        compLeaveEarnedMinutes:Number(settlement.total_comp_leave_earned_minutes)||0,
        compLeaveUsedMinutes:Number(settlement.total_comp_leave_used_minutes)||0
      };
    }
    return rows.reduce((acc,r)=>{
      acc.overtimePayMinutes+=Number(r.overtimePayMinutes)||0;
      acc.compLeaveEarnedMinutes+=Number(r.compLeaveEarnedMinutes)||0;
      acc.compLeaveUsedMinutes+=Number(r.compLeaveUsedMinutes)||0;
      return acc;
    },{overtimePayMinutes:0,compLeaveEarnedMinutes:0,compLeaveUsedMinutes:0});
  }

  async function closePayrollMonth(){
    if(!overtimeCanReview()){toast('目前帳號沒有薪資結算權限','warn');return;}
    if(payrollSettlementForMonth()){toast('本月份已完成結算','warn');return;}
    const {start,end}=overtimeMonthRange(state.overtimeMonth);
    const pendingOvertime=state.overtimeRows.filter(r=>r.work_date>=start && r.work_date<=end && r.status==='pending').length;
    const pendingLeave=state.compLeaveRows.filter(r=>r.leave_date>=start && r.leave_date<=end && r.status==='pending').length;
    if(pendingOvertime+pendingLeave>0){
      toast(`本月尚有 ${pendingOvertime+pendingLeave} 筆待審核申請，請先完成審核再結算`,'warn');
      return;
    }

    const rows=payrollBuildRows();
    if(!rows.length){toast('本月尚無可結算資料','warn');return;}
    const totals=payrollTotals(rows);
    if(!confirm(`確定將「${payrollMonthLabel(state.overtimeMonth)}」標記為已結算嗎？\n\n結算後會保存目前報表快照並鎖定，不再因後續補登資料而改變本次結算結果。`)) return;

    const payload={
      settlement_month:start,
      summary:{
        version:1,
        month:state.overtimeMonth,
        generatedAt:new Date().toISOString(),
        staff:rows,
        totals
      },
      total_overtime_pay_minutes:totals.overtimePayMinutes,
      total_comp_leave_earned_minutes:totals.compLeaveEarnedMinutes,
      total_comp_leave_used_minutes:totals.compLeaveUsedMinutes,
      settled_by:state.user.id,
      settled_by_name:state.profile.display_name
    };

    showLoading(true);
    try{
      const {error}=await db.from('payroll_month_settlements').insert(payload);
      if(error) throw error;
      await loadOvertimeRows();
      state.overtimeView='payroll';
      renderOvertimePage();
      toast('本月薪資結算已完成並鎖定','success');
    }catch(err){
      console.error(err);
      const message=String(err?.message||err);
      toast(message.includes('duplicate')?'本月份已完成結算':'薪資結算失敗：'+message,'error');
    }finally{showLoading(false);}
  }

  function openOvertimeModal(existing=null){

    state.workflowEditingOvertimeId=existing?.id||null;
    const title=existing?'編輯申請':'新增申請';

    const updateFields=()=>{
      const selected=!!typeEl?.value;

      $('otTimeFields')?.classList.toggle('hidden',!selected || holiday);

  function updateOvertimeCalc(){
    const type=$('otType')?.value;

  async function saveOvertime(existing,status){
    const type=$('otType').value;
    const workDate=$('otWorkDate').value;
    const reason=$('otReason').value.trim();
    const notes=$('otNotes').value.trim();
    const start=overtimeSelectedTime('otStart');
    const end=overtimeSelectedTime('otEnd');
    const timeBased=['overtime','holiday_overtime'].includes(type);
    const compensationMethod=type==='holiday_visit_comp'?'comp_leave':$('otCompensation').value;
    const overtimeMinutes=timeBased?overtimeMinutesBetween(start,end):0;
    const caseCount=type==='holiday_visit_comp'?Math.max(0,Number($('otCaseCount')?.value)||0):null;
    const compMinutes=type==='holiday_visit_comp'?caseCount*120:0;

    if(!type){toast('請選擇申請類型','warn');return false;}
    if(!workDate){toast('請選擇日期','warn');return false;}
    if(!reason){toast('請填寫加班事由','warn');return false;}
    if(type!=='holiday_visit_comp' && !compensationMethod){toast('請選擇加班處理方式','warn');return false;}
    if(timeBased && (!overtimeValid24h(start)||!overtimeValid24h(end))){toast('請選擇完整的開始及結束時間','warn');return false;}
    if(timeBased && (!['00','10','20','30','40','50'].includes(String(start).slice(3,5)) || !['00','10','20','30','40','50'].includes(String(end).slice(3,5)))){toast('加班時間請以10分鐘為間隔','warn');return false;}
    if(timeBased && overtimeCrossesMidnight(start,end)){toast('跨午夜加班請分成兩筆申請：第一天申請至 00:00（視同 24:00），第二天改以隔日 00:00 起另行申請。','warn');return false;}
    if(timeBased && overtimeMinutes<=0){toast('結束時間須晚於開始時間','warn');return false;}
    if(type==='holiday_visit_comp' && (caseCount<1 || caseCount>4)){toast('假日家訪案數請選擇1至4案','warn');return false;}

    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      work_date:workDate,
      request_type:type,
      start_time:timeBased?start:null,
      end_time:timeBased?end:null,
      overtime_minutes:overtimeMinutes,
      compensation_method:compensationMethod,
      reason,
      holiday_case_count:type==='holiday_visit_comp'?caseCount:null,
      comp_minutes:compMinutes,
      notes:notes||null,
      status,
      submitted_at:status==='pending'?new Date().toISOString():null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:new Date().toISOString()
    };

    showLoading(true);
    try{
      let error;
      if(existing){
        ({error}=await db.from('overtime_requests').update(payload).eq('id',existing.id));
      }else{
        payload.created_by=state.user.id;
        ({error}=await db.from('overtime_requests').insert(payload));
      }
      if(error) throw error;
      state.overtimePageFormDirty=false;
      closeModal();
      await loadOvertimeRows();
      renderOvertimePage();
      toast(status==='pending'?'申請已送出審核':'草稿已儲存','success');
      return true;
    }catch(err){
      console.error(err);
      toast('儲存失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

    if(minutes<=0){toast('結束時間須晚於開始時間','warn');return false;}
    if(!reason){toast('請填寫換休事由','warn');return false;}
    if(status==='pending' && minutes>sourceAvailable.total){
      toast(`所選日期目前合計可用換休僅 ${overtimeDurationLabel(sourceAvailable.total)}，請調整換休起訖時間`,'warn');
      return false;
    }

    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      leave_date:leaveDate,
      start_time:start,
      end_time:end,
      minutes,
      reason,
      notes:notes||null,
      status,
      submitted_at:status==='pending'?new Date().toISOString():null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:new Date().toISOString()
    };

    showLoading(true);
    try{
      let error;
      if(existing){
        ({error}=await db.from('comp_leave_usages').update(payload).eq('id',existing.id));
      }else{
        payload.created_by=state.user.id;
        ({error}=await db.from('comp_leave_usages').insert(payload));
      }
      if(error) throw error;
      state.overtimePageFormDirty=false;
      closeModal();
      await loadOvertimeRows();
      state.overtimeView='usage';
      renderOvertimePage();
      toast(status==='pending'?'換休使用申請已送出審核':'換休使用草稿已儲存','success');
      return true;
    }catch(err){
      console.error(err);
      toast('儲存換休使用申請失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

// V5.2 加班／換休流程補強：只保留編輯追蹤、重疊檢核與撤回按鈕
// 月份篩選由 overtime-month-filter-patch 處理；撤回動作由 overtime-withdraw-patch 處理。
// 本檔不再包裝 navigate、不再執行資料重載、不再保留舊的 withdrawn 流程。
(() => {
  Object.assign(state,{workflowEditingOvertimeId:null,workflowEditingLeaveId:null});

  const active=new Set(['draft','pending','approved']);
  const ownO=()=> (state.overtimeRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile?.id));
  const ownL=()=> (state.compLeaveRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile?.id));
  const norm=v=>{
    const x=String(v||'').slice(0,5);
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(x)?x:null;
  };
  const sel=p=>{
    const h=document.getElementById(p+'Hour')?.value||'';
    const m=document.getElementById(p+'Minute')?.value||'';
    return h&&m?`${h}:${m}`:null;
  };
  const min=v=>{
    const x=norm(v);
    if(!x)return null;
    const [h,m]=x.split(':').map(Number);
    return h*60+m;
  };
  const interval=(a,b)=>{
    const start=min(a),rawEnd=min(b);
    if(!Number.isFinite(start)||!Number.isFinite(rawEnd))return null;
    const end=rawEnd===0&&start>0?1440:rawEnd;
    return end>start?[start,end]:null;
  };
  const overlap=(a,b,c,d)=>{
    const x=interval(a,b),y=interval(c,d);
    return !!x&&!!y&&x[0]<y[1]&&x[1]>y[0];
  };

  function addWithdrawButtons(){
    if(state.page!=='overtime')return;
    const view=state.overtimeView;
    if(!['mine','usage'].includes(view))return;

    const rows=view==='mine'?ownO():ownL();
    const domRows=[...document.querySelectorAll('#otMain .ot-table tbody tr')];
    if(!domRows.length||domRows.length!==rows.length)return;

    rows.forEach((row,index)=>{
      if(row.status!=='pending')return;
      const box=domRows[index]?.querySelector('.ot-actions');
      if(!box||box.querySelector('[data-wf-withdraw]'))return;
      const btn=document.createElement('button');
      btn.className='btn sm';
      btn.type='button';
      btn.textContent='撤回申請';
      btn.title='撤回待審核申請';
      btn.dataset.wfWithdraw=`${view==='mine'?'overtime':'leave'}:${row.id}`;
      box.appendChild(btn);
    });
  }

  function inferO(){
    if(state.workflowEditingOvertimeId)return state.workflowEditingOvertimeId;
    const d=document.getElementById('otWorkDate')?.value;
    const t=document.getElementById('otType')?.value;
    const a=sel('otStart');
    const b=sel('otEnd');
    const c=Number(document.getElementById('otCaseCount')?.value||0);
    const x=ownO()
      .filter(r=>['draft','rejected'].includes(r.status)&&r.work_date===d&&r.request_type===t)
      .filter(r=>t==='holiday_visit_comp'
        ? Number(r.holiday_case_count||0)===c
        : norm(r.start_time)===a&&norm(r.end_time)===b);
    return x.length===1?x[0].id:null;
  }

  function inferL(){
    if(state.workflowEditingLeaveId)return state.workflowEditingLeaveId;
    const d=document.getElementById('clDate')?.value;
    const a=sel('clStart');
    const b=sel('clEnd');
    const x=ownL().filter(r=>
      ['draft','rejected'].includes(r.status)&&
      r.leave_date===d&&
      norm(r.start_time)===a&&
      norm(r.end_time)===b
    );
    return x.length===1?x[0].id:null;
  }

  function validOvertime(){
    const d=document.getElementById('otWorkDate')?.value;
    const t=document.getElementById('otType')?.value;
    if(!d||!t)return true;

    const id=inferO();
    const rows=ownO().filter(r=>
      active.has(r.status)&&
      String(r.id)!==String(id||'')&&
      r.work_date===d
    );

    if(t==='holiday_visit_comp'){
      const c=Number(document.getElementById('otCaseCount')?.value||0);
      const same=rows.filter(r=>r.request_type==='holiday_visit_comp');
      if(same.some(r=>Number(r.holiday_case_count||0)===c)){
        toast('同一天相同家訪案數的假日家訪換休已存在，請勿重複申請','warn');
        return false;
      }
      return !same.length||confirm('同一天已經有另一筆假日家訪換休申請。\n\n若這不是重複申請，確定要繼續嗎？');
    }

    const a=sel('otStart');
    const b=sel('otEnd');
    if(!a||!b)return true;

    const timedTypes=new Set([
      'overtime',
      'holiday_overtime',
      'rest_day_overtime',
      'national_holiday_overtime'
    ]);
    const timed=rows.filter(r=>timedTypes.has(r.request_type));

    if(timed.some(r=>norm(r.start_time)===a&&norm(r.end_time)===b)){
      toast('同一天相同起訖時間已存在加班申請，請勿重複送出','warn');
      return false;
    }

    const hit=timed.find(r=>overlap(a,b,r.start_time,r.end_time));
    if(hit){
      toast(`這段加班時間與既有申請（${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');
      return false;
    }
    return true;
  }

  function validLeave(){
    const d=document.getElementById('clDate')?.value;
    const a=sel('clStart');
    const b=sel('clEnd');
    if(!d||!a||!b)return true;

    const id=inferL();
    const hit=ownL().find(r=>
      active.has(r.status)&&
      String(r.id)!==String(id||'')&&
      r.leave_date===d&&
      overlap(a,b,r.start_time,r.end_time)
    );
    if(hit){
      toast(`換休時段與既有申請 ${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)} 重疊，請調整時間`,'warn');
      return false;
    }
    return true;
  }

  document.addEventListener('click',e=>{
    const b=e.target.closest?.('button');
    if(!b)return;

    if(b.id==='otAddBtn')state.workflowEditingOvertimeId=null;
    if(b.dataset.otEdit)state.workflowEditingOvertimeId=b.dataset.otEdit;
    if(b.id==='clAddBtn')state.workflowEditingLeaveId=null;
    if(b.dataset.clEdit)state.workflowEditingLeaveId=b.dataset.clEdit;

    const modal=b.closest('.modal,.ot-page-form');
    if(!modal)return;
    const submit=b.id==='modalConfirm'||b.textContent.trim()==='儲存草稿';
    if(!submit)return;

    const ok=modal.querySelector('#otWorkDate')
      ? validOvertime()
      : modal.querySelector('#clDate')
        ? validLeave()
        : true;
    if(!ok){
      e.preventDefault();
      e.stopImmediatePropagation();
    }
  },true);

  window.AttendanceV2Lifecycle?.register('overtime:render','overtime-workflow-withdraw',addWithdrawButtons);
  addWithdrawButtons();
})();



// V5.2 加班／換休：待審核申請撤回（直接刪除，不追蹤原因）
// 本檔只處理撤回，不再重載加班資料、不包裝 navigate、不覆寫換休餘額。
(() => {
  const style=document.createElement('style');
  style.id='v52-overtime-withdraw-style';
  style.textContent='[data-wf-withdraw]{white-space:nowrap}';
  document.head.appendChild(style);

  function ownRow(kind,id){
    const rows=kind==='leave'?(state.compLeaveRows||[]):(state.overtimeRows||[]);
    return rows.find(r=>String(r.id)===String(id)&&String(r.applicant_staff_id)===String(state.profile?.id))||null;
  }

  function openWithdrawModal(kind,id){
    const row=ownRow(kind,id);
    if(!row){
      toast('找不到這筆申請，請重新整理後再試','warn');
      return;
    }
    if(row.status!=='pending'){
      toast('只有待審核中的申請可以撤回','warn');
      return;
    }

    const date=kind==='leave'?row.leave_date:row.work_date;
    const time=row.start_time&&row.end_time
      ? `${String(row.start_time).slice(0,5)}–${String(row.end_time).slice(0,5)}`
      : '—';
    openModal({
      title:'撤回申請',
      html:`
        <div style="font-size:14px;font-weight:800;color:#344054">確定撤回此筆申請？</div>
        <div class="ot-detail" style="margin-top:10px">日期：${escapeHTML(date||'—')}　｜　時間：${escapeHTML(time)}</div>`,
      confirmText:'確認撤回',
      onConfirm:()=>submitWithdraw(kind,id)
    });
  }

  async function submitWithdraw(kind,id){
    const row=ownRow(kind,id);
    if(!row||row.status!=='pending'){
      toast('此申請目前已不是待審核狀態，請重新整理後再確認','warn');
      return false;
    }

    showLoading(true);
    try{
      const table=kind==='leave'?'comp_leave_usages':'overtime_requests';
      const {data,error}=await db.from(table)
        .delete()
        .eq('id',id)
        .eq('applicant_staff_id',state.profile.id)
        .eq('status','pending')
        .select('id')
        .maybeSingle();
      if(error)throw error;
      if(!data)throw new Error('申請狀態已變更，請重新整理後再操作');

      closeModal();
      navigate('overtime');
      toast('申請已撤回','success');
      return true;
    }catch(err){
      console.error('撤回申請失敗',err);
      toast('撤回失敗：'+(err?.message||err),'error');
      return false;
    }finally{
      showLoading(false);
    }
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('[data-wf-withdraw]');
    if(!btn)return;
    e.preventDefault();
    e.stopPropagation();
    e.stopImmediatePropagation();

    const [kind,id]=String(btn.dataset.wfWithdraw||'').split(':');
    if(!['overtime','leave'].includes(kind)||!id){
      toast('撤回資料格式不正確，請重新整理頁面','warn');
      return;
    }
    openWithdrawModal(kind,id);
  },true);
})();




// V5.2 加班／換休資料完整性：月結鎖定、核准紀錄調整
(() => {
  const style=document.createElement('style');

    .ot-correction-table{width:100%;border-collapse:collapse}

  function canReviewIntegrity(){
    return !!state.profile && ['organization_manager','admin'].includes(state.profile.role);
  }

  function monthRange(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    const [y,m]=safe.split('-').map(Number);
    const start=`${y}-${String(m).padStart(2,'0')}-01`;
    const end=isoDate(new Date(y,m,0));
    return {start,end};
  }

  function monthLabel(month){
    const [y,m]=String(month||'').split('-').map(Number);
    return Number.isFinite(y)&&Number.isFinite(m)?`民國 ${y-1911} 年 ${m} 月`:String(month||'');
  }

  function dateText(value){
    return value?rocDate(String(value).slice(0,10)+'T00:00:00'):'—';
  }

  function typeText(row,kind){
    if(kind==='leave') return '換休使用';
    return ({overtime:'平日加班',holiday_overtime:'假日加班',holiday_visit_comp:'假日家訪換休'})[row.request_type]||row.request_type||'—';
  }

  function amountText(row,kind){
    if(kind==='leave') return `${Number(row.minutes)||0} 分鐘`;
    const mins=row.request_type==='holiday_visit_comp'?(Number(row.comp_minutes)||0):(Number(row.overtime_minutes)||0);
    return `${mins} 分鐘`;
  }

  function selectedMonthSettledFromState(){
    const start=monthRange(state.overtimeMonth).start;
    return (state.payrollSettlements||[]).some(r=>r.settlement_month===start);
  }

  function currentApprovedRows(){
    const {start,end}=monthRange(state.overtimeMonth);
    const overtime=(state.overtimeRows||[])
      .filter(r=>r.status==='approved'&&r.work_date>=start&&r.work_date<=end)
      .map(r=>({kind:'overtime',id:r.id,date:r.work_date,name:r.applicant_name||'—',row:r}));
    const leave=(state.compLeaveRows||[])
      .filter(r=>r.status==='approved'&&r.leave_date>=start&&r.leave_date<=end)
      .map(r=>({kind:'leave',id:r.id,date:r.leave_date,name:r.applicant_name||'—',row:r}));
    return [...overtime,...leave].sort((a,b)=>String(b.date).localeCompare(String(a.date))||String(a.name).localeCompare(String(b.name),'zh-Hant'));
  }

  function renderCorrectionRows(items,settled){
    if(!items.length) return '<div class="ot-empty"><div class="icon">✓</div><h3>本月沒有可調整的已核准紀錄</h3></div>';
    return `<div class="ot-table-shell"><table class="ot-correction-table"><thead><tr><th>日期</th><th>申請人</th><th>類型</th><th>認列</th><th>事由</th><th>操作</th></tr></thead><tbody>${items.map(item=>{
      const r=item.row;
      return `<tr><td>${dateText(item.date)}</td><td><strong>${escapeHTML(item.name)}</strong></td><td>${escapeHTML(typeText(r,item.kind))}</td><td>${escapeHTML(amountText(r,item.kind))}</td><td>${escapeHTML(r.reason||'—')}</td><td>${settled?'<span class="ot-status approved">已月結鎖定</span>':`<button class="btn sm" data-integrity-void="${item.kind}:${item.id}">作廢</button>`}</td></tr>`;
    }).join('')}</tbody></table></div>`;
  }

  function openCorrectionCenter(){
    if(!canReviewIntegrity()){toast('目前帳號沒有調整核准紀錄的權限','warn');return;}
    const settled=selectedMonthSettledFromState();
    const items=currentApprovedRows();
    openModal({
      title:`核准紀錄調整｜${monthLabel(state.overtimeMonth)}`,
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">${settled
          ?'此月份已完成薪資結算，所有原始資料均已鎖定，不能直接作廢；若需調整，後續請使用「月結更正」流程。'
          :'此功能不會刪除原紀錄。作廢後原紀錄仍會保留並標示「已作廢」；如需更正內容，請再建立一筆正確申請重新送審。'}</div>
        <div style="margin-top:12px">${renderCorrectionRows(items,settled)}</div>`,
      large:true,
      showConfirm:false
    });
    document.querySelectorAll('[data-integrity-void]').forEach(btn=>btn.addEventListener('click',()=>{
      const [kind,id]=String(btn.dataset.integrityVoid||'').split(':');
      openVoidReason(kind,id);
    }));
  }

  function findItem(kind,id){
    return kind==='leave'
      ? (state.compLeaveRows||[]).find(r=>r.id===id)
      : (state.overtimeRows||[]).find(r=>r.id===id);
  }

  function openVoidReason(kind,id){
    const row=findItem(kind,id);
    if(!row) return;
    const d=kind==='leave'?row.leave_date:row.work_date;
    openModal({
      title:'作廢已核准紀錄',
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">將保留原紀錄，不會刪除。若本次是資料填錯，完成作廢後請重新新增正確資料送審。</div>
        <div class="field" style="margin-top:12px"><label>紀錄</label><input class="ot-readonly" disabled value="${escapeHTML(`${dateText(d)}｜${row.applicant_name||'—'}｜${typeText(row,kind)}`)}"></div>
        <div class="field"><label>作廢原因 *</label><textarea id="integrityVoidReason" placeholder="例如：核准後發現申請時間填寫錯誤，需重新申請"></textarea></div>`,
      confirmText:'確認作廢',
      onConfirm:()=>submitVoid(kind,id),
      large:true
    });
  }

  async function submitVoid(kind,id){
    const reason=String(document.getElementById('integrityVoidReason')?.value||'').trim();
    if(!reason){toast('請填寫作廢原因','warn');return false;}
    if(!confirm('確定要將這筆已核准紀錄標記為「已作廢」嗎？\n\n原紀錄會保留，不會刪除。')) return false;
    showLoading(true);
    try{
      const fn=kind==='leave'?'void_comp_leave_usage':'void_overtime_request';
      const args=kind==='leave'?{p_id:id,p_reason:reason}:{p_id:id,p_reason:reason};
      const {error}=await db.rpc(fn,args);
      if(error) throw error;
      const row=findItem(kind,id);
      if(row){
        row.status='voided';
        row.review_note=(row.review_note?row.review_note+'\n':'')+'作廢原因：'+reason;
      }
      closeModal();
      navigate('overtime');
      toast('紀錄已作廢；如需更正，請重新新增正確申請','success');
      return true;
    }catch(err){
      console.error(err);
      toast('作廢失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  async function checkDateLock(input){
    if(!input?.value) return;
    const field=input.closest('.field')||input.parentElement;
    let note=field?.querySelector?.('[data-payroll-lock-note]');
    if(!note&&field){
      note=document.createElement('div');
      note.dataset.payrollLockNote='1';
      note.className='ot-lock-note hidden';
      field.appendChild(note);
    }
    try{
      const {data,error}=await db.rpc('is_payroll_month_settled',{p_date:input.value});
      if(error) throw error;
      const locked=!!data;
      if(note){
        note.classList.toggle('hidden',!locked);
        note.textContent=locked?'此日期所屬月份已完成薪資結算，資料已鎖定，不能新增、修改或重新送審。':'';
      }
      const modal=input.closest('.modal,.ot-page-form');
      const confirmBtn=modal?.querySelector('#modalConfirm');
      const draftBtn=[...(modal?.querySelectorAll('.modal-foot .btn')||[])].find(b=>b.textContent.trim()==='儲存草稿');
      if(confirmBtn){confirmBtn.disabled=locked;confirmBtn.style.opacity=locked?'0.55':'';}
      if(draftBtn){draftBtn.disabled=locked;draftBtn.style.opacity=locked?'0.55':'';}
    }catch(err){
      console.error('檢查月結鎖定失敗',err);
    }
  }

  function attachDateGuard(input){
    if(!input||input.dataset.payrollDateGuard==='1') return;
    input.dataset.payrollDateGuard='1';
    input.addEventListener('change',()=>checkDateLock(input));
    checkDateLock(input);
  }

  function normalizeVoidedLabels(){
    document.querySelectorAll('.ot-status.voided').forEach(el=>{if(el.textContent.trim()!=='已作廢') el.textContent='已作廢';});
  }

  function ensureCorrectionButton(){
    if(state.page!=='overtime'||!canReviewIntegrity()) return;
    const reload=document.getElementById('otReload');
    if(!reload||document.getElementById('otCorrectionBtn')) return;
    const btn=document.createElement('button');
    btn.id='otCorrectionBtn';
    btn.className='btn';
    btn.textContent='↺ 核准紀錄調整';
    btn.title='處理尚未月結月份中的已核准紀錄調整';
    btn.addEventListener('click',openCorrectionCenter);
    reload.insertAdjacentElement('afterend',btn);
  }

  function enhance(){
    ensureCorrectionButton();
    normalizeVoidedLabels();
    attachDateGuard(document.getElementById('otWorkDate'));
    attachDateGuard(document.getElementById('clDate'));
  }

  window.AttendanceV2Lifecycle?.register('overtime:render','overtime-integrity',enhance);
  window.AttendanceV2Lifecycle?.register('overtime:form','overtime-integrity-form',enhance);
  enhance();
})();

(() => {
  const style=document.createElement('style');

  style.textContent=``;

  document.head.appendChild(style);

  Object.assign(state,{
    compLeaveExpiryStaff:state.compLeaveExpiryStaff||[],
    compLeaveExpiryLoaded:false,
    compLeaveExpiryLoading:false
  });

  function ceCanReview(){
    return !!state.profile&&['organization_manager','admin'].includes(state.profile.role);
  }

  function ceToday(){return isoDate(new Date());}

  function ceDuration(minutes){
    const n=Math.max(0,Number(minutes)||0);
    const h=Math.floor(n/60),m=n%60;
    if(h&&m)return `${h}小時${m}分`;
    if(h)return `${h}小時`;
    return `${m}分鐘`;
  }

  function ceDate(value){
    return value?rocDate(String(value).slice(0,10)+'T00:00:00'):'待設定';
  }

  function ceRole(role){return roleLabels?.[role]||role||'—';}

  function ceStaff(staffId){
    return (state.compLeaveExpiryStaff||[]).find(s=>String(s.id)===String(staffId))||null;
  }


  function ceIsPermanentHoliday(c){
    if(!c||c.source_type!=='overtime_request')return false;
    const source=(state.overtimeRows||[]).find(r=>String(r.id)===String(c.source_id));
    return source?.request_type==='holiday_visit_comp'||c.source_label==='假日家訪換休';
  }

  function ceDaysUntil(date){
    if(!date)return null;
    const a=new Date(ceToday()+'T00:00:00');
    const b=new Date(String(date).slice(0,10)+'T00:00:00');
    return Math.round((b-a)/86400000);
  }


  async function ceLoad(force=false){
    if(!state.profile||state.compLeaveExpiryLoading)return;
    if(!state.compLeaveCreditsLoaded)return;
    if(state.compLeaveExpiryLoaded&&!force){
      queueMicrotask(ceEnhance);
      return;
    }

    state.compLeaveExpiryLoading=true;
    try{
      const query=ceCanReview()
        ? db.from('staff_users').select('id,display_name,role,is_active,hire_date').eq('is_active',true).order('display_name')
        : db.from('staff_users').select('id,display_name,role,is_active,hire_date').eq('id',state.profile.id).limit(1);
      const {data,error}=await query;
      if(error)throw error;
      state.compLeaveExpiryStaff=data||[];
      state.compLeaveExpiryLoaded=true;
      queueMicrotask(ceEnhance);
    }catch(err){
      console.error('讀取換休期限設定失敗',err);
      toast('讀取換休期限設定失敗：'+(err?.message||err),'error');
    }finally{
      state.compLeaveExpiryLoading=false;
    }
  }


  function ceAddYearsClamped(date,years){
    const [y,m,d]=String(date||'').slice(0,10).split('-').map(Number);
    if(!y||!m||!d)return null;
    const targetY=y+Number(years||0);
    const lastDay=new Date(Date.UTC(targetY,m,0)).getUTCDate();
    return `${targetY}-${String(m).padStart(2,'0')}-${String(Math.min(d,lastDay)).padStart(2,'0')}`;
  }

  function ceAddDays(date,days){
    if(!date)return null;
    const d=new Date(String(date).slice(0,10)+'T00:00:00Z');
    d.setUTCDate(d.getUTCDate()+Number(days||0));
    return d.toISOString().slice(0,10);
  }

  function ceAnniversaryEnd(staffId){
    const hire=ceStaff(staffId)?.hire_date;
    if(!hire)return null;
    const t=ceToday();
    const hireYear=Number(String(hire).slice(0,4));
    const todayYear=Number(String(t).slice(0,4));
    let years=Math.max(0,todayYear-hireYear);
    let start=ceAddYearsClamped(hire,years);
    if(start&&start>t){
      years=Math.max(0,years-1);
      start=ceAddYearsClamped(hire,years);
    }
    return ceAddDays(ceAddYearsClamped(hire,years+1),-1);
  }

  function ceExpiryCountdownText(date){
    const d=ceDaysUntil(date);
    if(d===null)return '';
    if(d<0)return '已到期';
    if(d===0)return '今天到期';
    return `剩 ${d} 天`;
  }

  function ceAnniversaryReminder(){
    const summary=document.querySelector('#content .ot-summary-main');

    const expiry=ceAnniversaryEnd(state.profile?.id);
    const html=expiry
      ? `📅 本期一般加班換休到期日：<strong>${escapeHTML(ceDate(expiry))}</strong>（${escapeHTML(ceExpiryCountdownText(expiry))}）。到期未休依既有規則辦理結清；<span class="muted">假日家訪換休無到期限制。</span>`
      : '📅 本期一般加班換休到期日：<strong>尚未設定到職日</strong>，目前無法計算週年到期日。';
    if(existing){
      if(existing.innerHTML!==html)existing.innerHTML=html;
      return;
    }
    const box=document.createElement('div');
    box.id='ceAnniversaryReminder';
    box.className='attendance-expiry-reminder';
    box.innerHTML=html;
    summary.insertAdjacentElement('afterend',box);
  }

    if(state.page!=='overtime'||state.overtimeView!=='payroll'||!state.compLeaveExpiryLoaded||!ceCanReview())return;
    const main=document.getElementById('otMain');
    if(!main)return;

    document.getElementById('cePayrollExpiryPanel')?.remove();
    const month=String(state.overtimeMonth||'');
    if(!/^\d{4}-\d{2}$/.test(month))return;
    const start=month+'-01';
    const [y,m]=month.split('-').map(Number);
    const end=isoDate(new Date(y,m,0));
    const rows=(state.compLeaveCredits||[])
      .filter(c=>!ceIsPermanentHoliday(c)&&c.expires_on&&String(c.expires_on)>=start&&String(c.expires_on)<=end&&(Number(c.remaining_minutes)||0)>0);
    if(!rows.length)return;

    const total=rows.reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const names=new Map((state.compLeaveExpiryStaff||[]).map(s=>[String(s.id),s.display_name]));
    const html=`<section id="cePayrollExpiryPanel" class="ot-rule-card">
      <h3>本月一般加班換休到期結清</h3>
      <p>以下為本月到期仍未休完的一般加班換休；到期後會自動排除於可用換休餘額，並依原加班日薪資基數與倍率順位折算加班費。假日家訪換休無到期限制，不列入本表。</p>
      <div class="ot-rule-highlight">本月到期未休合計：<strong>${escapeHTML(ceDuration(total))}</strong>。</div>
      <div style="margin-top:12px"><div class="ot-table-shell"><table class="ot-table"><thead><tr><th>員工</th><th>來源日期</th><th>來源</th><th>到期日</th><th>待結清時數</th></tr></thead><tbody>${rows.map(c=>`<tr><td><strong>${escapeHTML(names.get(String(c.staff_id))||'—')}</strong></td><td>${ceDate(c.earned_date)}</td><td>${escapeHTML(c.source_label||'換休')}</td><td>${ceDate(c.expires_on)}</td><td><span class="ot-amount">${escapeHTML(ceDuration(c.remaining_minutes))}</span></td></tr>`).join('')}</tbody></table></div></div>

// V5.2 固定例假日設定＋申請提醒
(() => {
  const labels=['週日','週一','週二','週三','週四','週五','週六'];
  let staffCache=[];
  let staffLoaded=false;
  let staffLoading=false;

  function parseDate(value){
    if(!/^\d{4}-\d{2}-\d{2}$/.test(String(value||'')))return null;
    const d=new Date(String(value)+'T00:00:00');
    return Number.isNaN(d.getTime())?null:d;
  }
  function isRegularDay(date,weekday){
    const d=parseDate(date);
    return !!d && Number.isInteger(Number(weekday)) && d.getDay()===Number(weekday);
  }
  function currentStaff(){
    return staffCache.find(s=>String(s.id)===String(state.profile?.id))||null;
  }
  async function loadStaff(force=false){
    if(staffLoading)return;
    if(staffLoaded&&!force)return;
    staffLoading=true;
    try{
      const {data,error}=await db.from('staff_users')
        .select('id,display_name,role,is_active,regular_day_weekday')
        .eq('is_active',true)
        .order('display_name');
      if(error)throw error;
      staffCache=data||[];
      staffLoaded=true;
    }catch(err){
      console.error('讀取例假日設定失敗',err);
      toast('讀取例假日設定失敗：'+(err?.message||err),'error');
    }finally{
      staffLoading=false;
    }
  }

  function warningHTML(staff,date){
    if(!staff||staff.regular_day_weekday===null||staff.regular_day_weekday===undefined){
      return '<div class="rd-note">尚未設定固定例假日。</div>';
    }
    const label=labels[Number(staff.regular_day_weekday)]||'—';
    if(isRegularDay(date,staff.regular_day_weekday)){
      return '<div class="rd-warning">⚠ 此日期為你的固定例假日（'+escapeHTML(label)+'），原則上不得加班，請先確認實際排班與出勤原因。</div>';
    }
    return '<div class="rd-note">固定例假日：'+escapeHTML(label)+'</div>';
  }

  async function bindApplicationWarning(){
    const date=document.getElementById('otWorkDate');
    if(!date||date.dataset.rdBound==='1')return;
    date.dataset.rdBound='1';
    await loadStaff(false);
    const field=date.closest('.field');
    if(field&&!document.getElementById('rdApplicationWarning')){
      const host=document.createElement('div');
      host.id='rdApplicationWarning';
      host.style.marginTop='6px';
      field.appendChild(host);
    }
    const update=(notify=false)=>{
      const host=document.getElementById('rdApplicationWarning');
      const staff=currentStaff();
      if(host)host.innerHTML=warningHTML(staff,date.value);
      if(notify&&staff&&isRegularDay(date.value,staff.regular_day_weekday)){
        if(date.dataset.rdLastWarn!==date.value){
          date.dataset.rdLastWarn=date.value;
          toast('提醒：此日期為你設定的固定例假日，請確認後再申請。','warn');
        }
      }else if(!isRegularDay(date.value,staff?.regular_day_weekday)){
        date.dataset.rdLastWarn='';
      }
    };
    date.addEventListener('change',()=>update(true));
    update(false);
  }

  function enhanceOvertimePage(){
    if(state.page!=='overtime')return;
    bindApplicationWarning();
  }

  const style=document.createElement('style');
  style.id='v52-regular-day-warning-style';
  style.textContent='.rd-warning{border:1px solid #fecdca;background:#fef3f2;color:#b42318;border-radius:9px;padding:8px 10px;font-size:11px;font-weight:800;line-height:1.55}.rd-note{font-size:10.5px;color:#667085;line-height:1.5}';
  document.head.appendChild(style);

  window.AttendanceV2Lifecycle?.register('overtime:render','regular-day-warning',enhanceOvertimePage);
  window.AttendanceV2Lifecycle?.register('overtime:form','regular-day-warning-form',enhanceOvertimePage);

  window.v52RegularDayApi={loadStaff};
  enhanceOvertimePage();
})();



// V5.2 管理者假別餘額總覽：換休＋特休
(() => {
  const isManager=()=>!!state.profile&&['organization_manager','admin'].includes(state.profile.role);
  const today=()=>isoDate(new Date());

  function dur(minutes){
    const n=Math.max(0,Number(minutes)||0);
    const h=Math.floor(n/60),m=n%60;
    if(h&&m)return h+'小時'+m+'分';
    if(h)return h+'小時';
    return m+'分鐘';
  }
  function roleLabel(role){
    return ({admin:'系統管理員',organization_manager:'機構負責人',business_manager:'業務負責人',supervisor:'居家督導員'})[role]||role||'—';
  }
  function daysText(value){
    const n=Math.max(0,Number(value)||0);
    const rounded=Math.round(n*100)/100;
    return String(rounded).replace(/\.0+$/,'')+'天';
  }
  function currentCompBalance(staffId,date=today()){
    const credits=(state.compLeaveCredits||[])
      .filter(c=>String(c.staff_id)===String(staffId))
      .filter(c=>String(c.earned_date||'')<=date&&(!c.expires_on||String(c.expires_on)>=date));
    const ledgerRemaining=credits.reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const ordinaryRemaining=credits
      .filter(c=>!String(c.source_label||'').includes('假日家訪'))
      .reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const holidayVisitRemaining=credits
      .filter(c=>String(c.source_label||'').includes('假日家訪'))
      .reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const pending=(state.compLeaveRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='pending')
      .reduce((s,r)=>s+(Number(r.minutes)||0),0);
    return {
      credits,ledgerRemaining,ordinaryRemaining,holidayVisitRemaining,
      pending,available:Math.max(0,ledgerRemaining-pending)
    };
  }
  function currentAnnualBalance(staffId,credits,requests,date=today()){
    const current=credits.filter(c=>String(c.staff_id)===String(staffId)&&String(c.period_start||'')<=date&&String(c.period_end||'')>=date&&c.settlement_status==='active');
    const remaining=current.reduce((s,c)=>s+(Number(c.remaining_days)||0),0);
    const pending=requests
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='pending'&&current.some(c=>String(r.leave_date||'')>=String(c.period_start)&&String(r.leave_date||'')<=String(c.period_end)))
      .reduce((s,r)=>s+(Number(r.leave_days)||0),0);
    return {credits:current,remaining,pending,available:Math.max(0,remaining-pending)};
  }

  async function loadData(){
    const [staffRes,annualCreditRes,annualReqRes]=await Promise.all([
      db.from('staff_users').select('id,display_name,role,is_active').eq('is_active',true).order('display_name'),
      db.from('annual_leave_credits').select('id,staff_id,period_start,period_end,granted_days,remaining_days,granted_minutes,remaining_minutes,label,settlement_status,daily_hours_snapshot').order('period_end',{ascending:true}),
      db.from('annual_leave_requests').select('id,applicant_staff_id,leave_date,minutes,leave_days,status').eq('status','pending')
    ]);
    if(staffRes.error)throw staffRes.error;
    if(annualCreditRes.error)throw annualCreditRes.error;
    if(annualReqRes.error)throw annualReqRes.error;
    return {staff:staffRes.data||[],annualCredits:annualCreditRes.data||[],annualRequests:annualReqRes.data||[]};
  }

  function detailHTML(staff,comp,annual){
    const compRows=comp.credits.length
      ? '<div class="ot-table-shell"><table class="ot-table" style="min-width:620px"><thead><tr><th>取得日</th><th>來源</th><th>原始額度</th><th>剩餘</th><th>到期日</th></tr></thead><tbody>'+
        comp.credits.map(c=>'<tr><td>'+escapeHTML(rocDate(String(c.earned_date)+'T00:00:00'))+'</td><td>'+escapeHTML(c.source_label||c.source_type||'換休')+'</td><td>'+escapeHTML(dur(c.earned_minutes))+'</td><td><strong>'+escapeHTML(dur(c.remaining_minutes))+'</strong></td><td>'+escapeHTML(c.expires_on?rocDate(String(c.expires_on)+'T00:00:00'):'未設定')+'</td></tr>').join('')+
        '</tbody></table></div>'
      : '<div class="ot-empty"><p>目前沒有有效的換休額度。</p></div>';
    const annualRows=annual.credits.length
      ? '<div class="ot-table-shell"><table class="ot-table" style="min-width:620px"><thead><tr><th>額度</th><th>期間</th><th>原始額度</th><th>剩餘</th></tr></thead><tbody>'+
        annual.credits.map(c=>'<tr><td>'+escapeHTML(c.label||'特休額度')+'</td><td>'+escapeHTML(rocDate(String(c.period_start)+'T00:00:00'))+'～'+escapeHTML(rocDate(String(c.period_end)+'T00:00:00'))+'</td><td>'+escapeHTML(daysText(c.granted_days))+'<div class="ot-detail">生效時每日 '+escapeHTML(String(Number(c.daily_hours_snapshot)||'—'))+' 小時</div></td><td><strong>'+escapeHTML(daysText(c.remaining_days))+'</strong></td></tr>').join('')+
        '</tbody></table></div>'
      : '<div class="ot-empty"><p>目前沒有有效的特休額度。</p></div>';
    return '<div class="lab-detail-summary"><strong>'+escapeHTML(staff.display_name||'未命名')+'</strong><span>'+escapeHTML(roleLabel(staff.role))+'</span></div>'+
      '<section class="ot-rule-card"><h3>換休</h3><div class="lab-detail-metrics"><span>帳面剩餘 <strong>'+escapeHTML(dur(comp.ledgerRemaining))+'</strong></span><span>一般加班換休 <strong>'+escapeHTML(dur(comp.ordinaryRemaining))+'</strong></span><span>假日家訪換休 <strong>'+escapeHTML(dur(comp.holidayVisitRemaining))+'</strong></span><span>待審核 <strong>'+escapeHTML(dur(comp.pending))+'</strong></span><span>目前可用 <strong>'+escapeHTML(dur(comp.available))+'</strong></span></div>'+compRows+'</section>'+
      '<section class="ot-rule-card"><h3>特休</h3><div class="lab-detail-metrics"><span>剩餘 <strong>'+escapeHTML(daysText(annual.remaining))+'</strong></span><span>待審核 <strong>'+escapeHTML(daysText(annual.pending))+'</strong></span><span>目前可用 <strong>'+escapeHTML(daysText(annual.available))+'</strong></span></div>'+annualRows+'</section>';
  }

  async function openPage(){
    if(!isManager()){toast('目前帳號沒有假別餘額管理權限','warn');return;}
    showLoading(true);
    try{
      const data=await loadData();
      const d=today();
      const rows=data.staff.map(s=>{
        const comp=currentCompBalance(s.id,d);
        const annual=currentAnnualBalance(s.id,data.annualCredits,data.annualRequests,d);
        return {staff:s,comp,annual};
      });
      state.adminLeaveBalanceRows=rows;
      $('content').innerHTML=
        '<div class="ot-page-form lab-page">'+
          '<div class="ot-page-form-head"><div>'+
            '<button type="button" id="labBack" class="btn" style="margin-bottom:10px">← 返回加班休假</button>'+
            '<div class="ot-page-form-title">員工假別餘額</div>'+
            '<div class="ot-page-form-sub">顯示目前有效的換休與特休餘額；換休以小時／分鐘顯示，特休以天顯示，待審核額度另行列示。</div>'+
          '</div></div>'+
          '<div class="ot-page-form-card">'+
            (rows.length
              ? '<div class="ot-table-shell"><table class="ot-table lab-table"><thead><tr><th>員工</th><th>目前可用換休</th><th>待審核換休</th><th>目前剩餘特休</th><th>待審核特休</th><th>操作</th></tr></thead><tbody>'+
                rows.map(x=>'<tr><td><div class="ot-review-name">'+escapeHTML(x.staff.display_name||'未命名')+'</div><div class="ot-detail">'+escapeHTML(roleLabel(x.staff.role))+'</div></td><td><strong>'+escapeHTML(dur(x.comp.available))+'</strong><div class="ot-detail">一般 '+escapeHTML(dur(x.comp.ordinaryRemaining))+'｜假日家訪 '+escapeHTML(dur(x.comp.holidayVisitRemaining))+'</div></td><td>'+escapeHTML(dur(x.comp.pending))+'</td><td><strong>'+escapeHTML(daysText(x.annual.remaining))+'</strong></td><td>'+escapeHTML(daysText(x.annual.pending))+'</td><td><button type="button" class="btn sm" data-lab-detail="'+escapeHTML(String(x.staff.id))+'">查看明細</button></td></tr>').join('')+
                '</tbody></table></div>'
              : '<div class="ot-empty"><div class="icon">🕒</div><h3>目前沒有可顯示的員工資料</h3></div>')+
          '</div>'+
        '</div>';
      $('labBack')?.addEventListener('click',()=>navigate('overtime'));
      document.querySelectorAll('[data-lab-detail]').forEach(btn=>btn.addEventListener('click',()=>{
        const row=(state.adminLeaveBalanceRows||[]).find(x=>String(x.staff.id)===String(btn.dataset.labDetail));
        if(!row)return;
        openModal({title:'假別明細',html:detailHTML(row.staff,row.comp,row.annual),large:true,showConfirm:false});
      }));
      window.scrollTo({top:0,behavior:'auto'});
    }catch(err){
      console.error('讀取員工假別餘額失敗',err);
      toast('讀取員工假別餘額失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  function enhance(){
    if(!isManager()||state.page!=='overtime'||state.attendanceSection==='annual')return;
    const reload=document.getElementById('otReload');
    if(!reload||document.getElementById('labBalanceBtn'))return;
    const b=document.createElement('button');
    b.id='labBalanceBtn';
    b.className='btn';
    b.textContent='👥 假別餘額';
    b.addEventListener('click',openPage);
    reload.insertAdjacentElement('beforebegin',b);
  }

  const style=document.createElement('style');
  style.id='v52-admin-leave-balance-style';
  style.textContent='.lab-table{min-width:800px}.lab-detail-summary{display:flex;gap:10px;align-items:center;margin-bottom:12px}.lab-detail-summary strong{font-size:16px;color:#344054}.lab-detail-summary span{font-size:11px;color:#667085}.lab-detail-metrics{display:flex;gap:18px;flex-wrap:wrap;margin:0 0 12px;font-size:12px;color:#667085}.lab-detail-metrics strong{color:#344054}';
  document.head.appendChild(style);

  window.AttendanceV2Lifecycle?.register('overtime:render','admin-leave-balance',enhance);
  enhance();

  window.v52AdminLeaveBalanceApi={open:openPage};
})();
