(() => {
  // 上方導覽列已顯示模組名稱與說明，內容區不再重複顯示相同標題。
  if(!document.getElementById('v52-page-head-dedup-style')){
    const layoutStyle=document.createElement('style');
    layoutStyle.id='v52-page-head-dedup-style';
    layoutStyle.textContent=`
      .page-head > div:first-child:has(> h1){display:none!important}
      .page-head:has(> div:first-child:only-child > h1){display:none!important}
      .page-head:has(> .actions){justify-content:flex-end!important;align-items:center!important}
    `;
    document.head.appendChild(layoutStyle);
  }

  function payrollCurrentMonth(){
    const now=new Date();
    return `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}`;
  }

  function payrollSelectedMonthClosed(){
    const selected=String(state.overtimeMonth||'');
    return /^\d{4}-\d{2}$/.test(selected) && selected < payrollCurrentMonth();
  }

  function applyPayrollPeriodGuard(){
    const btn=document.getElementById('payrollCloseBtn');
    if(!btn) return;

    const canClose=payrollSelectedMonthClosed();
    const buttonText=canClose?'✓ 確認本月已結算':'本月尚未結束，暫不可結算';
    const buttonTitle=canClose?'確認薪資資料後進行月結':'僅能結算已完整結束的月份；當月請於次月1日起辦理結算。';

    btn.disabled=!canClose;
    btn.style.opacity=canClose?'':'0.55';
    btn.style.cursor=canClose?'':'not-allowed';
    if(btn.textContent!==buttonText) btn.textContent=buttonText;
    if(btn.title!==buttonTitle) btn.title=buttonTitle;

    const card=btn.closest('.ot-rule-card');
    if(!card) return;
    let notice=card.querySelector('[data-payroll-period-notice]');

    if(!canClose){
      if(!notice){
        notice=document.createElement('div');
        notice.dataset.payrollPeriodNotice='1';
        notice.className='ot-rule-highlight';
        card.appendChild(notice);
      }
      const [year,month]=String(state.overtimeMonth||'').split('-').map(Number);
      const nextMonth=new Date(year,month,1);
      const noticeText=`此月份尚未完整結束，需於 ${nextMonth.getFullYear()} 年 ${nextMonth.getMonth()+1} 月 1 日起才能進行月結。`;
      if(notice.textContent!==noticeText) notice.textContent=noticeText;
    }else if(notice){
      notice.remove();
    }
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('#payrollCloseBtn');
    if(!btn || payrollSelectedMonthClosed()) return;
    e.preventDefault();
    e.stopImmediatePropagation();
    toast('此月份尚未完整結束，請於次月1日起再進行薪資結算。','warn');
  },true);

  window.AttendanceV2Lifecycle?.register('overtime:render','payroll-period-guard',applyPayrollPeriodGuard);
  window.AttendanceV2Lifecycle?.register('overtime:form','payroll-period-guard-form',applyPayrollPeriodGuard);
  applyPayrollPeriodGuard();
})();

// 換休取得與換休使用更正均保存實際日期與必要來源資料；換休使用以30分鐘為單位。
(() => {

  correctionStyle.id='v52-payroll-correction-style';

  function pcCanReview(){
    return !!state.profile&&['organization_manager','admin'].includes(state.profile.role);
  }

  function pcMonthStart(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    return safe+'-01';
  }

  function pcSettlement(){
    const start=pcMonthStart(state.overtimeMonth);
    return (state.payrollSettlements||[]).find(r=>r.settlement_month===start)||null;
  }

  function pcCorrectionsForSettlement(settlementId){
    return (state.payrollCorrections||[]).filter(c=>String(c.settlement_id)===String(settlementId));
  }

  function pcSum(list,type){
    return list.filter(c=>c.adjustment_type===type).reduce((s,c)=>s+(Number(c.adjustment_minutes)||0),0);
  }

  function pcDuration(minutes){
    const n=Math.max(0,Number(minutes)||0),h=Math.floor(n/60),m=n%60;
    if(h&&m)return `${h}小時${m}分`;
    if(h)return `${h}小時`;
    return `${m}分鐘`;
  }

  function pcSigned(minutes){
    const n=Number(minutes)||0;
    if(!n)return '<span class="pc-adjust zero">0</span>';
    return `<span class="pc-adjust ${n>0?'plus':'minus'}">${n>0?'+':'−'}${escapeHTML(pcDuration(Math.abs(n)))}</span>`;
  }

  function pcTypeLabel(type){
    return ({overtime_pay:'加班計薪',comp_leave_earned:'換休取得',comp_leave_used:'換休使用'})[type]||type;
  }

  function pcRoleLabel(role){return roleLabels?.[role]||role||'—';}

  async function pcLoadData(force=false){
    if(!state.profile||state.payrollCorrectionLoading)return;
    if(state.payrollCorrectionsLoaded&&!force){
      queueMicrotask(pcEnhance);
      return;
    }

    state.payrollCorrectionLoading=true;
    try{
      const correctionPromise=db.from('payroll_settlement_corrections')
        .select('id,settlement_id,staff_id,staff_name,staff_role,adjustment_type,adjustment_minutes,reason,created_by,created_by_name,created_at,work_date,workday_type,start_time,end_time,day_offset_minutes,calculated_amount,wage_base_snapshot,agreed_daily_hours_snapshot,hourly_base_snapshot,rate_effective_from_snapshot,calculation_formula,calculation_version,related_source_type,related_source_id')
        .order('created_at',{ascending:false});
      const staffPromise=pcCanReview()
        ? db.from('staff_users').select('id,display_name,role,is_active').eq('is_active',true).order('display_name')
        : Promise.resolve({data:[],error:null});
      const [corrRes,staffRes]=await Promise.all([correctionPromise,staffPromise]);
      if(corrRes.error)throw corrRes.error;
      if(staffRes.error)throw staffRes.error;
      state.payrollCorrections=corrRes.data||[];
      state.overtimeStaffDirectory=staffRes.data||[];
      state.payrollCorrectionsLoaded=true;
      queueMicrotask(pcEnhance);
    }catch(err){
      console.error('讀取月結更正資料失敗',err);
      toast('讀取月結更正資料失敗：'+(err?.message||err),'error');
    }finally{
      state.payrollCorrectionLoading=false;
    }
  }

  function pcUpdateCard(label,value){
    document.querySelectorAll('#content .stat-card').forEach(card=>{
      const head=card.querySelector('.stat-head span:first-child');
      if(head?.textContent.trim()!==label)return;
      const val=card.querySelector('.stat-value');
      if(val&&val.textContent!==value)val.textContent=value;
    });
  }

  function pcCorrectedRows(settlement){
    const base=Array.isArray(settlement?.summary?.staff)?settlement.summary.staff:[];
    const map=new Map();
    base.forEach(r=>map.set(String(r.staffId),{
      staffId:String(r.staffId),name:r.name||'未命名',role:r.role||'',
      originalPay:Number(r.overtimePayMinutes)||0,
      originalEarned:Number(r.compLeaveEarnedMinutes)||0,
      originalUsed:Number(r.compLeaveUsedMinutes)||0,
      originalBalance:Number(r.compLeaveBalanceMinutes)||0,
      corrPay:0,corrEarned:0,corrUsed:0
    }));

    pcCorrectionsForSettlement(settlement.id).forEach(c=>{
      const key=String(c.staff_id);
      if(!map.has(key)){
        map.set(key,{
          staffId:key,name:c.staff_name||'未命名',role:c.staff_role||'',
          originalPay:0,originalEarned:0,originalUsed:0,originalBalance:0,
          corrPay:0,corrEarned:0,corrUsed:0
        });
      }
      const r=map.get(key);
      if(c.adjustment_type==='overtime_pay')r.corrPay+=Number(c.adjustment_minutes)||0;
      if(c.adjustment_type==='comp_leave_earned')r.corrEarned+=Number(c.adjustment_minutes)||0;
      if(c.adjustment_type==='comp_leave_used')r.corrUsed+=Number(c.adjustment_minutes)||0;
    });

    return [...map.values()].map(r=>({...r,
      finalPay:Math.max(0,r.originalPay+r.corrPay),
      finalEarned:Math.max(0,r.originalEarned+r.corrEarned),
      finalUsed:Math.max(0,r.originalUsed+r.corrUsed),
      finalBalance:Math.max(0,r.originalBalance+r.corrEarned-r.corrUsed)
    })).sort((a,b)=>String(a.name).localeCompare(String(b.name),'zh-Hant'));
  }

  function pcFlow(original,adjusted,finalValue){
    return `<div class="pc-flow"><span class="orig">${escapeHTML(pcDuration(original))}</span>${adjusted?`<span>${pcSigned(adjusted)}</span>`:''}<span class="arrow">→</span><span class="final">${escapeHTML(pcDuration(finalValue))}</span></div>`;
  }

  function pcUpdatePayrollUI(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll'||!state.payrollCorrectionsLoaded)return;
    const settlement=pcSettlement();
    if(!settlement)return;

    const corr=pcCorrectionsForSettlement(settlement.id);
    const payAdj=pcSum(corr,'overtime_pay');
    const earnedAdj=pcSum(corr,'comp_leave_earned');
    const usedAdj=pcSum(corr,'comp_leave_used');
    pcUpdateCard('本月待計薪加班',pcDuration((Number(settlement.total_overtime_pay_minutes)||0)+payAdj));
    pcUpdateCard('本月新增換休',pcDuration((Number(settlement.total_comp_leave_earned_minutes)||0)+earnedAdj));
    pcUpdateCard('本月使用換休',pcDuration((Number(settlement.total_comp_leave_used_minutes)||0)+usedAdj));

    const main=document.getElementById('otMain');
    if(!main)return;
    document.getElementById('payrollCorrectionPanel')?.remove();
    main.insertAdjacentHTML('beforeend',pcCorrectionPanelHTML(settlement));
    document.getElementById('pcAddBtn')?.addEventListener('click',pcOpenCorrectionModal);
  }

  function pcMinuteOptions(type,selected='00'){
    const vals=type==='comp_leave_used'?['00','30']:['00','10','20','30','40','50'];
    return vals.map(v=>`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`).join('');
  }


  function pcHourOptions(selected=''){
    let html='<option value="">時</option>';
    for(let i=0;i<24;i++){
      const v=String(i).padStart(2,'0');
      html+=`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`;
    }
    return html;
  }

  function pcTenMinuteOptions(selected='00'){
    return ['00','10','20','30','40','50'].map(v=>`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`).join('');
  }

  function pcThirtyMinuteOptions(selected='00'){
    return ['00','30'].map(v=>`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`).join('');
  }

  function pcSelectedTime(prefix){
    const h=document.getElementById(prefix+'Hour')?.value||'';
    const m=document.getElementById(prefix+'Minute')?.value||'';
    return h&&m?`${h}:${m}`:null;
  }

  function pcMinutesBetween(start,end){
    if(!start||!end)return 0;
    const [sh,sm]=start.split(':').map(Number),[eh,em]=end.split(':').map(Number);
    const a=sh*60+sm,b=eh*60+em;
    if(b===0&&a>0)return 1440-a;
    return b>a?b-a:0;
  }

  function pcCrossesMidnight(start,end){
    if(!start||!end)return false;
    const [sh,sm]=start.split(':').map(Number),[eh,em]=end.split(':').map(Number);
    const a=sh*60+sm,b=eh*60+em;
    return b<a&&b!==0;
  }

  function pcMoney(value){
    const n=Number(value);
    return Number.isFinite(n)?`NT$ ${n.toLocaleString('zh-TW',{minimumFractionDigits:2,maximumFractionDigits:2})}`:'—';
  }

  function pcWorkdayLabel(type){
    return ({overtime:'平日加班',rest_day_overtime:'休息日加班',national_holiday_overtime:'國定假日出勤',holiday_visit_comp:'假日家訪換休'})[type]||type||'—';
  }

  function pcEarnedSourceLabel(type){
    return ({overtime:'平日加班換休',rest_day_overtime:'休息日加班換休',national_holiday_overtime:'國定假日換休',holiday_visit_comp:'假日家訪換休'})[type]||type||'—';
  }

  function pcCorrectionMetaHTML(c){
    if(c.adjustment_type==='overtime_pay'&&c.work_date){
      return `加班：${rocDate(String(c.work_date)+'T00:00:00')}・${escapeHTML(pcWorkdayLabel(c.workday_type))}・${escapeHTML(String(c.start_time||'').slice(0,5))}～${escapeHTML(String(c.end_time||'').slice(0,5)==='00:00'?'24:00':String(c.end_time||'').slice(0,5))}<br>金額：${escapeHTML(pcMoney(c.calculated_amount))}・倍率順位第 ${Number(c.day_offset_minutes||0)+1}–${Number(c.day_offset_minutes||0)+Math.abs(Number(c.adjustment_minutes)||0)} 分鐘<br>`;
    }
    if(c.adjustment_type==='comp_leave_earned'&&c.work_date){
      if(c.workday_type==='holiday_visit_comp'){
        const cases=Math.max(1,Math.round(Math.abs(Number(c.adjustment_minutes)||0)/120));
        return `取得來源：${rocDate(String(c.work_date)+'T00:00:00')}・假日家訪換休・${cases}案×2小時・無到期限制<br>`;
      }
      const start=String(c.start_time||'').slice(0,5),end=String(c.end_time||'').slice(0,5)==='00:00'?'24:00':String(c.end_time||'').slice(0,5);
      const offset=Number(c.day_offset_minutes||0),mins=Math.abs(Number(c.adjustment_minutes)||0);
      return `取得來源：${rocDate(String(c.work_date)+'T00:00:00')}・${escapeHTML(pcEarnedSourceLabel(c.workday_type))}・${escapeHTML(start)}～${escapeHTML(end)}<br>原同日倍率順位第 ${offset+1}–${offset+mins} 分鐘；到期時依此來源資料折算<br>`;
    }
    if(c.adjustment_type==='comp_leave_used'&&c.work_date){
      const start=String(c.start_time||'').slice(0,5),end=String(c.end_time||'').slice(0,5)==='00:00'?'24:00':String(c.end_time||'').slice(0,5);
      const action=Number(c.adjustment_minutes)>0?'補登使用':'減少認列並還原原扣抵來源';
      return `使用：${rocDate(String(c.work_date)+'T00:00:00')}・${escapeHTML(start)}～${escapeHTML(end)}<br>${escapeHTML(action)}<br>`;
    }
    return '';
  }

  async function pcPreviewOvertimeCorrection(showError=false){
    const box=document.getElementById('pcCorrCalc');
    const settlement=pcSettlement();
    const staffId=document.getElementById('pcStaff')?.value;
    const date=document.getElementById('pcCorrDate')?.value;
    const dayType=document.getElementById('pcCorrDayType')?.value;
    const direction=Number(document.getElementById('pcDirection')?.value||1);
    const start=pcSelectedTime('pcCorrStart'),end=pcSelectedTime('pcCorrEnd');
    if(!settlement||!staffId||!date||!dayType||!start||!end){
      if(box)box.textContent='請填寫實際加班日期與起訖時間，系統會依同日實際加班順位試算。';
      return null;
    }
    if(pcCrossesMidnight(start,end)){
      if(box)box.textContent='跨午夜更正不可用一筆建立，請自行分成兩筆。';
      if(showError)toast('跨午夜更正請自行分成兩筆','warn');
      return null;
    }
    if(pcMinutesBetween(start,end)<=0){
      if(box)box.textContent='結束時間必須晚於開始時間；結束00:00視同當日24:00。';
      return null;
    }
    const {data,error}=await db.rpc('preview_payroll_overtime_correction',{
      p_settlement_id:settlement.id,p_staff_id:staffId,p_work_date:date,p_workday_type:dayType,
      p_start_time:start,p_end_time:end,p_direction:direction
    });
    if(error){
      if(box)box.textContent='無法試算：'+(error.message||error);
      if(showError)toast('無法建立加班計薪更正：'+(error.message||error),'warn');
      return null;
    }
    const row=Array.isArray(data)?data[0]:data;
    if(!row)return null;
    const endPos=Number(row.day_offset_minutes||0)+Number(row.correction_minutes||0);
    if(box)box.innerHTML=`自動計算：<strong>${escapeHTML(pcDuration(row.correction_minutes))}</strong>｜同日倍率順位第 ${Number(row.day_offset_minutes||0)+1}–${endPos} 分鐘｜預估更正金額 <strong>${escapeHTML(pcMoney(row.calculated_amount))}</strong>`;
    return row;
  }

  async function pcPreviewUsedCorrection(showError=false){
    const box=document.getElementById('pcUsedCalc');
    const settlement=pcSettlement();
    const staffId=document.getElementById('pcStaff')?.value;
    const date=document.getElementById('pcUsedDate')?.value;
    const direction=Number(document.getElementById('pcDirection')?.value||1);
    const start=pcSelectedTime('pcUsedStart'),end=pcSelectedTime('pcUsedEnd');
    if(!settlement||!staffId||!date||!start||!end){
      if(box)box.textContent='請填寫實際使用日期與起訖時間，系統會自動計算時數並檢查換休帳本。';
      return null;
    }
    if(pcCrossesMidnight(start,end)){
      if(box)box.textContent='跨午夜換休使用更正不可用一筆建立，請自行分成兩筆。';
      if(showError)toast('跨午夜換休使用更正請自行分成兩筆','warn');
      return null;
    }
    const mins=pcMinutesBetween(start,end);
    if(mins<=0||mins%30!==0){
      if(box)box.textContent='換休使用更正須以30分鐘為單位，且結束時間必須晚於開始時間。';
      return null;
    }
    const {data,error}=await db.rpc('preview_payroll_comp_leave_used_correction',{
      p_settlement_id:settlement.id,p_staff_id:staffId,p_leave_date:date,
      p_start_time:start,p_end_time:end,p_direction:direction
    });
    if(error){
      if(box)box.textContent='無法試算：'+(error.message||error);
      if(showError)toast('無法建立換休使用更正：'+(error.message||error),'warn');
      return null;
    }
    const row=Array.isArray(data)?data[0]:data;
    if(!row)return null;
    if(direction>0){
      if(box)box.innerHTML=`自動計算：<strong>${escapeHTML(pcDuration(row.correction_minutes))}</strong>｜該日目前可扣抵換休 <strong>${escapeHTML(pcDuration(row.available_minutes))}</strong>`;
    }else{
      if(box)box.innerHTML=`將整筆還原：<strong>${escapeHTML(row.related_source_label||'既有換休使用紀錄')}</strong>｜${escapeHTML(pcDuration(row.correction_minutes))}`;
    }
    return row;
  }

  function pcRefreshEarnedMeta(){
    const source=document.getElementById('pcEarnSource')?.value;
    const timed=document.getElementById('pcEarnTimedFields');

此類換休無到期限制。

建立後不可直接修改或刪除。`;
      }else{

${staff.display_name}｜${pcEarnedSourceLabel(source)}｜${rocDate(date+'T00:00:00')}

系統會保存原同日倍率順位，供日後換休到期未休折算。

建立後不可直接修改或刪除。`;

      const start=pcSelectedTime('pcUsedStart'),end=pcSelectedTime('pcUsedEnd');
      const {start:monthStart,end:monthEnd}=(()=>{const m=String(state.overtimeMonth||'');const [y,mo]=m.split('-').map(Number);return {start:`${m}-01`,end:isoDate(new Date(y,mo,0))};})();

建立後不可直接修改或刪除。`;
    }else{
      toast('不支援的更正項目','warn');
      return false;
    }

    if(!confirm(confirmText))return false;
    showLoading(true);
    try{
      const {error}=await db.from('payroll_settlement_corrections').insert(payload);

    }finally{
      showLoading(false);
    }
  }

    const dayEnd=offset+m;
    if(!rate||!Number.isFinite(monthly)||monthly<=0)return {ok:false,code:'missing_rate',text:'尚未設定有效薪資基數'};
    if(!Number.isFinite(hours)||hours<=0||hours>8)return {ok:false,code:'missing_hours',text:'尚未設定有效的約定每日工時'};
    if(!timedTypes.has(type)||type==='holiday_overtime')return {ok:false,code:'legacy',text:'需先確認假日類型'};
    if(m<=0)return {ok:false,code:'invalid',text:'工時資料不完整'};
    const h=monthly/30/hours;
    let amount=0,formula='';
    if(type==='overtime'){
      if(dayEnd>240)return {ok:false,code:'limit',text:'平日當日累計加班超過4小時，需人工確認'};
      const a=prRangeMinutes(offset,m,0,120),b=prRangeMinutes(offset,m,120,240);
      amount=h*(a/60)*(4/3)+h*(b/60)*(5/3);
      formula=`${monthly.toFixed(2)}÷30÷${hours}；同日依實際時間順序累計，本筆位於當日第${offset+1}–${dayEnd}分鐘；前2小時×4/3、其後2小時×5/3`;
    }else if(type==='rest_day_overtime'){
      if(dayEnd>720)return {ok:false,code:'limit',text:'休息日當日累計出勤超過12小時'};
      const a=prRangeMinutes(offset,m,0,120),b=prRangeMinutes(offset,m,120,480),c=prRangeMinutes(offset,m,480,720);
      amount=h*(a/60)*(4/3)+h*(b/60)*(5/3)+h*(c/60)*(8/3);
      formula=`${monthly.toFixed(2)}÷30÷${hours}；同日依實際時間順序累計，本筆位於當日第${offset+1}–${dayEnd}分鐘；前2小時×4/3、第3–8小時×5/3、第9–12小時×8/3`;
    }else if(type==='national_holiday_overtime'){
      const normalMinutes=Math.round(hours*60);
      if(dayEnd>normalMinutes+240)return {ok:false,code:'limit',text:'國定假日當日累計出勤超過約定每日工時加4小時，請重新確認'};
      amount=includeDayBase?monthly/30:0;
      const a=prRangeMinutes(offset,m,normalMinutes,normalMinutes+120),b=prRangeMinutes(offset,m,normalMinutes+120,normalMinutes+240);
      amount+=h*(a/60)*(4/3)+h*(b/60)*(5/3);
      formula=`${includeDayBase?monthly.toFixed(2)+'÷30為當日一次日工資；':'當日日工資已於同日其他領加班費紀錄認列；'}同日依實際時間順序累計，本筆位於當日第${offset+1}–${dayEnd}分鐘；約定每日工時${hours}小時，超過後前2小時×4/3、再2小時×5/3`;
    }
    return {ok:true,amount,hourly:h,monthly,hours,formula,dayOffset:offset,dayEnd};
  }

  function prPreviewPosition(staffId,date,start,existingId=null){
    const candidateStart=prNormTime(start);
    if(!staffId||!date||!candidateStart)return {offset:0,includeDayBase:true};
    const active=new Set(['draft','pending','approved']);
    const earlier=(state.overtimeRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.work_date===date&&active.has(r.status)&&String(r.id)!==String(existingId||'')&&timedTypes.has(r.request_type))
      .filter(r=>r.request_type!=='holiday_visit_comp'&&prNormTime(r.start_time)&&prNormTime(r.end_time))
      .filter(r=>prNormTime(r.start_time)<candidateStart)
      .sort((a,b)=>String(a.start_time||'').localeCompare(String(b.start_time||''))||String(a.end_time||'').localeCompare(String(b.end_time||''))||String(a.created_at||'').localeCompare(String(b.created_at||'')));
    let offset=0,dayBasePaid=false;
    earlier.forEach(r=>{
      offset+=Number(r.overtime_minutes)||prMinutes(r.start_time,r.end_time);
      if(r.compensation_method==='overtime_pay'&&['national_holiday_overtime'].includes(r.request_type))dayBasePaid=true;
    });
    return {offset,includeDayBase:!dayBasePaid};
  }

  async function prLoad(force=false){
    if(!state.profile||state.payrollRatesLoading)return;
    if(state.payrollRatesLoaded&&!force){prEnhance();return;}
    state.payrollRatesLoading=true;
    try{
      const ratesPromise=prCanReview()
        ? db.from('staff_payroll_rates')
            .select('id,staff_id,effective_from,monthly_wage_base,agreed_daily_hours,note,created_by,created_by_name,created_at')
            .order('effective_from',{ascending:false}).order('created_at',{ascending:false})
        : Promise.resolve({data:[],error:null});
      const staffPromise=prCanReview()
        ? db.from('staff_users').select('id,display_name,role,is_active,agreed_daily_hours').eq('is_active',true).order('display_name')
        : Promise.resolve({data:[{id:state.profile.id,display_name:state.profile.display_name,role:state.profile.role,is_active:true}],error:null});

        ? db.from('payroll_settlement_corrections').select('id,settlement_id,staff_id,adjustment_type,adjustment_minutes,work_date,workday_type,calculated_amount,reason,created_at')
        : Promise.resolve({data:[],error:null});
      const [ratesRes,staffRes,corrRes]=await Promise.all([ratesPromise,staffPromise,corrPromise]);
      if(ratesRes.error)throw ratesRes.error;if(staffRes.error)throw staffRes.error;if(corrRes.error)throw corrRes.error;
      state.payrollRateRows=ratesRes.data||[];
      state.payrollRateStaff=staffRes.data||[];
      state.payrollAmountCorrections=corrRes.data||[];
      state.payrollRatesLoaded=true;
      prEnhance();
    }catch(err){console.error('讀取薪資計算設定失敗',err);toast('讀取薪資計算設定失敗：'+(err?.message||err),'error');}

  function prStaffCalc(staffId,rateResolver=prRateFor){
    const {start,end}=prMonthRange();
    const rows=(state.overtimeRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='approved'&&r.work_date>=start&&r.work_date<=end&&timedTypes.has(r.request_type))
      .filter(r=>r.request_type!=='holiday_visit_comp')
      .sort((a,b)=>String(a.work_date).localeCompare(String(b.work_date))||String(a.start_time||'').localeCompare(String(b.start_time||''))||String(a.end_time||'').localeCompare(String(b.end_time||''))||String(a.created_at||'').localeCompare(String(b.created_at||'')));
    const details=[];
    let currentDate=null,dayOffset=0,dayBasePaid=false;
    rows.forEach(r=>{
      if(currentDate!==r.work_date){currentDate=r.work_date;dayOffset=0;dayBasePaid=false;}
      const minutes=Number(r.overtime_minutes)||0;
      const rowOffset=dayOffset;
      const isCash=r.compensation_method==='overtime_pay';
      if(isCash){
        const rate=rateResolver(staffId,r.work_date);
        const includeDayBase=!dayBasePaid;
        const c=prCalc(r.request_type,minutes,rate,rowOffset,includeDayBase);
        details.push({
          id:r.id,date:r.work_date,type:r.request_type,
          startTime:String(r.start_time||'').slice(0,5),endTime:String(r.end_time||'').slice(0,5),
          minutes,reason:r.reason||'',staffId,row:r,rate,calc:c,dayOffset:rowOffset
        });
        if(c.ok&&includeDayBase&&['national_holiday_overtime'].includes(r.request_type))dayBasePaid=true;
      }
      dayOffset+=minutes;
    });
    const ok=details.length>0&&details.every(d=>d.calc.ok);
    const total=details.filter(d=>d.calc.ok).reduce((s,d)=>s+d.calc.amount,0);
    return {details,ok,total,missing:details.filter(d=>!d.calc.ok)};
  }

  const newTypes=new Set(['rest_day_overtime','national_holiday_overtime']);

    const sel=document.getElementById('otType');
    const id=state.workflowEditingOvertimeId;

  }

  function defaultCompLeave(){
    const comp=document.getElementById('otCompensation');
    const type=document.getElementById('otType');
    const modal=comp?.closest('.modal,.ot-page-form');
    if(!comp||!type||!modal||!modal.querySelector('#otWorkDate'))return;

    if(comp.value!=='comp_leave'){
      comp.value='comp_leave';
      comp.dispatchEvent(new Event('change',{bubbles:true}));

      type.dataset.defaultCompLeaveBound='1';

            current.value='comp_leave';
            current.dispatchEvent(new Event('change',{bubbles:true}));
          }

      });
    }
  }

  const typeLabel=t=>({overtime:'平日加班',rest_day_overtime:'休息日加班',national_holiday_overtime:'國定假日出勤'})[t]||t||'—';

  function monthRange(){
    const month=/^\d{4}-\d{2}$/.test(state.overtimeMonth||'')?state.overtimeMonth:isoDate(new Date()).slice(0,7);
    const [y,m]=month.split('-').map(Number);
    return {start:`${y}-${String(m).padStart(2,'0')}-01`,end:isoDate(new Date(y,m,0)),month};
  }
  function settlement(){
    const {month}=monthRange();
    return (state.payrollSettlements||[]).find(s=>String(s.settlement_month)===month+'-01')||null;
  }
  function staffFor(id){return (state.personalHoursStaff||[]).find(s=>String(s.id)===String(id))||null;}
  function rateFor(staffId,date){
    return (state.personalHoursRates||[])
      .filter(r=>String(r.staff_id)===String(staffId)&&String(r.effective_from)<=String(date))
      .sort((a,b)=>String(b.effective_from).localeCompare(String(a.effective_from))||String(b.created_at).localeCompare(String(a.created_at)))[0]||null;
  }
  function calc(type,minutes,rate){
    const monthly=Number(rate?.monthly_wage_base),hours=Number(rate?.agreed_daily_hours),m=Number(minutes)||0;
    if(!rate||!Number.isFinite(monthly)||monthly<=0)return {ok:false,text:'尚未設定有效薪資基數'};
    if(!Number.isFinite(hours)||hours<=0||hours>8)return {ok:false,text:'尚未設定有效的約定每日工時'};
    if(!timedTypes.has(type))return {ok:false,text:'此日別不提供自動試算'};
    if(m<=0)return {ok:false,text:'工時資料不完整'};
    const hourly=monthly/30/hours;
    let amount=0,formula='';
    if(type==='overtime'){
      if(m>240)return {ok:false,text:'平日加班超過4小時，請重新確認'};
      const a=Math.min(m,120),b=Math.max(Math.min(m-120,120),0);
      amount=hourly*(a/60)*(4/3)+hourly*(b/60)*(5/3);
      formula=`${monthly.toFixed(2)}÷30÷${hours}；超過個人約定工時後，前2小時×4/3、其後2小時×5/3`;
    }else if(type==='rest_day_overtime'){
      if(m>720)return {ok:false,text:'休息日出勤超過12小時'};
      const a=Math.min(m,120),b=Math.max(Math.min(m-120,360),0),c=Math.max(Math.min(m-480,240),0);
      amount=hourly*(a/60)*(4/3)+hourly*(b/60)*(5/3)+hourly*(c/60)*(8/3);
      formula=`${monthly.toFixed(2)}÷30÷${hours}；前2小時×4/3、第3–8小時×5/3、第9–12小時×8/3`;
    }else if(type==='national_holiday_overtime'){
      const normalMinutes=Math.round(hours*60);
      if(m>normalMinutes+240)return {ok:false,text:'國定假日出勤超過約定每日工時加4小時，請重新確認'};
      amount=monthly/30;
      const extra=Math.max(m-normalMinutes,0),a=Math.min(extra,120),b=Math.max(Math.min(extra-120,120),0);
      amount+=hourly*(a/60)*(4/3)+hourly*(b/60)*(5/3);
      formula=`${monthly.toFixed(2)}÷30為1日工資；約定每日工時${hours}小時，超過後前2小時×4/3、再2小時×5/3`;
    }
    return {ok:true,amount,hourly,monthly,hours,formula};
  }

  async function load(force=false){
    if(!state.profile||state.personalHoursLoading)return;
    if(state.personalHoursLoaded&&!force){enhance();return;}
    state.personalHoursLoading=true;
    try{
      const staffQuery=canReview()
        ? db.from('staff_users').select('id,display_name,role,is_active,agreed_daily_hours').eq('is_active',true).order('display_name')
        : db.from('staff_users').select('id,display_name,role,is_active,agreed_daily_hours').eq('id',state.profile.id).limit(1);
      const rateQuery=canReview()
        ? db.from('staff_payroll_rates').select('id,staff_id,effective_from,monthly_wage_base,agreed_daily_hours,note,created_by_name,created_at').order('effective_from',{ascending:false}).order('created_at',{ascending:false})
        : Promise.resolve({data:[],error:null});
      const [staffRes,rateRes]=await Promise.all([staffQuery,rateQuery]);
      if(staffRes.error)throw staffRes.error;if(rateRes.error)throw rateRes.error;
      state.personalHoursStaff=staffRes.data||[];
      state.personalHoursRates=canReview()?(rateRes.data||[]):[];
      state.personalHoursLoaded=true;
      enhance();
    }catch(err){console.error('讀取個人約定工時失敗',err);toast('讀取個人約定工時失敗：'+(err?.message||err),'error');}
    finally{state.personalHoursLoading=false;}
  }

  function rateHistoryHTML(){
    const rows=state.personalHoursRates||[];
    if(!rows.length)return '<div class="ot-empty"><p>尚未設定任何薪資計算基數。</p></div>';
    return rows.map(r=>{
      const s=staffFor(r.staff_id),hours=Number(r.agreed_daily_hours),hourly=Number(r.monthly_wage_base)/30/hours;
      return `<div class="pr-rate-item"><div class="pr-rate-top"><div><div class="pr-rate-main">${escapeHTML(s?.display_name||'未命名')}｜${money(r.monthly_wage_base)}</div><div class="pr-rate-sub">生效：${rocDate(String(r.effective_from)+'T00:00:00')}｜約定每日工時：${escapeHTML(String(hours))}小時｜平日時薪基數：約 ${money(hourly)}/小時${r.note?`<br>備註：${escapeHTML(r.note)}`:''}</div></div><div class="pr-rate-sub">建立：${escapeHTML(r.created_by_name||'—')}・${rocDate(r.created_at)}</div></div></div>`;
    }).join('');
  }

  async function saveRateFromModal(){
    if(!canReview()){toast('目前帳號沒有薪資計算設定權限','warn');return;}
    const staffId=document.getElementById('prRateStaff')?.value;
    const date=document.getElementById('prRateDate')?.value;
    const amount=Number(document.getElementById('prRateAmount')?.value||0);
    const hours=Number(document.getElementById('phDailyHours')?.value||0);
    const note=String(document.getElementById('prRateNote')?.value||'').trim();
    if(!staffId||!date||!Number.isFinite(amount)||amount<=0||!Number.isFinite(hours)||hours<=0||hours>8){toast('請完整填寫員工、生效日期、月薪工資總額與約定每日工時','warn');return;}

    try{
      const {error}=await db.rpc('save_staff_payroll_rate',{
        p_staff_id:staffId,
        p_effective_from:date,
        p_monthly_wage_base:amount,
        p_agreed_daily_hours:hours,

      toast('新增薪資計算設定失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }

  function currentStaffHours(){return Number(staffFor(state.profile?.id)?.agreed_daily_hours)||null;}
  function selectedTime(prefix){const h=document.getElementById(prefix+'Hour')?.value||'',m=document.getElementById(prefix+'Minute')?.value||'';return h&&m?`${h}:${m}`:null;}
  function minutesBetween(a,b){if(!a||!b)return 0;const [ah,am]=a.split(':').map(Number),[bh,bm]=b.split(':').map(Number);const start=ah*60+am,end=bh*60+bm;if(end===0&&start>0)return 1440-start;return end>start?end-start:0;}
  function crossesMidnight(a,b){if(!a||!b)return false;const [ah,am]=a.split(':').map(Number),[bh,bm]=b.split(':').map(Number);const start=ah*60+am,end=bh*60+bm;return end<start&&end!==0;}

  function staffLiveCalc(staffId){
    if(typeof globalThis.v52OvertimePayrollStaffCalc==='function'){
      return globalThis.v52OvertimePayrollStaffCalc(staffId,rateFor);
    }
    const {start,end}=monthRange();
    const details=(state.overtimeRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='approved'&&r.compensation_method==='overtime_pay'&&r.work_date>=start&&r.work_date<=end&&timedTypes.has(r.request_type))
      .map(r=>{const rate=rateFor(staffId,r.work_date),c=calc(r.request_type,Number(r.overtime_minutes)||0,rate);return {row:r,rate,calc:c};});
    const missing=details.filter(x=>!x.calc.ok),total=details.filter(x=>x.calc.ok).reduce((s,x)=>s+x.calc.amount,0);
    return {details,missing,total,ok:missing.length===0};
  }
