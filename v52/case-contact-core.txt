  function ccDate(value){
    if(!value)return null;
    var d=new Date(String(value).length===10?value+'T00:00:00':value);
    return Number.isNaN(d.getTime())?null:d;
  }
  function ccIso(d){
    if(!d)return null;
    var y=d.getFullYear(),m=String(d.getMonth()+1).padStart(2,'0'),day=String(d.getDate()).padStart(2,'0');
    return y+'-'+m+'-'+day;
  }
  function ccAddMonths(dateStr,months){
    var d=ccDate(dateStr);if(!d)return null;
    var day=d.getDate(), target=new Date(d.getFullYear(),d.getMonth()+months,1);
    var last=new Date(target.getFullYear(),target.getMonth()+1,0).getDate();
    target.setDate(Math.min(day,last));
    return ccIso(target);
  }
  function ccDaysBetween(a,b){
    var da=ccDate(a),db=ccDate(b);if(!da||!db)return null;
    da.setHours(0,0,0,0);db.setHours(0,0,0,0);
    return Math.round((db-da)/86400000);
  }

  function ccCycle(caseRow,type){
    var today=ccIso(new Date()), months=type==='formal_phone'?1:3, warning=type==='formal_phone'?7:14;
    if(type==='home_visit' && caseRow.service_status==='suspended')return {key:'muted',text:'暫停服務中',due:null,last:ccLatest(caseRow.id,type)};
    if(caseRow.service_status==='closed')return {key:'muted',text:'已結案',due:null,last:ccLatest(caseRow.id,type)};
    var last=ccLatest(caseRow.id,type);
    var basis=last?.record_date||caseRow.open_date||null;
    if(!basis)return {key:'muted',text:'尚無起算日',due:null,last:last};
    var due=ccAddMonths(basis,months);
    var days=ccDaysBetween(today,due);
    if(days===null)return {key:'muted',text:'尚無期限',due:due,last:last};
    if(days<0)return {key:'overdue',text:'逾期 '+Math.abs(days)+' 天',due:due,last:last};
    if(days===0)return {key:'overdue',text:'今天到期',due:due,last:last};
    if(days<=warning)return {key:'due',text:'剩 '+days+' 天',due:due,last:last};
    return {key:'normal',text:'正常',due:due,last:last};
  }
  function ccAssessmentCycle(caseRow){
    if(caseRow.service_status==='suspended')return {key:'muted',text:'暫停服務中',due:null,last:ccLatestFullAssessment(caseRow.id)};
    if(caseRow.service_status==='closed')return {key:'muted',text:'已結案',due:null,last:ccLatestFullAssessment(caseRow.id)};
    var last=ccLatestFullAssessment(caseRow.id);
    var basis=last?.planned_date||caseRow.open_date||null;
    if(!basis)return {key:'muted',text:'尚無起算日',due:null,last:last};
    var due=ccAddMonths(basis,3),today=ccIso(new Date()),days=ccDaysBetween(today,due);
    if(days<0)return {key:'overdue',text:'逾期 '+Math.abs(days)+' 天',due:due,last:last};
    if(days===0)return {key:'overdue',text:'今天到期',due:due,last:last};
    if(days<=14)return {key:'due',text:'剩 '+days+' 天',due:due,last:last};
    return {key:'normal',text:'正常',due:due,last:last};
  }
  function ccWorstStatus(c){
    var h=ccCycle(c,'home_visit'),p=ccCycle(c,'formal_phone'),a=ccAssessmentCycle(c);
    var f=ccPendingFollowups(c.id), today=ccIso(new Date());
    var overdueF=f.some(function(x){return x.due_date&&x.due_date<today;});
    if(overdueF||[h,p,a].some(function(x){return x.key==='overdue';}))return {key:'overdue',text:'有逾期項目'};
    if([h,p,a].some(function(x){return x.key==='due';}))return {key:'due',text:'即將到期'};
    if(f.length)return {key:'followup',text:'待追蹤 '+f.length};
    if(c.service_status==='closed'||c.service_status==='suspended')return {key:'muted',text:c.service_status==='closed'?'已結案':'暫停服務'};
    return {key:'normal',text:'正常'};
  }

  async function ccLoadData(silent){
    if(!ccCanUse())return;
    if(!silent)showLoading(true);
    try{
      var results=await Promise.all([
        db.rpc('get_case_staff_directory'),
        db.from('care_cases').select('id,case_no,case_name,national_id,birth_date,gender,address,phone,supervisor_id,open_date,cms_level,service_status,service_usage_type,payment_method,notes,created_at').order('case_no',{ascending:true,nullsFirst:false}).order('case_name'),
        db.from('case_contact_records').select('*').order('record_date',{ascending:false}).order('created_at',{ascending:false}),
        db.from('case_approved_services').select('case_id,service_group,service_code,service_name,approved_quantity').eq('is_current',true).order('service_code'),
        db.from('case_followup_items').select('*').order('due_date',{ascending:true,nullsFirst:false}).order('created_at'),
        db.from('case_routine_plans').select('*').eq('status','planned').order('planned_date'),
        db.from('assessment_events').select('id,case_id,assessment_type,planned_date,status,counts_as_full_reassessment,combined_with_home_visit,completed_at,voided_at').neq('status','voided').order('planned_date',{ascending:false})
      ]);
      var bad=results.find(function(x){return x.error;});if(bad&&bad.error)throw bad.error;
      state.caseStaff=results[0].data||[];
      state.cases=results[1].data||[];
      state.caseContactRecords=results[2].data||[];
      state.caseServices=results[3].data||[];
      state.caseFollowupItems=results[4].data||[];
      state.caseRoutinePlans=results[5].data||[];
      state.caseAssessmentEvents=results[6].data||[];
      state.caseVisits=state.caseContactRecords;
      if(!state.caseSupervisorFilter)state.caseSupervisorFilter=state.profile?.can_supervise?'mine':'all';
      state.caseContactDataLoaded=true;
    }catch(err){
      console.error(err);toast('讀取家、電訪資料失敗：'+(err?.message||err),'error');
    }finally{if(!silent)showLoading(false);}
  }

    list.innerHTML='<div class="case-table-shell"><table class="case-table"><thead><tr><th>個案</th><th>負責督導</th><th>最近家訪</th><th>最近正式電訪</th><th>最近聯繫</th><th>期限／追蹤</th><th>狀態</th></tr></thead><tbody>'+
      rows.map(function(c){
        var h=ccLatest(c.id,'home_visit'),p=ccLatest(c.id,'formal_phone'),latest=ccLatestContact(c.id);
        var hc=ccCycle(c,'home_visit'),pc=ccCycle(c,'formal_phone'),ac=ccAssessmentCycle(c),f=ccPendingFollowups(c.id),status=ccWorstStatus(c);
        return '<tr>'+
          '<td><button class="case-name-btn" data-case-id="'+c.id+'">'+escapeHTML(c.case_name)+'</button><div class="visit-item-sub">'+escapeHTML(c.case_no||'—')+'｜'+escapeHTML(ccServiceSummary(c.id))+'</div></td>'+
          '<td>'+escapeHTML(ccStaffName(c.supervisor_id))+'</td>'+
          '<td>'+(h?ccDateText(h.record_date):'—')+'</td>'+
          '<td>'+(p?ccDateText(p.record_date):'—')+'</td>'+
          '<td>'+(latest?ccDateText(latest.record_date)+' '+escapeHTML(ccTypeLabel(latest.record_type)):'—')+'</td>'+
          '<td><div class="cc-status-grid"><span class="cc-chip '+hc.key+'">家訪 '+(hc.due?ccDateText(hc.due):hc.text)+'</span><span class="cc-chip '+pc.key+'">電訪 '+(pc.due?ccDateText(pc.due):pc.text)+'</span><span class="cc-chip '+ac.key+'">評估 '+(ac.due?ccDateText(ac.due):ac.text)+'</span>'+(f.length?'<span class="cc-chip followup">追蹤 '+f.length+'</span>':'')+'</div></td>'+
          '<td><span class="case-status '+status.key+'">'+escapeHTML(status.text)+'</span></td>'+
        '</tr>';
      }).join('')+'</tbody></table></div>';
    list.querySelectorAll('[data-case-id]').forEach(function(btn){btn.addEventListener('click',function(){if(typeof window.openCaseDetailPage==='function')window.openCaseDetailPage(btn.dataset.caseId,'visits');});});
  }

  var ccBaseNavigate=navigate;
  navigate=function(id){
    if(id!=='cases')return ccBaseNavigate(id);
    window.currentCaseDetailId=null;window.currentCaseDetailTab=null;
    var m=modules.find(function(x){return x.id==='cases';})||modules[0];
    state.page='cases';renderNav();$('topTitle').textContent=m.label;$('crumb').textContent=m.crumb;$('sidebar').classList.remove('open');
    window.scrollTo({top:0,behavior:'auto'});
    if(!ccCanUse()){
      $('content').innerHTML='<div class="empty"><div class="icon">🔒</div><h3>此帳號沒有個案管理權限</h3></div>';return;
    }
    $('content').innerHTML='<div class="empty compact"><div class="icon">☎️</div><h3>正在載入家、電訪資料</h3><p>請稍候。</p></div>';
    ccLoadData().then(ccRenderPage);
  };

  window.caseContactApi={
    canUse:ccCanUse,canEditCase:ccCanEditCase,staffName:ccStaffName,typeLabel:ccTypeLabel,reasonLabel:ccReasonLabel,
    dateOnly:ccDate,addMonths:ccAddMonths,loadData:ccLoadData,renderPage:ccRenderPage,
    latest:ccLatest,latestContact:ccLatestContact,pendingFollowups:ccPendingFollowups,
    cycle:ccCycle,assessmentCycle:ccAssessmentCycle,currentPlan:ccCurrentPlan,
    recordsForCase:function(caseId){return (state.caseContactRecords||[]).filter(function(r){return r.case_id===caseId;}).sort(function(a,b){return String(b.record_date).localeCompare(String(a.record_date))||new Date(b.created_at)-new Date(a.created_at);});},
    latestFullAssessment:ccLatestFullAssessment
  };

  window.caseVisitsApi={
    loadCaseVisitData:ccLoadData,
    pendingFollowUpsFor:ccPendingFollowups,
    caseStatusInfo:ccWorstStatus,
    caseTypeLabel:function(t){return ccTypeLabel(t==='home'?'home_visit':t==='phone'?'formal_phone':t);},
    caseStaffName:ccStaffName,
    isFollowUpResolved:function(){return false;},
    openVisitModal:function(c,type){
      var mapped=type==='home'?'home_visit':type==='phone'?'formal_phone':type||'home_visit';
      if(window.caseContactApi&&window.caseContactApi.openRecordForm)window.caseContactApi.openRecordForm(c,mapped);
    },
    renderCaseVisitsPage:ccRenderPage
  };
