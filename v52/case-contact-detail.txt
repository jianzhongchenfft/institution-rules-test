
// V5.2 家、電訪 V1：透過公開 hook 整合個案頁、紀錄檢視與列印
(function(){
  var api=window.caseContactApi;
  if(!api)return;

  function ccField(label,value,wide){
    var v=(value===null||value===undefined||value==='')?'—':value;
    return '<div class="case-data-item'+(wide?' wide':'')+'"><label>'+escapeHTML(label)+'</label><div>'+escapeHTML(String(v))+'</div></div>';
  }
  function ccEmpty(text){return '<div class="case-empty-inline">'+escapeHTML(text)+'</div>';}
  function ccPlanDate(v){return v?rocDate(v+'T00:00:00'):'日期未辨識';}
  function ccServiceStatus(c){
    var key=c?.service_status||'active';
    if(key==='suspended')return {key:'due',text:'暫停'};
    if(key==='closed')return {key:'overdue',text:'已結案'};
    return {key:'normal',text:'服務中'};
  }

  async function ccEnrichCaseDetailData(data){
    var records=api.recordsForCase(data.c.id);
    data.contactRecords=records;
    var ids=records.map(function(x){return x.id;});
    data.ccPeople=[];data.ccFormal=[];data.ccHome=[];data.ccPhone=[];data.ccSatisfaction=[];data.ccLinks=[];
    if(!ids.length)return data;
    var rs=await Promise.all([
      db.from('case_contact_record_people').select('*').in('record_id',ids).order('created_at'),
      db.from('case_contact_formal_details').select('*').in('record_id',ids),
      db.from('case_home_visit_details').select('*').in('record_id',ids),
      db.from('case_phone_details').select('*').in('record_id',ids),
      db.from('case_contact_satisfaction').select('*').in('record_id',ids),
      db.from('case_record_links').select('*').in('from_id',ids)
    ]);
    var bad=rs.find(function(x){return x.error;});if(bad&&bad.error)throw bad.error;
    data.ccPeople=rs[0].data||[];data.ccFormal=rs[1].data||[];data.ccHome=rs[2].data||[];
    data.ccPhone=rs[3].data||[];data.ccSatisfaction=rs[4].data||[];data.ccLinks=rs[5].data||[];
    return data;
  }

  function ccExtendTimeline(items,data){
    (data.contactRecords||[]).forEach(function(r){
      var d=(r.record_date||'1970-01-01')+'T12:00:00';
      items.push({date:d,title:api.typeLabel(r.record_type)+(r.status==='draft'?'（草稿）':r.status==='voided'?'（已作廢）':''),text:r.summary||api.reasonLabel(r.record_type,r.reason_code)||'聯繫紀錄',sort:new Date(d).getTime()});
    });
    return items;
  }

      '<br>制度期限：'+(obj.due?rocDate(obj.due+'T00:00:00'):'—')+
      '<br>預定執行日：'+(plan?rocDate(plan.planned_date+'T00:00:00'):'尚未安排')+

      (api.canEditCase(c)?'<div style="margin-top:8px;text-align:right"><button class="btn sm" data-cc-plan="'+planType+'">'+(plan?'修改預定日':'安排日期')+'</button></div>':'')+
      '</div>';
  }

  function ccCurrentMonthRange(){
    var now=new Date(),y=now.getFullYear(),m=now.getMonth();
    function iso(d){return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');}
    return {start:iso(new Date(y,m,1)),end:iso(new Date(y,m+1,0))};
  }

  function ccGeneralPhonePrint(data,startDate,endDate){
    var c=data.c,rows=(data.contactRecords||[]).filter(function(r){
      return r.record_type==='general_phone'&&r.status==='completed'
        &&(!startDate||r.record_date>=startDate)
        &&(!endDate||r.record_date<=endDate);
    });
    if(!rows.length){toast('此期間沒有可列印的一般電話聯繫紀錄。','warn');return false;}
    rows.sort(function(a,b){return String(a.record_date).localeCompare(String(b.record_date))||new Date(a.created_at)-new Date(b.created_at);});
    var w=window.open('','_blank');if(!w){toast('瀏覽器阻擋列印視窗，請允許彈出式視窗。','warn');return false;}
    var trs=rows.map(function(r){
      var people=(data.ccPeople||[]).filter(function(x){return x.record_id===r.id;}).map(function(x){return x.name_snapshot+(x.relationship_snapshot?'（'+x.relationship_snapshot+'）':'');}).join('、');
      var phone=(data.ccPhone||[]).find(function(x){return x.record_id===r.id;});
      var content=phone?.contact_content||r.summary||'—';
      return '<tr><td>'+rocDate(r.record_date+'T00:00:00')+'</td><td>'+escapeHTML(people||'—')+'</td><td>'+escapeHTML(api.reasonLabel(r.record_type,r.reason_code))+'</td><td>'+escapeHTML(content)+'</td><td>'+escapeHTML(r.performed_by_name_snapshot||api.staffName(r.performed_by_staff_id))+'</td></tr>';
    }).join('');
    var period=(startDate&&endDate)?rocDate(startDate+'T00:00:00')+' ～ '+rocDate(endDate+'T00:00:00'):'全部期間';
    w.document.write('<!doctype html><html><head><meta charset="utf-8"><title>個案聯繫紀錄明細</title><style>@page{size:A4 portrait;margin:12mm 14mm}body{font-family:"BiauKai","DFKai-SB",serif;font-size:12px;line-height:1.45;color:#222}h1{text-align:center;font-size:19px;margin:0 0 6px}.sub{text-align:center;margin-bottom:12px}table{width:100%;border-collapse:collapse}th,td{border:1px solid #555;padding:5px 6px;vertical-align:top}th{background:#f5f5f5}</style></head><body><h1>桃園市私立劉信子居家長照機構<br>個案聯繫紀錄明細</h1><div class="sub">個案：'+escapeHTML(c.case_name)+'　期間：'+escapeHTML(period)+'</div><table><thead><tr><th>日期</th><th>聯絡對象</th><th>聯繫原因</th><th>聯繫內容／結果</th><th>聯繫人員</th></tr></thead><tbody>'+trs+'</tbody></table></body></html>');
    w.document.close();setTimeout(function(){w.focus();w.print();},250);
    return true;
  }

        if(mode==='custom'){
          start=$('ccGeneralPrintStart')?.value||'';
          end=$('ccGeneralPrintEnd')?.value||'';
          if(!start||!end){toast('請選擇開始與結束日期','warn');return false;}

        }

    });

    });

  async function ccOpenRecordDetail(c,recordId){
    showLoading(true);
    try{
      var b=await api.loadBundle(recordId),m=b.main,people=(b.people||[]).map(function(x){return x.name_snapshot+(x.relationship_snapshot?'（'+x.relationship_snapshot+'）':'');}).join('、');
      var created=b.createdFollowups||[];
      var html='<div class="case-data-grid">'+ccField('類型',api.typeLabel(m.record_type))+ccField('日期',rocDate(m.record_date+'T00:00:00'))+ccField('狀態',api.recordStatusLabel(m.status))+ccField('執行人員',m.performed_by_name_snapshot||api.staffName(m.performed_by_staff_id))+ccField('負責督導',m.supervisor_name_snapshot||api.staffName(m.supervisor_staff_id))+ccField('原因',api.reasonLabel(m.record_type,m.reason_code))+ccField('聯絡／在場對象',people||'—',true)+(m.notes?ccField('補充備註',m.notes,true):'')+'</div>'+
        ccPhoneDetailHtml(b)+ccHomeDetailHtml(b)+ccFormalDetailHtml(b)+ccProcessedFollowupsHtml(b)+
        (created.length?'<div class="import-detail-block"><strong>本次建立的追蹤事項</strong><div class="cc-follow-list" style="margin-top:8px">'+created.map(function(x){return '<div class="cc-follow-row"><strong>'+escapeHTML(x.content)+'</strong><small>'+(x.due_date?'預計 '+rocDate(x.due_date+'T00:00:00'):'未設定日期')+'｜'+escapeHTML(x.status)+'</small></div>';}).join('')+'</div></div>':'')+
        '<div class="import-detail-block"><strong>紀錄摘要</strong><div class="visit-summary">'+escapeHTML(m.summary||'—')+'</div></div>'+
        (m.status==='voided'?'<div class="import-warning"><strong>此紀錄已作廢</strong><br>原因：'+escapeHTML(m.void_reason||'—')+'</div>':'')+
        '<div class="cc-record-actions" style="margin-top:14px">'+
          (m.status!=='voided'&&api.canEditCase(c)?'<button id="ccEditRecord" class="btn">編輯紀錄</button>':'')+
          (m.status==='completed'&&m.record_type!=='general_phone'?'<button id="ccPrintRecord" class="btn">列印</button>':'')+
          (m.status!=='voided'&&api.canEditCase(c)?'<button id="ccVoidRecord" class="btn">作廢</button>':'')+
        '</div>';
      openModal({title:api.typeLabel(m.record_type)+'｜'+c.case_name,large:true,showConfirm:false,html:html});
      $('ccEditRecord')?.addEventListener('click',function(){closeModal();api.openRecordForm(c,m.record_type,m.id);});
      $('ccVoidRecord')?.addEventListener('click',function(){closeModal();api.voidRecord(c,m.id);});
      $('ccPrintRecord')?.addEventListener('click',function(){ccPrintRecord(c,b);});
    }catch(e){console.error(e);toast('讀取紀錄失敗：'+(e?.message||e),'error');}
    finally{showLoading(false);}
  }

  }
