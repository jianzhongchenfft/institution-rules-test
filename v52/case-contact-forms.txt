  function ccPageDraftKey(c,type,recordId){
    return String(c.id)+'|'+String(type)+'|'+String(recordId||'new');
  }

  function ccPageControlsSnapshot(){
    var root=$('ccPageFormBody');
    if(!root)return [];
    return [...root.querySelectorAll('input,select,textarea')].map(function(el){
      return {value:el.value,checked:!!el.checked};
    });
  }

  function ccPageCaptureDraft(){
    var s=window.__ccPageFormState,root=$('ccPageFormBody');
    if(!s||!s.active||!root)return;
    var otherPeople=[...root.querySelectorAll('[data-cc-other-row]')].map(function(row){
      return {name:row.querySelector('[data-cc-other-name]')?.value||'',relation:row.querySelector('[data-cc-other-relation]')?.value||''};
    });
    var payload=ccReadDraftPayload();
    window.__ccPageDraftStore[s.key]={
      controls:ccPageControlsSnapshot(),
      other_people:otherPeople,
      new_followups:payload.new_followups||[],
      prior_followups:payload.prior_followups||[]
    };
  }

  function ccPageRestoreDraft(draft){
    if(!draft)return false;
    var other=$('ccOtherPeople');
    if(other&&Array.isArray(draft.other_people)){
      other.innerHTML='';
      draft.other_people.forEach(function(x){ccAddOtherPersonRow({name_snapshot:x.name||'',relationship_snapshot:x.relation||''});});
    }
    var newFollow=$('ccNewFollowups');
    if(newFollow&&Array.isArray(draft.new_followups)){
      newFollow.innerHTML='';
      draft.new_followups.forEach(function(x){ccAddFollowupRow(x);});
    }
    var controls=[...($('ccPageFormBody')?.querySelectorAll('input,select,textarea')||[])];
    (draft.controls||[]).forEach(function(x,i){
      var el=controls[i];if(!el)return;
      if(['checkbox','radio'].includes(el.type))el.checked=!!x.checked;
      else el.value=x.value??'';
    });
    (draft.prior_followups||[]).forEach(function(x){
      var row=document.querySelector('[data-cc-prior-follow="'+x.id+'"]');if(!row)return;
      var result=row.querySelector('[data-cc-follow-result]');if(result)result.value=x.result||'not_followed';
      var note=row.querySelector('[data-cc-follow-note]');if(note)note.value=x.note||'';
      var next=row.querySelector('[data-cc-follow-next]');if(next)next.value=x.next_due_date||'';
    });
    return true;
  }

  function ccPageSetDirty(dirty,restored){
    var s=window.__ccPageFormState;if(!s)return;
    s.dirty=!!dirty;
    var el=$('ccPageSaveState');if(!el)return;
    el.className='cc-page-form-status'+(restored?' restored':dirty?' dirty':'');
    el.textContent=restored?'已恢復未送出內容':dirty?'尚有未儲存變更':'目前內容尚未變更';
  }

  function ccPageMarkDirty(){
    var s=window.__ccPageFormState;if(!s||!s.active)return;
    ccPageSetDirty(true,false);
    ccPageCaptureDraft();
  }

  function ccPageClearDraft(){
    var s=window.__ccPageFormState;if(!s)return;
    delete window.__ccPageDraftStore[s.key];
    s.dirty=false;s.active=false;
  }

  function ccPageConfirmLeave(){
    var s=window.__ccPageFormState;
    if(!s||!s.active||!s.dirty)return true;
    ccPageCaptureDraft();
    return window.confirm('目前有尚未儲存的內容。\n\n離開後內容會暫存在目前瀏覽工作階段，重新整理網頁後暫存會清除。確定離開嗎？');
  }

  function ccPageLeaveToCase(c){
    if(!ccPageConfirmLeave())return;
    if(window.__ccPageFormState)window.__ccPageFormState.active=false;
    if(typeof window.openCaseDetailPage==='function')window.openCaseDetailPage(c.id,'visits',true);
    else navigate('cases');
  }

  function ccBindPageDraft(c,type,recordId){
    var key=ccPageDraftKey(c,type,recordId);
    window.__ccPageFormState={active:true,dirty:false,key:key,caseId:c.id,type:type,recordId:recordId||null};
    var root=$('ccPageFormBody');
    if(!root)return;
    var restored=ccPageRestoreDraft(window.__ccPageDraftStore[key]);
    ccToggleForm(type);
    ccPageSetDirty(restored,restored);
    root.addEventListener('input',ccPageMarkDirty);
    root.addEventListener('change',ccPageMarkDirty);
    root.addEventListener('click',function(e){
      if(e.target.closest?.('#ccAddFollowup,#ccAddOtherPerson,[data-cc-remove-follow],[data-cc-remove-other],#ccNotAskedAll,#ccBuildSummary')){
        setTimeout(ccPageMarkDirty,0);
      }
    });
    $('ccPageBack')?.addEventListener('click',function(){ccPageLeaveToCase(c);});
    $('ccPageCancel')?.addEventListener('click',function(){ccPageLeaveToCase(c);});
    $('ccPageSaveDraft')?.addEventListener('click',function(){ccSaveFromForm(type,'draft');});
    $('ccPageSubmit')?.addEventListener('click',function(){ccSaveFromForm(type,'completed');});
  }

  function ccDefaultTime(){
    var now=new Date(),h=now.getHours(),m=Math.floor(now.getMinutes()/10)*10;
    return String(h).padStart(2,'0')+':'+String(m).padStart(2,'0');
  }
  function ccTimeSelectHtml(value){
    var v=String(value||ccDefaultTime()),parts=v.split(':'),h=parts[0]||'00',m=parts[1]||'00';
    var hours=Array.from({length:24},function(_,i){return String(i).padStart(2,'0');});
    var mins=['00','10','20','30','40','50'];
    if(!mins.includes(m))m=String(Math.floor(Number(m||0)/10)*10).padStart(2,'0');
    return '<div class="cc-time-selects">'+
      '<select id="ccRecordHour">'+hours.map(function(x){return ccOpt(x,x,h);}).join('')+'</select>'+
      '<span class="cc-time-sep">時</span>'+
      '<select id="ccRecordMinute">'+mins.map(function(x){return ccOpt(x,x,m);}).join('')+'</select>'+
      '<span class="cc-time-sep">分</span>'+
    '</div>';
  }
  function ccReadTime(){
    var h=$('ccRecordHour')?.value,m=$('ccRecordMinute')?.value;
    return h!=null&&m!=null?h+':'+m:'';
  }

  async function ccLoadCaseContacts(caseId){
    var r=await db.from('case_contacts').select('*').eq('case_id',caseId).order('is_primary_contact',{ascending:false}).order('created_at');
    if(r.error)throw r.error;return r.data||[];
  }

  async function ccLoadBundle(recordId){
    var main=await db.from('case_contact_records').select('*').eq('id',recordId).maybeSingle();
    if(main.error)throw main.error;if(!main.data)throw new Error('找不到紀錄');
    var results=await Promise.all([
      db.from('case_contact_record_people').select('*').eq('record_id',recordId).order('is_primary',{ascending:false}).order('created_at'),
      db.from('case_contact_formal_details').select('*').eq('record_id',recordId).maybeSingle(),
      db.from('case_home_visit_details').select('*').eq('record_id',recordId).maybeSingle(),
      db.from('case_phone_details').select('*').eq('record_id',recordId).maybeSingle(),
      db.from('case_contact_satisfaction').select('*').eq('record_id',recordId),
      db.from('case_record_links').select('*').eq('from_id',recordId),
      db.from('case_followup_items').select('*').eq('source_id',recordId).order('created_at'),
      db.from('case_followup_updates').select('*').eq('source_id',recordId).order('created_at')
    ]);
    var bad=results.find(function(x){return x.error;});if(bad&&bad.error)throw bad.error;
    return {main:main.data,people:results[0].data||[],formal:results[1].data||null,home:results[2].data||null,phone:results[3].data||null,satisfaction:results[4].data||[],links:results[5].data||[],createdFollowups:results[6].data||[],processedFollowups:results[7].data||[]};
  }

  function ccBuildSummary(type){
    var date=$('ccRecordDate')?.value||'',reason=api.reasonLabel(type,$('ccReason')?.value),parts=[];
    if(type==='home_visit')parts.push((date?rocDate(date+'T00:00:00'):'')+'進行'+reason+'。');
    else if(type==='formal_phone')parts.push((date?rocDate(date+'T00:00:00'):'')+'進行'+reason+'。');
    else parts.push((date?rocDate(date+'T00:00:00'):'')+'進行電話聯繫，原因為'+reason+'。');
    if(type!=='home_visit'){
      if($('ccContactSuccess')?.value==='false')parts.push('本次未成功聯繫'+($('ccNeedsRecontact')?.value==='true'?'，後續將再次聯繫。':'。'));
      else parts.push('本次已成功聯繫。');
    }
    if(type!=='general_phone'&&$('ccContactSuccess')?.value!=='false'){
      var su=$('ccServiceUsage')?.value;
      if(su==='normal')parts.push('目前居家服務依安排正常使用。');
      else if(su)parts.push('目前服務使用情形有需留意事項'+($('ccServiceUsageNote')?.value.trim()?'，'+$('ccServiceUsageNote').value.trim():'')+'。');
      if($('ccMajorEvent')?.value==='true'&&$('ccMajorEventNote')?.value.trim())parts.push('近期重大事件：'+$('ccMajorEventNote').value.trim()+'。');
      if($('ccConcern')?.value==='true'&&$('ccConcernNote')?.value.trim())parts.push('個案／家屬主要關切：'+$('ccConcernNote').value.trim()+'。');
      if($('ccRequiresAction')?.value==='true'&&$('ccActionNote')?.value.trim())parts.push('後續處理：'+$('ccActionNote').value.trim()+'。');
    }
    var txt=$('ccContactContent')?.value.trim();if(txt)parts.push(txt+(txt.endsWith('。')?'':'。'));
    return parts.join('');
  }

  function ccReadPeople(type,c,contacts){
    var rows=[];
    if(type==='home_visit'){
      if($('ccClientPresent')?.checked)rows.push({person_type:'client',name_snapshot:c.case_name,relationship_snapshot:'本人',is_primary:rows.length===0});
      document.querySelectorAll('[data-cc-contact-id]:checked').forEach(function(el){
        var x=contacts.find(function(z){return z.id===el.dataset.ccContactId;});if(x)rows.push({person_type:x.is_primary_caregiver?'primary_caregiver':'family',case_contact_id:x.id,name_snapshot:x.contact_name,relationship_snapshot:x.relationship||null,is_primary:rows.length===0});
      });
      document.querySelectorAll('[data-cc-other-row]').forEach(function(row){
        var name=row.querySelector('[data-cc-other-name]')?.value.trim()||'';
        var relation=row.querySelector('[data-cc-other-relation]')?.value.trim()||'';
        if(name)rows.push({person_type:'other',name_snapshot:name,relationship_snapshot:relation||null,is_primary:rows.length===0});
      });
    }else{
      var v=$('ccPhonePerson')?.value||'client';
      if(v==='client')rows.push({person_type:'client',name_snapshot:c.case_name,relationship_snapshot:'本人',is_primary:true});
      else if(v.indexOf('contact:')===0){
        var id=v.split(':')[1],x=contacts.find(function(z){return z.id===id;});
        if(x)rows.push({person_type:x.is_primary_caregiver?'primary_caregiver':'family',case_contact_id:x.id,name_snapshot:x.contact_name,relationship_snapshot:x.relationship||null,is_primary:true});
      }else{
        var name=$('ccPhoneOtherName')?.value.trim();if(name)rows.push({person_type:'other',name_snapshot:name,relationship_snapshot:null,is_primary:true});
      }
    }
    return rows;
  }

  function ccValidate(type,status,contacts,c){
    if(!$('ccRecordDate')?.value)return '請填寫日期';
    var visitTime=ccReadTime();
    if(!visitTime)return '請選擇時間';
    if(!$('ccReason')?.value)return '請選擇原因';
    if($('ccReason')?.value==='other'&&!$('ccReasonNote')?.value.trim())return '選擇其他原因時請補充說明';
    var people=ccReadPeople(type,c,contacts);if(!people.length)return type==='home_visit'?'請至少選擇一名在場／受訪對象':'請選擇聯絡對象';
    if(status==='draft')return null;
    if(type!=='home_visit'&&$('ccContactSuccess')?.value==='false'){
      if(!$('ccFailureReason')?.value)return '請選擇未成功聯繫原因';
      if($('ccFailureReason')?.value==='other'&&!$('ccFailureNote')?.value.trim())return '選擇其他未成功原因時請補充說明';
      if($('ccNeedsRecontact')?.value==='true'&&!$('ccRecontactDate')?.value)return '請填寫預計再次聯繫日期';
      return null;
    }
    if(type==='general_phone'){
      if(!$('ccContactContent')?.value.trim())return '請填寫聯繫內容／結果';
      return null;
    }
    if(!$('ccServiceUsage')?.value)return '請選擇服務使用情形';
    if($('ccServiceUsage')?.value!=='normal'&&!$('ccServiceUsageNote')?.value.trim())return '服務使用非正常時請填寫狀況說明';
    if(!$('ccServiceFit')?.value)return '請選擇現行服務是否符合需求';
    if($('ccMajorEvent')?.value==='true'){
      if(!document.querySelectorAll('[data-cc-major]:checked').length)return '請選擇重大事件類型';
      if(!$('ccMajorEventNote')?.value.trim())return '請簡述近期重大事件';
    }
    if($('ccConcern')?.value==='true'&&!$('ccConcernNote')?.value.trim())return '請填寫個案／家屬主要關切事項';
    var sats=[...document.querySelectorAll('[data-cc-sat]')];
    if(sats.some(function(x){return !x.value;}))return '請完成六項服務滿意度；若本次未詢問可使用「本次未進行滿意度調查」';
    for(var sat of sats){
      if(['dissatisfied','very_dissatisfied'].includes(sat.value)){
        var note=document.querySelector('[data-cc-sat-note="'+sat.dataset.ccSat+'"]')?.value.trim();
        if(!note)return '選擇「不滿意」或「非常不滿意」時，請填寫該項原因';
      }
    }
    if($('ccAssessmentAction')?.value==='completed'&&!$('ccAssessmentLink')?.value)return '請選擇已完成的評估紀錄';
    if(type==='home_visit'){
      if(!$('ccScheduleExec')?.value||!$('ccServiceExec')?.value)return '請完成家訪現場服務執行情形';
      if($('ccExecDifficulty')?.value==='true'&&!$('ccExecDifficultyNote')?.value.trim())return '請填寫服務執行困難原因';
    }
    return null;
  }

  async function ccSaveHistoryIfNeeded(bundle){
    if(!bundle||bundle.main.status!=='completed')return;
    var r=await db.from('case_contact_record_history').select('version_no').eq('record_id',bundle.main.id).order('version_no',{ascending:false}).limit(1);
    if(r.error)throw r.error;
    var version=(r.data&&r.data[0]?r.data[0].version_no:0)+1;
    var ins=await db.from('case_contact_record_history').insert({record_id:bundle.main.id,version_no:version,snapshot:bundle,changed_by_staff_id:state.profile.id});
    if(ins.error)throw ins.error;
  }

  async function ccProcessPriorFollowups(recordId){
    var rows=[...document.querySelectorAll('[data-cc-prior-follow]')];
    for(var row of rows){
      var result=row.querySelector('[data-cc-follow-result]')?.value||'not_followed';
      if(result==='not_followed')continue;
      var itemId=row.dataset.ccPriorFollow,note=row.querySelector('[data-cc-follow-note]')?.value.trim()||null,next=row.querySelector('[data-cc-follow-next]')?.value||null;
      var ins=await db.from('case_followup_updates').insert({followup_item_id:itemId,update_date:$('ccRecordDate').value,result_code:result,note:note,next_due_date:next,source_type:'case_contact_record',source_id:recordId,updated_by_staff_id:state.profile.id});
      if(ins.error)throw ins.error;
      var status=['completed','improved'].includes(result)?'completed':result==='close_report'?'closed':'in_progress';
      var payload={status:status,updated_at:new Date().toISOString()};
      if(status==='completed'||status==='closed'){payload.completed_at=new Date().toISOString();payload.result=note||ccFollowResultLabel(result);}
      else payload.due_date=next||null;
      var up=await db.from('case_followup_items').update(payload).eq('id',itemId);if(up.error)throw up.error;
    }
  }

  async function ccInsertNewFollowups(c,recordId,type){
    var rows=[...document.querySelectorAll('[data-cc-new-follow-content]')].map(function(input){
      var wrap=input.closest('.cc-follow-row');
      return {content:input.value.trim(),due_date:wrap.querySelector('[data-cc-new-follow-date]')?.value||null,responsible_staff_id:wrap.querySelector('[data-cc-new-follow-staff]')?.value||state.profile.id};
    }).filter(function(x){return x.content;});
    if($('ccAssessmentAction')?.value==='schedule_later')rows.push({content:'安排評估',due_date:null,responsible_staff_id:state.profile.id});
    if(!rows.length)return;
    var payload=rows.map(function(x){return {case_id:c.id,content:x.content,due_date:x.due_date,responsible_staff_id:x.responsible_staff_id,status:'pending',source_type:type,source_id:recordId,created_by_staff_id:state.profile.id};});
    var r=await db.from('case_followup_items').insert(payload);if(r.error)throw r.error;
  }
