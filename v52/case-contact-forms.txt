  async function ccLoadCaseContacts(caseId){
    var r=await db.from('case_contacts').select('*').eq('case_id',caseId).order('is_primary_contact',{ascending:false}).order('created_at');

  async function ccLoadBundle(recordId){
    var main=await db.from('case_contact_records').select('*').eq('id',recordId).maybeSingle();
    if(main.error)throw main.error;if(!main.data)throw new Error('找不到紀錄');

      db.from('case_contact_record_people').select('*').eq('record_id',recordId).order('is_primary',{ascending:false}).order('created_at'),

  function ccToggleForm(type){
    function toggle(id,on){var el=$(id);if(el)el.style.display=on?'':'none';}
    var reason=$('ccReason');toggle('ccReasonNoteWrap',reason&&reason.value==='other');
    if(type!=='general_phone'){
      toggle('ccServiceUsageNoteWrap',$('ccServiceUsage')?.value!=='normal');
      toggle('ccAdjustWrap',$('ccServiceFit')?.value!=='fit');
      toggle('ccMajorEventWrap',$('ccMajorEvent')?.value==='true');
      toggle('ccConcernWrap',$('ccConcern')?.value==='true');
      toggle('ccAssessmentLinkWrap',$('ccAssessmentAction')?.value==='completed');
      toggle('ccActionWrap',$('ccRequiresAction')?.value==='true');
      var badSat=[...document.querySelectorAll('[data-cc-sat]')].some(function(x){return ['neutral','dissatisfied','very_dissatisfied'].includes(x.value);});
      toggle('ccSatNoteWrap',badSat);
    }
    if(type==='home_visit'){
      toggle('ccVisitLocationNoteWrap',$('ccVisitLocation')?.value==='other');
      toggle('ccExecDifficultyWrap',$('ccExecDifficulty')?.value==='true');
    }else{
      var success=$('ccContactSuccess')?.value!=='false';
      toggle('ccFailureReasonWrap',!success);
      toggle('ccRecontactWrap',!success);
      toggle('ccRecontactDateWrap',!success&&$('ccNeedsRecontact')?.value==='true');
      toggle('ccPhoneOtherWrap',$('ccPhonePerson')?.value==='other');
      var formalBody=$('ccFormalBody');if(formalBody)formalBody.style.display=success?'':'none';
    }
  }

  function ccBuildSummary(type){
    var date=$('ccRecordDate')?.value||'',reason=api.reasonLabel(type,$('ccReason')?.value),parts=[];
    if(type==='home_visit')parts.push((date?rocDate(date+'T00:00:00'):'')+'進行'+reason+'。');
    else if(type==='formal_phone')parts.push((date?rocDate(date+'T00:00:00'):'')+'進行'+reason+'。');
    else parts.push((date?rocDate(date+'T00:00:00'):'')+'進行電話聯繫，原因為'+reason+'。');
    if(type!=='home_visit'){
      if($('ccContactSuccess')?.value==='false')parts.push('本次未成功聯繫'+($('ccNeedsRecontact')?.value==='true'?'，後續將再次聯繫。':'。'));
      else parts.push('本次已成功聯繫。');
    }
    if(type!=='general_phone'&&$('ccContactSuccess')?.value!=='false'){
      var su=$('ccServiceUsage')?.value;
      if(su==='normal')parts.push('目前居家服務依安排正常使用。');
      else if(su)parts.push('目前服務使用情形有需留意事項'+($('ccServiceUsageNote')?.value.trim()?'，'+$('ccServiceUsageNote').value.trim():'')+'。');
      if($('ccMajorEvent')?.value==='true'&&$('ccMajorEventNote')?.value.trim())parts.push('近期重大事件：'+$('ccMajorEventNote').value.trim()+'。');
      if($('ccConcern')?.value==='true'&&$('ccConcernNote')?.value.trim())parts.push('個案／家屬主要關切：'+$('ccConcernNote').value.trim()+'。');
      if($('ccRequiresAction')?.value==='true'&&$('ccActionNote')?.value.trim())parts.push('後續處理：'+$('ccActionNote').value.trim()+'。');
    }
    var txt=$('ccContactContent')?.value.trim();if(txt)parts.push(txt+(txt.endsWith('。')?'':'。'));
    return parts.join('');
  }

  function ccReadDraftPayload(){
    var newFollowups=[...document.querySelectorAll('[data-cc-new-follow-content]')].map(function(input){
      var wrap=input.closest('.cc-follow-row');
      return {content:input.value.trim(),due_date:wrap.querySelector('[data-cc-new-follow-date]')?.value||null,responsible_staff_id:wrap.querySelector('[data-cc-new-follow-staff]')?.value||state.profile.id};
    }).filter(function(x){return x.content;});
    var prior=[...document.querySelectorAll('[data-cc-prior-follow]')].map(function(row){
      return {id:row.dataset.ccPriorFollow,result:row.querySelector('[data-cc-follow-result]')?.value||'not_followed',note:row.querySelector('[data-cc-follow-note]')?.value||'',next_due_date:row.querySelector('[data-cc-follow-next]')?.value||null};
    });
    return {new_followups:newFollowups,prior_followups:prior};
  }
  function ccApplyDraftPayload(payload){
    if(!payload)return;
    (payload.new_followups||[]).forEach(function(x){ccAddFollowupRow(x);});
    (payload.prior_followups||[]).forEach(function(x){
      var row=document.querySelector('[data-cc-prior-follow="'+x.id+'"]');if(!row)return;
      var result=row.querySelector('[data-cc-follow-result]');if(result)result.value=x.result||'not_followed';
      var note=row.querySelector('[data-cc-follow-note]');if(note)note.value=x.note||'';
      var next=row.querySelector('[data-cc-follow-next]');if(next)next.value=x.next_due_date||'';
    });
  }

  function ccReadPeople(type,c,contacts){
    var rows=[];
    if(type==='home_visit'){
      if($('ccClientPresent')?.checked)rows.push({person_type:'client',name_snapshot:c.case_name,relationship_snapshot:'本人',is_primary:rows.length===0});
      document.querySelectorAll('[data-cc-contact-id]:checked').forEach(function(el){
        var x=contacts.find(function(z){return z.id===el.dataset.ccContactId;});if(x)rows.push({person_type:x.is_primary_caregiver?'primary_caregiver':'family',case_contact_id:x.id,name_snapshot:x.contact_name,relationship_snapshot:x.relationship||null,is_primary:rows.length===0});
      });
      var other=$('ccOtherPerson')?.value.trim();if(other)rows.push({person_type:'other',name_snapshot:other,relationship_snapshot:$('ccOtherRelation')?.value.trim()||null,is_primary:rows.length===0});
    }else{
      var v=$('ccPhonePerson')?.value||'client';
      if(v==='client')rows.push({person_type:'client',name_snapshot:c.case_name,relationship_snapshot:'本人',is_primary:true});
      else if(v.indexOf('contact:')===0){
        var id=v.split(':')[1],x=contacts.find(function(z){return z.id===id;});
        if(x)rows.push({person_type:x.is_primary_caregiver?'primary_caregiver':'family',case_contact_id:x.id,name_snapshot:x.contact_name,relationship_snapshot:x.relationship||null,is_primary:true});
      }else{
        var name=$('ccPhoneOtherName')?.value.trim();if(name)rows.push({person_type:'other',name_snapshot:name,relationship_snapshot:null,is_primary:true});
      }
    }
    return rows;
  }

  function ccValidate(type,status,contacts,c){
    if(!$('ccRecordDate')?.value)return '請填寫日期';
    if(!$('ccReason')?.value)return '請選擇原因';
    if($('ccReason')?.value==='other'&&!$('ccReasonNote')?.value.trim())return '選擇其他原因時請補充說明';
    var people=ccReadPeople(type,c,contacts);if(!people.length)return type==='home_visit'?'請至少選擇一名在場／受訪對象':'請選擇聯絡對象';
    if(status==='draft')return null;
    if(type!=='home_visit'&&$('ccContactSuccess')?.value==='false'){
      if(!$('ccFailureReason')?.value)return '請選擇未成功聯繫原因';
      if($('ccNeedsRecontact')?.value==='true'&&!$('ccRecontactDate')?.value)return '請填寫預計再次聯繫日期';
      return null;
    }
    if(type==='general_phone'){
      if(!$('ccContactContent')?.value.trim())return '請填寫聯繫內容／結果';
      return null;
    }
    if(!$('ccServiceUsage')?.value)return '請選擇服務使用情形';
    if($('ccServiceUsage')?.value!=='normal'&&!$('ccServiceUsageNote')?.value.trim())return '服務使用非正常時請填寫狀況說明';
    if(!$('ccServiceFit')?.value)return '請選擇現行服務是否符合需求';
    if($('ccMajorEvent')?.value==='true'){
      if(!document.querySelectorAll('[data-cc-major]:checked').length)return '請選擇重大事件類型';
      if(!$('ccMajorEventNote')?.value.trim())return '請簡述近期重大事件';
    }
    if($('ccConcern')?.value==='true'&&!$('ccConcernNote')?.value.trim())return '請填寫個案／家屬主要關切事項';
    var sats=[...document.querySelectorAll('[data-cc-sat]')];
    if(sats.some(function(x){return !x.value;}))return '請完成六項服務滿意度；若本次未詢問可使用「本次未進行滿意度調查」';
    if(sats.some(function(x){return ['dissatisfied','very_dissatisfied'].includes(x.value);})&&!$('ccSatNote')?.value.trim())return '滿意度有不滿意項目時，請填寫服務意見／補充說明';
    if($('ccAssessmentAction')?.value==='completed'&&!$('ccAssessmentLink')?.value)return '請選擇已完成的評估紀錄';
    if(type==='home_visit'){
      if(!$('ccScheduleExec')?.value||!$('ccServiceExec')?.value)return '請完成家訪現場服務執行情形';
      if($('ccExecDifficulty')?.value==='true'&&!$('ccExecDifficultyNote')?.value.trim())return '請填寫服務執行困難原因';
    }
    return null;
  }

  async function ccSaveHistoryIfNeeded(bundle){
    if(!bundle||bundle.main.status!=='completed')return;
    var r=await db.from('case_contact_record_history').select('version_no').eq('record_id',bundle.main.id).order('version_no',{ascending:false}).limit(1);
    if(r.error)throw r.error;
    var version=(r.data&&r.data[0]?r.data[0].version_no:0)+1;
    var ins=await db.from('case_contact_record_history').insert({record_id:bundle.main.id,version_no:version,snapshot:bundle,changed_by_staff_id:state.profile.id});
    if(ins.error)throw ins.error;
  }

  async function ccProcessPriorFollowups(recordId){
    var rows=[...document.querySelectorAll('[data-cc-prior-follow]')];
    for(var row of rows){
      var result=row.querySelector('[data-cc-follow-result]')?.value||'not_followed';
      if(result==='not_followed')continue;
      var itemId=row.dataset.ccPriorFollow,note=row.querySelector('[data-cc-follow-note]')?.value.trim()||null,next=row.querySelector('[data-cc-follow-next]')?.value||null;
      var ins=await db.from('case_followup_updates').insert({followup_item_id:itemId,update_date:$('ccRecordDate').value,result_code:result,note:note,next_due_date:next,source_type:'case_contact_record',source_id:recordId,updated_by_staff_id:state.profile.id});
      if(ins.error)throw ins.error;
      var status=['completed','improved'].includes(result)?'completed':result==='close_report'?'closed':'in_progress';
      var payload={status:status,updated_at:new Date().toISOString()};
      if(status==='completed'||status==='closed'){payload.completed_at=new Date().toISOString();payload.result=note||ccFollowResultLabel(result);}
      else payload.due_date=next||null;
      var up=await db.from('case_followup_items').update(payload).eq('id',itemId);if(up.error)throw up.error;
    }
  }
