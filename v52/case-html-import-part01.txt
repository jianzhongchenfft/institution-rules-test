  function htmlCleanText(value=''){
    return String(value).replace(/\u00a0/g,' ').replace(/[ \t]+/g,' ').replace(/\r/g,'').trim();
  }
  function htmlRocToIso(value){
    const s=htmlCleanText(value);
    if(!s)return '';
    let m=s.match(/(\d{1,4})\s*[年\/\.\-]\s*(\d{1,2})\s*[月\/\.\-]\s*(\d{1,2})/);
    if(!m)return '';
    let y=Number(m[1]); if(y<1911)y+=1911;
    const mo=String(Number(m[2])).padStart(2,'0'), d=String(Number(m[3])).padStart(2,'0');
    return `${y}-${mo}-${d}`;
  }
  function htmlNum(value){
    const m=String(value||'').replace(/,/g,'').match(/-?\d+(?:\.\d+)?/); return m?Number(m[0]):null;
  }
