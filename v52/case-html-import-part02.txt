        if(cells[i].tagName!=='TH')continue;
        let td=null; for(let j=i+1;j<cells.length;j++){if(cells[j].tagName==='TH')break;if(cells[j].tagName==='TD'){td=cells[j];break;}}
        if(td)pairs.push({label:htmlCleanText(cells[i].textContent).replace(/^＊/,'').trim(),value:htmlCleanText(td.textContent)});
      }
    });
    return pairs;
  }
  function htmlFindPair(pairs, labels){
    for(const target of labels){
      const exact=pairs.find(x=>x.label===target); if(exact?.value)return exact.value;
      const fuzzy=pairs.find(x=>x.label.includes(target)); if(fuzzy?.value)return fuzzy.value;
    }
    return '';
  }
  function htmlLabeledBlock(doc,label){
    const th=[...doc.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes(label));
    if(!th)return '';
    const td=th.nextElementSibling?.tagName==='TD'?th.nextElementSibling:th.parentElement?.querySelector('td');
    return htmlCleanText(td?.textContent||'');
  }
  function htmlSectionText(text, startLabel, nextLabels=[]){
    const start=text.indexOf(startLabel); if(start<0)return '';
    let end=text.length;
    nextLabels.forEach(l=>{const i=text.indexOf(l,start+startLabel.length);if(i>=0&&i<end)end=i;});
    return htmlCleanText(text.slice(start+startLabel.length,end)).replace(/^\s*[：:]\s*/,'');
  }
  function htmlFullContactName(raw, role){
    const re=role==='primary'
      ?/(?:主要照顧者|案妻為主要照顧者)[：:]?\s*([^\n。]+)/
      :/(?:次要照顧者)[：:]?\s*([^\n。]+)/;
    const m=raw.match(re); if(!m)return '';
    const line=m[1];
    const n=line.match(/[-－]\s*([^\(（\s，,]+)\s*[\(（]/) || line.match(/^([^\(（，,]+)\s*[\(（]/);
    return htmlCleanText(n?.[1]||'').replace(/^案(?:妻|夫|女|子|妹|兄|姐|弟)[-－]?/,'');
  }
