  }
  function htmlServiceGroup(code){
    code=String(code||'').toUpperCase();
    if(code.startsWith('BA'))return 'B'; if(code.startsWith('CA'))return 'C'; if(code.startsWith('DA'))return 'D';
    if(code.startsWith('EA')||code.startsWith('FA'))return 'EF'; if(code.startsWith('GA'))return 'G'; if(code.startsWith('SC'))return 'S';
    return code.replace(/\d.*$/,'')||'OTHER';
  }
  function htmlParseServices(doc){
    const out=[]; const seen=new Set();
    doc.querySelectorAll('table').forEach(table=>{
      const headers=[...table.querySelectorAll('thead th')].map(x=>htmlCleanText(x.textContent));
      if(!headers.some(x=>x.includes('服務項目'))||!headers.some(x=>x.includes('服務數量')))return;
      table.querySelectorAll('tbody tr').forEach(tr=>{
        const cells=[...tr.querySelectorAll(':scope > td')].map(x=>htmlCleanText(x.textContent));
        if(cells.length<4)return;
        const serviceText=cells[1]||'';
        const m=serviceText.match(/([A-Z]{1,3}\d+(?:-\d+)?)\s*(?:\[([^\]]+)\])?/i); if(!m)return;
        const code=m[1].toUpperCase(), key=code+'|'+(cells[3]||''); if(seen.has(key))return; seen.add(key);
        out.push({service_group:htmlServiceGroup(code),service_code:code,service_name:htmlCleanText(m[2]||serviceText.replace(m[1],'')),unit_price:htmlNum(cells[2]),approved_quantity:htmlNum(cells[3]),subtotal:htmlNum(cells[4]),valid_from:'',valid_to:''});
      });
    });
    return out;
  }

    }
    if(services.some(x=>x.service_group==='G')){
      const panels=[...doc.querySelectorAll('.panel, .panel-body, div')].filter(x=>/居家喘息服務|喘息服務\(G\)/.test(htmlCleanText(x.textContent)));
