  function htmlParseDocument(source){
    const doc=new DOMParser().parseFromString(source,'text/html');
    const pairs=htmlTablePairs(doc); const allText=htmlCleanText(doc.body?.textContent||'');
    const headMatch=allText.match(/個案姓名\/身分證字號[：:]\s*([^\/\n]+)\/([A-Z]\d{9})/i);
    let caseName=htmlFindPair(pairs,['個案姓名']) || htmlCleanText(headMatch?.[1]||'');
    let nationalId=htmlFindPair(pairs,['個案身分證']) || htmlCleanText(headMatch?.[2]||'');
    nationalId=(nationalId.match(/[A-Z]\d{9}/i)||[])[0]||nationalId;
    const birth=htmlFindPair(pairs,['個案生日','出生日期']);
    const phone=(htmlFindPair(pairs,['個案電話']).match(/0\d[\d\-\s]{7,14}/)||[])[0]||'';
    let gender=(pairs.find(x=>x.label==='性別'||x.label==='個案性別')?.value||'').replace(/^\d+\.?\s*/,''); if(!/^(男|女)$/.test(gender))gender='';
    const address=htmlFindPair(pairs,['居住(通訊)地址','居住地址','通訊地址','現居地址']);
    let cms=(allText.match(/CMS\s*(?:第)?\s*([1-8])\s*級/i)||[])[1]||htmlFindPair(pairs,['CMS等級','失能等級']).match(/[1-8]/)?.[0]||'';
    let aUnit=htmlFindPair(pairs,['A單位名稱']);
    let manager=htmlFindPair(pairs,['A個管姓名']);
    const managerTh=[...doc.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes('A個管姓名'));
    const managerTable=managerTh?.closest('table');
    const managerPhoneTh=managerTable?[...managerTable.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes('聯絡電話')):null;
    const managerPhoneCell=managerPhoneTh?.nextElementSibling?.tagName==='TD'?managerPhoneTh.nextElementSibling:managerPhoneTh?.parentElement?.querySelector('td');
    let managerPhone=(htmlCleanText(managerPhoneCell?.textContent||'').match(/0\d[\d#\-\s]{7,20}/)||[])[0]||'';
    let assessor=htmlFindPair(pairs,['承辦人']);
    const topAssessor=(allText.match(/評估照專[：:]\s*([^\n]+)/)||[])[1]; if(topAssessor)assessor=htmlCleanText(topAssessor);
    const assessment=htmlRocToIso(htmlFindPair(pairs,['處理時間','評估日期']));
    const approval=htmlRocToIso(htmlFindPair(pairs,['近一次照顧計畫送審日期','審核日期']));
    const identityType=htmlFindPair(pairs,['長照資格','補助身分','身分類別']);
    let livesAlone=null; const living=htmlFindPair(pairs,['目前居住狀況','居住狀況','居住型態']); if(living){if(/獨居/.test(living))livesAlone=true;else if(/同住|與.+同住/.test(living))livesAlone=false;}
    const carePlan=htmlParseCarePlan(doc,allText);
    const services=htmlParseServices(doc); const budgets=htmlParseBudgets(doc,services);
    const contacts=htmlParseContacts(doc,pairs,carePlan.raw_plan_text||allText);
    const ca=(source.match(/ca110id=(\d+)/)||[])[1]||'';
    return {case_name:caseName,national_id:nationalId.toUpperCase(),birth_date:htmlRocToIso(birth),gender,address,phone:phone.replace(/\s/g,''),lives_alone:livesAlone,cms_level:cms,identity_type:identityType,assessment_date:assessment,plan_approval_date:approval,a_unit_name:aUnit,case_manager_name:manager,case_manager_phone:managerPhone.replace(/\s/g,''),assessor_name:assessor,source_ca110_id:ca,contacts,services,budgets,care_plan:carePlan,warnings:[]};
  }
  async function htmlSha256(file){
    const b=await file.arrayBuffer(); const h=await crypto.subtle.digest('SHA-256',b); return [...new Uint8Array(h)].map(x=>x.toString(16).padStart(2,'0')).join('');
  }
