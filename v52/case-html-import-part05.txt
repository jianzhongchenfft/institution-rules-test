  async function htmlResolveMatch(d){
    d._match=null; d._potential=[]; d._caseNoConflict=null;
    const byNo=await db.from('care_cases').select('id,case_no,case_name,national_id,birth_date,supervisor_id').ilike('case_no',d.case_no).limit(2); if(byNo.error)throw byNo.error;
    if(byNo.data?.[0])d._caseNoConflict=byNo.data[0];
    if(d.national_id){const r=await db.from('care_cases').select('id,case_no,case_name,national_id,birth_date,supervisor_id').ilike('national_id',d.national_id).limit(2);if(r.error)throw r.error;if(r.data?.[0])d._match=r.data[0];}
    if(!d._match && d.case_name && d.birth_date){const r=await db.from('care_cases').select('id,case_no,case_name,national_id,birth_date,supervisor_id').eq('case_name',d.case_name).eq('birth_date',d.birth_date).limit(5);if(r.error)throw r.error;d._potential=r.data||[];}
    if(!d._match && d._caseNoConflict)d._match=d._caseNoConflict;
  }
