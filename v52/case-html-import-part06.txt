  async function htmlSavePreview(d,exactMatchId){
    const caseNo=$('hiCaseNo').value.trim(), name=$('hiName').value.trim(); const supervisor=state.profile?.role==='supervisor'?state.profile.id:$('hiSupervisor').value;
    if(!caseNo||!name||!supervisor){toast('個案編號、姓名與負責督導為必填','warn');return false;}

    const file=d._file; showLoading(true); let path='';
    try{
      const safeNo=caseNo.replace(/[^A-Za-z0-9_-]/g,'_'); const ext=/\.htm$/i.test(file.name)?'htm':'html'; path=`${safeNo}/${Date.now()}-${crypto.randomUUID()}.${ext}`;

      if(d.lives_alone!==null)payload.lives_alone=d.lives_alone;
      const r=await db.rpc('import_case_html_data',{payload}); if(r.error)throw r.error;
      if(r.data?.action==='duplicate'){await db.storage.from(CASE_HTML_BUCKET).remove([path]);closeModal();toast('這份 HTML 已經匯入過，未重複建立資料。','warn');navigate('cases');return true;}
      closeModal();toast(r.data?.action==='updated'?'既有個案已更新':'個案已建立','success');navigate('cases');return true;
    }catch(err){if(path)try{await db.storage.from(CASE_HTML_BUCKET).remove([path]);}catch(_){ }console.error(err);toast('HTML 匯入失敗：'+(err?.message||err),'error');return false;}finally{showLoading(false);}
  }

        db.from('case_approved_services').select('*').eq('case_id',caseId).eq('is_current',true).order('service_group').order('service_code'),
