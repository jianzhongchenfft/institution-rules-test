  function htmlCleanText(value=''){
    return String(value).replace(/\u00a0/g,' ').replace(/[ \t]+/g,' ').replace(/\r/g,'').trim();
  }
  function htmlRocToIso(value){
    const s=htmlCleanText(value);
    if(!s)return '';
    let m=s.match(/(\d{1,4})\s*[年\/\.\-]\s*(\d{1,2})\s*[月\/\.\-]\s*(\d{1,2})/);
    if(!m)return '';
    let y=Number(m[1]); if(y<1911)y+=1911;
    const mo=String(Number(m[2])).padStart(2,'0'), d=String(Number(m[3])).padStart(2,'0');
    return `${y}-${mo}-${d}`;
  }
  function htmlNum(value){
    const m=String(value||'').replace(/,/g,'').match(/-?\d+(?:\.\d+)?/); return m?Number(m[0]):null;
  }
  function htmlTablePairs(doc){
    const pairs=[];
    doc.querySelectorAll('tr').forEach(tr=>{
      const cells=[...tr.children].filter(x=>/^(TH|TD)$/.test(x.tagName));
      for(let i=0;i<cells.length;i++){

        if(cells[i].tagName!=='TH')continue;
        let td=null; for(let j=i+1;j<cells.length;j++){if(cells[j].tagName==='TH')break;if(cells[j].tagName==='TD'){td=cells[j];break;}}
        if(td)pairs.push({label:htmlCleanText(cells[i].textContent).replace(/^＊/,'').trim(),value:htmlCleanText(td.textContent)});
      }
    });
    return pairs;
  }
  function htmlFindPair(pairs, labels){
    for(const target of labels){
      const exact=pairs.find(x=>x.label===target); if(exact?.value)return exact.value;
      const fuzzy=pairs.find(x=>x.label.includes(target)); if(fuzzy?.value)return fuzzy.value;
    }
    return '';
  }
  function htmlLabeledBlock(doc,label){
    const th=[...doc.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes(label));
    if(!th)return '';
    const td=th.nextElementSibling?.tagName==='TD'?th.nextElementSibling:th.parentElement?.querySelector('td');
    return htmlCleanText(td?.textContent||'');
  }
  function htmlSectionText(text, startLabel, nextLabels=[]){
    const start=text.indexOf(startLabel); if(start<0)return '';
    let end=text.length;
    nextLabels.forEach(l=>{const i=text.indexOf(l,start+startLabel.length);if(i>=0&&i<end)end=i;});
    return htmlCleanText(text.slice(start+startLabel.length,end)).replace(/^\s*[：:]\s*/,'');
  }
  function htmlFullContactName(raw, role){
    const re=role==='primary'
      ?/(?:主要照顧者|案妻為主要照顧者)[：:]?\s*([^\n。]+)/
      :/(?:次要照顧者)[：:]?\s*([^\n。]+)/;
    const m=raw.match(re); if(!m)return '';
    const line=m[1];
    const n=line.match(/[-－]\s*([^\(（\s，,]+)\s*[\(（]/) || line.match(/^([^\(（，,]+)\s*[\(（]/);
    return htmlCleanText(n?.[1]||'').replace(/^案(?:妻|夫|女|子|妹|兄|姐|弟)[-－]?/,'');
  }

  function htmlServiceGroup(code){
    code=String(code||'').toUpperCase();
    if(code.startsWith('BA'))return 'B'; if(code.startsWith('CA'))return 'C'; if(code.startsWith('DA'))return 'D';
    if(code.startsWith('EA')||code.startsWith('FA'))return 'EF'; if(code.startsWith('GA'))return 'G'; if(code.startsWith('SC'))return 'S';
    return code.replace(/\d.*$/,'')||'OTHER';
  }
  function htmlParseServices(doc){
    const out=[]; const seen=new Set();
    doc.querySelectorAll('table').forEach(table=>{
      const headers=[...table.querySelectorAll('thead th')].map(x=>htmlCleanText(x.textContent));
      if(!headers.some(x=>x.includes('服務項目'))||!headers.some(x=>x.includes('服務數量')))return;
      table.querySelectorAll('tbody tr').forEach(tr=>{
        const cells=[...tr.querySelectorAll(':scope > td')].map(x=>htmlCleanText(x.textContent));
        if(cells.length<4)return;
        const serviceText=cells[1]||'';
        const m=serviceText.match(/([A-Z]{1,3}\d+(?:-\d+)?)\s*(?:\[([^\]]+)\])?/i); if(!m)return;
        const code=m[1].toUpperCase(), key=code+'|'+(cells[3]||''); if(seen.has(key))return; seen.add(key);
        out.push({service_group:htmlServiceGroup(code),service_code:code,service_name:htmlCleanText(m[2]||serviceText.replace(m[1],'')),unit_price:htmlNum(cells[2]),approved_quantity:htmlNum(cells[3]),subtotal:htmlNum(cells[4]),valid_from:'',valid_to:''});
      });
    });
    return out;
  }
  function htmlParseBudgets(doc,services){
    const out=[]; const cats=new Set();
    const whole=htmlCleanText(doc.body?.textContent||'');
    if(services.some(x=>['B','C'].includes(x.service_group))){
      const m=whole.match(/BC碼給付額度[：:]\s*([\d,.]+).*?B碼擬定總計[：:]\s*([\d,.]+).*?當月剩餘額度[：:]\s*([\d,.]+)/s);
      if(m){out.push({category:'BC',benefit_limit:htmlNum(m[1]),planned_total:htmlNum(m[2]),remaining_amount:htmlNum(m[3]),valid_from:'',valid_to:''});cats.add('BC');}
    }
    if(services.some(x=>x.service_group==='G')){
      const gTable=[...doc.querySelectorAll('table')].find(t=>/GA\d+/i.test(htmlCleanText(t.textContent)) && /服務數量/.test(htmlCleanText(t.textContent)));
      const gPanel=gTable?.closest('.panel') || gTable?.parentElement;
      const txt=htmlCleanText(gPanel?.textContent||'');
      const m=txt.match(/給付額度[：:]\s*([\d,.]+).*?擬定總計[：:]\s*([\d,.]+).*?剩餘額度[：:]\s*([\d,.]+)/s);
      if(m){out.push({category:'G',benefit_limit:htmlNum(m[1]),planned_total:htmlNum(m[2]),remaining_amount:htmlNum(m[3]),valid_from:'',valid_to:''});cats.add('G');}
    }
    return out;
  }
  function htmlParseCarePlan(doc,allText){
    const goal=htmlLabeledBlock(doc,'計畫目標');
    const exec=htmlLabeledBlock(doc,'計畫執行規劃');
    const brief=htmlLabeledBlock(doc,'計畫簡述');
    const note=htmlLabeledBlock(doc,'其他備註');
    const combined=[brief,goal,exec,note].filter(Boolean).join('\n\n');
    const phone=htmlRocToIso((goal.match(/電聯日期[：:]\s*([^\n]+)/)||[])[1]||'');
    const home=htmlRocToIso((goal.match(/家訪日期[：:]\s*([^\n]+)/)||[])[1]||'');
    const subjectDate=htmlRocToIso((allText.match(/AA01\s*[\(（]\s*([^\)）]+)/)||[])[1]||'');
    const planDate=home||subjectDate||'';
    const visitWith=htmlCleanText((goal.match(/偕同訪視者[：:]\s*([^\n]+)/)||[])[1]||'');

    const sections={}; if(brief)sections['計畫簡述']=brief;if(goal)sections['計畫目標']=goal;if(exec)sections['計畫執行規劃']=exec;if(note)sections['其他備註']=note;
    return {plan_date:planDate,phone_contact_date:phone,home_visit_date:home,visit_with:visitWith,sections,raw_plan_text:combined};
  }
  function htmlParseDocument(source){
    const doc=new DOMParser().parseFromString(source,'text/html');
    const pairs=htmlTablePairs(doc); const allText=htmlCleanText(doc.body?.textContent||'');
    const headMatch=allText.match(/個案姓名\/身分證字號[：:]\s*([^\/\n]+)\/([A-Z]\d{9})/i);
    let caseName=htmlFindPair(pairs,['個案姓名']) || htmlCleanText(headMatch?.[1]||'');
    let nationalId=htmlFindPair(pairs,['個案身分證']) || htmlCleanText(headMatch?.[2]||'');
    nationalId=(nationalId.match(/[A-Z]\d{9}/i)||[])[0]||nationalId;
    const birth=htmlFindPair(pairs,['個案生日','出生日期']);
    const phone=(htmlFindPair(pairs,['個案電話']).match(/0\d[\d\-\s]{7,14}/)||[])[0]||'';
    let gender=(pairs.find(x=>x.label==='性別'||x.label==='個案性別')?.value||'').replace(/^\d+\.?\s*/,''); if(!/^(男|女)$/.test(gender))gender='';
    const address=htmlFindPair(pairs,['居住(通訊)地址','居住地址','通訊地址','現居地址']);
    let cms=(allText.match(/CMS\s*(?:第)?\s*([1-8])\s*級/i)||[])[1]||htmlFindPair(pairs,['CMS等級','失能等級']).match(/[1-8]/)?.[0]||'';
    let aUnit=htmlFindPair(pairs,['A單位名稱']);
    let manager=htmlFindPair(pairs,['A個管姓名']);
    const managerTh=[...doc.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes('A個管姓名'));
    const managerTable=managerTh?.closest('table');
    const managerPhoneTh=managerTable?[...managerTable.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes('聯絡電話')):null;
    const managerPhoneCell=managerPhoneTh?.nextElementSibling?.tagName==='TD'?managerPhoneTh.nextElementSibling:managerPhoneTh?.parentElement?.querySelector('td');
    let managerPhone=(htmlCleanText(managerPhoneCell?.textContent||'').match(/0\d[\d#\-\s]{7,20}/)||[])[0]||'';
    let assessor=htmlFindPair(pairs,['承辦人']);
    const topAssessor=(allText.match(/評估照專[：:]\s*([^\n]+)/)||[])[1]; if(topAssessor)assessor=htmlCleanText(topAssessor);
    const assessment=htmlRocToIso(htmlFindPair(pairs,['處理時間','評估日期']));
    const approval=htmlRocToIso(htmlFindPair(pairs,['近一次照顧計畫送審日期','審核日期']));
    const identityType=htmlFindPair(pairs,['長照資格','補助身分','身分類別']);

    const carePlan=htmlParseCarePlan(doc,allText);
    const services=htmlParseServices(doc); const budgets=htmlParseBudgets(doc,services);
    const contacts=htmlParseContacts(doc,pairs,carePlan.raw_plan_text||allText);
    const ca=(source.match(/ca110id=(\d+)/)||[])[1]||'';
    return {case_name:caseName,national_id:nationalId.toUpperCase(),birth_date:htmlRocToIso(birth),gender,address,phone:phone.replace(/\s/g,''),lives_alone:livesAlone,cms_level:cms,identity_type:identityType,assessment_date:assessment,plan_approval_date:approval,a_unit_name:aUnit,case_manager_name:manager,case_manager_phone:managerPhone.replace(/\s/g,''),assessor_name:assessor,source_ca110_id:ca,contacts,services,budgets,care_plan:carePlan,warnings:[]};
  }
  async function htmlSha256(file){
    const b=await file.arrayBuffer(); const h=await crypto.subtle.digest('SHA-256',b); return [...new Uint8Array(h)].map(x=>x.toString(16).padStart(2,'0')).join('');
  }
