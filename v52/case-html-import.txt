  function htmlCleanText(value=''){
    return String(value).replace(/\u00a0/g,' ').replace(/[ \t]+/g,' ').replace(/\r/g,'').trim();
  }
  function htmlRocToIso(value){
    const s=htmlCleanText(value);
    if(!s)return '';
    let m=s.match(/(\d{1,4})\s*[年\/\.\-]\s*(\d{1,2})\s*[月\/\.\-]\s*(\d{1,2})/);
    if(!m)return '';
    let y=Number(m[1]); if(y<1911)y+=1911;
    const mo=String(Number(m[2])).padStart(2,'0'), d=String(Number(m[3])).padStart(2,'0');
    return `${y}-${mo}-${d}`;
  }
  function htmlNum(value){
    const m=String(value||'').replace(/,/g,'').match(/-?\d+(?:\.\d+)?/); return m?Number(m[0]):null;
  }
  function htmlTablePairs(doc){
    const pairs=[];
    doc.querySelectorAll('tr').forEach(tr=>{
      const cells=[...tr.children].filter(x=>/^(TH|TD)$/.test(x.tagName));
      for(let i=0;i<cells.length;i++){

        if(cells[i].tagName!=='TH')continue;
        let td=null; for(let j=i+1;j<cells.length;j++){if(cells[j].tagName==='TH')break;if(cells[j].tagName==='TD'){td=cells[j];break;}}
        if(td)pairs.push({label:htmlCleanText(cells[i].textContent).replace(/^＊/,'').trim(),value:htmlCleanText(td.textContent)});
      }
    });
    return pairs;
  }
  function htmlFindPair(pairs, labels){
    for(const target of labels){
      const exact=pairs.find(x=>x.label===target); if(exact?.value)return exact.value;
      const fuzzy=pairs.find(x=>x.label.includes(target)); if(fuzzy?.value)return fuzzy.value;
    }
    return '';
  }
  function htmlLabeledBlock(doc,label){
    const th=[...doc.querySelectorAll('th')].find(x=>htmlCleanText(x.textContent).includes(label));
    if(!th)return '';
    const td=th.nextElementSibling?.tagName==='TD'?th.nextElementSibling:th.parentElement?.querySelector('td');
    return htmlCleanText(td?.textContent||'');
  }
  function htmlSectionText(text, startLabel, nextLabels=[]){
    const start=text.indexOf(startLabel); if(start<0)return '';
    let end=text.length;
    nextLabels.forEach(l=>{const i=text.indexOf(l,start+startLabel.length);if(i>=0&&i<end)end=i;});
    return htmlCleanText(text.slice(start+startLabel.length,end)).replace(/^\s*[：:]\s*/,'');
  }
  function htmlFullContactName(raw, role){
    const re=role==='primary'
      ?/(?:主要照顧者|案妻為主要照顧者)[：:]?\s*([^\n。]+)/
      :/(?:次要照顧者)[：:]?\s*([^\n。]+)/;
    const m=raw.match(re); if(!m)return '';
    const line=m[1];
    const n=line.match(/[-－]\s*([^\(（\s，,]+)\s*[\(（]/) || line.match(/^([^\(（，,]+)\s*[\(（]/);
    return htmlCleanText(n?.[1]||'').replace(/^案(?:妻|夫|女|子|妹|兄|姐|弟)[-－]?/,'');
  }

  function htmlServiceGroup(code){
    code=String(code||'').toUpperCase();
    if(code.startsWith('BA'))return 'B'; if(code.startsWith('CA'))return 'C'; if(code.startsWith('DA'))return 'D';
    if(code.startsWith('EA')||code.startsWith('FA'))return 'EF'; if(code.startsWith('GA'))return 'G'; if(code.startsWith('SC'))return 'S';
    return code.replace(/\d.*$/,'')||'OTHER';
  }
  function htmlParseServices(doc){
    const out=[]; const seen=new Set();
    doc.querySelectorAll('table').forEach(table=>{
      const headers=[...table.querySelectorAll('thead th')].map(x=>htmlCleanText(x.textContent));
      if(!headers.some(x=>x.includes('服務項目'))||!headers.some(x=>x.includes('服務數量')))return;
      table.querySelectorAll('tbody tr').forEach(tr=>{
        const cells=[...tr.querySelectorAll(':scope > td')].map(x=>htmlCleanText(x.textContent));
        if(cells.length<4)return;
        const serviceText=cells[1]||'';
        const m=serviceText.match(/([A-Z]{1,3}\d+(?:-\d+)?)\s*(?:\[([^\]]+)\])?/i); if(!m)return;
        const code=m[1].toUpperCase(), key=code+'|'+(cells[3]||''); if(seen.has(key))return; seen.add(key);
        out.push({service_group:htmlServiceGroup(code),service_code:code,service_name:htmlCleanText(m[2]||serviceText.replace(m[1],'')),unit_price:htmlNum(cells[2]),approved_quantity:htmlNum(cells[3]),subtotal:htmlNum(cells[4]),valid_from:'',valid_to:''});
      });
    });
    return out;
  }
  function htmlParseBudgets(doc,services){
    const out=[]; const cats=new Set();
    const whole=htmlCleanText(doc.body?.textContent||'');
    if(services.some(x=>['B','C'].includes(x.service_group))){
      const m=whole.match(/BC碼給付額度[：:]\s*([\d,.]+).*?B碼擬定總計[：:]\s*([\d,.]+).*?當月剩餘額度[：:]\s*([\d,.]+)/s);
      if(m){out.push({category:'BC',benefit_limit:htmlNum(m[1]),planned_total:htmlNum(m[2]),remaining_amount:htmlNum(m[3]),valid_from:'',valid_to:''});cats.add('BC');}
    }
    if(services.some(x=>x.service_group==='G')){
      const gTable=[...doc.querySelectorAll('table')].find(t=>/GA\d+/i.test(htmlCleanText(t.textContent)) && /服務數量/.test(htmlCleanText(t.textContent)));
      const gPanel=gTable?.closest('.panel') || gTable?.parentElement;
      const txt=htmlCleanText(gPanel?.textContent||'');
      const m=txt.match(/給付額度[：:]\s*([\d,.]+).*?擬定總計[：:]\s*([\d,.]+).*?剩餘額度[：:]\s*([\d,.]+)/s);
      if(m){out.push({category:'G',benefit_limit:htmlNum(m[1]),planned_total:htmlNum(m[2]),remaining_amount:htmlNum(m[3]),valid_from:'',valid_to:''});cats.add('G');}
    }
    return out;
  }
  function htmlParseCarePlan(doc,allText){

  }

  async function htmlResolveMatch(d){
    d._match=null; d._potential=[]; d._caseNoConflict=null; d._existing=null;
    const fields='id,case_no,case_name,national_id,birth_date,supervisor_id,service_usage_type,payment_method';
    const byNo=await db.from('care_cases').select(fields).ilike('case_no',d.case_no).limit(2); if(byNo.error)throw byNo.error;
    if(byNo.data&&byNo.data[0])d._caseNoConflict=byNo.data[0];
    if(d.national_id){
      const r=await db.from('care_cases').select(fields).ilike('national_id',d.national_id).limit(2);
      if(r.error)throw r.error;
      if(r.data&&r.data[0])d._match=r.data[0];
    }
    if(!d._match && d.case_name && d.birth_date){
      const r=await db.from('care_cases').select(fields).eq('case_name',d.case_name).eq('birth_date',d.birth_date).limit(5);
      if(r.error)throw r.error;
      d._potential=r.data||[];
    }
    if(!d._match && d._caseNoConflict)d._match=d._caseNoConflict;
    if(d._match)d._existing=await htmlLoadExistingSnapshot(d._match.id);
  }

        contacts:d.contacts||[],
        services:d.services||[],
        budgets:d.budgets||[],
        care_plan:d.care_plan||{},
        source_ca110_id:d.source_ca110_id||'',
        source_hash:d.source_hash||'',
        file_name:file.name,
        storage_path:path,
        warnings:d.warnings||[],
        match_case_id:c.id
      };
      var r=await db.rpc('import_case_html_data',{payload:payload});
      if(r.error)throw r.error;
      if(r.data&&r.data.action==='duplicate'){
        await db.storage.from(CASE_HTML_BUCKET).remove([path]);
        await htmlReturnToCaseList('這份 HTML 已經上傳過，未重複建立版本。','warn');
      }else{
        await htmlReturnToCaseList(r.data&&r.data.plan_is_current?'照管平台照顧計畫已上傳，系統判定此版為目前最新版本':'照管平台照顧計畫已上傳並保存為歷史版本，目前最新版本不變','success');
      }
      return true;
    }catch(err){
      if(path){try{await db.storage.from(CASE_HTML_BUCKET).remove([path]);}catch(_e){}}
      toast('上傳照管平台照顧計畫失敗：'+(err&&err.message?err.message:err),'error');
      return false;
    }finally{showLoading(false);}
  }

  function htmlCaseServiceStatus(c){
    var key=c&&c.service_status?c.service_status:'active';
    if(key==='suspended')return {key:'due',text:'暫停'};
    if(key==='closed')return {key:'overdue',text:'已結案'};
    return {key:'normal',text:'服務中'};
  }

  async function htmlLoadCaseDetailData(caseId){
    var results=await Promise.all([
      db.from('care_cases').select('*').eq('id',caseId).maybeSingle(),
      db.from('case_contacts').select('*').eq('case_id',caseId).order('is_primary_contact',{ascending:false}).order('created_at'),
      db.from('case_approved_services').select('*').eq('case_id',caseId).order('is_current',{ascending:false}).order('service_group').order('service_code'),
      db.from('care_plans').select('*').eq('case_id',caseId).order('plan_date',{ascending:false,nullsFirst:false}).order('created_at',{ascending:false}),
      db.from('service_budgets').select('*').eq('case_id',caseId).order('is_current',{ascending:false}).order('category'),
      db.from('case_imports').select('*').eq('case_id',caseId).order('imported_at',{ascending:false}),
      db.from('case_visits').select('*').eq('case_id',caseId).order('visit_date',{ascending:false}).order('created_at',{ascending:false})
    ]);
    var err=results.find(function(x){return x.error;});
    if(err&&err.error)throw err.error;
    var data={
      c:results[0].data,contacts:results[1].data||[],services:results[2].data||[],plans:results[3].data||[],
      budgets:results[4].data||[],imports:results[5].data||[],visits:results[6].data||[]
    };
    if(!data.c)throw new Error('找不到個案資料');
    state.caseVisits=(state.caseVisits||[]).filter(function(v){return v.case_id!==caseId;}).concat(data.visits);
    return data;
  }

  function htmlBuildCaseTimeline(data){
    var items=[];
    var c=data.c;
    if(c.created_at)items.push({date:c.created_at,title:'建立個案',text:'個案主檔建立於系統。',sort:new Date(c.created_at).getTime()});
    (data.imports||[]).forEach(function(x){
      items.push({date:x.imported_at,title:x.import_type==='new_case'?'HTML 建立個案':'匯入照管平台資料',text:x.file_name||'HTML 匯入',sort:new Date(x.imported_at).getTime()});
    });
    (data.plans||[]).forEach(function(p){
      var d=p.plan_date?p.plan_date+'T12:00:00':p.created_at;
      items.push({date:d,title:'照管平台照顧計畫'+(p.is_current?'（目前使用）':p.is_voided?'（已作廢）':''),text:(p.plan_date?'計畫日期 '+htmlFormatPlanDate(p.plan_date):'未辨識計畫日期')+(p.is_voided&&p.void_reason?'｜作廢原因：'+p.void_reason:''),sort:new Date(d||0).getTime()});
      if(p.is_voided&&p.voided_at)items.push({date:p.voided_at,title:'作廢照管平台照顧計畫',text:'版本 '+htmlFormatPlanDate(p.plan_date)+'｜原因：'+(p.void_reason||'未填寫')+(p.voided_by_staff_id?'｜作廢者：'+htmlGetCaseVisitsApi().caseStaffName(p.voided_by_staff_id):''),sort:new Date(p.voided_at).getTime()});
    });
    (data.visits||[]).forEach(function(v){
      var d=(v.visit_date||'1970-01-01')+'T12:00:00';
      items.push({date:d,title:htmlGetCaseVisitsApi().caseTypeLabel(v.visit_type),text:v.summary||v.follow_up_item||'訪視紀錄',sort:new Date(d).getTime()});
    });
    return items.sort(function(a,b){return b.sort-a.sort;});
  }

    return '<div class="case-panel-card"><div class="case-panel-head"><div><strong>個案歷程</strong><div class="case-panel-note">目前整合個案建立、HTML 匯入、照管平台照顧計畫版本與家訪／電訪。</div></div></div><div class="case-timeline">'+rows+'</div></div>';
  }

  function htmlRenderDocumentsTab(data){
    var rows=data.imports.length?data.imports.map(function(x,i){
      return '<div class="import-history-row"><div><strong>'+escapeHTML(x.file_name)+'</strong><small>'+rocDate(x.imported_at)+'・'+(x.import_type==='new_case'?'建立個案':'更新個案')+(x.detected_plan_date?'・計畫 '+htmlFormatPlanDate(x.detected_plan_date):'')+'</small></div><button class="btn sm" data-document-index="'+i+'">查看原始 HTML</button></div>';
    }).join(''):htmlCaseEmpty('尚無原始 HTML 文件');
    return '<div class="case-panel-card"><div class="case-panel-head"><div><strong>文件</strong><div class="case-panel-note">目前此頁收錄照管平台匯入的原始 HTML；機構其他文件仍維持既有文件流程。</div></div></div><div class="import-history">'+rows+'</div></div>';
  }

  function htmlRenderCaseDetailPanel(){
    var data=htmlCaseDetailState.data;
    if(!data)return;
    var panel=$('caseDetailPanel');
    if(!panel)return;
    var tab=htmlCaseDetailState.tab;
    if(tab==='basic')panel.innerHTML=htmlRenderBasicTab(data);
    else if(tab==='platform-plan')panel.innerHTML=htmlRenderPlatformPlanTab(data);
    else if(tab==='services')panel.innerHTML=htmlRenderServicesTab(data);
    else if(tab==='visits')panel.innerHTML=htmlRenderVisitsTab(data);
    else if(tab==='history')panel.innerHTML=htmlRenderHistoryTab(data);
    else if(tab==='documents')panel.innerHTML=htmlRenderDocumentsTab(data);
    else panel.innerHTML=htmlRenderOverviewTab(data);

    document.querySelectorAll('[data-case-tab-jump]').forEach(function(btn){btn.addEventListener('click',function(){htmlSetCaseDetailTab(btn.dataset.caseTabJump);});});
    var edit=$('caseEditProfile');if(edit)edit.addEventListener('click',function(){htmlOpenCaseEdit(data.c.id);});
    var upload=$('caseUploadPlan');if(upload)upload.addEventListener('click',function(){htmlOpenExistingPlanUpload(data.c.id);});
    document.querySelectorAll('[data-plan-select]').forEach(function(btn){btn.addEventListener('click',function(){htmlCaseDetailState.selectedPlanId=btn.dataset.planSelect;htmlRenderCaseDetailPanel();});});
    document.querySelectorAll('[data-set-current-plan]').forEach(function(btn){btn.addEventListener('click',function(){htmlSetPreferredPlan(btn.dataset.setCurrentPlan);});});
    document.querySelectorAll('[data-edit-plan]').forEach(function(btn){btn.addEventListener('click',function(){htmlOpenPlanEdit(btn.dataset.editPlan);});});
    document.querySelectorAll('[data-void-plan]').forEach(function(btn){btn.addEventListener('click',function(){htmlOpenPlanVoid(btn.dataset.voidPlan);});});
    document.querySelectorAll('[data-add-visit]').forEach(function(btn){btn.addEventListener('click',function(){htmlGetCaseVisitsApi().openVisitModal(data.c);setTimeout(function(){var sel=$('visitType');if(sel)sel.value=btn.dataset.addVisit;},0);});});
    document.querySelectorAll('[data-document-index]').forEach(function(btn){btn.addEventListener('click',function(){var x=data.imports[Number(btn.dataset.documentIndex)];if(x)htmlViewOriginal(x.storage_path,x.file_name);});});
  }

  function htmlSetCaseDetailTab(tab){
    htmlCaseDetailState.tab=tab||'overview';
    window.currentCaseDetailTab=htmlCaseDetailState.tab;
    document.querySelectorAll('.case-detail-tab').forEach(function(btn){btn.classList.toggle('active',btn.dataset.caseTab===htmlCaseDetailState.tab);});
    htmlRenderCaseDetailPanel();
  }

  async function htmlOpenCaseDetailPage(caseId,tab,refresh){
    window.currentCaseDetailId=caseId;
    htmlCaseDetailState.caseId=caseId;
    htmlCaseDetailState.tab=tab||window.currentCaseDetailTab||'overview';
    window.currentCaseDetailTab=htmlCaseDetailState.tab;
    state.page='cases';
    $('topTitle').textContent='個案管理';
    $('crumb').textContent='個案管理 / 個案資料';
    if(!refresh)$('content').innerHTML='<div class="empty compact"><div class="icon">👤</div><h3>正在載入個案資料</h3><p>請稍候。</p></div>';
    showLoading(true);
    try{
      await htmlGetCaseVisitsApi().loadCaseVisitData(true);
      var data=await htmlLoadCaseDetailData(caseId);
      htmlCaseDetailState.data=data;
      var c=data.c,status=htmlCaseServiceStatus(c);
      var current=data.plans.find(function(p){return p.is_current;})||null;
      var firstValid=data.plans.find(function(p){return !p.is_voided;})||null;
      if(!htmlCaseDetailState.selectedPlanId||!data.plans.some(function(p){return p.id===htmlCaseDetailState.selectedPlanId;}))htmlCaseDetailState.selectedPlanId=current?current.id:firstValid?firstValid.id:data.plans[0]?data.plans[0].id:null;
      var tabs=[['overview','總覽'],['basic','基本資料'],['platform-plan','照管平台照顧計畫'],['services','服務核定'],['visits','家訪／電訪'],['history','個案歷程'],['documents','文件']];
      $('crumb').textContent='個案管理 / '+c.case_name;
      $('content').innerHTML=[
        '<div class="case-page">',
          '<div class="case-page-head">',
            '<div class="case-page-title-wrap"><button id="caseBackToList" class="btn case-back-btn">← 返回</button><div><div class="case-page-title">'+escapeHTML(c.case_name||'未命名個案')+'</div><div class="case-page-sub">',
              '<span class="case-meta-chip primary">'+escapeHTML(c.case_no||'未編號')+'</span>',
              '<span class="case-meta-chip">'+escapeHTML(status.text)+'</span>',
              '<span class="case-meta-chip">CMS '+escapeHTML(c.cms_level||'—')+'</span>',
              '<span class="case-meta-chip">督導 '+escapeHTML(htmlGetCaseVisitsApi().caseStaffName(c.supervisor_id))+'</span>',
              '<span class="case-meta-chip">'+escapeHTML(c.service_usage_type||'服務類別未設定')+'</span>',
            '</div></div></div>',
            '<div class="case-page-actions">'+(htmlCanEditCase(c)?'<button id="caseHeaderEdit" class="btn">編輯個案</button><button id="caseHeaderUpload" class="btn primary">＋ 上傳照管平台照顧計畫</button>':'')+'</div>',
          '</div>',
          '<div class="case-detail-tabs">'+tabs.map(function(t){return '<button class="case-detail-tab '+(htmlCaseDetailState.tab===t[0]?'active':'')+'" data-case-tab="'+t[0]+'">'+t[1]+'</button>';}).join('')+'</div>',
          '<div id="caseDetailPanel" class="case-tab-panel"></div>',
        '</div>'
      ].join('');
      $('caseBackToList').addEventListener('click',function(){window.currentCaseDetailId=null;window.currentCaseDetailTab=null;htmlCaseDetailState.data=null;navigate('cases');});
      var headEdit=$('caseHeaderEdit');if(headEdit)headEdit.addEventListener('click',function(){htmlOpenCaseEdit(c.id);});
      var headUpload=$('caseHeaderUpload');if(headUpload)headUpload.addEventListener('click',function(){htmlOpenExistingPlanUpload(c.id);});
      document.querySelectorAll('[data-case-tab]').forEach(function(btn){btn.addEventListener('click',function(){htmlSetCaseDetailTab(btn.dataset.caseTab);});});
      htmlRenderCaseDetailPanel();
      window.scrollTo({top:0,behavior:'auto'});
    }catch(err){
      console.error(err);
      $('content').innerHTML='<div class="empty"><div class="icon">⚠️</div><h3>個案資料載入失敗</h3><p>'+escapeHTML(err&&err.message?err.message:String(err))+'</p><button class="btn" onclick="navigate(\'cases\')">返回個案列表</button></div>';
    }finally{showLoading(false);}
  }

  async function htmlRefreshCaseDetail(caseId,message,type,tab){

    closeModal();
    return htmlOpenCaseDetailPage(caseId,tab||window.currentCaseDetailTab||'overview',true);
  }
