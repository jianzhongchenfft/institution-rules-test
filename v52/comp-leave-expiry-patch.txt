// V5.2 換休期限／到期管理：到職週年制、FIFO、到期提醒
(() => {
  const style=document.createElement('style');
  style.id='v52-comp-leave-expiry-style';
  style.textContent=`
    .ce-chip{display:inline-flex;align-items:center;border-radius:999px;padding:4px 8px;font-size:10.5px;font-weight:900;white-space:nowrap}
    .ce-chip.ok{background:#eef7f1;color:#237a4b}.ce-chip.soon{background:#fffaeb;color:#b54708}.ce-chip.expired{background:#fef3f2;color:#b42318}.ce-chip.missing{background:#f2f4f7;color:#667085}.ce-chip.used{background:#f8fafc;color:#98a2b3}
    .ce-note{margin-top:7px;font-size:11px;line-height:1.55;color:#667085}
    .ce-warn{margin-top:7px;border:1px solid #fedf89;background:#fffaeb;color:#93370d;border-radius:8px;padding:7px 9px;font-size:11px;line-height:1.55}
    .ce-hire-grid{display:grid;gap:8px}.ce-hire-row{display:grid;grid-template-columns:minmax(130px,1fr) minmax(150px,190px);gap:10px;align-items:center;border-bottom:1px solid var(--line);padding:8px 0}.ce-hire-row:last-child{border-bottom:0}.ce-hire-name{font-weight:900;color:#344054}.ce-hire-role{font-size:11px;color:#667085;margin-top:2px}
    @media(max-width:560px){.ce-hire-row{grid-template-columns:1fr}.ce-hire-row input{width:100%}}
  `;
  document.head.appendChild(style);

  Object.assign(state,{
    compLeaveCredits:state.compLeaveCredits||[],
    compLeaveExpiryStaff:state.compLeaveExpiryStaff||[],
    compLeaveExpiryLoaded:false,
    compLeaveExpiryLoading:false
  });

  function ceCanReview(){return !!state.profile&&['organization_manager','admin'].includes(state.profile.role);}
  function ceToday(){return isoDate(new Date());}
  function ceDuration(minutes){
    const n=Math.max(0,Number(minutes)||0),h=Math.floor(n/60),m=n%60;
    if(h&&m)return `${h}小時${m}分`;
    if(h)return `${h}小時`;
    return `${m}分鐘`;
  }
  function ceDate(value){return value?rocDate(String(value).slice(0,10)+'T00:00:00'):'待設定';}
  function ceRole(role){return roleLabels?.[role]||role||'—';}
  function ceStaff(staffId){return (state.compLeaveExpiryStaff||[]).find(s=>String(s.id)===String(staffId))||null;}
  function ceCredits(staffId){return (state.compLeaveCredits||[]).filter(c=>String(c.staff_id)===String(staffId));}
  function cePending(staffId){return (state.compLeaveRows||[]).filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='pending').reduce((s,r)=>s+(Number(r.minutes)||0),0);}
  function ceDaysUntil(date){
    if(!date)return null;
    const a=new Date(ceToday()+'T00:00:00'),b=new Date(String(date).slice(0,10)+'T00:00:00');
    return Math.round((b-a)/86400000);
  }
  function ceBalance(staffId,asOf=ceToday()){
    const credits=ceCredits(staffId);
    const eligible=credits.filter(c=>String(c.earned_date)<=asOf&&(!c.expires_on||String(c.expires_on)>=asOf));
    const remaining=eligible.reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const pending=cePending(staffId);
    const expired=credits.filter(c=>c.expires_on&&String(c.expires_on)<asOf&&(Number(c.remaining_minutes)||0)>0).reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const missing=credits.filter(c=>!c.expires_on&&(Number(c.remaining_minutes)||0)>0).reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const soon=credits.filter(c=>{
      if(!c.expires_on||(Number(c.remaining_minutes)||0)<=0)return false;
      const d=Math.round((new Date(String(c.expires_on)+'T00:00:00')-new Date(asOf+'T00:00:00'))/86400000);
      return d>=0&&d<=30;
    }).reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    return {remaining,pending,requestable:Math.max(0,remaining-pending),expired,missing,soon};
  }

  async function ceLoad(force=false){
    if(!state.profile||state.compLeaveExpiryLoading)return;
    if(state.compLeaveExpiryLoaded&&!force){setTimeout(ceEnhance,20);return;}
    state.compLeaveExpiryLoading=true;
    try{
      const creditsPromise=db.from('comp_leave_credits')
        .select('id,staff_id,source_type,source_id,earned_date,earned_minutes,remaining_minutes,expires_on,source_label,created_at,updated_at')
        .order('earned_date',{ascending:true})
        .order('created_at',{ascending:true});
      const staffPromise=ceCanReview()
        ? db.from('staff_users').select('id,display_name,role,is_active,hire_date').eq('is_active',true).order('display_name')
        : db.from('staff_users').select('id,display_name,role,is_active,hire_date').eq('id',state.profile.id).limit(1);
      const [cr,sr]=await Promise.all([creditsPromise,staffPromise]);
      if(cr.error)throw cr.error;
      if(sr.error)throw sr.error;
      state.compLeaveCredits=cr.data||[];
      state.compLeaveExpiryStaff=sr.data||[];
      state.compLeaveExpiryLoaded=true;
      setTimeout(ceEnhance,20);
    }catch(err){
      console.error('讀取換休期限資料失敗',err);
      toast('讀取換休期限資料失敗：'+(err?.message||err),'error');
    }finally{state.compLeaveExpiryLoading=false;}
  }

  function ceUpdateCard(label,value,foot){
    document.querySelectorAll('#content .stat-card').forEach(card=>{
      const head=card.querySelector('.stat-head span:first-child');
      if(head?.textContent.trim()!==label)return;
      const val=card.querySelector('.stat-value');
      if(val)val.textContent=value;
      if(foot){const f=card.querySelector('.stat-foot');if(f)f.textContent=foot;}
    });
  }

  function ceUpdateCards(){
    if(state.page!=='overtime'||!state.profile||!state.compLeaveExpiryLoaded)return;
    const b=ceBalance(state.profile.id);
    ceUpdateCard('目前剩餘換休',ceDuration(b.remaining),'僅計尚未到期的換休來源');
    ceUpdateCard('可再申請換休',ceDuration(b.requestable),'尚未到期餘額－待審核使用');
    if(state.overtimeView==='balance'){
      ceUpdateCard('可再申請換休',ceDuration(b.requestable),'已扣除到期與待審核時數');
    }
  }

  function ceStatus(c){
    const rem=Number(c.remaining_minutes)||0;
    if(rem<=0)return {cls:'used',text:'已用完'};
    if(!c.expires_on)return {cls:'missing',text:'待設定到職日'};
    const d=ceDaysUntil(c.expires_on);
    if(d<0)return {cls:'expired',text:'已到期，待結清'};
    if(d<=30)return {cls:'soon',text:`${d}天內到期`};
    return {cls:'ok',text:'有效'};
  }

  function ceBalancePanel(){
    if(state.page!=='overtime'||state.overtimeView!=='balance'||!state.compLeaveExpiryLoaded)return;
    const main=document.getElementById('otMain');
    if(!main)return;
    document.getElementById('ceExpiryPanel')?.remove();
    const staffId=state.profile?.id;
    const credits=ceCredits(staffId).slice().sort((a,b)=>String(a.earned_date).localeCompare(String(b.earned_date))||String(a.created_at).localeCompare(String(b.created_at)));
    const b=ceBalance(staffId);
    const missingHire=!ceStaff(staffId)?.hire_date;
    const html=`<section id="ceExpiryPanel" class="ot-rule-card">
      <h3>換休期限與剩餘明細</h3>
      <p>換休依取得先後順序使用；到期日以員工到職週年制的當期特休年度末日計算。</p>
      ${missingHire?'<div class="ce-warn">目前尚未設定到職日，因此部分換休無法計算法定最終到期日。請由機構負責人或系統管理員完成設定。</div>':''}
      ${b.soon>0?`<div class="ce-warn">未來30天內有 <strong>${escapeHTML(ceDuration(b.soon))}</strong> 換休即將到期，建議優先安排。</div>`:''}
      ${b.expired>0?`<div class="ot-rule-highlight">已有 <strong>${escapeHTML(ceDuration(b.expired))}</strong> 到期未休，不再列入可用換休，請於薪資作業中辦理結清。</div>`:''}
      <div style="margin-top:12px">${credits.length?`<div class="ot-table-shell"><table class="ot-table"><thead><tr><th>取得日期</th><th>來源</th><th>取得時數</th><th>剩餘</th><th>到期日</th><th>狀態</th></tr></thead><tbody>${credits.map(c=>{const s=ceStatus(c);return `<tr><td>${ceDate(c.earned_date)}</td><td>${escapeHTML(c.source_label||'換休')}</td><td>${escapeHTML(ceDuration(c.earned_minutes))}</td><td><strong>${escapeHTML(ceDuration(c.remaining_minutes))}</strong></td><td>${c.expires_on?ceDate(c.expires_on):'待設定'}</td><td><span class="ce-chip ${s.cls}">${escapeHTML(s.text)}</span></td></tr>`;}).join('')}</tbody></table></div>`:'<div class="ot-empty"><div class="icon">🕒</div><h3>尚無換休來源</h3><p>核准為換休的加班或假日家訪換休會自動建立來源。</p></div>'}</div>
    </section>`;
    main.insertAdjacentHTML('beforeend',html);
  }

  function cePayrollPanel(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll'||!state.compLeaveExpiryLoaded||!ceCanReview())return;
    const main=document.getElementById('otMain');
    if(!main)return;
    document.getElementById('cePayrollExpiryPanel')?.remove();
    const month=String(state.overtimeMonth||'');
    if(!/^\d{4}-\d{2}$/.test(month))return;
    const start=month+'-01';
    const [y,m]=month.split('-').map(Number);
    const end=isoDate(new Date(y,m,0));
    const rows=(state.compLeaveCredits||[]).filter(c=>c.expires_on&&String(c.expires_on)>=start&&String(c.expires_on)<=end&&(Number(c.remaining_minutes)||0)>0);
    if(!rows.length)return;
    const total=rows.reduce((s,c)=>s+(Number(c.remaining_minutes)||0),0);
    const names=new Map((state.compLeaveExpiryStaff||[]).map(s=>[String(s.id),s.display_name]));
    const html=`<section id="cePayrollExpiryPanel" class="ot-rule-card">
      <h3>本月到期換休待結清</h3>
      <p>以下為本月到期仍未休完的換休來源；到期後已自動排除於可用換休餘額。</p>
      <div class="ot-rule-highlight">本月到期未休合計：<strong>${escapeHTML(ceDuration(total))}</strong>。目前系統尚未保存個別員工加班費計算基數，因此先列示待結清時數，不自動計算金額。</div>
      <div style="margin-top:12px"><div class="ot-table-shell"><table class="ot-table"><thead><tr><th>員工</th><th>來源日期</th><th>來源</th><th>到期日</th><th>待結清時數</th></tr></thead><tbody>${rows.map(c=>`<tr><td><strong>${escapeHTML(names.get(String(c.staff_id))||'—')}</strong></td><td>${ceDate(c.earned_date)}</td><td>${escapeHTML(c.source_label||'換休')}</td><td>${ceDate(c.expires_on)}</td><td><span class="ot-amount">${escapeHTML(ceDuration(c.remaining_minutes))}</span></td></tr>`).join('')}</tbody></table></div></div>
    </section>`;
    main.insertAdjacentHTML('beforeend',html);
  }

  function ceEnsureSettingsButton(){
    if(state.page!=='overtime'||!ceCanReview())return;
    const reload=document.getElementById('otReload');
    if(!reload||document.getElementById('ceHireSettingsBtn'))return;
    const btn=document.createElement('button');
    btn.id='ceHireSettingsBtn';btn.className='btn';btn.textContent='📅 換休期限設定';btn.title='設定員工到職日，以計算週年制換休到期日';
    btn.addEventListener('click',ceOpenHireSettings);
    const anchor=document.getElementById('otCorrectionBtn')||reload;
    anchor.insertAdjacentElement('afterend',btn);
  }

  function ceOpenHireSettings(){
    if(!ceCanReview()){toast('目前帳號沒有設定換休期限的權限','warn');return;}
    const staff=(state.compLeaveExpiryStaff||[]).filter(s=>s.is_active!==false);
    openModal({
      title:'換休期限設定｜到職週年制',
      html:`<div class="ot-rule-highlight" style="margin-top:0">系統依到職日計算每筆換休所屬特休週年年度末日，並以該日作為最終到期日。假日家訪換休採相同到期規則。</div>
      <div class="ce-hire-grid" style="margin-top:12px">${staff.map(s=>`<div class="ce-hire-row"><div><div class="ce-hire-name">${escapeHTML(s.display_name||'—')}</div><div class="ce-hire-role">${escapeHTML(ceRole(s.role))}</div></div><input type="date" data-ce-hire-id="${escapeHTML(String(s.id))}" data-ce-original="${escapeHTML(s.hire_date||'')}" value="${escapeHTML(s.hire_date||'')}"></div>`).join('')}</div>`,
      confirmText:'儲存到職日',
      onConfirm:ceSaveHireSettings,
      large:true
    });
  }

  async function ceSaveHireSettings(){
    const inputs=[...document.querySelectorAll('[data-ce-hire-id]')];
    const changed=inputs.filter(i=>String(i.value||'')!==String(i.dataset.ceOriginal||''));
    if(!changed.length){closeModal();toast('到職日沒有變更','success');return true;}
    showLoading(true);
    try{
      for(const input of changed){
        const {error}=await db.from('staff_users').update({hire_date:input.value||null,updated_at:new Date().toISOString()}).eq('id',input.dataset.ceHireId);
        if(error)throw error;
      }
      closeModal();
      state.compLeaveExpiryLoaded=false;
      await ceLoad(true);
      toast('到職日已更新，換休到期日已重新計算','success');
      return true;
    }catch(err){
      console.error(err);toast('儲存到職日失敗：'+(err?.message||err),'error');return false;
    }finally{showLoading(false);}
  }

  function ceSelectedLeaveMinutes(){
    const sh=Number(document.getElementById('clStartHour')?.value||0),sm=Number(document.getElementById('clStartMinute')?.value||0),eh=Number(document.getElementById('clEndHour')?.value||0),em=Number(document.getElementById('clEndMinute')?.value||0);
    return Math.max(0,(eh*60+em)-(sh*60+sm));
  }

  function ceEnhanceLeaveModal(){
    const date=document.getElementById('clDate');
    if(!date||date.dataset.ceExpiryBound==='1'||!state.compLeaveExpiryLoaded)return;
    date.dataset.ceExpiryBound='1';
    const modal=date.closest('.modal');
    if(!modal)return;
    const fields=[...modal.querySelectorAll('.field')];
    const balanceField=fields.find(f=>f.querySelector('label')?.textContent.trim()==='目前可再申請');
    const balanceInput=balanceField?.querySelector('input');
    let note=balanceField?.querySelector('[data-ce-leave-note]');
    if(!note&&balanceField){note=document.createElement('div');note.dataset.ceLeaveNote='1';note.className='ce-note';balanceField.appendChild(note);}
    const refresh=()=>{
      const b=ceBalance(state.profile.id,date.value||ceToday());
      if(balanceInput)balanceInput.value=ceDuration(b.requestable);
      if(note){
        if(b.missing>0)note.textContent='部分換休尚未設定到職日，請洽管理者完成期限設定。';
        else if(b.requestable<=0)note.textContent='此日期沒有可使用且尚未到期的換休時數。';
        else note.textContent='已依取得先後順序與到期日計算可使用時數。';
      }
    };
    ['clDate','clStartHour','clStartMinute','clEndHour','clEndMinute'].forEach(id=>document.getElementById(id)?.addEventListener('change',refresh));
    const confirmBtn=modal.querySelector('#modalConfirm');
    if(confirmBtn&&!confirmBtn.dataset.ceExpiryGuard){
      confirmBtn.dataset.ceExpiryGuard='1';
      confirmBtn.addEventListener('click',e=>{
        const b=ceBalance(state.profile.id,date.value||ceToday());
        const mins=ceSelectedLeaveMinutes();
        if(mins>0&&mins>b.requestable){
          e.preventDefault();e.stopImmediatePropagation();
          toast(`此日期可使用且尚未到期的換休僅剩 ${ceDuration(b.requestable)}，請調整換休時間`,'warn');
        }
      },true);
    }
    refresh();
  }

  function ceEnhance(){
    if(state.page!=='overtime'||!state.compLeaveExpiryLoaded)return;
    ceEnsureSettingsButton();
    ceUpdateCards();
    ceBalancePanel();
    cePayrollPanel();
    ceEnhanceLeaveModal();
  }

  const baseNavigateForExpiry=navigate;
  navigate=function(id){
    const result=baseNavigateForExpiry(id);
    if(id==='overtime')setTimeout(()=>ceLoad(true),120);
    return result;
  };

  let scheduled=false;
  const observer=new MutationObserver(mutations=>{
    const relevant=mutations.some(m=>Array.from(m.addedNodes||[]).some(n=>n.nodeType===1&&(n.id==='otMain'||n.id==='clDate'||n.id==='otReload'||n.matches?.('.modal,.ot-summary')||n.querySelector?.('#otMain,#clDate,#otReload,.ot-summary'))));
    if(!relevant||scheduled)return;
    scheduled=true;
    setTimeout(()=>{scheduled=false;if(state.page==='overtime'){if(!state.compLeaveExpiryLoaded)ceLoad();else ceEnhance();}},30);
  });
  observer.observe(document.body,{childList:true,subtree:true});
  if(state.profile&&state.page==='overtime')setTimeout(()=>ceLoad(true),120);
})();
