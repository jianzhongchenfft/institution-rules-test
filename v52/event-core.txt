const compareEventSortValues=(a,b,key)=>{
  if(key==='status'){
    return (EVENT_STATUS_SORT_ORDER[a.status]??99)-(EVENT_STATUS_SORT_ORDER[b.status]??99);
  }

  if(key==='date'){
    return String(a.event_date||'').localeCompare(String(b.event_date||''));
  }

  if(key==='next_followup'){
    const av=a.next_followup_date||null;
    const bv=b.next_followup_date||null;
    if(!av&&!bv)return 0;
    if(!av)return 1;
    if(!bv)return -1;
    return String(av).localeCompare(String(bv));
  }

  if(key==='category'){
    return String(a.category||'').localeCompare(String(b.category||''),'zh-Hant');
  }

  if(key==='responsible'){
    return staffName(a.responsible_staff_id).localeCompare(staffName(b.responsible_staff_id),'zh-Hant');
  }

  return 0;
};

const sortEvents=(events,key,direction)=>{
  if(!key||!direction){
    return events.sort((a,b)=>
      (isEventOverdue(b)-isEventOverdue(a))||
      String(b.event_date||'').localeCompare(String(a.event_date||''))
    );
  }

  const sign=direction==='desc'?-1:1;
  return events.sort((a,b)=>{
    // 下次追蹤沒有日期時，不論正反向都固定放最後。
    if(key==='next_followup'){
      const aEmpty=!a.next_followup_date;
      const bEmpty=!b.next_followup_date;
      if(aEmpty!==bEmpty)return aEmpty?1:-1;
    }

    const primary=compareEventSortValues(a,b,key);
    if(primary!==0)return primary*sign;

    return String(b.event_date||'').localeCompare(String(a.event_date||''));
  });
};
