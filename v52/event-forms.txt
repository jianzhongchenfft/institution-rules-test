const fieldValue=id=>$(id).value.trim();
async function saveInitialReport(e,a){
  const eventTime=readOptionalTime('th','tmin');
  const nextDays=Number(fieldValue('nday'));
  if(a!=='pending_close'&&(!Number.isInteger(nextDays)||nextDays<1))return toast('請填正確的追蹤天數','warn');

  const cat=fieldValue('cat');
  const sensitive=isSensitiveCategory(cat);
  const victim=sensitive?readSensitiveParty('victim'):{identity_type:null,person_kind:null,person_id:null,name:null};
  const actor=sensitive?readSensitiveParty('actor'):{identity_type:null,person_kind:null,person_id:null,name:null};
  const reportingStatus=sensitive?fieldValue('srs'):null;
  let reportedAt=null;
  if(sensitive&&reportingStatus==='reported'){
    const rd=fieldValue('srDate'),rh=fieldValue('srHour'),rm=fieldValue('srMin');
    if(!rd||!rh||!fieldValue('srAgency'))return toast('請填妥外部通報日期、時間與通報單位','warn');
    reportedAt=new Date(`${rd}T${rh}:${rm||'00'}:00`).toISOString();
  }

  if(sensitive&&a!=='draft'){
    if(!victim.identity_type||!victim.name||!actor.identity_type||!actor.name){
      return toast('請填妥雙方身分與姓名','warn');
    }
    if(cat==='性騷擾'&&victim.identity_type!=='staff'&&actor.identity_type!=='staff'){
      return toast('性騷擾事件至少一方需為本機構工作人員','warn');
    }
  }

  const p={
    event_id:e?.id||null,
    event_date:fieldValue('d'),
    event_time:eventTime,
    category:cat,
    subject:fieldValue('sub'),
    case_id:fieldValue('ca')||null,
    care_worker_id:fieldValue('wo')||null,
    reporter_type:fieldValue('rt')||null,
    reporter_name:fieldValue('rn')||null,
    responsible_staff_id:fieldValue('ow'),
    event_description:fieldValue('de'),
    completed_actions:fieldValue('dn'),
    pending_tasks:fieldValue('td'),
    requires_followup:a==='pending_followup',
    next_followup_date:a==='pending_close'?null:addDays(isoDate(new Date()),nextDays),
    line_notify:$('ln').checked,
    sensitive_victim_identity_type:victim.identity_type,
    sensitive_victim_person_kind:victim.person_kind,
    sensitive_victim_person_id:victim.person_id,
    sensitive_victim_name:victim.name,
    sensitive_actor_identity_type:actor.identity_type,
    sensitive_actor_person_kind:actor.person_kind,
    sensitive_actor_person_id:actor.person_id,
    sensitive_actor_name:actor.name,
    sensitive_immediate_protection_actions:sensitive?fieldValue('sip'):null,
    sensitive_external_reporting_status:reportingStatus,
    sensitive_external_reported_at:reportedAt,
    sensitive_external_reporting_agency:sensitive&&reportingStatus==='reported'?fieldValue('srAgency')||null:null,
    sensitive_external_reporting_method:sensitive&&reportingStatus==='reported'?fieldValue('srMethod')||null:null,
    sensitive_external_reporting_note:sensitive&&reportingStatus==='reported'?fieldValue('srNote')||null:null
  };

  if(!p.subject)return toast('請填主旨','warn');
  if(a!=='draft'&&(!p.event_description||!p.completed_actions||!p.pending_tasks))return toast('請填妥初報三欄','warn');

  showLoading(true);
  try{
    const r=await db.rpc('save_tracking_event_draft',{payload:p});
    if(r.error)throw r.error;
    if(a!=='draft'){
      const q=await db.rpc('submit_tracking_event',{p_event_id:r.data,p_target_status:a,p_line_notify:p.line_notify});
      if(q.error)throw q.error;
    }
    let ae=null;
    try{await uploadEventAttachments(r.data,'initial',null,'att')}catch(err){ae=err}
    closeModal();
    await loadEventData();
    renderEventDetail(r.data);
    if(ae)toast('事件已儲存，但附件上傳失敗：'+ae.message,'warn');
  }catch(x){
    toast('儲存失敗：'+x.message,'error');
  }finally{
    showLoading(false);
  }
}

      toast('續報失敗：'+x.message,'error');
    }finally{
      showLoading(false);
    }
  };
}

async function reopenEvent(e){
  let x=prompt('重新開啟原因');
  if(!x)return;
  let r=await db.rpc('reopen_tracking_event',{p_event_id:e.id,p_reason:x});
  if(r.error)return toast(r.error.message,'error');
  await loadEventData();renderEventDetail(e.id);
}

}
