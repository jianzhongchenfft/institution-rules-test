const eventFormDateTimeParts=value=>{
  let d=value?new Date(value):new Date();
  if(Number.isNaN(d.getTime()))d=new Date();
  return {
    date:`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`,
    time:`${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`
  };
};
function eventFillTimeFields(prefix,value=null){
  const p=eventFormDateTimeParts(value);
  return `
    <div class="event-field">
      <label>填表日期 *</label>
      <input id="${prefix}Date" type="date" value="${p.date}">
    </div>
    <div class="event-field">
      <label>填表時間 *</label>
      <input id="${prefix}Time" type="time" step="60" value="${p.time}">
    </div>
  `;
}
function readEventFormFilledAt(prefix){
  const date=fieldValue(prefix+'Date');
  const time=fieldValue(prefix+'Time');
  if(!date||!time)return null;
  const d=new Date(`${date}T${time}:00`);
  return Number.isNaN(d.getTime())?null:d.toISOString();
}

const fieldValue=id=>$(id).value.trim();
async function saveInitialReport(e,a){
  const eventTime=readOptionalTime('th','tmin');
  const formFilledAt=readEventFormFilledAt('ifill');
  if(!formFilledAt)return toast('請填妥填表日期與時間','warn');

  const nextDays=Number(fieldValue('nday'));
  if(a!=='pending_close'&&(!Number.isInteger(nextDays)||nextDays<1))return toast('請填正確的追蹤天數','warn');

  const cat=fieldValue('cat');
  const sensitive=isSensitiveCategory(cat);
  const victim=sensitive?readSensitiveParty('victim'):{identity_type:null,person_kind:null,person_id:null,name:null};
  const actor=sensitive?readSensitiveParty('actor'):{identity_type:null,person_kind:null,person_id:null,name:null};
  const reportingStatus=sensitive?fieldValue('srs'):null;
  let reportedAt=null;
  if(sensitive&&reportingStatus==='reported'){
    const rd=fieldValue('srDate'),rh=fieldValue('srHour'),rm=fieldValue('srMin');
    if(!rd||!rh||!fieldValue('srAgency'))return toast('請填妥外部通報日期、時間與通報單位','warn');
    reportedAt=new Date(`${rd}T${rh}:${rm||'00'}:00`).toISOString();
  }

  if(sensitive&&a!=='draft'){
    if(!victim.identity_type||!victim.name||!actor.identity_type||!actor.name){
      return toast('請填妥雙方身分與姓名','warn');
    }
    if(cat==='性騷擾'&&victim.identity_type!=='staff'&&actor.identity_type!=='staff'){
      return toast('性騷擾事件至少一方需為本機構工作人員','warn');
    }
  }

  const p={
    event_id:e?.id||null,
    event_date:fieldValue('d'),
    event_time:eventTime,
    category:cat,
    subject:fieldValue('sub'),
    case_id:fieldValue('ca')||null,
    care_worker_id:fieldValue('wo')||null,
    reporter_type:fieldValue('rt')||null,
    reporter_name:fieldValue('rn')||null,
    responsible_staff_id:fieldValue('ow'),
    event_description:fieldValue('de'),
    completed_actions:fieldValue('dn'),
    pending_tasks:fieldValue('td'),
    form_filled_at:formFilledAt,
    requires_followup:a==='pending_followup',
    next_followup_date:a==='pending_close'?null:addDays(isoDate(new Date()),nextDays),
    line_notify:$('ln').checked,
    sensitive_victim_identity_type:victim.identity_type,
    sensitive_victim_person_kind:victim.person_kind,
    sensitive_victim_person_id:victim.person_id,
    sensitive_victim_name:victim.name,
    sensitive_actor_identity_type:actor.identity_type,
    sensitive_actor_person_kind:actor.person_kind,
    sensitive_actor_person_id:actor.person_id,
    sensitive_actor_name:actor.name,
    sensitive_immediate_protection_actions:sensitive?fieldValue('sip'):null,
    sensitive_external_reporting_status:reportingStatus,
    sensitive_external_reported_at:reportedAt,
    sensitive_external_reporting_agency:sensitive&&reportingStatus==='reported'?fieldValue('srAgency')||null:null,
    sensitive_external_reporting_method:sensitive&&reportingStatus==='reported'?fieldValue('srMethod')||null:null,
    sensitive_external_reporting_note:sensitive&&reportingStatus==='reported'?fieldValue('srNote')||null:null
  };

  if(!p.subject)return toast('請填主旨','warn');
  if(a!=='draft'&&(!p.event_description||!p.completed_actions||!p.pending_tasks))return toast('請填妥初報三欄','warn');

  showLoading(true);
  try{
    const r=await db.rpc('save_tracking_event_draft',{payload:p});
    if(r.error)throw r.error;
    if(a!=='draft'){
      const q=await db.rpc('submit_tracking_event',{p_event_id:r.data,p_target_status:a,p_line_notify:p.line_notify});
      if(q.error)throw q.error;
    }
    let ae=null;
    try{await uploadEventAttachments(r.data,'initial',null,'att')}catch(err){ae=err}
    closeModal();
    await loadEventData();
    renderEventDetail(r.data);
    if(ae)toast('事件已儲存，但附件上傳失敗：'+ae.message,'warn');
  }catch(x){
    toast('儲存失敗：'+x.message,'error');
  }finally{
    showLoading(false);
  }
}

  const initial=getInitialReport(e.id);
  if(initial?.pending_tasks?.trim()){
    pendingTaskEntries.push({label:'初報',text:initial.pending_tasks.trim()});
  }
  getFollowups(e.id).forEach((x,n)=>{
    if(!x.is_voided&&x.pending_tasks?.trim()){
      pendingTaskEntries.push({label:`續報 ${n+1}`,text:x.pending_tasks.trim()});
    }
  });

  const warning=pendingTaskEntries.length?`
    <div class="event-field full">
      <div style="padding:14px 16px;border:1px solid #f5dca8;background:#fff8e8;border-radius:10px;line-height:1.65">
        <div><b>目前紀錄中仍有後續待辦：</b></div>
        <div style="margin-top:8px;display:grid;gap:8px">
          ${pendingTaskEntries.map(x=>`<div><b>${escapeHTML(x.label)}：</b><span style="white-space:pre-wrap">${escapeHTML(x.text)}</span></div>`).join('')}
        </div>
        <label class="event-check" style="margin-top:10px!important">
          <input id="ack" type="checkbox">
          確認以上待辦均已完成或不再需要處理
        </label>
      </div>
    </div>`:''; 

  openModal({
    title:'事件結報',
    large:true,
    showConfirm:false,
    html:`<div class="event-form">
      <div class="event-section-title">結報基本資料</div>

      <div class="event-field">
        <label>結報日期 *</label>
        <input id="cd" type="date" value="${isoDate(new Date())}">
      </div>

      ${eventFillTimeFields('cfill')}

      ${warning}

      <div class="event-section-title">結報內容</div>

      <div class="event-field full">
        <label>1　結案判定 *</label>
        <select id="cw">
          <option value="">請選擇</option>
          <option value="已完成處理，無後續待辦">已完成處理，無後續待辦</option>
          <option value="狀況已穩定，無需繼續追蹤">狀況已穩定，無需繼續追蹤</option>
          <option value="近期未再發生類似事件，無需繼續追蹤">近期未再發生類似事件，無需繼續追蹤</option>
          <option value="已完成轉介／交由其他單位處理">已完成轉介／交由其他單位處理</option>
          <option value="服務已終止／轉出，無需繼續追蹤">服務已終止／轉出，無需繼續追蹤</option>
          <option value="其他">其他</option>
        </select>
      </div>

      <div class="event-field full" id="cwOtherField" style="display:none">
        <label>其他結案判定 *</label>
        <input id="cwOther">
      </div>

      <div class="event-field full">
        <label>2　結案說明 *</label>
        <textarea id="cf" class="event-text-long"></textarea>
      </div>

      <div class="event-field full">
        <label class="event-check"><input id="cln" type="checkbox" checked> LINE 通知</label>
      </div>

      <div class="event-field full event-file">
        <label>附件</label>
        <input id="catt" type="file" multiple accept="${EVENT_ATTACHMENT_ACCEPT}">
      </div>
    </div>`
  });

  document.querySelector('#modalRoot .modal')?.classList.add('event-modal');

  const closeFoot=document.querySelector('#modalRoot .modal-foot');
  const closeCancel=$('modalCancel');
  if(closeFoot&&closeCancel){
    const submitBtn=document.createElement('button');
    submitBtn.id='cs';
    submitBtn.className='btn primary';
    submitBtn.textContent='完成結報';
    closeFoot.insertBefore(submitBtn,closeCancel.nextSibling);
  }

  const toggleOther=()=>{
    const isOther=fieldValue('cw')==='其他';
    if($('cwOtherField'))$('cwOtherField').style.display=isOther?'block':'none';
    if(!isOther&&$('cwOther'))$('cwOther').value='';
  };
  $('cw')?.addEventListener('change',toggleOther);
  toggleOther();

  $('cs').onclick=async()=>{
    const decision=fieldValue('cw')==='其他'?fieldValue('cwOther'):fieldValue('cw');
    const formFilledAt=readEventFormFilledAt('cfill');
    if(!formFilledAt)return toast('請填妥填表日期與時間','warn');

    const p={
      event_id:e.id,
      close_date:fieldValue('cd'),
      closure_decision:decision,
      final_status:fieldValue('cf'),
      form_filled_at:formFilledAt,
      line_notify:$('cln').checked
    };

    if(!p.final_status||!p.closure_decision)return toast('請填妥結報內容','warn');
    if(pendingTaskEntries.length&&!$('ack')?.checked)return toast('請先確認尚未完成的後續待辦','warn');

    showLoading(true);
    try{
      const r=await db.rpc('close_tracking_event',{payload:p});
      if(r.error)throw r.error;
      let ae=null;
      try{await uploadEventAttachments(e.id,'closure',r.data,'catt')}catch(err){ae=err}
      closeModal();
      await loadEventData();
      renderEventDetail(e.id);
      if(ae)toast('結報已完成，但附件上傳失敗：'+ae.message,'warn');
    }catch(x){
      toast('結報失敗：'+x.message,'error');
    }finally{
      showLoading(false);
    }
  };
}

    };

async function reopenEvent(e){
  let x=prompt('重新開啟原因');
  if(!x)return;
  let r=await db.rpc('reopen_tracking_event',{p_event_id:e.id,p_reason:x});
  if(r.error)return toast(r.error.message,'error');
  await loadEventData();renderEventDetail(e.id);
}

  if(r.error)return toast(r.error.message,'error');
  await loadEventData();renderEventDetail(e.id);
}
