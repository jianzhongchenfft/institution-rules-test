  function status(c){

    if(same&&same.status==='draft')return '草稿';
    if(ev&&!same)return '待擬定';
    if(latest&&latest.status==='completed')return '執行中';
    if(latest&&latest.status==='draft')return '草稿';
    return ev?'待擬定':'尚無完成評估';
  }

  function positiveMonths(x){var v=Number(x);return Number.isInteger(v)&&v>=1&&v<=120?v:0;}
  function monthsFor(goal,planDate){
    if(Object.prototype.hasOwnProperty.call(goal,'review_months'))return positiveMonths(goal.review_months);
    // Older drafts used an actual date: preserve the approximate cycle instead of discarding it.
    if(goal.review_date&&planDate){
      var a=String(planDate).split('-').map(Number),b=String(goal.review_date).split('-').map(Number);
      if(a.length===3&&b.length===3){
        var diff=(b[0]-a[0])*12+(b[1]-a[1]);
        if(diff>=1&&diff<=120)return diff;
      }
    }
    return 3;
  }
  function dueFromMonths(date,months){
    var m=positiveMonths(months),p=String(date||'').split('-').map(Number);
    if(!m||p.length!==3||p.some(function(x){return !Number.isInteger(x);}))return '';
    var first=new Date(Date.UTC(p[0],p[1]-1+m,1));
    var yy=first.getUTCFullYear(),mm=first.getUTCMonth();
    var day=Math.min(p[2],new Date(Date.UTC(yy,mm+1,0)).getUTCDate());
    return String(yy).padStart(4,'0')+'-'+String(mm+1).padStart(2,'0')+'-'+String(day).padStart(2,'0');
  }
  function syncReviewDates(plan){
    (plan.plan_content.problems||[]).forEach(function(p){

        if(m){g.review_months=m;g.review_date=dueFromMonths(plan.plan_date,m);}
        else g.review_date='';
      });
    });
  }

  function normalizeCaregiver(plan){
    var body=plan.plan_content||{},care=body.caregiver;
    if(!care){body.caregiver={};return false;}
    var oldFields=['guidance','communication','need_transfer','transfer_detail','next_review_date','notes'];
    if(!oldFields.some(function(k){return Object.prototype.hasOwnProperty.call(care,k);}))return false;
    if(!String(care.guidance_communication||'').trim()){
      var messages=[];
      if(String(care.guidance||'').trim())messages.push('照顧技巧指導：'+String(care.guidance).trim());
      if(String(care.communication||'').trim())messages.push('溝通方式與說明：'+String(care.communication).trim());
      care.guidance_communication=messages.join('\n');
    }
    var actions=String(care.overall_plan||'').trim();
    if(!actions){
      var oldActions=[];
      if(String(care.transfer_detail||'').trim())oldActions.push(String(care.transfer_detail).trim());
      else if(care.need_transfer==='yes')oldActions.push('原紀錄勾選需轉介，待補充後續處理');
      if(String(care.next_review_date||'').trim())oldActions.push('原預定完整評值日期：'+care.next_review_date);
      actions=oldActions.join('\n');
    }
    var note=String(care.notes||'').trim();
    if(note && !actions.includes(note)) actions+=(actions?'\n':'')+'原補充備註：'+note;
    care.overall_plan=actions;
    oldFields.forEach(function(k){delete care[k];});
    return true;
  }

  function onField(e){
    var path=e.target.dataset.path;if(!path)return;
    var segments=path.split('.'),obj=L.editing.plan_content;
    for(var i=0;i<segments.length-1;i++)obj=obj[segments[i]];
    obj[segments[segments.length-1]]=e.target.value;L.dirty=true;
    if(path.endsWith('.review_months'))updateDueLabels();
  }
  function handle(btn){
    var a=btn.dataset.action,i=Number(btn.dataset.i),j=Number(btn.dataset.j),k=Number(btn.dataset.k);
    var list=L.editing.plan_content.problems,prob=list[i],goal=prob&&prob.goals[j];
    if(a==='copyPrevious')return copyPrevious();
    if(a==='assessmentFindings'){
      var summary=summaryFor(L.editing.assessment_event_id);
      var source=String(summary&&summary.key_findings||'').trim();
      if(!source){toast('本次整體評估沒有可帶入的主要發現','error');return;}
      if(!prob.rationale||confirm('確定以本次評估主要發現取代原問題說明？'))prob.rationale=source;
    }
    if(a==='addProblem')list.push(newProblem());
    if(a==='removeProblem'&&confirm('確定刪除此照顧問題及其目標？'))list.splice(i,1);
    if(a==='addGoal')prob.goals.push(newGoal());
    if(a==='removeGoal'&&confirm('確定刪除此照顧目標？'))prob.goals.splice(j,1);
    if(a==='addMeasure')goal.measures.push(newMeasure());
    if(a==='removeMeasure')goal.measures.splice(k,1);
    if(a==='suggestGoal'&&goalSuggestions(prob.title).length){
      goalSuggestions(prob.title).forEach(function(text){
        if(prob.goals.some(function(x){return x.title===text;}))return;
        var g=prob.goals.find(function(x){return !String(x.title||'').trim();});
        if(!g){g=newGoal();prob.goals.push(g);}g.title=text;
      });
    }
    if(a==='suggestMeasure'){
      var suggestion=measureSuggestion(prob.title,goal.measures[k]);
      if(!suggestion){toast('此照顧問題與服務項目目前沒有適用的建議措施，請自行填寫。','error');return;}
      var current=goal.measures[k];
      if(!String(current.text||'').trim()||confirm('此措施已有填寫內容，確定改為建議文字嗎？'))current.text=suggestion;
      else return;
    }
    L.dirty=true;renderForm(true);
  }
  function validate(p){
    if(!p.plan_date)return '請選擇計畫日期';
    var problems=p.plan_content.problems||[];
    if(!problems.length)return '請至少新增一項照顧問題';
    for(var i=0;i<problems.length;i++){
      var x=problems[i];
      if(!String(x.category||'').trim()||!String(x.title||'').trim()||!String(x.rationale||'').trim())return '請完成照顧問題 '+(i+1)+' 的名稱及評估依據';
      if(!x.goals||!x.goals.length)return '照顧問題 '+(i+1)+' 需要至少一個照顧目標';
      for(var j=0;j<x.goals.length;j++){
        var g=x.goals[j];
        if(!String(g.title||'').trim()||!String(g.criteria||'').trim())return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 尚未完成目標及評值標準';
        if(!positiveMonths(monthsFor(g,p.plan_date)))return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 請輸入有效評值月份（1～120）';
        if(!g.measures||!g.measures.length||g.measures.some(function(m){return !String(m.text||'').trim()||!String(m.executor||'').trim();}))return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 尚未完成執行措施及執行者';
        for(var k=0;k<g.measures.length;k++){
          var m=g.measures[k],mode=measureMode(m);
          if(mode==='pending'&&!String(m.proposed_service||m.service_code||'').trim())
            return '照顧問題 '+(i+1)+' 的措施 '+(k+1)+' 請填寫擬新增或調整的服務項目';
          if(mode==='service'&&!serviceForCase(p.case_id).some(function(s){return s.service_code===m.service_code;}))
            return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 有措施未連結有效的核定BA／GA服務項目';
          if(!['service','non_service','pending'].includes(mode))return '請選擇每項措施的服務項目／措施類型';
        }
      }
    }
    return '';
  }
  async function save(finish){
    var p=L.editing;if(!p||p.status==='completed')return;
    if(finish){var err=validate(p);if(err){toast(err,'error');return;}}
    syncReviewDates(p);
    // Older pending drafts could store proposals in service_code: keep the proposal, but never treat as approved service.
    (p.plan_content.problems||[]).forEach(function(pr){
      (pr.goals||[]).forEach(function(g){(g.measures||[]).forEach(function(m){
        if(measureMode(m)==='pending'){
          m.proposed_service=m.proposed_service||m.service_code||'';
          m.service_code='';
        }
      });});
    });
    showLoading(true);
    try{
      var data={case_id:p.case_id,assessment_event_id:p.assessment_event_id,plan_date:p.plan_date,plan_content:p.plan_content,status:finish?'completed':'draft'};
      var q=p.id?db.from('individual_care_plans').update(data).eq('id',p.id).select().single():db.from('individual_care_plans').insert(data).select().single();
      var rs=await q;if(rs.error)throw rs.error;
      var idx=L.plans.findIndex(function(x){return x.id===rs.data.id;});
      if(idx>=0)L.plans[idx]=rs.data;else L.plans.unshift(rs.data);
      L.dirty=false;
      toast(finish?'照顧計畫已完成':'草稿已儲存','success');
      L.editing=null;renderCase();
    }catch(e){
      console.error(e);
      var raw=String(e.message||e);
      var reason=raw.includes('CARE_PLAN_SERVICE_NOT_APPROVED')
        ?'所選服務不在本案目前有效的BA／GA核定項目內，請重新確認個管師核定內容'
        :raw.includes('CARE_PLAN_PENDING_DETAIL_REQUIRED')
          ?'待與A單位討論的項目必須填寫擬新增／調整的服務'
          :raw.includes('CARE_PLAN_PENDING_CANNOT_LINK_CODE')
            ?'待協調服務不能填入已核定服務碼'
          :raw.includes('CARE_PLAN_NON_SERVICE_CANNOT_LINK_CODE')
            ?'非服務措施不能連結服務碼，請檢查措施類型'
            :raw;
      toast('儲存照顧計畫失敗：'+reason,'error');
    }
    finally{showLoading(false);}
  }
  function leaveOK(){return !L.dirty||confirm('目前有尚未儲存的變更，確定離開嗎？');}
  async function start(){
    if(!usable()){shell('計畫擬定','', '', '<div class="icp-card">此帳號沒有使用權限。</div>');return;}
    showLoading(true);
    try{await load();renderList();}
    catch(e){console.error(e);shell('計畫擬定','','<button id="icpRetry" class="btn">重試</button>','<div class="icp-card">載入失敗：'+h(e.message||e)+'</div>');$('icpRetry').onclick=start;}
    finally{showLoading(false);}
  }

  };
