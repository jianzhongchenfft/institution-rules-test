  function status(c){
    var ev=eventFor(c)[0],ps=plansFor(c.id),same=ev?ps.find(function(p){return p.assessment_event_id===ev.id;}):null,latest=ps[0];
    if(same&&same.status==='draft')return '草稿';
    if(ev&&!same)return '待擬定';
    if(latest&&latest.status==='completed')return '執行中';
    if(latest&&latest.status==='draft')return '草稿';
    return ev?'待擬定':'尚無完成評估';

  function onField(e){
    var path=e.target.dataset.path;if(!path)return;
    var segments=path.split('.'),obj=L.editing.plan_content;
    for(var i=0;i<segments.length-1;i++)obj=obj[segments[i]];
    obj[segments[segments.length-1]]=e.target.value;L.dirty=true;
    if(path==='caregiver.need_transfer')renderForm();
  }
  function handle(btn){
    var a=btn.dataset.action,i=Number(btn.dataset.i),j=Number(btn.dataset.j),k=Number(btn.dataset.k);
    var list=L.editing.plan_content.problems,prob=list[i],goal=prob&&prob.goals[j];
    if(a==='copyPrevious')return copyPrevious();
    if(a==='addProblem')list.push(newProblem());
    if(a==='removeProblem'&&confirm('確定刪除此照顧問題及其目標？'))list.splice(i,1);
    if(a==='addGoal')prob.goals.push(newGoal());
    if(a==='removeGoal'&&confirm('確定刪除此照顧目標？'))prob.goals.splice(j,1);
    if(a==='addMeasure')goal.measures.push(newMeasure());
    if(a==='removeMeasure')goal.measures.splice(k,1);
    if(a==='suggestGoal'&&SUGGEST[prob.title]){
      SUGGEST[prob.title].goals.forEach(function(text){if(!prob.goals.some(function(x){return x.title===text;})){var g=newGoal();g.title=text;prob.goals.push(g);}});
    }
    if(a==='suggestMeasure'&&SUGGEST[prob.title]){
      SUGGEST[prob.title].measures.forEach(function(text){if(!goal.measures.some(function(x){return x.text===text;})){var m=newMeasure();m.text=text;goal.measures.push(m);}});
    }
    L.dirty=true;renderForm();
  }
  function validate(p){
    if(!p.plan_date)return '請選擇計畫日期';
    var problems=p.plan_content.problems||[];
    if(!problems.length)return '請至少新增一項照顧問題';
    for(var i=0;i<problems.length;i++){
      var x=problems[i];
      if(!String(x.category||'').trim()||!String(x.title||'').trim()||!String(x.rationale||'').trim())return '請完成照顧問題 '+(i+1)+' 的分類、名稱及評估依據';
      if(!x.goals||!x.goals.length)return '照顧問題 '+(i+1)+' 需要至少一個照顧目標';
      for(var j=0;j<x.goals.length;j++){
        var g=x.goals[j];
        if(!String(g.title||'').trim()||!String(g.criteria||'').trim())return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 尚未完成目標及評值標準';
        if(!g.measures||!g.measures.length||g.measures.some(function(m){return !String(m.text||'').trim()||!String(m.executor||'').trim();}))return '照顧問題 '+(i+1)+' 的目標 '+(j+1)+' 尚未完成執行措施及執行者';
      }
    }
    if(p.plan_content.caregiver.need_transfer==='yes'&&!String(p.plan_content.caregiver.transfer_detail||'').trim())return '請填寫轉介或後續處理內容';
    return '';
  }
  async function save(finish){
    var p=L.editing;if(!p||p.status==='completed')return;
    if(finish){var err=validate(p);if(err){toast(err,'error');return;}}
    showLoading(true);
    try{
      var data={case_id:p.case_id,assessment_event_id:p.assessment_event_id,plan_date:p.plan_date,plan_content:p.plan_content,status:finish?'completed':'draft'};
      var q=p.id?db.from('individual_care_plans').update(data).eq('id',p.id).select().single():db.from('individual_care_plans').insert(data).select().single();
      var rs=await q;if(rs.error)throw rs.error;
      var idx=L.plans.findIndex(function(x){return x.id===rs.data.id;});
      if(idx>=0)L.plans[idx]=rs.data;else L.plans.unshift(rs.data);
      L.dirty=false;
      toast(finish?'照顧計畫已完成':'草稿已儲存','success');
      L.editing=null;renderCase();
    }catch(e){console.error(e);toast('儲存照顧計畫失敗：'+(e.message||e),'error');}
    finally{showLoading(false);}
  }
