// V5.2 跨午夜加班前端整合：跨日儲存、重疊檢核、審核翌日日別與分日薪資計算
(() => {
  const timedTypes = new Set(['overtime','holiday_overtime','rest_day_overtime','national_holiday_overtime','regular_day_off_overtime']);
  const supportedDirectTypes = new Set(['overtime','rest_day_overtime','national_holiday_overtime']);
  const activeStatuses = new Set(['draft','pending','approved']);

  const typeLabel = type => ({
    overtime:'平日加班',
    holiday_overtime:'假日加班（舊分類）',
    rest_day_overtime:'休息日加班',
    national_holiday_overtime:'國定假日出勤',
    regular_day_off_overtime:'例假日（跨午夜延續）',
    holiday_visit_comp:'假日家訪換休'
  })[type] || type || '—';

  const canReview = () => !!state.profile && ['organization_manager','admin'].includes(state.profile.role);
  const norm = value => {
    const x=String(value||'').slice(0,5);
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(x)?x:null;
  };
  const toMin = value => {
    const x=norm(value);
    if(!x)return null;
    const [h,m]=x.split(':').map(Number);
    return h*60+m;
  };
  const selected = prefix => {
    const h=document.getElementById(prefix+'Hour')?.value||'';
    const m=document.getElementById(prefix+'Minute')?.value||'';
    return h&&m?`${h}:${m}`:null;
  };
  const nextDate = date => {
    const m=String(date||'').match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if(!m)return '';
    const d=new Date(Date.UTC(Number(m[1]),Number(m[2])-1,Number(m[3])+1));
    return d.toISOString().slice(0,10);
  };
  const duration = (start,end) => {
    const s=toMin(start),e=toMin(end);
    if(!Number.isFinite(s)||!Number.isFinite(e)||s===e)return 0;
    return e>s?e-s:(1440-s)+e;
  };
  const crosses = (start,end) => {
    const s=toMin(start),e=toMin(end);
    return Number.isFinite(s)&&Number.isFinite(e)&&e<s;
  };
  const hasNextSegment = (start,end) => crosses(start,end) && toMin(end)>0;
  const durationLabel = minutes => {
    const n=Math.max(0,Number(minutes)||0),h=Math.floor(n/60),m=n%60;
    return h&&m?`${h}小時${m}分`:h?`${h}小時`:`${m}分鐘`;
  };
  const money = n => `NT$ ${(Number(n)||0).toLocaleString('zh-TW',{minimumFractionDigits:2,maximumFractionDigits:2})}`;

  function segments(row){
    if(typeof globalThis.v52OvertimeSegments==='function'){
      return globalThis.v52OvertimeSegments(row)||[];
    }
    if(!row||row.request_type==='holiday_visit_comp')return [];
    const start=norm(row.start_time),end=norm(row.end_time),s=toMin(start),e=toMin(end);
    if(!start||!end||!Number.isFinite(s)||!Number.isFinite(e)||s===e)return [];
    const common={requestId:row.id,staffId:row.applicant_staff_id,compensationMethod:row.compensation_method,reason:row.reason||'',createdAt:row.created_at||'',row};
    if(e>s)return [{...common,segmentIndex:1,date:row.work_date,type:row.request_type,startTime:start,endTime:end,startMin:s,minutes:e-s}];
    const out=[{...common,segmentIndex:1,date:row.work_date,type:row.request_type,startTime:start,endTime:'24:00',startMin:s,minutes:1440-s}];
    if(e>0)out.push({...common,segmentIndex:2,date:nextDate(row.work_date),type:row.next_day_request_type||null,startTime:'00:00',endTime:end,startMin:0,minutes:e});
    return out;
  }

  function candidateSegments(date,type,start,end){
    return segments({
      id:'__candidate__',
      applicant_staff_id:state.profile?.id,
      work_date:date,
      request_type:type,
      start_time:start,
      end_time:end,
      compensation_method:document.getElementById('otCompensation')?.value||null,
      next_day_request_type:null
    });
  }

  function segmentsOverlap(a,b){
    if(!a||!b||a.date!==b.date)return false;
    const ae=Number(a.startMin)+Number(a.minutes),be=Number(b.startMin)+Number(b.minutes);
    return Number(a.startMin)<be && ae>Number(b.startMin);
  }

  function findOverlap(date,type,start,end,existingId=null){
    if(!date||!timedTypes.has(type)||!norm(start)||!norm(end)||duration(start,end)<=0)return null;
    const candidate=candidateSegments(date,type,start,end);
    const own=(state.overtimeRows||[]).filter(r=>
      String(r.applicant_staff_id)===String(state.profile?.id)&&
      activeStatuses.has(r.status)&&
      String(r.id)!==String(existingId||'')&&
      timedTypes.has(r.request_type)&&
      r.request_type!=='holiday_visit_comp'
    );
    for(const row of own){
      const existing=segments(row);
      if(candidate.some(a=>existing.some(b=>segmentsOverlap(a,b))))return row;
    }
    return null;
  }

  function rateFor(staffId,date){
    const rows=[...(state.personalHoursRates||[]),...(state.payrollRateRows||[])];
    const seen=new Set();
    return rows
      .filter(r=>{
        const key=String(r.id||`${r.staff_id}|${r.effective_from}|${r.created_at}`);
        if(seen.has(key))return false;
        seen.add(key);
        return String(r.staff_id)===String(staffId)&&String(r.effective_from)<=String(date);
      })
      .sort((a,b)=>String(b.effective_from).localeCompare(String(a.effective_from))||String(b.created_at||'').localeCompare(String(a.created_at||'')))[0]||null;
  }

  function monthRange(){
    const month=/^\d{4}-\d{2}$/.test(state.overtimeMonth||'')?state.overtimeMonth:isoDate(new Date()).slice(0,7);
    const [y,m]=month.split('-').map(Number);
    return {start:`${y}-${String(m).padStart(2,'0')}-01`,end:isoDate(new Date(y,m,0))};
  }

  function payrollPreviewPosition(staffId,date,start,existingId=null){
    const candidateStart=toMin(start);
    if(!staffId||!date||!Number.isFinite(candidateStart))return {offset:0,includeDayBase:true};
    const earlier=(state.overtimeRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&activeStatuses.has(r.status)&&String(r.id)!==String(existingId||'')&&timedTypes.has(r.request_type)&&r.request_type!=='holiday_visit_comp')
      .flatMap(r=>segments(r))
      .filter(seg=>seg.date===date&&Number(seg.startMin)<candidateStart)
      .sort((a,b)=>Number(a.startMin)-Number(b.startMin)||String(a.createdAt||'').localeCompare(String(b.createdAt||''))||String(a.requestId).localeCompare(String(b.requestId))||Number(a.segmentIndex)-Number(b.segmentIndex));
    let offset=0,dayBasePaid=false;
    earlier.forEach(seg=>{
      offset+=Number(seg.minutes)||0;
      if(seg.compensationMethod==='overtime_pay'&&seg.type==='national_holiday_overtime')dayBasePaid=true;
    });
    return {offset,includeDayBase:!dayBasePaid};
  }

  function payrollStaffCalc(staffId,rateResolver=rateFor){
    const {start,end}=monthRange();
    const calcFn=globalThis.v52OvertimePayrollCalcSegment;
    const segs=(state.overtimeRows||[])
      .filter(r=>String(r.applicant_staff_id)===String(staffId)&&r.status==='approved'&&timedTypes.has(r.request_type)&&r.request_type!=='holiday_visit_comp')
      .flatMap(r=>segments(r))
      .filter(seg=>seg.date>=start&&seg.date<=end&&Number(seg.minutes)>0)
      .sort((a,b)=>String(a.date).localeCompare(String(b.date))||Number(a.startMin)-Number(b.startMin)||String(a.createdAt||'').localeCompare(String(b.createdAt||''))||String(a.requestId).localeCompare(String(b.requestId))||Number(a.segmentIndex)-Number(b.segmentIndex));

    const details=[];
    let currentDate=null,dayOffset=0,dayBasePaid=false;
    segs.forEach(seg=>{
      if(currentDate!==seg.date){currentDate=seg.date;dayOffset=0;dayBasePaid=false;}
      const minutes=Number(seg.minutes)||0;
      const rowOffset=dayOffset;
      if(seg.compensationMethod==='overtime_pay'){
        let rate=null,calc;
        if(!seg.type){
          calc={ok:false,code:'missing_day_type',text:'跨午夜後段尚未確認日別'};
        }else if(seg.type==='regular_day_off_overtime'){
          calc={ok:false,code:'regular_day',text:'跨午夜後段屬例假日，需人工確認處理'};
        }else if(typeof calcFn!=='function'){
          calc={ok:false,code:'calc_unavailable',text:'薪資試算元件尚未就緒'};
        }else{
          rate=rateResolver(staffId,seg.date);
          const includeDayBase=!dayBasePaid;
          calc=calcFn(seg.type,minutes,rate,rowOffset,includeDayBase);
          if(calc?.ok&&includeDayBase&&seg.type==='national_holiday_overtime')dayBasePaid=true;
        }
        details.push({
          id:seg.requestId,
          segmentIndex:seg.segmentIndex,
          date:seg.date,
          type:seg.type,
          startTime:seg.startTime,
          endTime:seg.endTime,
          minutes,
          reason:seg.reason||'',
          staffId,
          row:seg.row,
          rate,
          calc,
          dayOffset:rowOffset
        });
      }
      dayOffset+=minutes;
    });
    const missing=details.filter(d=>!d.calc?.ok);
    const total=details.filter(d=>d.calc?.ok).reduce((sum,d)=>sum+Number(d.calc.amount||0),0);
    return {details,missing,total,ok:missing.length===0};
  }

  globalThis.v52OvertimePayrollPreviewPosition=payrollPreviewPosition;
  globalThis.v52OvertimePayrollStaffCalc=payrollStaffCalc;

  function refreshCrossPreview(){
    const modal=document.getElementById('otWorkDate')?.closest('.modal');
    if(!modal)return;
    const type=document.getElementById('otType')?.value;
    const date=document.getElementById('otWorkDate')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    if(!timedTypes.has(type)||!date||!norm(start)||!norm(end)||!crosses(start,end))return;
    let box=modal.querySelector('#prAmountPreview');
    const anchor=document.getElementById('otMinuteCalc')?.closest('.field');
    if(!box&&anchor){box=document.createElement('div');box.id='prAmountPreview';box.className='pr-preview';anchor.insertAdjacentElement('afterend',box);}
    const total=duration(start,end),firstMinutes=1440-toMin(start),secondMinutes=toMin(end);
    if(document.getElementById('otMinuteCalc')){
      document.getElementById('otMinuteCalc').textContent=`${total}分鐘（${durationLabel(total)}）；跨午夜，00:00後歸翌日`;
    }
    if(!box)return;
    const comp=document.getElementById('otCompensation')?.value;
    if(comp==='comp_leave'){
      box.textContent=secondMinutes>0
        ? `本次跨午夜共 ${durationLabel(total)}；核准時會於 00:00 拆成 ${date} ${durationLabel(firstMinutes)} 與 ${nextDate(date)} ${durationLabel(secondMinutes)}，分日建立換休來源。`
        : `本次至翌日 00:00，共 ${durationLabel(total)}；全部歸屬 ${date}。`;
      return;
    }
    if(!canReview()){
      box.textContent=secondMinutes>0
        ? `本次跨午夜共 ${durationLabel(total)}；00:00 後歸翌日，翌日日別由審核者確認後分日計薪。`
        : `本次至翌日 00:00，共 ${durationLabel(total)}；實際加班費由機構負責人於薪資作業中計算。`;
      return;
    }
    const calcFn=globalThis.v52OvertimePayrollCalcSegment;
    if(typeof calcFn!=='function'){box.textContent='跨午夜薪資試算元件尚未就緒';return;}
    const pos=payrollPreviewPosition(state.profile?.id,date,start,state.workflowEditingOvertimeId||null);
    const first=calcFn(type,firstMinutes,rateFor(state.profile?.id,date),pos.offset,pos.includeDayBase);
    if(!first?.ok){box.textContent=first?.text||'跨午夜前段薪資資料不完整';return;}
    box.innerHTML=secondMinutes>0
      ? `跨午夜前段預估加班費：<strong>${money(first.amount)}</strong>（${escapeHTML(date)}，${durationLabel(firstMinutes)}）<br>翌日 ${escapeHTML(nextDate(date))} 00:00–${escapeHTML(end)} 共 ${durationLabel(secondMinutes)}，日別待審核確認後另依翌日有效薪資設定計算。`
      : `預估加班費：<strong>${money(first.amount)}</strong>；本次至翌日 00:00，全部歸屬 ${escapeHTML(date)}。`;
  }

  async function saveCrossMidnight(status){
    const type=document.getElementById('otType')?.value;
    const workDate=document.getElementById('otWorkDate')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    const minutes=duration(start,end);
    const reason=String(document.getElementById('otReason')?.value||'').trim();
    const notes=String(document.getElementById('otNotes')?.value||'').trim();
    const compensationMethod=document.getElementById('otCompensation')?.value;
    const existingId=state.workflowEditingOvertimeId||null;

    if(!supportedDirectTypes.has(type)){toast('此申請類型目前不支援跨午夜直接送出','warn');return false;}
    if(!workDate||!norm(start)||!norm(end)||!crosses(start,end)||minutes<=0||!reason||!compensationMethod){toast('請完整填寫日期、時間、處理方式與加班事由','warn');return false;}
    if(!['00','10','20','30','40','50'].includes(start.slice(3,5))||!['00','10','20','30','40','50'].includes(end.slice(3,5))){toast('加班時間請以10分鐘為間隔','warn');return false;}

    const hit=findOverlap(workDate,type,start,end,existingId);
    if(hit){
      toast(`這段加班時間與既有申請（${hit.work_date} ${String(hit.start_time||'').slice(0,5)}–${crosses(hit.start_time,hit.end_time)?'翌日 ':''}${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');
      return false;
    }

    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      work_date:workDate,
      request_type:type,
      start_time:start,
      end_time:end,
      overtime_minutes:minutes,
      next_day_request_type:null,
      compensation_method:compensationMethod,
      reason,
      holiday_case_count:null,
      comp_minutes:0,
      notes:notes||null,
      status,
      submitted_at:status==='pending'?new Date().toISOString():null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:new Date().toISOString()
    };

    showLoading(true);
    try{
      let error;
      if(existingId){
        ({error}=await db.from('overtime_requests').update(payload).eq('id',existingId));
      }else{
        payload.created_by=state.user.id;
        ({error}=await db.from('overtime_requests').insert(payload));
      }
      if(error)throw error;
      closeModal();
      toast(status==='pending'?'跨午夜申請已送出審核':'跨午夜草稿已儲存','success');
      setTimeout(()=>document.getElementById('otReload')?.click(),0);
      return true;
    }catch(err){
      console.error('儲存跨午夜加班失敗',err);
      toast('儲存失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  function openCrossMidnightApproval(row){
    if(!canReview()){toast('目前帳號沒有審核權限','warn');return;}
    const start=String(row.start_time||'').slice(0,5),end=String(row.end_time||'').slice(0,5);
    const secondMinutes=toMin(end)||0;
    openModal({
      title:`確認跨午夜翌日日別｜${row.applicant_name||'申請人'}`,
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">
          本筆加班為 ${escapeHTML(row.work_date)} ${escapeHTML(start)} 至翌日 ${escapeHTML(end)}。<br>
          00:00 前 ${durationLabel(1440-toMin(start))} 歸 ${escapeHTML(row.work_date)}；00:00 後 ${durationLabel(secondMinutes)} 歸 ${escapeHTML(nextDate(row.work_date))}。
        </div>
        <div class="field" style="margin-top:12px">
          <label>翌日 ${escapeHTML(nextDate(row.work_date))} 日別 *</label>
          <select id="cmNextDayType">
            <option value="">請選擇</option>
            <option value="overtime">平日加班</option>
            <option value="rest_day_overtime">休息日加班</option>
            <option value="national_holiday_overtime">國定假日出勤</option>
            <option value="regular_day_off_overtime">例假日（僅前一日跨午夜延續）</option>
          </select>
          <div class="help">翌日若為例假日，本筆仍可作為前一日跨午夜延續紀錄，但薪資月結需再人工確認處理。</div>
        </div>`,
      confirmText:'確認日別並核准',
      onConfirm:async()=>{
        const nextType=document.getElementById('cmNextDayType')?.value;
        if(!nextType){toast('請選擇翌日日別','warn');return false;}
        showLoading(true);
        try{
          const {error}=await db.from('overtime_requests').update({
            status:'approved',
            next_day_request_type:nextType,
            review_note:null,
            reviewed_by:state.user.id,
            reviewed_at:new Date().toISOString(),
            updated_at:new Date().toISOString()
          }).eq('id',row.id).eq('status','pending');
          if(error)throw error;
          closeModal();
          toast(`跨午夜申請已核准；翌日依「${typeLabel(nextType)}」認列`,'success');
          setTimeout(()=>document.getElementById('otReload')?.click(),0);
          return true;
        }catch(err){
          console.error('核准跨午夜申請失敗',err);
          toast('審核失敗：'+(err?.message||err),'error');
          return false;
        }finally{showLoading(false);}
      }
    });
  }

  window.addEventListener('click',e=>{
    const btn=e.target.closest?.('button');
    if(!btn)return;

    if(btn.dataset.otApprove){
      const row=(state.overtimeRows||[]).find(r=>String(r.id)===String(btn.dataset.otApprove));
      if(row&&row.status==='pending'&&hasNextSegment(row.start_time,row.end_time)){
        e.preventDefault();
        e.stopImmediatePropagation();
        openCrossMidnightApproval(row);
      }
      return;
    }

    const modal=btn.closest?.('.modal');
    if(!modal||!modal.querySelector('#otWorkDate'))return;
    const submit=btn.id==='modalConfirm'||btn.textContent.trim()==='儲存草稿';
    if(!submit)return;

    const type=document.getElementById('otType')?.value;
    const date=document.getElementById('otWorkDate')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    if(!date||!timedTypes.has(type)||!norm(start)||!norm(end))return;

    const existingId=state.workflowEditingOvertimeId||null;
    const hit=findOverlap(date,type,start,end,existingId);
    if(hit){
      e.preventDefault();
      e.stopImmediatePropagation();
      toast(`這段加班時間與既有申請（${hit.work_date} ${String(hit.start_time||'').slice(0,5)}–${crosses(hit.start_time,hit.end_time)?'翌日 ':''}${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');
      return;
    }

    if(crosses(start,end)&&type==='regular_day_off_overtime'){
      e.preventDefault();
      e.stopImmediatePropagation();
      toast('此日為例假日，不得申請加班。若為前一日加班延續至跨午夜，請改以前一日提出申請，系統將於 00:00 自動拆分計算。','warn');
      return;
    }

    if(crosses(start,end)&&supportedDirectTypes.has(type)){
      e.preventDefault();
      e.stopImmediatePropagation();
      const status=btn.textContent.trim()==='儲存草稿'?'draft':'pending';
      saveCrossMidnight(status);
    }
  },true);

  window.addEventListener('change',e=>{
    if(['otWorkDate','otType','otStartHour','otStartMinute','otEndHour','otEndMinute','otCompensation'].includes(e.target?.id)){
      setTimeout(refreshCrossPreview,60);
    }
  },true);

  let queued=false;
  const observer=new MutationObserver(mutations=>{
    if(queued||!mutations.some(m=>m.addedNodes?.length))return;
    queued=true;
    setTimeout(()=>{
      queued=false;
      refreshCrossPreview();
    },80);
  });
  observer.observe(document.body,{childList:true,subtree:true});

  globalThis.v52CrossMidnight={duration,crosses,hasNextSegment,segments,payrollStaffCalc};
  setTimeout(refreshCrossPreview,100);
})();
