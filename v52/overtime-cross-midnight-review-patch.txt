// V5.2 跨午夜加班：翌日日別由審核者確認
(() => {
  function cmCanReview(){
    return !!state.profile && ['organization_manager','admin'].includes(state.profile.role);
  }
  function cmNorm(value){
    const x=String(value||'').slice(0,5);
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(x)?x:null;
  }
  function cmMinute(value){
    const x=cmNorm(value);if(!x)return null;
    const [h,m]=x.split(':').map(Number);return h*60+m;
  }
  function cmHasNextDay(row){
    const s=cmMinute(row?.start_time),e=cmMinute(row?.end_time);
    return Number.isFinite(s)&&Number.isFinite(e)&&e<s&&e>0;
  }
  function cmNextDate(date){
    if(!/^\d{4}-\d{2}-\d{2}$/.test(String(date||'')))return '';
    const [y,m,d]=String(date).split('-').map(Number);
    const x=new Date(y,m-1,d+1);
    return `${x.getFullYear()}-${String(x.getMonth()+1).padStart(2,'0')}-${String(x.getDate()).padStart(2,'0')}`;
  }
  function cmTypeLabel(type){
    return ({
      overtime:'平日加班',
      rest_day_overtime:'休息日加班',
      national_holiday_overtime:'國定假日出勤',
      regular_day_off_overtime:'例假日延續（需人工確認）'
    })[type]||type||'待確認';
  }
  function cmOpen(row){
    const next=cmNextDate(row.work_date);
    const nextMinutes=cmMinute(row.end_time)||0;
    openModal({
      title:'跨午夜翌日日別確認',
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">本申請跨越午夜。申請人不自行選擇翌日倍率，由審核者依實際排班／假別確認翌日日別後再核准。</div>
        <div class="form-grid" style="margin-top:12px">
          <div class="field"><label>申請人</label><input class="ot-readonly" disabled value="${escapeHTML(row.applicant_name||'—')}"></div>
          <div class="field"><label>原申請</label><input class="ot-readonly" disabled value="${escapeHTML(String(row.work_date))} ${escapeHTML(cmNorm(row.start_time)||'—')}–翌日 ${escapeHTML(cmNorm(row.end_time)||'—')}"></div>
          <div class="field"><label>翌日日期</label><input class="ot-readonly" disabled value="${escapeHTML(next)}"></div>
          <div class="field"><label>翌日延續時數</label><input class="ot-readonly" disabled value="${Math.floor(nextMinutes/60)}小時${nextMinutes%60?String(nextMinutes%60)+'分':''}"></div>
          <div class="field full"><label>翌日日別 *</label><select id="cmNextDayType">
            <option value="">請確認翌日日別</option>
            <option value="overtime">平日加班</option>
            <option value="rest_day_overtime">休息日加班</option>
            <option value="national_holiday_overtime">國定假日出勤</option>
            <option value="regular_day_off_overtime">例假日延續</option>
          </select><div id="cmNextDayHelp" class="help">系統會把 00:00 後的時數歸到翌日，並使用翌日有效的薪資與當日倍率順位。</div></div>
        </div>
        <div id="cmRegularWarn" class="pr-legal-warn hidden">翌日若屬例假日，系統會保留實際跨午夜紀錄，但不會自動完成該月加班費結算，需先人工確認處理。</div>`,
      confirmText:'確認日別並核准',
      onConfirm:()=>cmApprove(row),
      large:true
    });
    document.getElementById('cmNextDayType')?.addEventListener('change',e=>{
      document.getElementById('cmRegularWarn')?.classList.toggle('hidden',e.target.value!=='regular_day_off_overtime');
    });
  }
  async function cmApprove(row){
    if(!cmCanReview()){toast('目前帳號沒有審核權限','warn');return false;}
    const type=document.getElementById('cmNextDayType')?.value;
    if(!['overtime','rest_day_overtime','national_holiday_overtime','regular_day_off_overtime'].includes(type||'')){
      toast('請先確認翌日日別','warn');return false;
    }
    if(!confirm(`確定將翌日確認為「${cmTypeLabel(type)}」並核准這筆跨午夜申請嗎？`))return false;
    showLoading(true);
    try{
      const {error}=await db.from('overtime_requests').update({
        next_day_request_type:type,
        status:'approved',
        review_note:null,
        reviewed_by:state.user.id,
        reviewed_at:new Date().toISOString(),
        updated_at:new Date().toISOString()
      }).eq('id',row.id).eq('status','pending');
      if(error)throw error;
      closeModal();
      state.overtimeView='review';
      navigate('overtime');
      toast('跨午夜翌日日別已確認，申請已核准','success');
      return true;
    }catch(err){
      console.error(err);
      toast('核准跨午夜申請失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('[data-ot-approve]');
    if(!btn||!cmCanReview())return;
    const row=(state.overtimeRows||[]).find(r=>String(r.id)===String(btn.dataset.otApprove));
    if(!row||row.status!=='pending'||!cmHasNextDay(row)||row.next_day_request_type)return;
    e.preventDefault();
    e.stopImmediatePropagation();
    cmOpen(row);
  },true);
})();
