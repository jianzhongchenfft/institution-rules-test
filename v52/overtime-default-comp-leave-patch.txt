// V5.2 加班申請：新申請預設處理方式為換休
(() => {
  function applyDefaultCompLeave(){
    const comp=document.getElementById('otCompensation');
    const modal=comp?.closest('.modal');
    if(!comp||!modal||!modal.querySelector('#otWorkDate'))return;
    if(comp.dataset.defaultCompLeaveApplied==='1')return;
    if(state.workflowEditingOvertimeId)return;
    if(![...comp.options].some(o=>o.value==='comp_leave'))return;

    comp.value='comp_leave';
    comp.dataset.defaultCompLeaveApplied='1';
    comp.dispatchEvent(new Event('change',{bubbles:true}));
  }

  let queued=false;
  const observer=new MutationObserver(ms=>{
    if(queued)return;
    const hit=ms.some(m=>Array.from(m.addedNodes||[]).some(n=>n.nodeType===1&&(n.id==='otCompensation'||n.querySelector?.('#otCompensation'))));
    if(!hit)return;
    queued=true;
    setTimeout(()=>{queued=false;applyDefaultCompLeave();},80);
  });
  observer.observe(document.body,{childList:true,subtree:true});
})();
