// V5.2 跨午夜加班：送出時自動拆成兩筆單日申請
// 不包裝 navigate、不重載模組、不使用 MutationObserver；只處理跨午夜送出與成組審核。
(() => {
  const timedTypes=new Set(['overtime','rest_day_overtime','national_holiday_overtime']);
  const activeStatuses=new Set(['draft','pending','approved']);

  const norm=v=>{
    const x=String(v||'').slice(0,5);
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(x)?x:null;
  };
  const toMin=v=>{
    const x=norm(v);
    if(!x)return null;
    const [h,m]=x.split(':').map(Number);
    return h*60+m;
  };
  const selected=p=>{
    const h=document.getElementById(p+'Hour')?.value||'';
    const m=document.getElementById(p+'Minute')?.value||'';
    return h&&m?`${h}:${m}`:null;
  };
  const nextDate=date=>{
    const m=String(date||'').match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if(!m)return '';
    const d=new Date(Date.UTC(Number(m[1]),Number(m[2])-1,Number(m[3])+1));
    return d.toISOString().slice(0,10);
  };
  const durationLabel=minutes=>{
    const n=Math.max(0,Number(minutes)||0),h=Math.floor(n/60),m=n%60;
    if(h&&m)return `${h}小時${m}分`;
    if(h)return `${h}小時`;
    return `${m}分鐘`;
  };
  const typeLabel=type=>({
    overtime:'平日加班',
    rest_day_overtime:'休息日加班',
    national_holiday_overtime:'國定假日出勤'
  })[type]||type||'—';

  function crosses(start,end){
    const s=toMin(start),e=toMin(end);
    return Number.isFinite(s)&&Number.isFinite(e)&&e<s;
  }

  function duration(start,end){
    const s=toMin(start),e=toMin(end);
    if(!Number.isFinite(s)||!Number.isFinite(e)||s===e)return 0;
    return e>s?e-s:(1440-s)+e;
  }

  function currentExistingId(){
    return state.workflowEditingOvertimeId||null;
  }

  function rowInterval(row){
    if(!row||!timedTypes.has(row.request_type)||!activeStatuses.has(row.status))return null;
    const start=toMin(row.start_time),end=toMin(row.end_time);
    if(!Number.isFinite(start)||!Number.isFinite(end)||start===end)return null;
    if(end>start)return {date:row.work_date,start,end};
    if(end===0)return {date:row.work_date,start,end:1440};
    return null;
  }

  function splitOverlap(date,start,end,excludeId=null){
    const s=toMin(start),e=toMin(end);
    if(!date||!Number.isFinite(s)||!Number.isFinite(e)||e>=s)return null;
    const candidates=[{date,start:s,end:1440}];
    if(e>0)candidates.push({date:nextDate(date),start:0,end:e});

    return (state.overtimeRows||[]).find(row=>{
      if(String(row.applicant_staff_id)!==String(state.profile?.id))return false;
      if(String(row.id)===String(excludeId||''))return false;
      const x=rowInterval(row);
      if(!x)return false;
      return candidates.some(c=>c.date===x.date&&c.start<x.end&&c.end>x.start);
    })||null;
  }

  function previewSplit(){
    const date=document.getElementById('otWorkDate')?.value;
    const type=document.getElementById('otType')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    if(!date||!timedTypes.has(type)||!norm(start)||!norm(end)||!crosses(start,end))return;

    const total=duration(start,end);
    const first=1440-toMin(start);
    const second=toMin(end);
    const calc=document.getElementById('otMinuteCalc');
    if(calc){
      calc.textContent=second>0
        ? `${total}分鐘（${durationLabel(total)}）；送出時自動拆成 ${date} ${start}–24:00、${nextDate(date)} 00:00–${end}`
        : `${first}分鐘（${durationLabel(first)}）；本筆計至 ${date} 24:00`;
    }

    const modal=document.getElementById('otWorkDate')?.closest('.modal');
    const payBox=modal?.querySelector('#prAmountPreview');
    if(payBox){
      payBox.textContent=second>0
        ? '跨午夜會拆成兩筆單日申請；翌日日別由審核者確認後，各筆依所屬日期的薪資基數與同日累計倍率計算。'
        : '本筆工作至午夜 24:00，全部歸屬所選日期。';
    }
  }

  function basePayload(status){
    const type=document.getElementById('otType')?.value;
    const workDate=document.getElementById('otWorkDate')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    const compensationMethod=document.getElementById('otCompensation')?.value;
    const reason=String(document.getElementById('otReason')?.value||'').trim();
    const notes=String(document.getElementById('otNotes')?.value||'').trim();
    return {type,workDate,start,end,compensationMethod,reason,notes,status};
  }

  function validateBase(v){
    if(!v.workDate||!timedTypes.has(v.type)||!norm(v.start)||!norm(v.end)||!v.compensationMethod||!v.reason){
      toast('請完整填寫日期、時間、處理方式與加班事由','warn');
      return false;
    }
    if(!['00','10','20','30','40','50'].includes(v.start.slice(3,5))||!['00','10','20','30','40','50'].includes(v.end.slice(3,5))){
      toast('加班時間請以10分鐘為間隔','warn');
      return false;
    }
    return true;
  }

  async function saveToMidnight(status){
    const v=basePayload(status);
    if(!validateBase(v))return false;
    const endMin=toMin(v.end),startMin=toMin(v.start);
    if(endMin!==0||!Number.isFinite(startMin)||startMin<=0){return false;}

    const existingId=currentExistingId();
    const hit=splitOverlap(v.workDate,v.start,v.end,existingId);
    if(hit){
      toast(`這段加班時間與既有申請（${hit.work_date} ${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');
      return false;
    }

    const now=new Date().toISOString();
    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      work_date:v.workDate,
      request_type:v.type,
      start_time:v.start,
      end_time:'00:00',
      overtime_minutes:1440-startMin,
      compensation_method:v.compensationMethod,
      reason:v.reason,
      holiday_case_count:null,
      comp_minutes:0,
      notes:v.notes||null,
      status,
      submitted_at:status==='pending'?now:null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:now,
      split_group_id:null,
      split_part:null,
      split_day_type_confirmed:true
    };

    showLoading(true);
    try{
      let error;
      if(existingId){
        ({error}=await db.from('overtime_requests').update(payload).eq('id',existingId));
      }else{
        payload.created_by=state.user.id;
        ({error}=await db.from('overtime_requests').insert(payload));
      }
      if(error)throw error;
      closeModal();
      toast(status==='pending'?'申請已送出審核':'草稿已儲存','success');
      setTimeout(()=>document.getElementById('otReload')?.click(),0);
      return true;
    }catch(err){
      console.error('儲存午夜結束加班失敗',err);
      toast('儲存失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  async function submitSplit(){
    const v=basePayload('pending');
    if(!validateBase(v))return false;
    if(!crosses(v.start,v.end)||toMin(v.end)===0)return false;
    if(currentExistingId()){
      toast('既有單筆申請不可直接改成跨午夜；請關閉此視窗後另外新增一筆跨午夜申請。','warn');
      return false;
    }

    const hit=splitOverlap(v.workDate,v.start,v.end,null);
    if(hit){
      toast(`跨午夜時段與既有申請（${hit.work_date} ${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');
      return false;
    }

    const firstMinutes=1440-toMin(v.start),secondMinutes=toMin(v.end);
    const secondDate=nextDate(v.workDate);
    const ok=confirm(
      `此加班跨越午夜，系統會自動拆成兩筆申請：\n\n`+
      `1. ${v.workDate} ${v.start}–24:00（${durationLabel(firstMinutes)}）\n`+
      `2. ${secondDate} 00:00–${v.end}（${durationLabel(secondMinutes)}）\n\n`+
      `翌日日別將由審核者確認。確定送出嗎？`
    );
    if(!ok)return false;

    const groupId=crypto.randomUUID();
    const now=new Date().toISOString();
    const common={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      compensation_method:v.compensationMethod,
      reason:v.reason,
      holiday_case_count:null,
      comp_minutes:0,
      notes:v.notes||null,
      status:'pending',
      submitted_at:now,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:now,
      created_by:state.user.id,
      split_group_id:groupId
    };
    const rows=[
      {
        ...common,
        work_date:v.workDate,
        request_type:v.type,
        start_time:v.start,
        end_time:'00:00',
        overtime_minutes:firstMinutes,
        split_part:1,
        split_day_type_confirmed:true
      },
      {
        ...common,
        work_date:secondDate,
        request_type:v.type,
        start_time:'00:00',
        end_time:v.end,
        overtime_minutes:secondMinutes,
        split_part:2,
        split_day_type_confirmed:false
      }
    ];

    showLoading(true);
    try{
      const {error}=await db.from('overtime_requests').insert(rows);
      if(error)throw error;
      closeModal();
      toast('跨午夜加班已自動拆成兩筆並送出審核','success');
      setTimeout(()=>document.getElementById('otReload')?.click(),0);
      return true;
    }catch(err){
      console.error('送出跨午夜拆分申請失敗',err);
      toast('送出失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  function groupRows(row){
    if(!row?.split_group_id)return [];
    return (state.overtimeRows||[])
      .filter(r=>String(r.split_group_id)===String(row.split_group_id))
      .sort((a,b)=>Number(a.split_part||0)-Number(b.split_part||0));
  }

  function openGroupApproval(row){
    const rows=groupRows(row),first=rows.find(r=>Number(r.split_part)===1),second=rows.find(r=>Number(r.split_part)===2);
    if(!first||!second||rows.length!==2){toast('跨午夜申請資料不完整，請重新整理後再試','warn');return;}
    openModal({
      title:`核准跨午夜申請｜${row.applicant_name||'申請人'}`,
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">
          此次申請已拆成兩筆，核准時會一次處理兩筆：<br>
          1. ${escapeHTML(first.work_date)}　${escapeHTML(String(first.start_time||'').slice(0,5))}–24:00　${durationLabel(first.overtime_minutes)}<br>
          2. ${escapeHTML(second.work_date)}　00:00–${escapeHTML(String(second.end_time||'').slice(0,5))}　${durationLabel(second.overtime_minutes)}
        </div>
        <div class="field" style="margin-top:12px">
          <label>翌日 ${escapeHTML(second.work_date)} 日別 *</label>
          <select id="splitSecondType">
            <option value="">請確認翌日日別</option>
            <option value="overtime">平日加班</option>
            <option value="rest_day_overtime">休息日加班</option>
            <option value="national_holiday_overtime">國定假日出勤</option>
          </select>
          <div class="help">若翌日為例假日，請先退回本組申請，暫不直接核准。</div>
        </div>`,
      confirmText:'兩筆一起核准',
      onConfirm:async()=>{
        const secondType=document.getElementById('splitSecondType')?.value;
        if(!secondType){toast('請先確認翌日日別','warn');return false;}
        showLoading(true);
        try{
          const {data,error}=await db.rpc('review_split_overtime_group',{
            p_group_id:row.split_group_id,
            p_action:'approved',
            p_second_request_type:secondType,
            p_review_note:null
          });
          if(error)throw error;
          if(Number(data)!==2)throw new Error('未完整核准兩筆申請');
          closeModal();
          toast(`跨午夜兩筆申請已核准；翌日依「${typeLabel(secondType)}」認列`,'success');
          setTimeout(()=>document.getElementById('otReload')?.click(),0);
          return true;
        }catch(err){
          console.error('核准跨午夜拆分申請失敗',err);
          toast('審核失敗：'+(err?.message||err),'error');
          return false;
        }finally{showLoading(false);}
      }
    });
  }

  function openGroupReject(row){
    const rows=groupRows(row),first=rows.find(r=>Number(r.split_part)===1),second=rows.find(r=>Number(r.split_part)===2);
    if(!first||!second||rows.length!==2){toast('跨午夜申請資料不完整，請重新整理後再試','warn');return;}
    openModal({
      title:`退回跨午夜申請｜${row.applicant_name||'申請人'}`,
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">此操作會將同一組跨午夜的兩筆申請一起退回。</div>
        <div class="field" style="margin-top:12px"><label>退回原因 *</label><textarea id="splitRejectReason" placeholder="請填寫退回原因"></textarea></div>`,
      confirmText:'兩筆一起退回',
      onConfirm:async()=>{
        const reason=String(document.getElementById('splitRejectReason')?.value||'').trim();
        if(!reason){toast('請填寫退回原因','warn');return false;}
        showLoading(true);
        try{
          const {data,error}=await db.rpc('review_split_overtime_group',{
            p_group_id:row.split_group_id,
            p_action:'rejected',
            p_second_request_type:null,
            p_review_note:reason
          });
          if(error)throw error;
          if(Number(data)!==2)throw new Error('未完整退回兩筆申請');
          closeModal();
          toast('跨午夜兩筆申請已一起退回','success');
          setTimeout(()=>document.getElementById('otReload')?.click(),0);
          return true;
        }catch(err){
          console.error('退回跨午夜拆分申請失敗',err);
          toast('退回失敗：'+(err?.message||err),'error');
          return false;
        }finally{showLoading(false);}
      }
    });
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('button');
    if(!btn)return;

    if(btn.dataset.otApprove||btn.dataset.otReject||btn.dataset.otEdit){
      const id=btn.dataset.otApprove||btn.dataset.otReject||btn.dataset.otEdit;
      const row=(state.overtimeRows||[]).find(r=>String(r.id)===String(id));
      if(row?.split_group_id){
        e.preventDefault();
        e.stopPropagation();
        e.stopImmediatePropagation();
        if(btn.dataset.otApprove)openGroupApproval(row);
        else if(btn.dataset.otReject)openGroupReject(row);
        else toast('跨午夜拆分申請不可單筆編輯；若退回後需修改，請重新新增一筆跨午夜申請。','warn');
        return;
      }
    }

    const modal=btn.closest?.('.modal');
    if(!modal||!modal.querySelector('#otWorkDate'))return;
    const submit=btn.id==='modalConfirm'||btn.textContent.trim()==='儲存草稿';
    if(!submit)return;

    const type=document.getElementById('otType')?.value;
    const start=selected('otStart'),end=selected('otEnd');
    if(!timedTypes.has(type)||!norm(start)||!norm(end)||!crosses(start,end))return;

    e.preventDefault();
    e.stopPropagation();
    e.stopImmediatePropagation();

    const status=btn.textContent.trim()==='儲存草稿'?'draft':'pending';
    if(toMin(end)===0){
      saveToMidnight(status);
      return;
    }
    if(status==='draft'){
      toast('跨午夜申請暫不支援草稿；請直接送出審核，系統會自動拆成兩筆。','warn');
      return;
    }
    submitSplit();
  },true);

  document.addEventListener('change',e=>{
    if(['otWorkDate','otType','otStartHour','otStartMinute','otEndHour','otEndMinute','otCompensation'].includes(e.target?.id)){
      setTimeout(previewSplit,0);
    }
  },true);
})();
