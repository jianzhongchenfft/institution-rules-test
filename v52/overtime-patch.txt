  function overtimeMonthRange(month){
    const safe = /^\d{4}-\d{2}$/.test(month || '') ? month : isoDate(new Date()).slice(0,7);
    const [year,mon] = safe.split('-').map(Number);
    const start = `${year}-${String(mon).padStart(2,'0')}-01`;
    const next = new Date(year, mon, 1);
    const end = isoDate(new Date(next.getFullYear(), next.getMonth(), 0));
    return {start,end};
  }

  function overtimeNormalizeTime(value){
    const raw=String(value||'').trim().replace(/：/g,':');
    const match=raw.match(/^(\d{1,2}):(\d{1,2})$/);
    if(!match) return raw;
    const h=Number(match[1]),m=Number(match[2]);
    if(h>23 || m>59) return raw;
    return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`;
  }

  function overtimeValid24h(value){
    return /^([01]\d|2[0-3]):[0-5]\d$/.test(String(value||''));
  }

  function overtimeMinutesBetween(start,end){
    const s=overtimeNormalizeTime(start),e=overtimeNormalizeTime(end);
    if(!overtimeValid24h(s) || !overtimeValid24h(e)) return 0;
    const toMin = value => {
      const [h,m] = String(value).split(':').map(Number);
      return h*60 + m;
    };
    return Math.max(0, toMin(e)-toMin(s));
  }

  function overtimeTimeParts(value){
    const normalized=overtimeNormalizeTime(value);
    if(!overtimeValid24h(normalized)) return {hour:'',minute:''};
    const [hour,minute]=normalized.split(':');
    return {hour,minute};
  }

  function overtimeTimeOptions(max,selected,placeholder){
    let html=`<option value="">${placeholder}</option>`;
    for(let i=0;i<=max;i++){
      const value=String(i).padStart(2,'0');
      html+=`<option value="${value}" ${value===selected?'selected':''}>${value}</option>`;
    }
    return html;
  }

  function overtimeDurationLabel(minutes){
    const n = Number(minutes) || 0;
    const h = Math.floor(n/60);
    const m = n%60;
    if(h && m) return `${h}小時${m}分`;
    if(h) return `${h}小時`;
    return `${m}分鐘`;
  }

  }

  function compLeaveApprovedUsedMinutes(staffId){
    return state.compLeaveRows
      .filter(r=>r.applicant_staff_id===staffId && r.status==='approved')
      .reduce((sum,r)=>sum+(Number(r.minutes)||0),0);
  }

  function compLeavePendingMinutes(staffId){
    return state.compLeaveRows
      .filter(r=>r.applicant_staff_id===staffId && r.status==='pending')
      .reduce((sum,r)=>sum+(Number(r.minutes)||0),0);
  }

  function compLeaveBalance(staffId){
    const earned=compLeaveEarnedMinutes(staffId);
    const used=compLeaveApprovedUsedMinutes(staffId);
    const pending=compLeavePendingMinutes(staffId);
    return {
      earned,
      used,
      pending,
      remaining:Math.max(0,earned-used),
      requestable:Math.max(0,earned-used-pending)
    };
  }

  function compLeaveDurationParts(minutes){
    const n=Math.max(0,Number(minutes)||0);
    return {hour:String(Math.floor(n/60)).padStart(2,'0'),minute:String(n%60).padStart(2,'0')};
  }

  async function loadOvertimeRows(){
    if(!state.profile) return;
    showLoading(true);
    try{
      const [overtimeResult,leaveResult] = await Promise.all([
        db.from('overtime_requests')
          .select('id,applicant_staff_id,applicant_name,applicant_role,work_date,request_type,start_time,end_time,overtime_minutes,compensation_method,reason,holiday_case_count,comp_minutes,notes,status,review_note,submitted_at,reviewed_at,created_at,updated_at')
          .order('work_date',{ascending:false})
          .order('created_at',{ascending:false}),
        db.from('comp_leave_usages')
          .select('id,applicant_staff_id,applicant_name,applicant_role,leave_date,start_time,end_time,minutes,reason,notes,status,review_note,submitted_at,reviewed_at,created_at,updated_at')
          .order('leave_date',{ascending:false})
          .order('created_at',{ascending:false})
      ]);
      if(overtimeResult.error) throw overtimeResult.error;
      if(leaveResult.error) throw leaveResult.error;
      state.overtimeRows = overtimeResult.data || [];
      state.compLeaveRows = leaveResult.data || [];
      state.overtimeLoaded = true;
      state.compLeaveLoaded = true;
    }catch(err){
      console.error(err);
      state.overtimeRows = [];
      state.compLeaveRows = [];
      toast('讀取加班／換休資料失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  function overtimeSummary(){
    const {start,end}=overtimeMonthRange(state.overtimeMonth);
    const own=overtimeOwnRows().filter(r=>r.work_date>=start && r.work_date<=end && r.status==='approved');
    const overtimeMinutes=own.filter(r=>['overtime','holiday_overtime'].includes(r.request_type)).reduce((sum,r)=>sum+(Number(r.overtime_minutes)||0),0);
    const compMinutes=own.filter(r=>r.request_type==='holiday_visit_comp').reduce((sum,r)=>sum+(Number(r.comp_minutes)||0),0);
    const pendingOvertime=state.overtimeRows.filter(r=>r.status==='pending').length;
    const pendingLeave=state.compLeaveRows.filter(r=>r.status==='pending').length;
    const ownPending=overtimeOwnRows().filter(r=>r.status==='pending').length+compLeaveOwnRows().filter(r=>r.status==='pending').length;
    const balance=compLeaveBalance(state.profile?.id);
    return {overtimeMinutes,compMinutes,pending:pendingOvertime+pendingLeave,pendingOvertime,pendingLeave,ownPending,balance};
  }

  function openOvertimeModal(existing=null){
    if(!overtimeCanSubmit()){ toast('目前帳號沒有加班申請權限','warn'); return; }
    openModal({
      title:existing?'編輯加班／換休申請':'新增加班／換休申請',
      html:overtimeFormHTML(existing),
      confirmText:'送出審核',
      onConfirm:()=>saveOvertime(existing,'pending'),
      large:true
    });
    const foot=$('modalRoot').querySelector('.modal-foot');
    if(foot && !existing){
      const draft=el('button',{class:'btn',text:'儲存草稿',onClick:()=>saveOvertime(null,'draft')});
      foot.insertBefore(draft,$('modalConfirm'));
    }
    const typeEl=$('otType');
    const compensationEl=$('otCompensation');
    const updateFields=()=>{
      const selected=!!typeEl.value;
      const holiday=typeEl.value==='holiday_visit_comp';
      $('otTimeFields').classList.toggle('hidden',!selected || holiday);
      $('otHolidayFields').classList.toggle('hidden',!holiday);
      if(!selected){
        compensationEl.value='';
        compensationEl.disabled=true;
        $('otCompensationHelp').textContent='請先選擇申請類型。';
      }else if(holiday){
        compensationEl.value='comp_leave';
        compensationEl.disabled=true;
        $('otCompensationHelp').textContent='假日家訪換休固定為換休。';
      }else{
        compensationEl.disabled=false;
        $('otCompensationHelp').textContent='請選擇本次加班要領加班費或換休。';
      }
      updateOvertimeCalc();
    };
    typeEl.addEventListener('change',()=>{
      if(typeEl.value!=='holiday_visit_comp') compensationEl.value='';
      updateFields();
    });
    ['otStartHour','otStartMinute','otEndHour','otEndMinute'].forEach(id=>$(id)?.addEventListener('change',updateOvertimeCalc));
    $('otCaseCount')?.addEventListener('change',updateOvertimeCalc);
    updateFields();
  }

  function updateOvertimeCalc(){
    const type=$('otType')?.value;
    if(type==='holiday_visit_comp'){
      const count=Math.max(0,Number($('otCaseCount')?.value)||0);
      if($('otCompCalc')) $('otCompCalc').textContent=count?`${count*2}小時`:'請輸入家訪案數';
    }else{
      const start=overtimeSelectedTime('otStart');
      const end=overtimeSelectedTime('otEnd');
      const minutes=overtimeMinutesBetween(start,end);
      const valid=overtimeValid24h(start)&&overtimeValid24h(end);
      if($('otMinuteCalc')) $('otMinuteCalc').textContent=!valid?'請選擇完整的開始與結束時間':minutes?`${minutes}分鐘（${overtimeDurationLabel(minutes)}）`:'結束時間須晚於開始時間';
    }
  }

  async function saveOvertime(existing,status){
    const type=$('otType').value;
    const workDate=$('otWorkDate').value;
    const reason=$('otReason').value.trim();
    const notes=$('otNotes').value.trim();
    const start=overtimeSelectedTime('otStart');
    const end=overtimeSelectedTime('otEnd');
    const timeBased=['overtime','holiday_overtime'].includes(type);
    const compensationMethod=type==='holiday_visit_comp'?'comp_leave':$('otCompensation').value;
    const overtimeMinutes=timeBased?overtimeMinutesBetween(start,end):0;
    const caseCount=type==='holiday_visit_comp'?Math.max(0,Number($('otCaseCount')?.value)||0):null;
    const compMinutes=type==='holiday_visit_comp'?caseCount*120:0;

    if(!type){toast('請選擇申請類型','warn');return false;}
    if(!workDate){toast('請選擇日期','warn');return false;}
    if(!reason){toast('請填寫加班事由','warn');return false;}
    if(type!=='holiday_visit_comp' && !compensationMethod){toast('請選擇加班處理方式','warn');return false;}
    if(timeBased && (!overtimeValid24h(start)||!overtimeValid24h(end))){toast('請選擇完整的開始及結束時間','warn');return false;}
    if(timeBased && (!['00','10','20','30','40','50'].includes(String(start).slice(3,5)) || !['00','10','20','30','40','50'].includes(String(end).slice(3,5)))){toast('加班時間請以10分鐘為間隔','warn');return false;}
    if(timeBased && overtimeMinutes<=0){toast('結束時間須晚於開始時間','warn');return false;}
    if(type==='holiday_visit_comp' && (caseCount<1 || caseCount>4)){toast('假日家訪案數請選擇1至4案','warn');return false;}

    const payload={
      applicant_staff_id:state.profile.id,
      applicant_name:state.profile.display_name,
      applicant_role:state.profile.role,
      work_date:workDate,
      request_type:type,
      start_time:timeBased?start:null,
      end_time:timeBased?end:null,
      overtime_minutes:overtimeMinutes,
      compensation_method:compensationMethod,
      reason,
      holiday_case_count:type==='holiday_visit_comp'?caseCount:null,
      comp_minutes:compMinutes,
      notes:notes||null,
      status,
      submitted_at:status==='pending'?new Date().toISOString():null,
      review_note:null,
      reviewed_at:null,
      reviewed_by:null,
      updated_at:new Date().toISOString()
    };

    showLoading(true);
    try{
      let error;
      if(existing){
        ({error}=await db.from('overtime_requests').update(payload).eq('id',existing.id));
      }else{
        payload.created_by=state.user.id;
        ({error}=await db.from('overtime_requests').insert(payload));
      }
      if(error) throw error;
      closeModal();
      await loadOvertimeRows();
      renderOvertimePage();
      toast(status==='pending'?'申請已送出審核':'草稿已儲存','success');
      return true;
    }catch(err){
      console.error(err);
      toast('儲存失敗：'+(err?.message||err),'error');
      return false;
    }finally{showLoading(false);}
  }

  async function reviewCompLeaveUsage(id,status,note=null){
    if(!overtimeCanReview()){toast('目前帳號沒有審核權限','warn');return;}

    if(!row || row.status!=='pending'){toast('此換休使用申請目前不是待審核狀態','warn');return;}
    if(status==='approved' && !confirm(`確定核准「${row.applicant_name}」使用 ${overtimeDurationLabel(row.minutes)} 換休嗎？`)) return;
    showLoading(true);
    try{
      const {error}=await db.from('comp_leave_usages').update({
        status,
        review_note:note||null,
        reviewed_by:state.user.id,
        reviewed_at:new Date().toISOString(),
        updated_at:new Date().toISOString()
      }).eq('id',id).eq('status','pending');
      if(error) throw error;
      closeModal();
      await loadOvertimeRows();
      renderOvertimePage();
      toast(status==='approved'?'換休使用申請已核准':'換休使用申請已退回','success');
    }catch(err){
      console.error(err);
      toast('審核換休使用申請失敗，請確認換休餘額是否足夠：'+(err?.message||err),'error');

    if(!row) return;
    openModal({
      title:'退回換休使用申請',
      html:`<div class="field"><label>退回原因 *</label><textarea id="clRejectReason" placeholder="請說明需要申請人補充或修正的內容"></textarea></div>`,
      confirmText:'確認退回',
      onConfirm:async()=>{
        const note=$('clRejectReason').value.trim();
        if(!note){toast('請填寫退回原因','warn');return false;}
        await reviewCompLeaveUsage(id,'rejected',note);
        return true;
      }
    });
  }

  async function reviewOvertime(id,status,note=null){
    if(!overtimeCanReview()){toast('目前帳號沒有審核權限','warn');return;}
    const row=state.overtimeRows.find(r=>r.id===id);
    if(!row || row.status!=='pending'){toast('此申請目前不是待審核狀態','warn');return;}
    if(status==='approved' && !confirm(`確定核准「${row.applicant_name}」${overtimeTypeLabel(row.request_type)}申請嗎？`)) return;
    showLoading(true);
    try{
      const {error}=await db.from('overtime_requests').update({
        status,
        review_note:note||null,
        reviewed_by:state.user.id,
        reviewed_at:new Date().toISOString(),
        updated_at:new Date().toISOString()
      }).eq('id',id).eq('status','pending');
      if(error) throw error;
      closeModal();
      await loadOvertimeRows();
      renderOvertimePage();
      toast(status==='approved'?'申請已核准':'申請已退回','success');
    }catch(err){
      console.error(err);
      toast('審核失敗：'+(err?.message||err),'error');
    }finally{showLoading(false);}
  }

  function openOvertimeRejectModal(id){
    const row=state.overtimeRows.find(r=>r.id===id);
    if(!row) return;
    openModal({
      title:'退回加班／換休申請',
      html:`<div class="field"><label>退回原因 *</label><textarea id="otRejectReason" placeholder="請說明需要申請人補充或修正的內容"></textarea></div>`,
      confirmText:'確認退回',
      onConfirm:async()=>{
        const note=$('otRejectReason').value.trim();
        if(!note){toast('請填寫退回原因','warn');return false;}
        await reviewOvertime(id,'rejected',note);
        return true;
      }
    });
  }

  const baseNavigateForOvertime=navigate;
  navigate=function(id){
    if(id!=='overtime') return baseNavigateForOvertime(id);
    const m=modules.find(x=>x.id==='overtime')||modules[0];
    state.page='overtime';
    if(overtimeCanReview() && !overtimeCanSubmit()) state.overtimeView='review';
    renderNav();
    $('topTitle').textContent=m.label;
    $('crumb').textContent=m.crumb;
    $('sidebar').classList.remove('open');
    window.scrollTo({top:0,behavior:'auto'});
    $('content').innerHTML='<div class="empty compact"><div class="icon">⏱</div><h3>正在載入加班／換休資料</h3><p>請稍候。</p></div>';
    loadOvertimeRows().then(renderOvertimePage);
  };
