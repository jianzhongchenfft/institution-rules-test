// V5.2 新假別編輯補強＋新申請預設換休
(() => {
  const newTypes=new Set(['rest_day_overtime','national_holiday_overtime']);

  function restore(){
    const sel=document.getElementById('otType');
    const id=state.workflowEditingOvertimeId;
    if(!sel||!id)return;
    const row=(state.overtimeRows||[]).find(r=>String(r.id)===String(id));
    if(!row||!newTypes.has(row.request_type)||sel.value===row.request_type)return;
    if([...sel.options].some(o=>o.value===row.request_type)){
      sel.value=row.request_type;
      sel.dispatchEvent(new Event('change',{bubbles:true}));
    }
  }

  function defaultCompLeave(){
    const comp=document.getElementById('otCompensation');
    const type=document.getElementById('otType');
    const modal=comp?.closest('.modal,.ot-page-form');
    if(!comp||!type||!modal||!modal.querySelector('#otWorkDate'))return;
    if(state.workflowEditingOvertimeId)return;
    if(![...comp.options].some(o=>o.value==='comp_leave'))return;

    // 新增申請一律預設「換休」。
    // 其他模組可能在自動判斷申請類型時觸發 change 並把處理方式清空，
    // 因此除了開啟視窗時設定外，也監聽申請類型變更後重新套用預設值。
    if(comp.value!=='comp_leave'){
      comp.value='comp_leave';
      comp.dispatchEvent(new Event('change',{bubbles:true}));
    }

    if(type.dataset.defaultCompLeaveBound!=='1'){
      type.dataset.defaultCompLeaveBound='1';
      type.addEventListener('change',()=>{
        if(state.workflowEditingOvertimeId)return;
        setTimeout(()=>{
          const current=document.getElementById('otCompensation');
          if(!current||![...current.options].some(o=>o.value==='comp_leave'))return;
          if(current.value!=='comp_leave'){
            current.value='comp_leave';
            current.dispatchEvent(new Event('change',{bubbles:true}));
          }
        },0);
      });
    }
  }

  function apply(){
    restore();
    defaultCompLeave();
  }

  window.AttendanceV2Lifecycle?.register('overtime:form','overtime-payroll-edit-fix',apply);
  apply();
})();
