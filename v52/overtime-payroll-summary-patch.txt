// V5.2 薪資計算補強：新假別統計＋月結金額快照顯示
(() => {
  const timed=new Set(['overtime','holiday_overtime','rest_day_overtime','national_holiday_overtime']);
  const dur=n=>{n=Math.max(0,Number(n)||0);const h=Math.floor(n/60),m=n%60;return h&&m?`${h}小時${m}分`:h?`${h}小時`:`${m}分鐘`;};
  const money=n=>`NT$ ${(Number(n)||0).toLocaleString('zh-TW',{minimumFractionDigits:2,maximumFractionDigits:2})}`;
  const typeLabel=t=>({overtime:'平日加班',rest_day_overtime:'休息日加班',national_holiday_overtime:'國定假日出勤',holiday_visit_comp:'假日家訪換休'})[t]||t||'—';

  function monthStart(){
    const month=/^\d{4}-\d{2}$/.test(state.overtimeMonth||'')?state.overtimeMonth:isoDate(new Date()).slice(0,7);
    return month+'-01';
  }
  function settlement(){return (state.payrollSettlements||[]).find(s=>String(s.settlement_month)===monthStart())||null;}
  function snapshotVersion(s){return Number(s?.summary?.version)||1;}
  function snapshotStaff(s,staffId){
    const rows=Array.isArray(s?.summary?.staff)?s.summary.staff:[];
    return rows.find(r=>String(r.staffId)===String(staffId))||null;
  }
  function correctionAmount(s,staffId=null){
    return (state.payrollAmountCorrections||[])
      .filter(c=>String(c.settlement_id)===String(s?.id)&&c.adjustment_type==='overtime_pay'&&(!staffId||String(c.staff_id)===String(staffId)))
      .reduce((sum,c)=>sum+(Number(c.calculated_amount)||0),0);
  }

  function updateMineCard(){
    if(state.page!=='overtime'||!state.profile)return;
    const month=/^\d{4}-\d{2}$/.test(state.overtimeMonth||'')?state.overtimeMonth:isoDate(new Date()).slice(0,7);
    const [y,m]=month.split('-').map(Number),start=`${y}-${String(m).padStart(2,'0')}-01`,end=isoDate(new Date(y,m,0));
    const mins=(state.overtimeRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile.id)&&r.status==='approved'&&r.work_date>=start&&r.work_date<=end&&timed.has(r.request_type)).reduce((s,r)=>s+(Number(r.overtime_minutes)||0),0);
    const card=[...document.querySelectorAll('#content .stat-card')].find(c=>c.querySelector('.stat-head span:first-child')?.textContent.trim()==='本月我的加班');
    if(!card)return;const val=card.querySelector('.stat-value');if(val)val.textContent=dur(mins);
  }

  function applySnapshotMoney(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll')return;
    const s=settlement();
    if(!s)return;
    const version=snapshotVersion(s);
    const section=[...document.querySelectorAll('#otMain .ot-rule-card')].find(x=>x.querySelector('h3')?.textContent.trim()==='員工月結摘要');
    const table=section?.querySelector('table');
    if(table){
      const hr=table.querySelector('thead tr');
      if(hr&&!hr.querySelector('[data-pr-money-head]')){
        const th=document.createElement('th');th.dataset.prMoneyHead='1';th.textContent='系統計算加班費';hr.children[1]?.insertAdjacentElement('afterend',th);
      }
      [...table.querySelectorAll('tbody tr')].forEach(tr=>{
        const btn=tr.querySelector('[data-payroll-detail]');if(!btn)return;
        const staffId=btn.dataset.payrollDetail;
        let td=tr.querySelector('[data-pr-money-cell]');
        if(!td){td=document.createElement('td');td.dataset.prMoneyCell='1';tr.children[1]?.insertAdjacentElement('afterend',td);}
        if(version>=2){
          const row=snapshotStaff(s,staffId),original=Number(row?.overtimePayAmount)||0,corr=correctionAmount(s,staffId),finalAmount=original+corr;
          const expired=version>=6?(Number(row?.expiredCompLeavePayoutAmount)||0):0;
          td.dataset.snapshotLocked='1';
          td.innerHTML=`<span class="pr-money">${money(finalAmount)}</span><div class="ot-detail">原月結快照 ${money(original)}${expired?`・含到期換休折算 ${money(expired)}`:''}${corr?`・更正 ${corr>0?'+':''}${money(corr).replace('NT$ ','')}`:''}</div>`;
        }else{
          td.dataset.snapshotLocked='1';
          td.innerHTML='<span class="pr-money missing">舊月結</span><div class="ot-detail">此月份建立於金額快照功能前，僅保留當時的時數快照，不以目前薪資重新計算。</div>';
        }
      });
    }

    const main=document.getElementById('otMain');if(!main)return;
    let note=document.getElementById('payrollAmountSnapshotNotice');
    if(!note){
      note=document.createElement('section');note.id='payrollAmountSnapshotNotice';note.className='ot-rule-card';
      const correctionPanel=document.getElementById('payrollCorrectionPanel');
      if(correctionPanel)correctionPanel.insertAdjacentElement('beforebegin',note);else main.appendChild(note);
    }
    if(version>=2){
      const original=Number(s.summary?.totals?.overtimePayAmount)||0,corr=correctionAmount(s),finalAmount=original+corr;
      const expired=version>=6?(Number(s.summary?.totals?.expiredCompLeavePayoutAmount)||0):0;
      const expiredMinutes=version>=6?(Number(s.summary?.totals?.expiredCompLeavePayoutMinutes)||0):0;
      note.innerHTML=`<h3>🔒 月結金額快照</h3><p>本月份結算時的薪資基數、加班日別、計算公式與加班費金額均已鎖定保存。後續調薪或修改薪資基數不會改變本月原始結算。${version>=6?'換休到期未休部分會依原加班日期的薪資基數與原同日倍率順位自動折算。':''}</p><div class="ot-rule-highlight">原月結加班費：<strong>${money(original)}</strong>${expired?`　｜　其中到期換休折算：<strong>${money(expired)}</strong>（${dur(expiredMinutes)}）`:''}${corr?`　｜　月結後更正：<strong>${corr>0?'+':''}${money(corr).replace('NT$ ','')}</strong>`:''}　｜　目前認列：<strong>${money(finalAmount)}</strong></div>`;
    }else{
      note.innerHTML='<h3>🔒 舊版月結快照</h3><p>此月份是在「金額快照」功能上線前完成月結，因此保留原有時數快照，但不會用現在的薪資基數回頭重算金額。若需調整，請使用月結後更正單。</p>';
    }
  }

  function simpleItems(items,kind){
    if(!Array.isArray(items)||!items.length)return '<div class="ot-empty"><p>無資料。</p></div>';
    return `<div class="ot-table-shell"><table class="ot-table"><thead><tr><th>日期</th><th>類型／時間</th><th>時數</th><th>事由</th></tr></thead><tbody>${items.map(i=>`<tr><td>${rocDate(String(i.date)+'T00:00:00')}</td><td>${kind==='used'?'換休使用':escapeHTML(typeLabel(i.type))}<div class="ot-detail">${i.startTime||''}${i.startTime&&i.endTime?'–':''}${i.endTime||''}</div></td><td>${escapeHTML(dur(i.minutes))}</td><td>${escapeHTML(i.reason||'—')}</td></tr>`).join('')}</tbody></table></div>`;
  }

  function paySnapshotItems(items,emptyText='本月沒有領加班費的紀錄。'){
    if(!Array.isArray(items)||!items.length)return `<div class="ot-empty"><p>${escapeHTML(emptyText)}</p></div>`;
    return `<div class="ot-table-shell"><table class="ot-table"><thead><tr><th>來源日期</th><th>類型／時間</th><th>時數</th><th>薪資基數</th><th>加班費</th><th>計算依據</th></tr></thead><tbody>${items.map(i=>`<tr><td>${rocDate(String(i.date)+'T00:00:00')}${i.expiryDate?`<div class="ot-detail">到期：${rocDate(String(i.expiryDate)+'T00:00:00')}</div>`:''}</td><td>${escapeHTML(typeLabel(i.type))}<div class="ot-detail">${escapeHTML(i.startTime||'')}–${escapeHTML(i.endTime||'')}</div></td><td>${escapeHTML(dur(i.minutes))}${Number(i.usedMinutes)>0?`<div class="ot-detail">原取得 ${escapeHTML(dur(i.sourceEarnedMinutes))}，已休 ${escapeHTML(dur(i.usedMinutes))}</div>`:''}</td><td>${money(i.wageBase)}<div class="ot-detail">生效：${i.rateEffectiveFrom?rocDate(String(i.rateEffectiveFrom)+'T00:00:00'):'—'}<br>時薪基數 ${money(i.hourlyBase)}</div></td><td><strong>${money(i.calculatedAmount)}</strong></td><td><div class="ot-detail">${escapeHTML(i.calculationFormula||'—')}</div></td></tr>`).join('')}</tbody></table></div>`;
  }

  function openSnapshotDetail(staffId){
    const s=settlement();if(!s)return;
    const row=snapshotStaff(s,staffId);if(!row){toast('找不到本次月結的員工快照','warn');return;}
    const version=snapshotVersion(s),corr=correctionAmount(s,staffId),original=Number(row.overtimePayAmount)||0;
    const pay=version>=2?paySnapshotItems(row.overtimePayItems):simpleItems(row.overtimePayItems,'pay');
    const expired=version>=6?paySnapshotItems(row.expiredCompLeavePayoutItems,'本月沒有到期未休換休折算加班費。'):'';
    const expiredAmount=version>=6?(Number(row.expiredCompLeavePayoutAmount)||0):0;
    const expiredMinutes=version>=6?(Number(row.expiredCompLeavePayoutMinutes)||0):0;
    openModal({title:`薪資結算明細｜${row.name||'未命名'}`,html:`
      ${version>=2?`<div class="ot-rule-highlight" style="margin-top:0">原月結加班費：<strong>${money(original)}</strong>${expiredAmount?`｜其中到期換休折算：<strong>${money(expiredAmount)}</strong>（${dur(expiredMinutes)}）`:''}${corr?`｜月結後更正：<strong>${corr>0?'+':''}${money(corr).replace('NT$ ','')}</strong>`:''}｜目前認列：<strong>${money(original+corr)}</strong></div>`:'<div class="ot-rule-highlight" style="margin-top:0">此為舊版月結，只保存原始時數，不使用目前薪資回推歷史金額。</div>'}
      <section class="ot-rule-card"><h3>直接領加班費${version>=2?'（月結快照）':''}</h3>${pay}</section>
      ${version>=6?`<section class="ot-rule-card"><h3>換休到期未休折算加班費</h3><p style="margin-bottom:10px">依原加班日期的薪資基數與原同日倍率順位折算；已使用的換休視為先抵原加班前段。</p>${expired}</section>`:''}
      <section class="ot-rule-card"><h3>本月新增換休</h3>${simpleItems(row.compLeaveEarnedItems,'earned')}</section>
      <section class="ot-rule-card"><h3>本月使用換休</h3>${simpleItems(row.compLeaveUsedItems,'used')}</section>
      <div class="ot-rule-highlight">月底換休餘額：<strong>${dur(row.compLeaveBalanceMinutes)}</strong></div>`,large:true,showConfirm:false});
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('[data-payroll-detail]');
    if(!btn||state.page!=='overtime'||state.overtimeView!=='payroll'||!settlement())return;
    e.preventDefault();e.stopImmediatePropagation();openSnapshotDetail(btn.dataset.payrollDetail);
  },true);

  function update(){updateMineCard();applySnapshotMoney();}
  let q=false;const obs=new MutationObserver(ms=>{if(q||!ms.some(m=>m.addedNodes?.length))return;q=true;setTimeout(()=>{q=false;update();},30);});obs.observe(document.body,{childList:true,subtree:true});
  setTimeout(update,250);
})();
