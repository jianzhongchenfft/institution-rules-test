// V5.2 例假日政策補強：一般申請與月結更正皆不開放例假日加班
(() => {
  function cleanCorrectionType(){
    const sel=document.getElementById('prCorrDayType');
    if(!sel)return;
    const opt=[...sel.options].find(o=>o.value==='regular_day_off_overtime');
    if(opt)opt.remove();
    if(sel.value==='regular_day_off_overtime')sel.value='rest_day_overtime';
  }

  window.addEventListener('click',e=>{
    const b=e.target.closest?.('button');
    if(!b)return;
    const modal=b.closest?.('.modal');
    if(!modal)return;
    const corr=modal.querySelector('#prCorrDayType');
    if(corr?.value==='regular_day_off_overtime'&&(b.id==='modalConfirm'||b.textContent.trim()==='建立更正單')){
      e.preventDefault();
      e.stopImmediatePropagation();
      toast('例假日不開放加班計薪更正，請確認實際出勤日別。','warn');
    }
  },true);

  let queued=false;
  const observer=new MutationObserver(ms=>{
    if(queued)return;
    if(!ms.some(m=>Array.from(m.addedNodes||[]).some(n=>n.nodeType===1&&(n.id==='prCorrDayType'||n.querySelector?.('#prCorrDayType')))))return;
    queued=true;
    setTimeout(()=>{queued=false;cleanCorrectionType();},20);
  });
  observer.observe(document.body,{childList:true,subtree:true});
  cleanCorrectionType();
})();
