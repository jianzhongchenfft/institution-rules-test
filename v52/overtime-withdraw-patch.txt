// V5.2 加班／換休：待審核申請撤回（撤回原因選填）
(() => {
  const style=document.createElement('style');
  style.id='v52-overtime-withdraw-style';
  style.textContent=`
    [data-wf-withdraw]{white-space:nowrap}
    .wd-note{margin-top:10px;color:#667085;font-size:11.5px;line-height:1.6}
  `;
  document.head.appendChild(style);

  function normalizeButtons(){
    document.querySelectorAll('[data-wf-withdraw]').forEach(btn=>{
      btn.textContent='撤回申請';
      btn.title='僅待審核中的申請可由本人撤回';
    });
  }

  function ownRow(kind,id){
    const rows=kind==='leave'?(state.compLeaveRows||[]):(state.overtimeRows||[]);
    return rows.find(r=>String(r.id)===String(id)&&String(r.applicant_staff_id)===String(state.profile?.id))||null;
  }

  function openWithdrawModal(kind,id){
    const row=ownRow(kind,id);
    if(!row){toast('找不到這筆申請，請重新整理後再試','warn');return;}
    if(row.status!=='pending'){
      toast('只有待審核中的申請可以撤回','warn');
      return;
    }

    const date=kind==='leave'?row.leave_date:row.work_date;
    const time=row.start_time&&row.end_time?`${String(row.start_time).slice(0,5)}–${String(row.end_time).slice(0,5)}`:'—';
    openModal({
      title:'撤回待審核申請',
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">
          確定要撤回這筆待審核申請嗎？撤回後原申請會保留並標示「已撤回」，不會刪除；如仍需申請，請重新建立一筆新申請。
        </div>
        <div class="ot-detail" style="margin-top:10px">日期：${escapeHTML(date||'—')}　｜　時間：${escapeHTML(time)}</div>
        <div class="field" style="margin-top:14px">
          <label>撤回原因（選填）</label>
          <textarea id="wdReason" placeholder="例如：時間填寫錯誤，先撤回後重新申請"></textarea>
          <div class="wd-note">未填寫原因也可以撤回；系統仍會保留撤回人與撤回時間。</div>
        </div>`,
      confirmText:'確認撤回',
      onConfirm:()=>submitWithdraw(kind,id)
    });
  }

  async function submitWithdraw(kind,id){
    const row=ownRow(kind,id);
    if(!row||row.status!=='pending'){
      toast('此申請目前已不是待審核狀態，請重新整理後再確認','warn');
      return false;
    }

    const input=String(document.getElementById('wdReason')?.value||'').trim();
    // 現行資料庫政策要求 withdraw_reason 必須有值；使用者未填時以「未填寫」保存，介面仍視為選填。
    const reason=input||'未填寫';
    const table=kind==='leave'?'comp_leave_usages':'overtime_requests';
    showLoading(true);
    try{
      const now=new Date().toISOString();
      const {data,error}=await db.from(table)
        .update({
          status:'withdrawn',
          withdraw_reason:reason,
          withdrawn_by:state.user.id,
          withdrawn_at:now,
          updated_at:now
        })
        .eq('id',id)
        .eq('applicant_staff_id',state.profile.id)
        .eq('status','pending')
        .select('id,status')
        .maybeSingle();
      if(error)throw error;
      if(!data)throw new Error('申請狀態已變更，請重新整理後再操作');

      closeModal();
      await navigate('overtime');
      toast('申請已撤回，原紀錄已保留','success');
      return true;
    }catch(err){
      console.error('撤回申請失敗',err);
      toast('撤回失敗：'+(err?.message||err),'error');
      return false;
    }finally{
      showLoading(false);
    }
  }

  // 攔截舊版「撤回」按鈕事件，改用本版流程；使用 capture 確保不會同時觸發舊版必填原因流程。
  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('[data-wf-withdraw]');
    if(!btn)return;
    e.preventDefault();
    e.stopPropagation();
    e.stopImmediatePropagation();
    const [kind,id]=String(btn.dataset.wfWithdraw||'').split(':');
    if(!['overtime','leave'].includes(kind)||!id){toast('撤回資料格式不正確，請重新整理頁面','warn');return;}
    openWithdrawModal(kind,id);
  },true);

  let queued=false;
  const observer=new MutationObserver(ms=>{
    if(queued||!ms.some(m=>m.addedNodes?.length))return;
    queued=true;
    setTimeout(()=>{queued=false;normalizeButtons();},20);
  });
  observer.observe(document.body,{childList:true,subtree:true});

  normalizeButtons();
})();
