// V5.2 加班／換休流程補強：月份篩選、重複防呆、待審核撤回
(() => {
  const s=document.createElement('style');
  s.id='v52-overtime-workflow-style';
  s.textContent='.ot-status.withdrawn{background:#f2f4f7;color:#667085}.wf-month-empty{margin-top:10px}';
  document.head.appendChild(s);

  Object.assign(state,{workflowEditingOvertimeId:null,workflowEditingLeaveId:null});
  const active=new Set(['draft','pending','approved']);
  const ownO=()=> (state.overtimeRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile?.id));
  const ownL=()=> (state.compLeaveRows||[]).filter(r=>String(r.applicant_staff_id)===String(state.profile?.id));
  const norm=v=>{const x=String(v||'').slice(0,5);return /^([01]\d|2[0-3]):[0-5]\d$/.test(x)?x:null;};
  const sel=p=>{const h=document.getElementById(p+'Hour')?.value||'',m=document.getElementById(p+'Minute')?.value||'';return h&&m?`${h}:${m}`:null;};
  const min=v=>{const x=norm(v);if(!x)return null;const [h,m]=x.split(':').map(Number);return h*60+m;};
  const overlap=(a,b,c,d)=>{const x=[min(a),min(b),min(c),min(d)];return x.every(Number.isFinite)&&x[0]<x[3]&&x[1]>x[2];};

  function monthRange(){
    const month=/^\d{4}-\d{2}$/.test(state.overtimeMonth||'')?state.overtimeMonth:isoDate(new Date()).slice(0,7);
    const [y,m]=month.split('-').map(Number);
    return {start:`${y}-${String(m).padStart(2,'0')}-01`,end:isoDate(new Date(y,m,0))};
  }
  function inMonth(d){const {start,end}=monthRange();return !!d&&String(d)>=start&&String(d)<=end;}

  function monthControls(){
    const input=document.getElementById('otMonth'),label=document.querySelector('label[for="otMonth"]');
    if(!input)return;
    const show=['mine','usage','payroll'].includes(state.overtimeView);
    input.style.display=show?'':'none'; if(label)label.style.display=show?'':'none';
  }

  function addWithdraw(tr,kind,row){
    const span=tr.querySelector('.ot-status');
    if(row.status==='withdrawn'&&span){span.classList.add('withdrawn');span.textContent='已撤回';}
    if(row.status!=='pending')return;
    const box=tr.querySelector('.ot-actions'); if(!box||box.querySelector('[data-wf-withdraw]'))return;
    const b=document.createElement('button');b.className='btn sm';b.type='button';b.textContent='撤回';b.dataset.wfWithdraw=`${kind}:${row.id}`;
    b.addEventListener('click',()=>openWithdraw(kind,row.id)); box.appendChild(b);
  }

  function applyMonthFilter(){
    if(state.page!=='overtime')return; monthControls();
    const view=state.overtimeView,old=document.getElementById('wfMonthEmpty');
    if(!['mine','usage'].includes(view)){old?.remove();return;}
    const rows=view==='mine'?ownO():ownL(),trs=[...document.querySelectorAll('#otMain .ot-table tbody tr')];
    if(!trs.length||trs.length!==rows.length)return;
    let shown=0;
    rows.forEach((r,i)=>{const ok=inMonth(view==='mine'?r.work_date:r.leave_date);trs[i].style.display=ok?'':'none';if(ok)shown++;addWithdraw(trs[i],view==='mine'?'overtime':'leave',r);});
    const shell=document.querySelector('#otMain .ot-table-shell'); if(shell)shell.style.display=shown?'':'none';
    if(!shown&&rows.length&&!old){const d=document.createElement('div');d.id='wfMonthEmpty';d.className='ot-empty wf-month-empty';d.innerHTML='<div class="icon">📅</div><h3>本月份沒有紀錄</h3><p>請切換其他月份查看歷史申請。</p>';document.getElementById('otMain')?.appendChild(d);} else if(shown)old?.remove();
  }

  function openWithdraw(kind,id){
    const row=(kind==='leave'?state.compLeaveRows:state.overtimeRows||[]).find(r=>String(r.id)===String(id));
    if(!row||row.status!=='pending'){toast('此申請目前已不是待審核狀態','warn');return;}
    openModal({title:'撤回待審核申請',html:'<div class="ot-rule-highlight" style="margin-top:0">撤回後原申請仍會保留並標示「已撤回」，不會刪除。</div><div class="field" style="margin-top:12px"><label>撤回原因 *</label><textarea id="wfWithdrawReason" placeholder="例如：時間填寫錯誤，先撤回後重新申請"></textarea></div>',confirmText:'確認撤回',onConfirm:()=>submitWithdraw(kind,id)});
  }
  async function submitWithdraw(kind,id){
    const reason=String(document.getElementById('wfWithdrawReason')?.value||'').trim();if(!reason){toast('請填寫撤回原因','warn');return false;}
    if(!confirm('確定撤回這筆待審核申請嗎？\n\n原紀錄會保留，不會刪除。'))return false;
    const table=kind==='leave'?'comp_leave_usages':'overtime_requests';showLoading(true);
    try{
      const {data,error}=await db.from(table).update({status:'withdrawn',withdraw_reason:reason,withdrawn_by:state.user.id,withdrawn_at:new Date().toISOString(),updated_at:new Date().toISOString()}).eq('id',id).eq('status','pending').select('id').maybeSingle();
      if(error)throw error;if(!data)throw new Error('申請狀態已變更，請重新整理後再操作');closeModal();navigate('overtime');toast('申請已撤回並保留紀錄','success');return true;
    }catch(err){console.error(err);toast('撤回失敗：'+(err?.message||err),'error');return false;}finally{showLoading(false);}
  }

  function inferO(){
    if(state.workflowEditingOvertimeId)return state.workflowEditingOvertimeId;
    const d=document.getElementById('otWorkDate')?.value,t=document.getElementById('otType')?.value,a=sel('otStart'),b=sel('otEnd'),c=Number(document.getElementById('otCaseCount')?.value||0);
    const x=ownO().filter(r=>['draft','rejected'].includes(r.status)&&r.work_date===d&&r.request_type===t).filter(r=>t==='holiday_visit_comp'?Number(r.holiday_case_count||0)===c:norm(r.start_time)===a&&norm(r.end_time)===b);
    return x.length===1?x[0].id:null;
  }
  function inferL(){
    if(state.workflowEditingLeaveId)return state.workflowEditingLeaveId;
    const d=document.getElementById('clDate')?.value,a=sel('clStart'),b=sel('clEnd');
    const x=ownL().filter(r=>['draft','rejected'].includes(r.status)&&r.leave_date===d&&norm(r.start_time)===a&&norm(r.end_time)===b);return x.length===1?x[0].id:null;
  }

  function validOvertime(){
    const d=document.getElementById('otWorkDate')?.value,t=document.getElementById('otType')?.value;if(!d||!t)return true;
    const id=inferO(),rows=ownO().filter(r=>active.has(r.status)&&String(r.id)!==String(id||'')&&r.work_date===d);
    if(t==='holiday_visit_comp'){
      const c=Number(document.getElementById('otCaseCount')?.value||0),same=rows.filter(r=>r.request_type==='holiday_visit_comp');
      if(same.some(r=>Number(r.holiday_case_count||0)===c)){toast('同一天相同家訪案數的假日家訪換休已存在，請勿重複申請','warn');return false;}
      return !same.length||confirm('同一天已經有另一筆假日家訪換休申請。\n\n若這不是重複申請，確定要繼續嗎？');
    }
    const a=sel('otStart'),b=sel('otEnd');if(!a||!b)return true;
    const timedTypes=new Set(['overtime','holiday_overtime','rest_day_overtime','national_holiday_overtime','regular_day_off_overtime']);
    const timed=rows.filter(r=>timedTypes.has(r.request_type));
    if(timed.some(r=>norm(r.start_time)===a&&norm(r.end_time)===b)){toast('同一天相同起訖時間已存在加班申請，請勿重複送出','warn');return false;}
    const hit=timed.find(r=>overlap(a,b,r.start_time,r.end_time));
    if(hit){toast(`這段加班時間與既有申請（${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)}）重疊，請調整時間後再申請。`,'warn');return false;}
    return true;
  }
  function validLeave(){
    const d=document.getElementById('clDate')?.value,a=sel('clStart'),b=sel('clEnd');if(!d||!a||!b)return true;const id=inferL();
    const hit=ownL().find(r=>active.has(r.status)&&String(r.id)!==String(id||'')&&r.leave_date===d&&overlap(a,b,r.start_time,r.end_time));
    if(hit){toast(`換休時段與既有申請 ${String(hit.start_time||'').slice(0,5)}–${String(hit.end_time||'').slice(0,5)} 重疊，請調整時間`,'warn');return false;}return true;
  }

  document.addEventListener('click',e=>{
    const b=e.target.closest?.('button');if(!b)return;
    if(b.id==='otAddBtn')state.workflowEditingOvertimeId=null;if(b.dataset.otEdit)state.workflowEditingOvertimeId=b.dataset.otEdit;
    if(b.id==='clAddBtn')state.workflowEditingLeaveId=null;if(b.dataset.clEdit)state.workflowEditingLeaveId=b.dataset.clEdit;
    const modal=b.closest('.modal');if(!modal)return;const submit=b.id==='modalConfirm'||b.textContent.trim()==='儲存草稿';if(!submit)return;
    const ok=modal.querySelector('#otWorkDate')?validOvertime():modal.querySelector('#clDate')?validLeave():true;
    if(!ok){e.preventDefault();e.stopImmediatePropagation();}
  },true);

  function enhance(){if(state.page==='overtime')applyMonthFilter();}
  const base=navigate;navigate=function(id){const r=base(id);if(id==='overtime'){state.workflowEditingOvertimeId=null;state.workflowEditingLeaveId=null;setTimeout(enhance,220);}return r;};
  let queued=false;const obs=new MutationObserver(ms=>{if(queued)return;if(!ms.some(m=>m.addedNodes?.length))return;queued=true;setTimeout(()=>{queued=false;enhance();},25);});obs.observe(document.body,{childList:true,subtree:true});
  if(state.profile&&state.page==='overtime')setTimeout(enhance,220);
})();
