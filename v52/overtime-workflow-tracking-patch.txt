// 固定「週一至週五／週六／週日」日別判斷已移除；例假提醒改讀員工主檔設定。
(() => {
  window.addEventListener('click',e=>{
    const b=e.target.closest?.('button');
    if(!b)return;
    if(b.id==='otAddBtn')state.workflowEditingOvertimeId=null;
    if(b.dataset.otEdit)state.workflowEditingOvertimeId=b.dataset.otEdit;
    if(b.id==='clAddBtn')state.workflowEditingLeaveId=null;
    if(b.dataset.clEdit)state.workflowEditingLeaveId=b.dataset.clEdit;
  },true);
})();
