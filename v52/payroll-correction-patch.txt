// V5.2 月結後更正單：只處理已月結報表的附加式更正
// 本檔不再計算或覆寫「目前剩餘換休」，也不再攔截「使用換休」申請流程。
(() => {
  const correctionStyle=document.createElement('style');
  correctionStyle.id='v52-payroll-correction-style';
  correctionStyle.textContent=`
    .pc-adjust{font-weight:900;white-space:nowrap}
    .pc-adjust.plus{color:#237a4b}.pc-adjust.minus{color:#b42318}.pc-adjust.zero{color:#667085}
    .pc-flow{display:flex;align-items:center;gap:5px;flex-wrap:wrap;white-space:nowrap}
    .pc-flow .orig{color:#667085}.pc-flow .arrow{color:#98a2b3}.pc-flow .final{font-weight:900;color:#344054}
    .pc-history{display:grid;gap:7px;margin-top:10px}
    .pc-history-item{border:1px solid var(--line);border-radius:10px;padding:10px 11px;background:#fff}
    .pc-history-top{display:flex;gap:10px;justify-content:space-between;align-items:flex-start;flex-wrap:wrap}
    .pc-history-main{font-size:12px;font-weight:900;color:#344054}.pc-history-sub{font-size:11px;color:#667085;line-height:1.55;margin-top:4px}
    @media(max-width:560px){.pc-flow{white-space:normal}}
  `;
  document.head.appendChild(correctionStyle);

  Object.assign(state,{
    payrollCorrections:state.payrollCorrections||[],
    payrollCorrectionsLoaded:false,
    overtimeStaffDirectory:state.overtimeStaffDirectory||[],
    payrollCorrectionLoading:false
  });

  function pcCanReview(){
    return !!state.profile&&['organization_manager','admin'].includes(state.profile.role);
  }

  function pcMonthStart(month){
    const safe=/^\d{4}-\d{2}$/.test(month||'')?month:isoDate(new Date()).slice(0,7);
    return safe+'-01';
  }

  function pcSettlement(){
    const start=pcMonthStart(state.overtimeMonth);
    return (state.payrollSettlements||[]).find(r=>r.settlement_month===start)||null;
  }

  function pcCorrectionsForSettlement(settlementId){
    return (state.payrollCorrections||[]).filter(c=>String(c.settlement_id)===String(settlementId));
  }

  function pcSum(list,type){
    return list.filter(c=>c.adjustment_type===type).reduce((s,c)=>s+(Number(c.adjustment_minutes)||0),0);
  }

  function pcDuration(minutes){
    const n=Math.max(0,Number(minutes)||0),h=Math.floor(n/60),m=n%60;
    if(h&&m)return `${h}小時${m}分`;
    if(h)return `${h}小時`;
    return `${m}分鐘`;
  }

  function pcSigned(minutes){
    const n=Number(minutes)||0;
    if(!n)return '<span class="pc-adjust zero">0</span>';
    return `<span class="pc-adjust ${n>0?'plus':'minus'}">${n>0?'+':'−'}${escapeHTML(pcDuration(Math.abs(n)))}</span>`;
  }

  function pcTypeLabel(type){
    return ({overtime_pay:'加班計薪',comp_leave_earned:'換休取得',comp_leave_used:'換休使用'})[type]||type;
  }

  function pcRoleLabel(role){return roleLabels?.[role]||role||'—';}

  async function pcLoadData(force=false){
    if(!state.profile||state.payrollCorrectionLoading)return;
    if(state.payrollCorrectionsLoaded&&!force){
      queueMicrotask(pcEnhance);
      return;
    }

    state.payrollCorrectionLoading=true;
    try{
      const correctionPromise=db.from('payroll_settlement_corrections')
        .select('id,settlement_id,staff_id,staff_name,staff_role,adjustment_type,adjustment_minutes,reason,created_by,created_by_name,created_at')
        .order('created_at',{ascending:false});
      const staffPromise=pcCanReview()
        ? db.from('staff_users').select('id,display_name,role,is_active').eq('is_active',true).order('display_name')
        : Promise.resolve({data:[],error:null});
      const [corrRes,staffRes]=await Promise.all([correctionPromise,staffPromise]);
      if(corrRes.error)throw corrRes.error;
      if(staffRes.error)throw staffRes.error;
      state.payrollCorrections=corrRes.data||[];
      state.overtimeStaffDirectory=staffRes.data||[];
      state.payrollCorrectionsLoaded=true;
      queueMicrotask(pcEnhance);
    }catch(err){
      console.error('讀取月結更正資料失敗',err);
      toast('讀取月結更正資料失敗：'+(err?.message||err),'error');
    }finally{
      state.payrollCorrectionLoading=false;
    }
  }

  function pcUpdateCard(label,value,foot=null){
    document.querySelectorAll('#content .stat-card').forEach(card=>{
      const head=card.querySelector('.stat-head span:first-child');
      if(head?.textContent.trim()!==label)return;
      const val=card.querySelector('.stat-value');
      if(val&&val.textContent!==value)val.textContent=value;
      if(foot){
        const f=card.querySelector('.stat-foot');
        if(f&&f.textContent!==foot)f.textContent=foot;
      }
    });
  }

  function pcCorrectedRows(settlement){
    const base=Array.isArray(settlement?.summary?.staff)?settlement.summary.staff:[];
    const map=new Map();
    base.forEach(r=>map.set(String(r.staffId),{
      staffId:String(r.staffId),name:r.name||'未命名',role:r.role||'',
      originalPay:Number(r.overtimePayMinutes)||0,
      originalEarned:Number(r.compLeaveEarnedMinutes)||0,
      originalUsed:Number(r.compLeaveUsedMinutes)||0,
      originalBalance:Number(r.compLeaveBalanceMinutes)||0,
      corrPay:0,corrEarned:0,corrUsed:0
    }));

    pcCorrectionsForSettlement(settlement.id).forEach(c=>{
      const key=String(c.staff_id);
      if(!map.has(key)){
        map.set(key,{
          staffId:key,name:c.staff_name||'未命名',role:c.staff_role||'',
          originalPay:0,originalEarned:0,originalUsed:0,originalBalance:0,
          corrPay:0,corrEarned:0,corrUsed:0
        });
      }
      const r=map.get(key);
      if(c.adjustment_type==='overtime_pay')r.corrPay+=Number(c.adjustment_minutes)||0;
      if(c.adjustment_type==='comp_leave_earned')r.corrEarned+=Number(c.adjustment_minutes)||0;
      if(c.adjustment_type==='comp_leave_used')r.corrUsed+=Number(c.adjustment_minutes)||0;
    });

    return [...map.values()].map(r=>({...r,
      finalPay:Math.max(0,r.originalPay+r.corrPay),
      finalEarned:Math.max(0,r.originalEarned+r.corrEarned),
      finalUsed:Math.max(0,r.originalUsed+r.corrUsed),
      finalBalance:Math.max(0,r.originalBalance+r.corrEarned-r.corrUsed)
    })).sort((a,b)=>String(a.name).localeCompare(String(b.name),'zh-Hant'));
  }

  function pcFlow(original,adjusted,finalValue){
    return `<div class="pc-flow"><span class="orig">${escapeHTML(pcDuration(original))}</span>${adjusted?`<span>${pcSigned(adjusted)}</span>`:''}<span class="arrow">→</span><span class="final">${escapeHTML(pcDuration(finalValue))}</span></div>`;
  }

  function pcCorrectionPanelHTML(settlement){
    const corrections=pcCorrectionsForSettlement(settlement.id);
    const rows=pcCorrectedRows(settlement);
    return `
      <section id="payrollCorrectionPanel" class="ot-rule-card">
        <div style="display:flex;justify-content:space-between;gap:12px;align-items:flex-start;flex-wrap:wrap">
          <div><h3>月結後更正</h3><p>原月結資料保持鎖定；後續調整以獨立更正單追加，不覆蓋原始結算。</p></div>
          ${pcCanReview()?'<button id="pcAddBtn" class="btn primary">＋ 新增月結更正</button>':''}
        </div>
        <div class="ot-rule-highlight">更正單建立後不可修改或刪除；若更正單本身填錯，請再新增一筆反向更正，保留完整稽核軌跡。</div>
        <div style="margin-top:14px"><h3>更正後月結摘要</h3>
          ${rows.length?`<div class="ot-table-shell"><table class="ot-table"><thead><tr><th>員工</th><th>加班計薪</th><th>換休取得</th><th>換休使用</th><th>更正後月底換休餘額</th></tr></thead><tbody>${rows.map(r=>`<tr><td><div class="ot-review-name">${escapeHTML(r.name)}</div><div class="ot-detail">${escapeHTML(pcRoleLabel(r.role))}</div></td><td>${pcFlow(r.originalPay,r.corrPay,r.finalPay)}</td><td>${pcFlow(r.originalEarned,r.corrEarned,r.finalEarned)}</td><td>${pcFlow(r.originalUsed,r.corrUsed,r.finalUsed)}</td><td><strong>${escapeHTML(pcDuration(r.finalBalance))}</strong></td></tr>`).join('')}</tbody></table></div>`:'<div class="ot-empty"><p>目前沒有可顯示的結算資料。</p></div>'}
        </div>
        <div style="margin-top:14px"><h3>更正紀錄</h3>
          ${corrections.length?`<div class="pc-history">${corrections.map(c=>`<div class="pc-history-item"><div class="pc-history-top"><div><div class="pc-history-main">${escapeHTML(c.staff_name)}｜${escapeHTML(pcTypeLabel(c.adjustment_type))}　${pcSigned(c.adjustment_minutes)}</div><div class="pc-history-sub">原因：${escapeHTML(c.reason)}<br>更正人：${escapeHTML(c.created_by_name||'—')}・${rocDate(c.created_at)}</div></div></div></div>`).join('')}</div>`:'<div class="ot-empty"><div class="icon">🧾</div><h3>尚無月結更正</h3><p>若結算後才發現漏算或誤算，可由機構負責人建立更正單。</p></div>'}
        </div>
      </section>`;
  }

  function pcUpdatePayrollUI(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll'||!state.payrollCorrectionsLoaded)return;
    const settlement=pcSettlement();
    if(!settlement)return;

    const corr=pcCorrectionsForSettlement(settlement.id);
    const payAdj=pcSum(corr,'overtime_pay');
    const earnedAdj=pcSum(corr,'comp_leave_earned');
    const usedAdj=pcSum(corr,'comp_leave_used');
    pcUpdateCard('本月待計薪加班',pcDuration((Number(settlement.total_overtime_pay_minutes)||0)+payAdj),payAdj?'原月結＋月結更正':'僅計已核准且選擇領加班費');
    pcUpdateCard('本月新增換休',pcDuration((Number(settlement.total_comp_leave_earned_minutes)||0)+earnedAdj),earnedAdj?'原月結＋月結更正':'已核准且選擇換休');
    pcUpdateCard('本月使用換休',pcDuration((Number(settlement.total_comp_leave_used_minutes)||0)+usedAdj),usedAdj?'原月結＋月結更正':'僅計已核准使用');

    const main=document.getElementById('otMain');
    if(!main)return;
    document.getElementById('payrollCorrectionPanel')?.remove();
    main.insertAdjacentHTML('beforeend',pcCorrectionPanelHTML(settlement));
    document.getElementById('pcAddBtn')?.addEventListener('click',pcOpenCorrectionModal);
  }

  function pcMinuteOptions(type,selected='00'){
    const vals=type==='comp_leave_used'?['00','30']:['00','10','20','30','40','50'];
    return vals.map(v=>`<option value="${v}" ${v===selected?'selected':''}>${v}</option>`).join('');
  }

  function pcOpenCorrectionModal(){
    if(!pcCanReview()){
      toast('目前帳號沒有建立月結更正的權限','warn');
      return;
    }
    const settlement=pcSettlement();
    if(!settlement){
      toast('只有已完成月結的月份才能建立更正單','warn');
      return;
    }
    const staff=(state.overtimeStaffDirectory||[]).filter(s=>s.is_active!==false);
    if(!staff.length){
      toast('目前沒有可選擇的人員','warn');
      return;
    }

    openModal({
      title:'新增月結更正單',
      html:`
        <div class="ot-rule-highlight" style="margin-top:0">更正月份：<strong>${escapeHTML(state.overtimeMonth)}</strong>。原月結不會被修改，本次只新增一筆調整紀錄。</div>
        <div class="form-grid" style="margin-top:12px">
          <div class="field"><label>員工 *</label><select id="pcStaff">${staff.map(s=>`<option value="${escapeHTML(s.id)}">${escapeHTML(s.display_name)}｜${escapeHTML(pcRoleLabel(s.role))}</option>`).join('')}</select></div>
          <div class="field"><label>更正項目 *</label><select id="pcType"><option value="overtime_pay">加班計薪</option><option value="comp_leave_earned">換休取得</option><option value="comp_leave_used">換休使用</option></select></div>
          <div class="field"><label>調整方向 *</label><select id="pcDirection"><option value="1">增加認列</option><option value="-1">減少認列</option></select></div>
          <div class="field"><label>更正時數 *</label><div class="ot-time-pair"><input id="pcHours" type="number" min="0" max="999" step="1" value="0"><select id="pcMinutes">${pcMinuteOptions('overtime_pay')}</select></div><div id="pcMinuteHelp" class="help">加班計薪／換休取得以10分鐘為間隔。</div></div>
          <div class="field full"><label>更正原因 *</label><textarea id="pcReason" placeholder="例如：薪資結算後發現加班漏列30分鐘"></textarea></div>
        </div>`,
      confirmText:'建立更正單',
      onConfirm:pcSubmitCorrection,
      large:true
    });

    document.getElementById('pcType')?.addEventListener('change',()=>{
      const type=document.getElementById('pcType').value;
      const min=document.getElementById('pcMinutes');
      min.innerHTML=pcMinuteOptions(type,'00');
      document.getElementById('pcMinuteHelp').textContent=type==='comp_leave_used'
        ? '換休使用以30分鐘為間隔。'
        : '加班計薪／換休取得以10分鐘為間隔。';
    });
  }

  async function pcSubmitCorrection(){
    const settlement=pcSettlement();
    if(!settlement){toast('找不到本月份的月結資料','error');return false;}

    const staffId=document.getElementById('pcStaff')?.value;
    const staff=(state.overtimeStaffDirectory||[]).find(s=>String(s.id)===String(staffId));
    const type=document.getElementById('pcType')?.value;
    const direction=Number(document.getElementById('pcDirection')?.value||1);
    const hours=Math.max(0,Number(document.getElementById('pcHours')?.value)||0);
    const minutes=Math.max(0,Number(document.getElementById('pcMinutes')?.value)||0);
    const total=(hours*60+minutes)*direction;
    const reason=String(document.getElementById('pcReason')?.value||'').trim();

    if(!staff){toast('請選擇員工','warn');return false;}
    if(!['overtime_pay','comp_leave_earned','comp_leave_used'].includes(type)){toast('請選擇更正項目','warn');return false;}
    if(total===0){toast('更正時數不可為0','warn');return false;}
    if(!reason){toast('請填寫更正原因','warn');return false;}
    if(!confirm(`確定建立這筆月結更正嗎？\n\n${staff.display_name}｜${pcTypeLabel(type)}｜${direction>0?'+':'−'}${pcDuration(Math.abs(total))}\n\n建立後不可直接修改或刪除。`))return false;

    showLoading(true);
    try{
      const {error}=await db.from('payroll_settlement_corrections').insert({
        settlement_id:settlement.id,
        staff_id:staff.id,
        staff_name:staff.display_name,
        staff_role:staff.role,
        adjustment_type:type,
        adjustment_minutes:total,
        reason,
        created_by:state.user.id,
        created_by_name:state.profile.display_name
      });
      if(error)throw error;
      closeModal();
      await pcLoadData(true);
      toast('月結更正單已建立','success');
      return true;
    }catch(err){
      console.error(err);
      toast('建立月結更正失敗：'+(err?.message||err),'error');
      return false;
    }finally{
      showLoading(false);
    }
  }

  function pcEnhance(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll')return;
    if(!state.payrollCorrectionsLoaded){
      pcLoadData();
      return;
    }
    pcUpdatePayrollUI();
  }

  document.addEventListener('click',e=>{
    if(e.target.closest?.('#otReload')&&state.page==='overtime'&&state.overtimeView==='payroll'){
      setTimeout(()=>pcLoadData(true),0);
    }
  },true);

  let scheduled=false;
  const observer=new MutationObserver(mutations=>{
    if(scheduled)return;
    const relevant=mutations.some(m=>Array.from(m.addedNodes||[]).some(n=>
      n.nodeType===1&&(n.id==='otMain'||n.querySelector?.('#otMain'))
    ));
    if(!relevant)return;
    scheduled=true;
    queueMicrotask(()=>{
      scheduled=false;
      pcEnhance();
    });
  });
  observer.observe(document.body,{childList:true,subtree:true});
})();
