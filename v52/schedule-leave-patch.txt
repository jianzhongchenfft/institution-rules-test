  function leaveISO(date){
    const d = new Date(date);
    const y = d.getFullYear();
    const m = String(d.getMonth()+1).padStart(2,'0');
    const day = String(d.getDate()).padStart(2,'0');
    return `${y}-${m}-${day}`;
  }

  function leaveTime(value){
    return String(value || '').slice(0,5);
  }

  function leaveDateLabel(value){
    const d = new Date(value + 'T00:00:00');
    return `${d.getMonth()+1}/${d.getDate()}`;
  }

  function leaveWorkerName(id){
    return leaveDraft?.workers?.find(x=>x.id===id)?.worker_name || '未指定';
  }

  function leaveCaseName(id){
    return leaveDraft?.cases?.find(x=>x.id===id)?.case_name || '未找到個案';
  }

  function leaveItemsFor(scheduleId){
    return (leaveDraft?.items || []).filter(x=>x.schedule_id===scheduleId);
  }

  function leaveScheduleOverlaps(row,startDate,startTime,endDate,endTime,isAllDay){
    const rowStart = new Date(`${row.service_date}T${leaveTime(row.start_time)}:00`);
    const rowEnd = new Date(`${row.service_date}T${leaveTime(row.end_time)}:00`);
    const leaveStart = isAllDay
      ? new Date(startDate+'T00:00:00')
      : new Date(`${startDate}T${startTime}:00`);
    const leaveEnd = isAllDay
      ? new Date(endDate+'T23:59:59')
      : new Date(`${endDate}T${endTime}:00`);
    return rowStart < leaveEnd && rowEnd > leaveStart;
  }

    const checked = $('leaveAllDay')?.checked;

  async function findAffectedSchedules(){
    const workerId = $('leaveWorker')?.value;
    const startDate = $('leaveStartDate')?.value;
    const startTime = $('leaveStartTime')?.value;
    const endDate = $('leaveEndDate')?.value;
    const endTime = $('leaveEndTime')?.value;
    const isAllDay = !!$('leaveAllDay')?.checked;
    const reason = $('leaveReason')?.value?.trim() || null;

    if(!workerId || !startDate || !endDate){
      toast('請選擇居服員並填寫請假日期。','error');
      return;
    }
    if(!isAllDay && (!startTime || !endTime)){
      toast('請填寫請假開始與結束時間。','error');
      return;
    }
    if(endDate < startDate || (!isAllDay && endDate===startDate && endTime<=startTime)){
      toast('請假結束時間必須晚於開始時間。','error');
      return;
    }

    const btn = $('modalConfirm');
    if(btn) btn.disabled = true;
    showLoading(true);

    try{
      const schedulesRes = await db.from('service_schedules')
        .select('id,service_date,start_time,end_time,case_id,care_worker_id,original_care_worker_id,staffing_status,schedule_status,notes')
        .eq('care_worker_id',workerId)
        .neq('schedule_status','cancelled')
        .gte('service_date',startDate)
        .lte('service_date',endDate)
        .order('service_date')
        .order('start_time');
      if(schedulesRes.error) throw schedulesRes.error;

      const affected = (schedulesRes.data || []).filter(row=>
        leaveScheduleOverlaps(row,startDate,startTime,endDate,endTime,isAllDay)
      );

      const caseIds = [...new Set(affected.map(x=>x.case_id).filter(Boolean))];
      const scheduleIds = affected.map(x=>x.id);

      const [workersRes,casesRes,itemsRes] = await Promise.all([
        db.from('care_workers').select('id,worker_name,employment_status').eq('employment_status','active').order('worker_name'),
        caseIds.length ? db.from('care_cases').select('id,case_name').in('id',caseIds) : Promise.resolve({data:[],error:null}),
        scheduleIds.length ? db.from('schedule_service_items').select('schedule_id,service_code,quantity').in('schedule_id',scheduleIds).order('service_code') : Promise.resolve({data:[],error:null})
      ]);
      if(workersRes.error) throw workersRes.error;
      if(casesRes.error) throw casesRes.error;
      if(itemsRes.error) throw itemsRes.error;

      leaveDraft = {
        workerId,startDate,startTime,endDate,endTime,isAllDay,reason,
        schedules:affected,
        workers:workersRes.data || [],
        cases:casesRes.data || [],
        items:itemsRes.data || []
      };

      openLeaveAffectedModal();
    }catch(err){
      console.error(err);
      toast('讀取受影響班表失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(btn && document.body.contains(btn)) btn.disabled = false;
    }
  }

  function leaveSelectedActions(){
    return (leaveDraft?.schedules || []).map(row=>{
      const action=document.querySelector(`[data-leave-action="${row.id}"]`)?.value || 'pending';
      const substituteWorkerId=document.querySelector(`[data-leave-substitute="${row.id}"]`)?.value || null;
      return {row,action,substituteWorkerId};
    });
  }

  async function leaveConflictCount(actions){
    const substitutions=actions.filter(x=>x.action==='substitute' && x.substituteWorkerId);
    if(!substitutions.length) return 0;

    const affectedIds=new Set(actions.map(x=>x.row.id));
    let count=0;

    const groups=new Map();
    substitutions.forEach(item=>{
      if(!groups.has(item.substituteWorkerId)) groups.set(item.substituteWorkerId,[]);
      groups.get(item.substituteWorkerId).push(item);
    });

    for(const [workerId,group] of groups){
      const dates=group.map(x=>x.row.service_date).sort();
      const res=await db.from('service_schedules')
        .select('id,service_date,start_time,end_time,schedule_status,staffing_status')
        .eq('care_worker_id',workerId)
        .neq('schedule_status','cancelled')
        .neq('staffing_status','pending_substitute')
        .gte('service_date',dates[0])
        .lte('service_date',dates[dates.length-1]);
      if(res.error) throw res.error;

      group.forEach(item=>{
        const conflict=(res.data||[]).some(existing=>
          !affectedIds.has(existing.id) &&
          existing.service_date===item.row.service_date &&
          existing.start_time<item.row.end_time &&
          existing.end_time>item.row.start_time
        );
        if(conflict) count++;
      });
    }
    return count;
  }

  async function applyLeaveAndSubstitution(){
    if(!leaveDraft) return;
    const actions=leaveSelectedActions();

    if(actions.some(x=>x.action==='substitute' && !x.substituteWorkerId)){
      toast('請為所有「指定代班」的班表選擇代班居服員。','error');
      return;
    }

    const btn=$('modalConfirm');
    if(btn) btn.disabled=true;
    showLoading(true);

    try{
      const conflictCount=await leaveConflictCount(actions);
      if(conflictCount){
        const ok=confirm(
          '⚠️ 代班時間衝突\n\n' +
          '有 '+conflictCount+' 筆代班與新居服員既有班表時間重疊。\n\n' +
          '仍要套用本次請假／代班處理嗎？'
        );
        if(!ok) return;
      }

      const leaveRes=await db.from('care_worker_leaves')
        .insert({
          care_worker_id:leaveDraft.workerId,
          start_date:leaveDraft.startDate,
          start_time:leaveDraft.isAllDay?null:leaveDraft.startTime,
          end_date:leaveDraft.endDate,
          end_time:leaveDraft.isAllDay?null:leaveDraft.endTime,
          is_all_day:leaveDraft.isAllDay,
          reason:leaveDraft.reason,
          created_by:state.user.id
        })
        .select('id')
        .single();
      if(leaveRes.error) throw leaveRes.error;

      const leaveId=leaveRes.data.id;
      for(const item of actions){
        const base={
          original_care_worker_id:item.row.original_care_worker_id || item.row.care_worker_id,
          leave_request_id:leaveId,
          updated_by:state.user.id,
          updated_at:new Date().toISOString()
        };

        let payload;
        if(item.action==='pending'){
          payload={
            ...base,
            staffing_status:'pending_substitute'
          };
        }else if(item.action==='substitute'){
          payload={
            ...base,
            care_worker_id:item.substituteWorkerId,
            staffing_status:'substituted'
          };
        }else{
          payload={
            ...base,
            schedule_status:'cancelled',
            staffing_status:'normal'
          };
        }

        const updateRes=await db.from('service_schedules')
          .update(payload)
          .eq('id',item.row.id)
          .select('id');
        if(updateRes.error) throw updateRes.error;
        if(!(updateRes.data||[]).length) throw new Error('有班表未能完成異動。');
      }

      closeModal();
      leaveDraft=null;
      navigate('schedule');
      toast('請假／代班處理已完成。','success');
    }catch(err){
      console.error(err);
      toast('請假／代班處理失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(btn && document.body.contains(btn)) btn.disabled=false;
    }
  }

  function pendingWorkerName(id){
    return pendingDraft?.workers?.find(x=>x.id===id)?.worker_name || '未指定';
  }

  function pendingCaseName(id){
    return pendingDraft?.cases?.find(x=>x.id===id)?.case_name || '未找到個案';
  }

  function pendingItemsFor(scheduleId){
    return (pendingDraft?.items || []).filter(x=>x.schedule_id===scheduleId);
  }

  function pendingLeaveReason(leaveId){
    return pendingDraft?.leaves?.find(x=>x.id===leaveId)?.reason || '';
  }

      if(casesRes.error) throw casesRes.error;
      if(itemsRes.error) throw itemsRes.error;
      if(leavesRes.error) throw leavesRes.error;

            </div>
          </div>
          <div class="field full"><label>代班居服員 *</label>
            <select id="pendingSubstituteWorker">
              <option value="">請選擇代班居服員</option>
              ${candidates.map(w=>`<option value="${w.id}">${escapeHTML(w.worker_name)}</option>`).join('')}
            </select>
          </div>
        </div>
      `,
      onConfirm:()=>applyPendingSubstitute(scheduleId)
    });
  }

  async function refreshAfterPendingAction(message){
    if(typeof window.v52ScheduleRefresh==='function'){
      await window.v52ScheduleRefresh();
    }
    toast(message,'success');
    await openPendingSubstituteModal();
  }

  async function applyPendingSubstitute(scheduleId){
    const row=pendingDraft?.schedules?.find(x=>x.id===scheduleId);
    const workerId=$('pendingSubstituteWorker')?.value;
    if(!row || !workerId){
      toast('請選擇代班居服員。','error');
      return;
    }

    const btn=$('modalConfirm');
    if(btn) btn.disabled=true;
    showLoading(true);

    try{
      const conflictRes=await db.from('service_schedules')
        .select('id')
        .eq('service_date',row.service_date)
        .eq('care_worker_id',workerId)
        .neq('id',row.id)
        .neq('schedule_status','cancelled')
        .neq('staffing_status','pending_substitute')
        .lt('start_time',row.end_time)
        .gt('end_time',row.start_time)
        .limit(1);
      if(conflictRes.error) throw conflictRes.error;

      if((conflictRes.data||[]).length){
        const ok=confirm('⚠️ 排班時間重疊\n\n這位居服員在相同時段已有其他班表。\n\n仍要指定為代班人員嗎？');
        if(!ok) return;
      }

      const originalId=row.original_care_worker_id || row.care_worker_id;
      const updateRes=await db.from('service_schedules')
        .update({
          care_worker_id:workerId,
          original_care_worker_id:originalId,
          staffing_status:'substituted',
          updated_by:state.user.id,
          updated_at:new Date().toISOString()
        })
        .eq('id',row.id)
        .select('id');
      if(updateRes.error) throw updateRes.error;
      if(!(updateRes.data||[]).length) throw new Error('班表未能完成代班異動。');

      await refreshAfterPendingAction('已指定代班居服員。');
    }catch(err){
      console.error(err);
      toast('指定代班失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(btn && document.body.contains(btn)) btn.disabled=false;
    }
  }

  async function restorePendingSchedule(scheduleId){
    const row=pendingDraft?.schedules?.find(x=>x.id===scheduleId);
    if(!row) return;
    const originalId=row.original_care_worker_id || row.care_worker_id;
    const ok=confirm(
      '確定要解除待代班並恢復原排班嗎？\n\n' +
      leaveDateLabel(row.service_date)+' '+leaveTime(row.start_time)+'–'+leaveTime(row.end_time)+'\n' +
      pendingCaseName(row.case_id)+'\n原居服員：'+pendingWorkerName(originalId)
    );
    if(!ok) return;

    showLoading(true);
    try{
      const updateRes=await db.from('service_schedules')
        .update({
          care_worker_id:originalId,
          staffing_status:'normal',
          updated_by:state.user.id,
          updated_at:new Date().toISOString()
        })
        .eq('id',row.id)
        .select('id');
      if(updateRes.error) throw updateRes.error;
      if(!(updateRes.data||[]).length) throw new Error('班表未能恢復原排班。');

      await refreshAfterPendingAction('已解除待代班並恢復原排班。');
    }catch(err){
      console.error(err);
      toast('恢復原排班失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
    }
  }

  async function cancelPendingSchedule(scheduleId){

  document.addEventListener('click',event=>{
