  function scheduleMonday(date){
    const d = new Date(date);
    d.setHours(0,0,0,0);
    const offset = (d.getDay() + 6) % 7;
    d.setDate(d.getDate() - offset);
    return d;
  }

  function scheduleAddDays(date, days){
    const d = new Date(date);
    d.setDate(d.getDate() + days);
    return d;
  }

  function scheduleDateISO(date){
    return isoDate(date);
  }

  function scheduleDateLabel(date){
    return `${date.getMonth()+1}/${date.getDate()}`;
  }

  function scheduleWeekLabel(start){
    const end = scheduleAddDays(start, 6);
    return `${start.getFullYear()}年${start.getMonth()+1}月${start.getDate()}日－${end.getFullYear()}年${end.getMonth()+1}月${end.getDate()}日`;
  }

  function scheduleTimeLabel(value){
    return String(value || '').slice(0,5);
  }

  function scheduleMinutes(value){
    const [h,m] = String(value || '00:00').split(':').map(Number);
    return h * 60 + m;
  }

  function scheduleQty(value){
    const n = Number(value);
    if(!Number.isFinite(n)) return value || '1';
    return Number.isInteger(n) ? String(n) : String(n).replace(/0+$/,'').replace(/\.$/,'');
  }

  function scheduleStatusLabel(value){
    return ({planned:'已排定',confirmed:'已確認',completed:'已完成',cancelled:'已取消'})[value] || value || '—';
  }

  function canUseScheduleAdmin(){
    return !!state.profile && ['supervisor','business_manager','organization_manager','admin'].includes(state.profile.role);
  }

  function scheduleGroupFromCode(code){
    const match = String(code || '').trim().toUpperCase().match(/^[A-Z]+/);

  }

  async function saveNewSchedule(){
    const date = $('scheduleAddDate')?.value;
    const start = $('scheduleAddStart')?.value;
    const end = $('scheduleAddEnd')?.value;
    const caseId = $('scheduleAddCase')?.value;
    const workerId = $('scheduleAddWorker')?.value;
    const status = $('scheduleAddStatus')?.value || 'confirmed';
    const notes = $('scheduleAddNotes')?.value?.trim() || null;

    const items = [...document.querySelectorAll('#scheduleServiceRows .schedule-service-row')]
      .map(row=>({
        code: row.querySelector('.schedule-service-code')?.value?.trim().toUpperCase() || '',
        quantity: Number(row.querySelector('.schedule-service-qty')?.value || 0)
      }))
      .filter(x=>x.code);

    if(!date || !start || !end || !caseId || !workerId){
      toast('請完整填寫日期、時間、個案與居服員。','error');
      return;
    }
    if(end<=start){
      toast('結束時間必須晚於開始時間。','error');
      return;
    }
    if(!items.length){
      toast('請至少填寫 1 項服務碼。','error');
      return;
    }
    if(items.some(x=>!Number.isFinite(x.quantity) || x.quantity<=0)){
      toast('服務項目的支數必須大於 0。','error');
      return;
    }

    const confirmBtn = $('modalConfirm');
    if(confirmBtn) confirmBtn.disabled = true;
    showLoading(true);

    try{
      const conflictRes = await db.from('service_schedules')
        .select('id,start_time,end_time,schedule_status')
        .eq('service_date',date)
        .eq('care_worker_id',workerId)
        .neq('schedule_status','cancelled')
        .lt('start_time',end)
        .gt('end_time',start)
        .limit(1);

      if(conflictRes.error) throw conflictRes.error;
      if((conflictRes.data || []).length){
        toast('此居服員在這個時段已有其他班表，請先確認時間。','error');
        return;
      }

      const scheduleRes = await db.from('service_schedules')
        .insert({
          service_date:date,
          start_time:start,
          end_time:end,
          case_id:caseId,
          care_worker_id:workerId,
          schedule_status:status,
          notes,
          created_by:state.user.id,
          updated_by:state.user.id
        })
        .select('id')
        .single();

      if(scheduleRes.error) throw scheduleRes.error;

      const itemRows = items.map(x=>({
        schedule_id:scheduleRes.data.id,
        service_group:scheduleGroupFromCode(x.code),
        service_code:x.code,
        quantity:x.quantity
      }));

      const itemsRes = await db.from('schedule_service_items').insert(itemRows);
      if(itemsRes.error) throw itemsRes.error;

      state.scheduleWeekStart = scheduleDateISO(scheduleMonday(new Date(date+'T00:00:00')));
      closeModal();
      await loadScheduleWeek();
      renderSchedulePage();
      toast('班表已建立。','success');
    }catch(err){
      console.error(err);
      toast('建立班表失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(confirmBtn && document.body.contains(confirmBtn)) confirmBtn.disabled = false;
    }
  }

  async function loadScheduleWeek(){
    if(!canUseScheduleAdmin()) return;
    state.scheduleLoading = true;
    showLoading(true);

    try{
      const start = new Date(state.scheduleWeekStart + 'T00:00:00');
      const end = scheduleAddDays(start, 6);

      const [workersRes, schedulesRes] = await Promise.all([
        db.from('care_workers')
          .select('id,worker_name,employment_status')
          .eq('employment_status','active')
          .order('worker_name'),

        db.from('service_schedules')
          .select('id,service_date,start_time,end_time,case_id,care_worker_id,schedule_status,notes')
          .gte('service_date', scheduleDateISO(start))
          .lte('service_date', scheduleDateISO(end))
          .order('service_date')
          .order('start_time')
      ]);

      if(workersRes.error) throw workersRes.error;
      if(schedulesRes.error) throw schedulesRes.error;

      state.scheduleWorkers = workersRes.data || [];
      state.scheduleRows = schedulesRes.data || [];

      const scheduleIds = [...new Set(state.scheduleRows.map(x=>x.id))];
      const caseIds = [...new Set(state.scheduleRows.map(x=>x.case_id).filter(Boolean))];

      const [casesRes, itemsRes] = await Promise.all([
        caseIds.length
          ? db.from('care_cases').select('id,case_name').in('id',caseIds)
          : Promise.resolve({data:[],error:null}),

        scheduleIds.length
          ? db.from('schedule_service_items')
              .select('id,schedule_id,service_group,service_code,service_name,quantity')
              .in('schedule_id',scheduleIds)
              .order('service_code')
          : Promise.resolve({data:[],error:null})
      ]);

      if(casesRes.error) throw casesRes.error;
      if(itemsRes.error) throw itemsRes.error;

      state.scheduleCases = casesRes.data || [];
      state.scheduleItems = itemsRes.data || [];

    }catch(err){
      console.error(err);
      toast('讀取班表失敗：' + (err?.message || err), 'error');
    }finally{
      state.scheduleLoading = false;
      showLoading(false);
    }
  }

      renderSchedulePage();
    });

    $('scheduleReload').addEventListener('click', async()=>{
      await loadScheduleWeek();
      renderSchedulePage();
      toast('已重新整理班表','success');
    });
  }

  const baseNavigateForSchedule = navigate;

  navigate = function(id){
    if(id !== 'schedule'){
      return baseNavigateForSchedule(id);
    }

    const m = modules.find(x=>x.id==='schedule') || modules[0];

    state.page = 'schedule';

    renderNav();

    $('topTitle').textContent = m.label;
    $('crumb').textContent = m.crumb;

    $('sidebar').classList.remove('open');

    window.scrollTo({
      top:0,
      behavior:'auto'
    });

    if(!canUseScheduleAdmin()){
      renderSchedulePage();
      return;
    }

    $('content').innerHTML = `
      <div class="empty compact">
        <div class="icon">📅</div>
        <h3>正在載入班表</h3>
        <p>請稍候。</p>
      </div>
    `;

    loadScheduleWeek().then(renderSchedulePage);
  };
})();
