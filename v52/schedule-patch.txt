  function scheduleMonday(date){
    const d = new Date(date);
    d.setHours(0,0,0,0);
    const offset = (d.getDay() + 6) % 7;
    d.setDate(d.getDate() - offset);
    return d;
  }

  function scheduleAddDays(date, days){
    const d = new Date(date);
    d.setDate(d.getDate() + days);
    return d;
  }

  function scheduleDateISO(date){
    return isoDate(date);
  }

  function scheduleDateLabel(date){
    return `${date.getMonth()+1}/${date.getDate()}`;
  }

  function scheduleWeekLabel(start){
    const end = scheduleAddDays(start, 6);
    return `${start.getFullYear()}年${start.getMonth()+1}月${start.getDate()}日－${end.getFullYear()}年${end.getMonth()+1}月${end.getDate()}日`;
  }

  function scheduleTimeLabel(value){
    return String(value || '').slice(0,5);
  }

  function scheduleMinutes(value){
    const [h,m] = String(value || '00:00').split(':').map(Number);
    return h * 60 + m;
  }

  function scheduleQty(value){
    const n = Number(value);
    if(!Number.isFinite(n)) return value || '1';
    return Number.isInteger(n) ? String(n) : String(n).replace(/0+$/,'').replace(/\.$/,'');
  }

  function scheduleConflictIds(rows=state.scheduleRows){
    const ids = new Set();
    const active = rows.filter(x=>x.schedule_status!=='cancelled');
    for(let i=0;i<active.length;i++){
      for(let j=i+1;j<active.length;j++){
        const a = active[i], b = active[j];
        if(a.service_date!==b.service_date || a.care_worker_id!==b.care_worker_id) continue;
        if(a.start_time < b.end_time && b.start_time < a.end_time){
          ids.add(a.id);
          ids.add(b.id);
        }
      }
    }
    return ids;
  }

  function updateScheduleEndTime(){
    const startInput = $('scheduleAddStart');
    const endInput = $('scheduleAddEnd');
    if(!startInput || !endInput) return;

    const start = startInput.value;
    if(!start){
      endInput.value = '';
      return;
    }

    let totalMinutes = 0;
    document.querySelectorAll('#scheduleServiceRows .schedule-service-row').forEach(row=>{
      const code = row.querySelector('.schedule-service-code')?.value || '';
      const qty = Number(row.querySelector('.schedule-service-qty')?.value || 0);
      const info = scheduleServiceInfo(code);
      if(info && info.minutes>0 && qty>0) totalMinutes += info.minutes * qty;
    });

    if(totalMinutes<=0){
      endInput.value = '';
      return;
    }

    const [h,m] = start.split(':').map(Number);
    const total = h*60 + m + totalMinutes;
    const endH = Math.floor((total % (24*60))/60);
    const endM = total % 60;
    endInput.value = `${String(endH).padStart(2,'0')}:${String(endM).padStart(2,'0')}`;
  }

  function addScheduleServiceRow(code='', quantity=1){
    const wrap = $('scheduleServiceRows');
    if(!wrap) return;
    const row = document.createElement('div');
    row.className = 'schedule-service-row';
    const safeQty = Math.min(8,Math.max(1,Number(quantity)||1));
    row.innerHTML = `
      <select class="schedule-service-code">${scheduleServiceOptions(code)}</select>
      <span class="schedule-service-times">×</span>
      <select class="schedule-service-qty">${Array.from({length:8},(_,i)=>i+1).map(n=>`<option value="${n}" ${n===safeQty?'selected':''}>${n}</option>`).join('')}</select>
      <button type="button" class="btn schedule-remove-item" title="移除">×</button>
    `;

    row.querySelector('.schedule-service-code').addEventListener('change',updateScheduleEndTime);
    row.querySelector('.schedule-service-qty').addEventListener('change',updateScheduleEndTime);
    row.querySelector('.schedule-remove-item').addEventListener('click',()=>{
      if(wrap.querySelectorAll('.schedule-service-row').length<=1){
        row.querySelector('.schedule-service-code').value='';
        row.querySelector('.schedule-service-qty').value='1';
        updateScheduleEndTime();
        return;
      }
      row.remove();
      updateScheduleEndTime();
    });
    wrap.appendChild(row);
    updateScheduleEndTime();
  }

  function scheduleScopeLabel(scope){

  async function scheduleScopeRows(row,scope){
    if(!row?.recurrence_group_id || scope==='single') return [row];
    let query=db.from('service_schedules')
      .select('id,service_date,start_time,end_time,case_id,care_worker_id,schedule_status,recurrence_group_id,notes')
      .eq('recurrence_group_id',row.recurrence_group_id)
      .order('service_date')
      .order('start_time');
    if(scope==='future') query=query.gte('service_date',row.service_date);
    const res=await query;
    if(res.error) throw res.error;
    return res.data || [];
  }

  async function deleteSchedule(scheduleId){
    const row=state.scheduleRows.find(x=>x.id===scheduleId);
    if(!row){
      toast('找不到這筆班表，請重新整理後再試。','error');
      return;
    }

    const scope=row.recurrence_group_id ? ($('scheduleEditScope')?.value||'single') : 'single';
    const deleteBtn=$('scheduleDeleteBtn');
    if(deleteBtn) deleteBtn.disabled=true;
    showLoading(true);

    try{
      const targets=await scheduleScopeRows(row,scope);
      if(!targets.length) throw new Error('找不到要刪除的班表。');

      const rangeText=scheduleScopeLabel(scope);
      const ok=confirm(
        '確定要刪除班表嗎？\n\n' +
        '範圍：'+rangeText+'\n' +
        '共 '+targets.length+' 筆班表\n\n' +
        '刪除後無法復原。'
      );
      if(!ok) return;

      const ids=targets.map(x=>x.id);
      const res=await db.from('service_schedules')
        .delete()
        .in('id',ids)
        .select('id');

      if(res.error) throw res.error;
      if((res.data||[]).length!==ids.length){
        throw new Error('部分班表未能刪除，請重新整理後確認。');
      }

      closeModal();
      await loadScheduleWeek();

    }catch(err){
      console.error(err);
      toast('刪除班表失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(deleteBtn && document.body.contains(deleteBtn)) deleteBtn.disabled=false;
    }
  }

  async function saveScheduleEdit(scheduleId){
    const row=state.scheduleRows.find(x=>x.id===scheduleId);
    if(!row){
      toast('找不到這筆班表，請重新整理後再試。','error');
      return;
    }

    const scope=row.recurrence_group_id ? ($('scheduleEditScope')?.value||'single') : 'single';
    const date=$('scheduleAddDate')?.value;
    const start=$('scheduleAddStart')?.value;
    const end=$('scheduleAddEnd')?.value;
    const caseId=$('scheduleAddCase')?.value;
    const workerId=$('scheduleAddWorker')?.value;
    const status=$('scheduleAddStatus')?.value||'confirmed';
    const notes=$('scheduleAddNotes')?.value?.trim()||null;

    const items=[...document.querySelectorAll('#scheduleServiceRows .schedule-service-row')]
      .map(itemRow=>({
        code:itemRow.querySelector('.schedule-service-code')?.value?.trim().toUpperCase()||'',
        quantity:Number(itemRow.querySelector('.schedule-service-qty')?.value||0)
      }))
      .filter(x=>x.code);

    if(!date||!start||!caseId||!workerId){
      toast('請完整填寫日期、開始時間、個案與居服員。','error');
      return;
    }
    if(!end){
      toast('目前選擇的服務項目沒有可計算的服務時間，請確認服務碼別與支數。','error');
      return;
    }
    if(end<=start){
      toast('自動計算的結束時間異常，請確認服務項目。','error');
      return;
    }
    if(!items.length){
      toast('請至少填寫 1 項服務碼。','error');
      return;
    }
    if(items.some(x=>!Number.isInteger(x.quantity)||x.quantity<1||x.quantity>8)){
      toast('服務項目的支數必須介於 1～8。','error');
      return;
    }

    const confirmBtn=$('modalConfirm');
    if(confirmBtn) confirmBtn.disabled=true;
    showLoading(true);

    try{
      const targets=await scheduleScopeRows(row,scope);
      if(!targets.length) throw new Error('找不到要修改的班表。');

      // 單班可修改日期；整組／後續則保留每一班原日期。
      const proposed=targets.map(target=>({
        id:target.id,
        service_date:scope==='single'?date:target.service_date
      }));

      // 檢查修改後與其他班表是否重疊；被本次修改的班表彼此不算衝突。
      if(status!=='cancelled'){
        const targetIds=new Set(targets.map(x=>x.id));
        const dates=proposed.map(x=>x.service_date).sort();
        const existingRes=await db.from('service_schedules')
          .select('id,service_date,start_time,end_time,schedule_status')
          .eq('care_worker_id',workerId)
          .neq('schedule_status','cancelled')
          .gte('service_date',dates[0])
          .lte('service_date',dates[dates.length-1]);
        if(existingRes.error) throw existingRes.error;

        const proposedDates=new Set(dates);
        const conflicts=(existingRes.data||[]).filter(x=>
          !targetIds.has(x.id) &&
          proposedDates.has(x.service_date) &&
          x.start_time<end &&
          x.end_time>start
        );
        if(conflicts.length){
          const conflictDays=new Set(conflicts.map(x=>x.service_date)).size;
          const proceed=confirm(
            '⚠️ 排班時間重疊\n\n' +
            '修改範圍：'+scheduleScopeLabel(scope)+'\n' +
            '其中 '+conflictDays+' 個服務日會與此居服員其他班表時間重疊。\n\n' +
            '仍要儲存修改嗎？'
          );
          if(!proceed) return;
        }
      }

      let recurrenceValue=row.recurrence_group_id;
      if(row.recurrence_group_id && scope==='single'){
        // 單獨修改固定班表後，脫離原固定群組，避免日後整組修改覆蓋這個例外。
        recurrenceValue=null;
      }else if(row.recurrence_group_id && scope==='future'){
        // 從本班起切成新的固定群組，讓過去紀錄與新規則分開管理。
        recurrenceValue=crypto.randomUUID();
      }

      const updatePayload={
        start_time:start,
        end_time:end,
        case_id:caseId,
        care_worker_id:workerId,
        schedule_status:status,
        notes,
        updated_by:state.user.id,
        updated_at:new Date().toISOString()
      };

      if(scope==='single'){
        updatePayload.service_date=date;
        updatePayload.recurrence_group_id=recurrenceValue;
      }else if(scope==='future'){
        updatePayload.recurrence_group_id=recurrenceValue;
      }

      const ids=targets.map(x=>x.id);
      const scheduleRes=await db.from('service_schedules')
        .update(updatePayload)
        .in('id',ids)
        .select('id');

      if(scheduleRes.error) throw scheduleRes.error;
      if((scheduleRes.data||[]).length!==ids.length){
        throw new Error('部分班表未能更新，請重新整理後確認。');
      }

      const deleteItemsRes=await db.from('schedule_service_items')
        .delete()
        .in('schedule_id',ids);
      if(deleteItemsRes.error) throw deleteItemsRes.error;

      const itemRows=[];
      ids.forEach(id=>{
        items.forEach(x=>{
          const info=scheduleServiceInfo(x.code);
          itemRows.push({
            schedule_id:id,
            service_group:scheduleGroupFromCode(x.code),
            service_code:x.code,
            service_name:info?.name||null,
            quantity:x.quantity
          });
        });
      });

      const insertItemsRes=await db.from('schedule_service_items').insert(itemRows);
      if(insertItemsRes.error) throw insertItemsRes.error;

      const focusDate=scope==='single'?date:row.service_date;
      state.scheduleWeekStart=scheduleDateISO(scheduleMonday(new Date(focusDate+'T00:00:00')));
      closeModal();
      await loadScheduleWeek();
      renderSchedulePage();
      toast(scope==='single'?'班表已更新。':`已更新 ${ids.length} 筆班表。`,'success');
    }catch(err){
      console.error(err);
      toast('更新班表失敗：'+(err?.message||err),'error');
    }finally{
      showLoading(false);
      if(confirmBtn && document.body.contains(confirmBtn)) confirmBtn.disabled=false;
    }
  }

        quantity: Number(row.querySelector('.schedule-service-qty')?.value || 0)
      }))

        .eq('care_worker_id',workerId)
        .neq('schedule_status','cancelled')

      if(confirmBtn && document.body.contains(confirmBtn)) confirmBtn.disabled = false;
    }
  }

  async function loadScheduleWeek(){
    if(!canUseScheduleAdmin()) return;
    state.scheduleLoading = true;
    showLoading(true);

    try{
      const start = new Date(state.scheduleWeekStart + 'T00:00:00');
      const end = scheduleAddDays(start, 6);

      const [workersRes, schedulesRes] = await Promise.all([
        db.from('care_workers')
          .select('id,worker_name,employment_status')
          .eq('employment_status','active')
          .order('worker_name'),

        db.from('service_schedules')
          .select('id,service_date,start_time,end_time,case_id,care_worker_id,schedule_status,recurrence_group_id,notes')
          .gte('service_date', scheduleDateISO(start))
          .lte('service_date', scheduleDateISO(end))
          .order('service_date')
          .order('start_time')
      ]);

      if(workersRes.error) throw workersRes.error;
      if(schedulesRes.error) throw schedulesRes.error;

      state.scheduleWorkers = workersRes.data || [];
      state.scheduleRows = schedulesRes.data || [];

      const scheduleIds = [...new Set(state.scheduleRows.map(x=>x.id))];
      const caseIds = [...new Set(state.scheduleRows.map(x=>x.case_id).filter(Boolean))];

      const [casesRes, itemsRes] = await Promise.all([
        caseIds.length
          ? db.from('care_cases').select('id,case_name').in('id',caseIds)
          : Promise.resolve({data:[],error:null}),

        scheduleIds.length
          ? db.from('schedule_service_items')
              .select('id,schedule_id,service_group,service_code,service_name,quantity')
              .in('schedule_id',scheduleIds)
              .order('service_code')
          : Promise.resolve({data:[],error:null})
      ]);

      if(casesRes.error) throw casesRes.error;
      if(itemsRes.error) throw itemsRes.error;

      state.scheduleCases = casesRes.data || [];
      state.scheduleItems = itemsRes.data || [];

    }catch(err){
      console.error(err);
      toast('讀取班表失敗：' + (err?.message || err), 'error');
    }finally{
      state.scheduleLoading = false;
      showLoading(false);
    }
  }
