  async function loadStaffManagement(force=false){
    if(state.staffMgmtLoading)return;
    if(state.staffMgmtLoaded&&!force)return;
    state.staffMgmtLoading=true;
    try{
      const [systemRes,workersRes,hoursRes]=await Promise.all([
        db.from('staff_users')
          .select('id,email,display_name,role,is_active,hire_date,agreed_daily_hours,can_supervise,regular_day_weekday,created_at,updated_at')
          .order('is_active',{ascending:false})
          .order('display_name',{ascending:true}),
        db.from('care_workers')
          .select('id,staff_user_id,worker_name,phone,employment_status,hire_date,leave_date,transportation,service_areas,notes,created_at,updated_at')
          .order('employment_status',{ascending:true})
          .order('worker_name',{ascending:true}),
        db.from('staff_work_hours_history')
          .select('id,staff_id,effective_from,agreed_daily_hours,note,created_by_name,created_at')
          .order('effective_from',{ascending:false})
          .order('created_at',{ascending:false})
      ]);
      if(systemRes.error)throw systemRes.error;
      if(workersRes.error)throw workersRes.error;
      if(hoursRes.error)throw hoursRes.error;
      state.staffMgmtSystem=systemRes.data||[];
      state.staffMgmtWorkers=workersRes.data||[];
      state.staffMgmtHoursHistory=hoursRes.data||[];
      state.staffMgmtLoaded=true;
    }catch(err){
      console.error('讀取人員管理資料失敗',err);
      toast('讀取人員資料失敗：'+(err?.message||err),'error');
      throw err;
    }finally{
      state.staffMgmtLoading=false;
    }
  }

  function filteredSystem(){
    const q=String(state.staffMgmtQuery||'').trim().toLowerCase();
    return (state.staffMgmtSystem||[]).filter(s=>{
      if(!state.staffMgmtShowInactive&&s.is_active===false)return false;
      if(!q)return true;
      return [s.display_name,s.email,roleText(s.role)].some(v=>String(v||'').toLowerCase().includes(q));
    });
  }

  function filteredWorkers(){
    const q=String(state.staffMgmtQuery||'').trim().toLowerCase();
    return (state.staffMgmtWorkers||[]).filter(w=>{
      if(!state.staffMgmtShowInactive&&w.employment_status!=='active')return false;
      if(!q)return true;
      return [w.worker_name,w.phone,w.service_areas,w.transportation,workerStatusText(w.employment_status)].some(v=>String(v||'').toLowerCase().includes(q));
    });
  }

  function systemCompleteness(s){
    const missing=[];
    if(!s.hire_date)missing.push('到職日');
    if(!(Number(s.agreed_daily_hours)>0))missing.push('每日工時');
    if(s.regular_day_weekday===null||s.regular_day_weekday===undefined)missing.push('固定例假');
    return missing;
  }

  function workHoursHistoryFor(staffId){
    return (state.staffMgmtHoursHistory||[])
      .filter(x=>String(x.staff_id)===String(staffId))
      .sort((a,b)=>String(b.effective_from).localeCompare(String(a.effective_from))||String(b.created_at).localeCompare(String(a.created_at)));
  }

  async function saveSystemStaff(s){
    if(!canEditSystem()){toast('目前帳號沒有修改此資料的權限','warn');return;}
    const hireDate=$('smHireDate')?.value||null;
    const dailyRaw=$('smDailyHours')?.value;
    const dailyHours=dailyRaw===''||dailyRaw===null?null:Number(dailyRaw);
    const hoursEffectiveFrom=$('smHoursEffectiveFrom')?.value||null;
    const weekday=Number($('smRegularDay')?.value);
    const oldHours=Number(s.agreed_daily_hours);
    const hoursChanged=(dailyHours===null&&s.agreed_daily_hours!==null&&s.agreed_daily_hours!==undefined)
      ||(dailyHours!==null&&(!Number.isFinite(oldHours)||Math.abs(dailyHours-oldHours)>0.0001));

    if(dailyHours!==null&&(!Number.isFinite(dailyHours)||dailyHours<=0||dailyHours>8)){
      toast('約定每日工時需大於0且不超過8小時','warn');
      return;
    }
    if(hoursChanged&&!hoursEffectiveFrom){
      toast('修改約定每日工時時，請填寫工時生效日','warn');
      return;
    }
    if(!Number.isInteger(weekday)||weekday<0||weekday>6){
      toast('請選擇固定例假日','warn');
      return;
    }

    showLoading(true);
    try{
      const {error:baseError}=await db.from('staff_users')
        .update({
          hire_date:hireDate,
          regular_day_weekday:weekday,
          updated_at:new Date().toISOString()
        })
        .eq('id',s.id);
      if(baseError)throw baseError;

      if(hoursChanged){
        if(dailyHours===null){
          throw new Error('約定每日工時不可清空；如需變更請輸入新的工時與生效日');
        }
        const {error:hoursError}=await db.rpc('save_staff_work_hours_timeline',{
          p_staff_id:s.id,
          p_effective_from:hoursEffectiveFrom,
          p_agreed_daily_hours:dailyHours,
          p_note:'由人員管理調整'
        });
        if(hoursError)throw hoursError;
      }

      state.staffMgmtLoaded=false;
      state.compLeaveExpiryLoaded=false;
      state.compLeaveExpiryStaff=[];
      state.annualAutoLoadedAt=0;
      state.personalHoursLoaded=false;
      state.payrollRatesLoaded=false;

      await loadStaffManagement(true);
      const updated=(state.staffMgmtSystem||[]).find(x=>String(x.id)===String(s.id));
      renderSystemDetail(updated?.id||s.id);
      toast(hoursChanged?'人員資料已更新，工時歷程與特休資格已同步檢查':'人員出勤基本資料已更新','success');
    }catch(err){
      console.error(err);
      toast('儲存人員資料失敗：'+(err?.message||err),'error');
    }finally{showLoading(false);}
  }

  async function saveWorker(w,isNew){
    if(!canEditWorkers()){toast('目前帳號沒有修改居服員資料的權限','warn');return;}
    const payload={
      worker_name:String($('smWorkerName')?.value||'').trim(),
      phone:String($('smWorkerPhone')?.value||'').trim()||null,
      employment_status:$('smWorkerStatus')?.value||'active',
      hire_date:$('smWorkerHire')?.value||null,
      leave_date:$('smWorkerLeave')?.value||null,
      transportation:String($('smWorkerTransport')?.value||'').trim()||null,
      service_areas:String($('smWorkerAreas')?.value||'').trim()||null,
      notes:String($('smWorkerNotes')?.value||'').trim()||null,
      updated_at:new Date().toISOString()
    };
    if(!payload.worker_name){toast('請填寫居服員姓名','warn');return;}
    if(!Object.prototype.hasOwnProperty.call(WORKER_STATUS,payload.employment_status)){
      toast('任職狀態不正確','warn');return;
    }

    showLoading(true);
    try{
      let result;
      if(isNew){
        const insertPayload={...payload};
        delete insertPayload.updated_at;
        result=await db.from('care_workers').insert(insertPayload).select('id').single();
      }else{
        result=await db.from('care_workers').update(payload).eq('id',w.id).select('id').single();
      }
      if(result.error)throw result.error;
      state.staffMgmtLoaded=false;
      await loadStaffManagement(true);
      if(isNew&&result.data?.id)renderWorkerForm(result.data.id);
      else renderWorkerForm(w.id);
      toast(isNew?'居服員已建立':'居服員資料已更新','success');
    }catch(err){
      console.error(err);
      toast((isNew?'建立':'儲存')+'居服員失敗：'+(err?.message||err),'error');
    }finally{showLoading(false);}
  }
