// 「內容」與「交辦內容」預設單行；輸入換行或自動折行時即時增高，刪字也會縮回。
function autoGrow(field){
 if(!field)return;
 const adjust=()=>{
  field.style.height='auto';
  field.style.height=Math.max(42,field.scrollHeight)+'px';
 };
 field.addEventListener('input',adjust);

}

async function load(){
 const [m,s]=await Promise.all([
  db.from('team_messages').select('*').order('created_at',{ascending:false}).limit(300),
  db.from('staff_users').select('id,display_name,role,is_active').eq('is_active',true).order('display_name')
 ]);
 if(m.error)throw m.error;if(s.error)throw s.error;
 const messages=m.data||[];let taskRows=[],recipientRows=[],updateRows=[];
 if(messages.length){
  const ids=messages.map(x=>x.id);
  const [a,b,c]=await Promise.all([
   db.from('team_message_tasks').select('*').in('message_id',ids).order('sort_order').limit(3000),
   db.from('team_message_recipients').select('*').in('message_id',ids).limit(3000),
   db.from('team_message_updates').select('*').in('message_id',ids).order('created_at').limit(5000)
  ]);
  if(a.error)throw a.error;if(b.error)throw b.error;if(c.error)throw c.error;
  taskRows=a.data||[];recipientRows=b.data||[];updateRows=c.data||[];
 }
 S.messages=messages;S.staff=(s.data||[]).filter(x=>ROLES.includes(x.role));
 S.tasks=taskRows;S.recipients=recipientRows;S.updates=updateRows;
 void sidebarCount();
}
async function refresh(){

 try{await load();if(S.view==='detail'&&S.id)detail(S.id);else listing();}
 catch(e){console.error(e);html(empty('訊息交辦載入失敗')+'<button class="btn" id="tm-retry">重試</button>');root().querySelector('#tm-retry').onclick=refresh;toast('載入失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
function card(m){
 const ts=tasks(m.id),count=readCount(m.id),own=myRead(m.id);
 const urgency=S.tab==='mine'&&pending(m)?'action':S.tab==='assigned'&&ready(m)?'closing':own&&!own.read_at?'unread':own?'read':'';
 const tag=S.tab==='mine'&&pending(m)?'待我辦理':S.tab==='assigned'&&ready(m)?'待我結案':'';
 const progress=ts.length?'分工完成 '+ts.filter(x=>x.is_done).length+'／'+ts.length:'無分工事項';
 return '<article class="tm-card tm-item '+urgency+'" role="button" tabindex="0" data-open="'+h(m.id)+'" aria-label="查看訊息：'+h(m.description.slice(0,60))+'">'+
  '<div class="tm-line"><div class="tm-meta"><span>建立者：'+h(name(m.created_by_staff_id))+'｜'+h(stamp(m.created_at))+'</span>'+
  '<span>已閱讀 '+count[0]+'／'+count[1]+' 人 · '+h(progress)+(own&&!own.read_at?' · 未閱讀':'')+'</span></div>'+
  '<span class="tm-pill '+status(m)+'">'+labels[status(m)]+'</span></div>'+
  '<div class="tm-main"><div class="tm-snippet">'+h(m.description)+'</div>'+

}
function listing(){
 S.view='list';S.id=null;S.editing=null;
 const pendingCount=S.messages.filter(pending).length,closeCount=S.messages.filter(ready).length;
 const rows=S.messages.filter(m=>{
  if(S.tab==='unread')return myRead(m.id)?.read_at===null;
  if(S.tab==='all')return true;
  if(S.tab==='mine')return isMine(m);
  if(S.tab==='assigned')return delegated(m);
  if(S.tab==='closed')return !!m.closed_at;
  return false;
 });
 if(S.tab==='mine')rows.sort((a,b)=>Number(pending(b))-Number(pending(a)));
 if(S.tab==='assigned')rows.sort((a,b)=>Number(ready(b))-Number(ready(a)));
 if(S.tab==='closed')rows.sort((a,b)=>new Date(b.closed_at)-new Date(a.closed_at));
 const tabs=[['unread','未閱讀'],['all','全部訊息'],['mine','我的待辦'],['assigned','我的交辦'],['closed','已結案']];
 html('<div class="tm-head"><div><h1>訊息交辦</h1><p>消息集中管理，事件回報依時間保留完整歷程。</p></div>'+
  '<button class="btn primary" id="tm-new">＋ 新增訊息</button></div>'+
  '<div class="tm-tabs">'+tabs.map(([id,label])=>{
   const n=id==='mine'?pendingCount:id==='assigned'?closeCount:null;
   return '<button class="tm-tab '+(id===S.tab?'active':'')+'" data-tab="'+id+'">'+label+
    (n===null?'':' <span class="tm-tab-count">'+n+'</span>')+'</button>';
  }).join('')+'<button class="btn sm" id="tm-refresh">重新整理</button></div>'+
  '<div class="tm-list">'+(rows.length?rows.map(card).join(''):empty('目前沒有符合條件的訊息'))+'</div>');
 root().querySelector('#tm-new').onclick=createForm;
 root().querySelector('#tm-refresh').onclick=refresh;
 root().querySelectorAll('[data-tab]').forEach(b=>b.onclick=()=>{S.tab=b.dataset.tab;listing();});
 root().querySelectorAll('[data-open]').forEach(el=>{
  el.onclick=()=>detail(el.dataset.open);
  el.onkeydown=e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();detail(el.dataset.open);}};
 });
}

function chosen(){

function renderChoices(){
 const target=root()?.querySelector('#tm-options');if(!target)return;
 const assignedPeople=assigned();
 target.innerHTML=S.staff.map(x=>{
  const required=x.id===me()||assignedPeople.has(x.id);
  return '<label class="tm-option"><input type="checkbox" data-recipient="'+h(x.id)+'"'+
   (required||S.chosen.has(x.id)?' checked':'')+(required?' disabled':'')+'>'+
   h(x.display_name)+(x.id===me()?'（建立者）':assignedPeople.has(x.id)?'（承辦人）':'')+'</label>';
 }).join('');
 target.querySelectorAll('[data-recipient]').forEach(el=>el.onchange=()=>{
  if(el.checked)S.chosen.add(el.dataset.recipient);else S.chosen.delete(el.dataset.recipient);
 });
}

async function createMessage(){
 if(S.busy)return;
 const description=root().querySelector('#tm-description').value.trim();
 const jobs=[...root().querySelectorAll('.tm-new-task')].map(el=>({
  description:el.querySelector('[data-work]').value.trim(),
  assignee_staff_id:el.querySelector('[data-person]').value
 }));
 if(!description){toast('請填寫訊息說明','error');return;}
 if(jobs.some(x=>!x.assignee_staff_id||!x.description)){toast('請完整填寫各項分工與承辦人','error');return;}
 const btn=root().querySelector('#tm-submit');btn.disabled=true;S.busy=true;
 try{
  const {data,error}=await db.rpc('create_team_message',{
   p_description:description,p_tasks:jobs,p_recipient_ids:chosen()
  });
  if(error)throw error;
  await load();toast('訊息建立成功','success');detail(data);
 }catch(e){console.error(e);btn.disabled=false;toast('建立失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}

  '<div class="tm-compose"><div class="tm-small-head">新增回報（選填）</div>'+
  '<div class="tm-field"><textarea id="tm-compose" rows="3" maxlength="2000" placeholder="例如：已聯繫案主，對方表示同意。"></textarea></div>'+
  '<button class="btn primary" id="tm-send">送出回報</button></div>');
}
function dirty(){
 if(S.view==='create')return createDirty();
 if(S.view!=='detail'||!root())return false;
 if(root().querySelector('#tm-compose')?.value.trim())return true;
 if(S.editing){
  const original=S.updates.find(x=>x.id===S.editing);
  const edited=root().querySelector('[data-report="'+S.editing+'"] [data-edit-body]')?.value;
  return edited!==undefined&&edited!==original?.body;
 }
 return false;
}
function detail(id){
 const m=S.messages.find(x=>x.id===id);if(!m){listing();return;}
 S.view='detail';S.id=id;
 const jobs=tasks(id),canClose=!m.closed_at&&jobs.length&&jobs.every(x=>x.is_done);
 html('<div class="tm-head"><div><h1>訊息詳情</h1><p>建立者：'+h(name(m.created_by_staff_id))+
 '｜'+h(stamp(m.created_at))+'</p></div><button class="btn" id="tm-back">返回列表</button></div>'+
 '<section class="tm-card"><div class="tm-line"><strong>內容</strong>'+
 '<span class="tm-pill '+status(m)+'">'+labels[status(m)]+'</span></div>'+
 '<div class="tm-description">'+h(m.description)+'</div>'+
 (m.closed_at?'<p class="tm-muted">結案時間：'+h(stamp(m.closed_at))+'</p>':'')+'</section>'+
 readPanel(id)+taskPanel(m)+historyPanel(m)+
 (canClose?'<div class="tm-summary"><strong>所有分工已完成，待建立者結案。</strong>'+
  (m.created_by_staff_id===me()?'<div class="tm-actions"><button class="btn primary" id="tm-close">確認結案</button></div>':
  '<p class="tm-muted">僅建立者可以結案。</p>')+'</div>':''));
 root().querySelector('#tm-back').onclick=()=>{if(!dirty()||confirm('尚有未儲存的回報，確定返回？'))listing();};
 root().querySelector('#tm-read')?.addEventListener('change',()=>markRead(id));
 root().querySelectorAll('[data-done]').forEach(input=>input.onchange=()=>setDone(input));
 root().querySelector('#tm-send')?.addEventListener('click',()=>sendReport(id));
 root().querySelectorAll('[data-edit]').forEach(btn=>btn.onclick=()=>{
  if(dirty()){toast('請先送出或清除未儲存的回報','error');return;}
  S.editing=btn.dataset.edit;detail(id);
 });
 root().querySelectorAll('[data-edit-save]').forEach(btn=>btn.onclick=()=>saveReport(id,btn.dataset.editSave));
 root().querySelectorAll('[data-edit-cancel]').forEach(btn=>btn.onclick=()=>{S.editing=null;detail(id);});
 root().querySelector('#tm-close')?.addEventListener('click',()=>closeMessage(id));
}
async function markRead(id){
 if(S.busy)return;
 const input=root().querySelector('#tm-read');if(!input)return;
 if(dirty()){input.checked=false;toast('請先處理未送出的回報','error');return;}
 input.disabled=true;S.busy=true;
 try{
  const {data,error}=await db.from('team_message_recipients').update({read_at:new Date().toISOString()})
   .eq('message_id',id).eq('staff_id',me()).is('read_at',null).select('staff_id');
  if(error)throw error;if(!data?.length)throw new Error('已確認閱讀或不需確認');
  await load();toast('已確認閱讀','success');detail(id);
 }catch(e){input.disabled=false;input.checked=false;toast('確認失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
async function setDone(input){
 const job=S.tasks.find(x=>x.id===input.dataset.done);
 if(!job||S.busy){input.checked=!input.checked;return;}
 if(dirty()){input.checked=!input.checked;toast('請先處理未送出的回報','error');return;}
 const completed=input.checked;input.disabled=true;S.busy=true;
 try{
  const {data,error}=await db.from('team_message_tasks').update({is_done:completed}).eq('id',job.id).select('id');
  if(error)throw error;if(!data?.length)throw new Error('只有承辦人可以修改尚未結案的分工');
  await load();toast(completed?'已標記完成':'已改為未完成','success');detail(job.message_id);
 }catch(e){input.disabled=false;input.checked=!completed;toast('更新失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
async function sendReport(id){
 if(S.busy)return;
 const area=root().querySelector('#tm-compose'),body=area?.value.trim();
 if(!body){toast('請輸入回報內容','error');return;}
 const button=root().querySelector('#tm-send');button.disabled=true;S.busy=true;
 try{
  const {data,error}=await db.from('team_message_updates').insert({message_id:id,staff_id:me(),body}).select('id');
  if(error)throw error;if(!data?.length)throw new Error('送出沒有成功');
  await load();toast('已新增回報','success');detail(id);
 }catch(e){button.disabled=false;toast('送出失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
async function saveReport(id,updateId){
 if(S.busy)return;
 const report=S.updates.find(x=>x.id===updateId);
 if(!report||report.staff_id!==me())return;
 const body=root().querySelector('[data-report="'+updateId+'"] [data-edit-body]')?.value.trim();
 if(!body){toast('回報內容不可空白','error');return;}
 const button=root().querySelector('[data-edit-save="'+updateId+'"]');button.disabled=true;S.busy=true;
 try{
  const {data,error}=await db.from('team_message_updates').update({body}).eq('id',updateId).eq('staff_id',me()).select('id');
  if(error)throw error;if(!data?.length)throw new Error('只有原輸入者可修改回報');
  S.editing=null;await load();toast('回報已修正','success');detail(id);
 }catch(e){button.disabled=false;toast('修改失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
async function closeMessage(id){
 if(S.busy)return;
 if(dirty()){toast('請先處理未送出的回報','error');return;}
 if(!confirm('所有分工已完成，確定結案？'))return;
 S.busy=true;
 try{
  const {data,error}=await db.from('team_messages').update({closed_at:new Date().toISOString()}).eq('id',id).select('id');
  if(error)throw error;if(!data?.length)throw new Error('僅建立者可在全部完成後結案');
  await load();toast('已結案','success');detail(id);
 }catch(e){toast('結案失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
let sidebarSeq=0;
async function sidebarCount(){
 if(!state.profile)return;
 const seq=++sidebarSeq,user=me();
 try{
  const {data,error}=await db.rpc('team_message_attention_count');
  if(error)throw error;if(sidebarSeq!==seq||user!==me())return;
  const button=[...$('nav').querySelectorAll('.nav-btn')].find(x=>x.querySelectorAll('span')[1]?.textContent==='訊息交辦');
  if(!button)return;
  let badge=button.querySelector('.tm-nav-count');
  const count=Math.max(0,Number(data)||0);
  if(!count){badge?.remove();return;}
  if(!badge){badge=document.createElement('span');badge.className='tm-nav-count';button.append(badge);}
  badge.textContent=count>99?'99+':String(count);
  badge.title='我的待辦事件＋待我結案事件：'+count;
 }catch(e){console.error('訊息交辦提醒讀取失敗',e);}
}
