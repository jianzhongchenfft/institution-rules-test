// 「內容」與「交辦內容」預設單行；輸入換行或自動折行時即時增高，刪字也會縮回。
function autoGrow(field){
 if(!field)return;
 const adjust=()=>{
  field.style.height='auto';
  field.style.height=Math.max(42,field.scrollHeight)+'px';
 };
 field.addEventListener('input',adjust);
 adjust();
}

function getTabRows(tab){
 const rows=S.messages.filter(m=>{
  if(tab==='unread')return !m.closed_at&&myRead(m.id)?.read_at===null;
  if(tab==='all')return true;
  if(tab==='mine')return pending(m);
  if(tab==='assigned')return delegated(m);
  if(tab==='favorites')return favorite(m.id);
  return false;
 });
 if(tab==='mine')rows.sort((a,b)=>Number(hasInstruction(b))-Number(hasInstruction(a)));
 if(tab==='assigned')rows.sort((a,b)=>Number(needsHelp(b))-Number(needsHelp(a))||Number(ready(b))-Number(ready(a)));
 return rows;
}

function browseNavigation(id){
 const tab=S.browseTab||S.tab,ids=S.browseIds.length?S.browseIds:getTabRows(tab).map(m=>m.id);
 const index=ids.indexOf(id),allowed=new Set(getTabRows(tab).map(m=>m.id));
 const seek=step=>{
  if(index<0)return null;
  for(let i=index+step;i>=0&&i<ids.length;i+=step){
   if(allowed.has(ids[i]))return ids[i];
  }
  return null;
 };
 return {tab,ids,index,previous:seek(-1),next:seek(1)};
}
async function load(){
 const [m,s,f]=await Promise.all([
  db.from('team_messages').select('*').order('created_at',{ascending:false}).limit(300),
  db.from('staff_users').select('id,display_name,role,is_active').eq('is_active',true).order('display_name'),
  db.from('team_message_favorites').select('message_id').eq('staff_id',me()).limit(5000)
 ]);
 if(m.error)throw m.error;if(s.error)throw s.error;if(f.error)throw f.error;
 const messages=m.data||[];let taskRows=[],recipientRows=[],updateRows=[];
 if(messages.length){
  const ids=messages.map(x=>x.id);
  const [a,b,c]=await Promise.all([
   db.from('team_message_tasks').select('*').in('message_id',ids).order('sort_order').limit(3000),
   db.from('team_message_recipients').select('*').in('message_id',ids).limit(3000),
   db.from('team_message_updates').select('*').in('message_id',ids).order('created_at').limit(5000)
  ]);
  if(a.error)throw a.error;if(b.error)throw b.error;if(c.error)throw c.error;
  taskRows=a.data||[];recipientRows=b.data||[];updateRows=c.data||[];
 }
 S.messages=messages;S.staff=(s.data||[]).filter(x=>ROLES.includes(x.role));
 S.tasks=taskRows;S.recipients=recipientRows;S.updates=updateRows;S.favorites=new Set((f.data||[]).map(x=>x.message_id));
 void sidebarCount();
}
async function toggleFavorite(id){
 if(S.busy)return;
 if(dirty()){toast('請先處理未儲存的內容','error');return;}
 const message=S.messages.find(m=>m.id===id);
 if(!message)return;
 S.busy=true;const adding=!favorite(id);
 try{
  let result;
  if(adding)result=await db.from('team_message_favorites').insert({message_id:id,staff_id:me()});
  else result=await db.from('team_message_favorites').delete().eq('message_id',id).eq('staff_id',me());
  if(result.error)throw result.error;
  await load();toast(adding?'已加入我的最愛':'已從我的最愛移除','success');
  if(S.view==='detail'&&S.id)detail(S.id);
  else listing();
 }catch(e){toast('收藏操作失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}
async function refresh(){
 if(S.busy)return;S.busy=true;
 try{await load();if(S.view==='detail'&&S.id)detail(S.id);else listing();}
 catch(e){console.error(e);html(empty('訊息交辦載入失敗')+'<button class="btn" id="tm-retry">重試</button>');root().querySelector('#tm-retry').onclick=refresh;toast('載入失敗：'+(e.message||e),'error');}
 finally{S.busy=false;}
}

function miniReaderNames(people){
 const names=people.map(p=>name(p.staff_id).trim());
 return names.map((full,i)=>{
  const units=Array.from(full);
  let count=1,short=units.slice(-count).join('');
  while(count<units.length&&names.some((other,j)=>j!==i&&Array.from(other).slice(-count).join('')===short)){
   count+=1;short=units.slice(-count).join('');
  }
  return short||full;
 });
}

 }

 if(S.managingTask){
  const task=S.tasks.find(t=>t.id===S.managingTask.id);
  if(S.managingTask.kind==='cancel')return false;
  if(task){
   const desc=root().querySelector('#tm-manage-description')?.value;
   const person=root().querySelector('#tm-manage-person')?.value;
   const notify=root().querySelector('#tm-manage-notify')?.checked;
   if(desc!==undefined&&(desc!==task.description||person!==task.assignee_staff_id||notify===false))return true;
  }
 }
 if(S.editing){
  const original=S.updates.find(x=>x.id===S.editing);
  const edited=root().querySelector('[data-report="'+S.editing+'"] [data-edit-body]')?.value;
  return edited!==undefined&&edited!==original?.body;
 }
 return false;
}

async function autoReadOnOpen(id){
 const current=myRead(id),user=me();
 if(!current||current.read_at||!user||S.autoReadPending.has(id))return;
 S.autoReadPending.add(id);
 try{
  const {data,error}=await db.from('team_message_recipients')
   .update({read_at:new Date().toISOString()})
   .eq('message_id',id).eq('staff_id',user).is('read_at',null)
   .select('read_at');
  if(error)throw error;
  const readAt=data?.[0]?.read_at;
  if(!readAt)return;
  const row=S.recipients.find(r=>r.message_id===id&&r.staff_id===user);
  if(row)row.read_at=readAt;
  if(S.view==='detail'&&S.id===id){
   const el=root()?.querySelector('[data-reader-staff="'+user+'"]');
   if(el){
    el.classList.remove('unread');el.classList.add('read');
    el.title=name(user)+'：已閱讀（'+stamp(readAt)+'）';
    el.setAttribute('aria-label',name(user)+'已閱讀');
   }
  }
 }catch(e){
  console.error('訊息自動確認閱讀失敗',e);
  toast('自動確認閱讀失敗：'+(e.message||e),'error');
 }finally{S.autoReadPending.delete(id);}
}
