// V5.2 全站介面文字精簡：操作頁只保留必要資訊，詳細規則留在規則/明細/警告
(() => {
  if(!document.getElementById('v52-ui-concise-copy-style')){
    const style=document.createElement('style');
    style.id='v52-ui-concise-copy-style';
    style.textContent=`
      /* 數據卡片的標題與數值已足夠辨識，不再顯示重複說明 */
      .stat-foot{display:none!important}
    `;
    document.head.appendChild(style);
  }

  const removeIfTextIncludes=(root,selector,needles)=>{
    root.querySelectorAll(selector).forEach(el=>{
      const text=(el.textContent||'').replace(/\s+/g,' ').trim();
      if(needles.some(x=>text.includes(x))) el.remove();
    });
  };

  function cleanOvertimePage(){
    if(state.page!=='overtime')return;

    // 「加班規則」頁本身就是詳細規則入口，不精簡其內容。
    if(state.overtimeView!=='rules'){
      const main=document.getElementById('otMain');
      if(main){
        // 移除操作頁中僅用來解釋用途的段落；表格、數值、狀態仍保留。
        removeIfTextIncludes(main,'.ot-rule-card > p',[
          '提供機構負責人發薪前核對',
          '換休依取得先後順序使用',
          '以下為本月到期仍未休完',
          '本月份結算時的薪資基數',
          '此月份是在「金額快照」功能上線前'
        ]);

        // 移除首頁式的大段公式/流程解說；詳細公式仍可在「查看明細」中查閱。
        removeIfTextIncludes(main,'.ot-rule-highlight',[
          '月薪工資總額 ÷ 30 ÷ 約定每日工時',
          '月薪工資總額 ÷ 240',
          '加班費依各員工',
          '加班費試算以',
          '目前先以「待計薪加班時數」',
          '此月份尚未完整結束，需於'
        ]);
      }
    }

    // 申請畫面只保留必要的個人正常工時，不再附帶制度說明。
    const hoursHint=document.getElementById('phApplicantHoursHint');
    if(hoursHint){
      const chip=hoursHint.querySelector('.ph-hours-chip');
      if(chip) hoursHint.innerHTML=`約定每日工時：<span class="ph-hours-chip">${chip.textContent.trim()}</span>`;
      else if((hoursHint.textContent||'').includes('尚未設定')) hoursHint.textContent='約定每日工時尚未設定';
    }

    // 薪資計算設定屬管理操作頁，欄位本身足以理解；移除上方與版本制說明文字。
    const rateModal=document.getElementById('prRateStaff')?.closest('.modal');
    if(rateModal){
      removeIfTextIncludes(rateModal,'.ot-rule-highlight',[
        '月薪工資總額',
        '薪資基數採「版本紀錄」方式保存'
      ]);
    }
  }

  function cleanGeneralPage(){
    // 全站只移除純重複的卡片說明；警告、錯誤、空白狀態與實際資料不動。
    document.querySelectorAll('.stat-foot').forEach(el=>el.setAttribute('aria-hidden','true'));
    cleanOvertimePage();
  }

  let queued=false;
  const observer=new MutationObserver(mutations=>{
    if(queued||!mutations.some(m=>m.addedNodes?.length))return;
    queued=true;
    setTimeout(()=>{queued=false;cleanGeneralPage();},30);
  });
  observer.observe(document.body,{childList:true,subtree:true});

  const baseNavigate=navigate;
  navigate=function(id){
    const r=baseNavigate(id);
    setTimeout(cleanGeneralPage,60);
    return r;
  };

  setTimeout(cleanGeneralPage,150);
})();
